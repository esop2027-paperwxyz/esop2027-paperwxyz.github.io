  IdD : ConD ∅ (A ẋ A) (A ẋ A)
  IdD = non A (end (pair (A [ ε ]) (A [ ε ]) vz vz))

  IdDs : ConDs (A ẋ A) (A ẋ A)
  IdDs = IdD ∷ []

module _ {A : Ty ∅ ℓA} where
  Id : (a b : Tm Γ (A [ ε ])) → Ty Γ ℓA
  Id a b = μ (IdDs A) ⟨ pair (A [ ε ]) (A [ ε ]) a b ⟩

  refI
    : {a : Tm Γ (A [ ε ])}
    → Tm Γ (Id a a)
  refI {a = a} =
    con# (# 0) (pair (A [ ε ]) (A [ ε ]) a a)
      (dpair (⇑ Eq (pair (A [ ε ]) (A [ ε ]) vz vz)
                   (pair (A [ ε ]) (A [ ε ]) (a [ wk ]t) (a [ wk ]t)))
        a
        (reflT' (pair (A [ ε ]) (A [ ε ]) a a)))
