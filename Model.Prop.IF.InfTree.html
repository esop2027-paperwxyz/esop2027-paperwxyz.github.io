data IT₀ : Set where
  leaf : (b : B.Bool) → IT₀
  node : (t : (a : ℕ) → IT₀)
    → IT₀

IT~ : IT₀ → IT₀ → Prop
IT~ (leaf b) (leaf c) = b B.~ c
IT~ (node t) (node u) = ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁) → IT~ (t a₀) (u a₁)
IT~ _        _        = ⊥ₚ

extIT : IT₀ → Prop
extIT (leaf _) = ⊤ₚ
extIT (node t) = ∃ (∀ a → extIT (t a)) λ ext → {a₀ a₁ : ℕ} (a₀₁ : a₀ N.≈ a₁)
  → IT~ (t a₀) (t a₁)

-- `IT~` is a PER on `IT₀`; every proof recurses on the tree.
IT~-refl : (t : IT₀) → (ext : extIT t) → IT~ t t
IT~-refl (leaf b) _        = B.ref b
IT~-refl (node t) (_ , t~) = t~

IT~-sym : ∀ {t u} → IT~ t u → IT~ u t
IT~-sym {leaf b} {leaf c} p = B.sym p
IT~-sym {node t} {node u} p {a₀} {a₁} = λ a₀₁ →
  IT~-sym {t a₁} {u a₀} (p {a₁} {a₀} (N.sym a₀₁))

IT~-trans : ∀ {t u v} → IT~ t u → IT~ u v → IT~ t v
IT~-trans {leaf b} {leaf c} {leaf d} p q = B.trans p q
IT~-trans {node t} {node u} {node v} p q {a₀} {a₂} =
  λ a₀₂ → IT~-trans {t a₀} {u a₂} (p a₀₂) (q N.refl)

-- The setoid of extensional trees, |W_B|.
∣IT∣ : Set
∣IT∣ = Σₚ IT₀ extIT

∣IT~∣ : ∣IT∣ → ∣IT∣ → Prop
∣IT~∣ t t' = IT~ (t .₁) (t' .₁)

IT : Ty Γ lzero
∣ IT ∣ γ    = ∣IT∣
IT .heq γ₀₁ = ∣IT~∣
IT .refl* (t , ext) = IT~-refl t ext
IT .sym*  {x₀ = t₀ , _} {x₁ = t₁ , _} = IT~-sym {t₀} {t₁}
IT .trans* {γ₀} {γ₁} {γ₂} {γ₀₁} {γ₁₂} {t₀ , _} {t₁ , _} {t₂ , _} p q =
  IT~-trans {t₀} {t₁} {t₂} p q
IT .coe _ t = t
IT .coh _ (t , ext) = IT~-refl t ext

-- Constructors of the algebra `\ws : P_B W ⇒ W`.
lea : Tm (Γ ▷ Bool) IT
∣ lea ∣   (_   , b)   = leaf b , tt
lea .resp (γ₀₁ , b₀₁) = b₀₁

nod : Tm (Γ ▷ Nat ⇒ IT) IT
∣ nod ∣ (γ , t , ext) = node (λ n → t n .₁) , (λ n → t n .₂) , ext
nod .resp {γ₀ , t₀} {γ₁ , t₁} (_ , t₀₁) = t₀₁ _ _

nodₑ : (t : Tm Γ (Nat ⇒ IT)) → Tm Γ IT
nodₑ t = nod ⟨ t ⟩t

module Elim₀
    (P : (t : IT₀) (ext : extIT t) → Set ℓ)
    -- `P~ v₀ v₁ t₀₁` — the values come first so the trees `t₀ t₁` are inferred
    -- from them (neither `extIT` nor `P` is invertible); the relation is last.
    (P~ : ∀ {t₀ t₁} {ext₀ : extIT t₀} {ext₁ : extIT t₁}
        → P t₀ ext₀ → P t₁ ext₁ → IT~ t₀ t₁ → Prop ℓ)
    (~refl : ∀ {t ext} (v : P t ext) → P~ v v (IT~-refl t ext))
    (~sym  : ∀ {t₀ t₁} {ext₀ : extIT t₀} {ext₁ : extIT t₁}
                 {v₀ : P t₀ ext₀} {v₁ : P t₁ ext₁} {t₀₁ : IT~ t₀ t₁}
             → P~ v₀ v₁ t₀₁ → P~ v₁ v₀ (IT~-sym {t₀} {t₁} t₀₁))
    (~trans : ∀ {t₀ t₁ t₂} {ext₀ : extIT t₀} {ext₁ : extIT t₁} {ext₂ : extIT t₂}
                  {v₀ : P t₀ ext₀} {v₁ : P t₁ ext₁} {v₂ : P t₂ ext₂}
                  {t₀₁ : IT~ t₀ t₁} {t₁₂ : IT~ t₁ t₂}
              → P~ v₀ v₁ t₀₁ → P~ v₁ v₂ t₁₂ → P~ v₀ v₂ (IT~-trans {t₀} t₀₁ t₁₂))
    (pleaf : (b : B.Bool) → P (leaf b) tt)
    (pleaf~
      : (b₀ b₁ : B.Bool) (b₀₁ : b₀ B.~ b₁) → P~ (pleaf b₀) (pleaf b₁) b₀₁)
    (pnode
      : ((t , t~) : Σₚ (ℕ → ∣IT∣)
            (λ t₀ → ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁) → IT~ (t₀ a₀ .₁) (t₀ a₁ .₁)))
      → (ih : Σₚ ((a : ℕ) → P (t a .₁) (t a .₂))
            (λ ih₀ → ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁)
                       → P~ (ih₀ a₀) (ih₀ a₁) (t~ a₀₁)))
      → P (node (λ a → t a .₁)) ((λ a → t a .₂) , t~))
    (pnode~
      : {tt@(t , t~) tt'@(t' , t'~) : Σₚ (ℕ → ∣IT∣)
            (λ t₀ → ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁) → IT~ (t₀ a₀ .₁) (t₀ a₁ .₁))}
      → {ih : Σₚ ((a : ℕ) → P (t a .₁) (t a .₂))
            (λ ih₀ → ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁) → P~ (ih₀ a₀) (ih₀ a₁) (t~ a₀₁))}
      → {ih' : Σₚ ((a : ℕ) → P (t' a .₁) (t' a .₂))
            (λ ih₀ → ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁) → P~ (ih₀ a₀) (ih₀ a₁) (t'~ a₀₁))}
      → (t₀₁ : ∀ {a a'} (a₀₁ : a N.≈ a') → IT~ (t a .₁) (t' a' .₁))
      → (ih₀₁ : ∀ {a a'} (a₀₁ : a N.≈ a')
                  → P~ (ih .₁ a) (ih' .₁ a') (t₀₁ a₀₁))
      → P~ (pnode tt ih) (pnode tt' ih') (λ {a₀} {a₁} a₀₁ → t₀₁ a₀₁))
    where

  -- The telescopic witness `isTel`.
  isElim : (t : IT₀) (ext : extIT t) (v : P t ext) → Set ℓ
  isElim (leaf b) _        v = LiftS lzero (P~ (pleaf b) v (IT~-refl (leaf b) tt))
  isElim (node t) (e , t~) v =
    Σₚ[ rec ∈ ((a : ℕ) → Σ (P (t a) (e a)) (isElim (t a) (e a))) ]
    -- isElim needs to be in Set
      ∃[ rec~ ∈ (∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁)
                  → P~ (rec a₀ .₁) (rec a₁ .₁) (t~ a₀₁)) ]
        (P~ (pnode ((λ a → t a , e a) , t~) ((λ a → rec a .₁) , rec~)) v
            (IT~-refl (node t) (e , t~)))

  uniq
    : (t₀ : IT₀) (ext₀ : extIT t₀) (v₀ : P t₀ ext₀)
      (t₁ : IT₀) (ext₁ : extIT t₁) (v₁ : P t₁ ext₁)
      (t₀₁ : IT~ t₀ t₁)
    → (tv₀ : isElim t₀ ext₀ v₀) (tv₁ : isElim t₁ ext₁ v₁) → P~ v₀ v₁ t₀₁
  uniq (leaf b)  _          v₀ (leaf c)  _          v₁ t₀₁ w₀ w₁ =
    ~trans (~sym (↓ w₀)) (~trans (pleaf~ b c t₀₁) (↓ w₁))
  uniq (node t) (ext₀ , t~) v₀ (node u) (ext₁ , u~) v₁ t₀₁ (rec₀ , rec₀~ , pnode~v₀) (rec₁ , rec₁~ , pnode~v₁) =
     ~trans (~sym pnode~v₀) (~trans
       (pnode~ t₀₁ λ {a₀} {a₁} a₀₁ →
         uniq _ _ _ _ _ _ _ (rec₀ a₀ .₂) (rec₁ a₁ .₂))
         (pnode~v₁)) 

  rec-isElim : (t : IT₀) (ext : extIT t) → Σ (P t ext) (isElim t ext)
  rec-isElim (leaf b) _        = pleaf b , ↑ (~refl (pleaf b))
  rec-isElim (node t) (e , t~) = v , (rec , rec~ , ~refl v)
    where
      rec : (a : ℕ) → Σ (P (t a) (e a)) (isElim (t a) (e a))
      rec a = rec-isElim (t a) (e a)
      rec~ : ∀ {a₀ a₁} (a₀₁ : a₀ N.≈ a₁) → P~ _ _ (t~ a₀₁)
      rec~ {a₀} {a₁} a₀₁ =
        uniq _ _ _ _ _ _ _ (rec a₀ .₂) (rec a₁ .₂)
      v : P (node t) (e , t~)
      v = pnode ((λ a → t a , e a) , t~) ((λ a → rec a .₁) , rec~)

module _
    {Γ : Con ℓΓ}
    (P : Ty (Γ ▷ IT) ℓ)
    (pleaf : Tm (Γ ▷ Bool) (P [ wk ▸ lea ]))
    (pnode
      : Tm (Γ ▷ Nat ⇒ IT ▷ Π Nat (P [ wk∘wk ▸ app' Nat IT vz ]))
           (P [ wk ▸ nod ] [ wk ]))
  where

  module ElimAt (γ : ∣ Γ ∣) = Elim₀
      (λ t ext → ∣ P ∣ (γ , (t , ext)))
      (λ v₀ v₁ t₀₁ → P .heq (Γ .refl γ , t₀₁) v₀ v₁)
      (λ v → P .refl* v)
      (λ p → P .sym* p)
      (λ p q → P .trans* p q)
      (λ b → ∣ pleaf ∣ (γ , b))
      (λ b₀ b₁ b₀₁ → pleaf .resp ((Γ .refl γ) , b₀₁))
      (λ tt ih → ∣ pnode ∣ ((γ , tt) , ih))
      (λ t₀₁ ih₀₁ → pnode .resp ((Γ .refl γ , λ _ _ → t₀₁) , λ _ _ → ih₀₁))

  elim₀₁
    : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
        (t₀ : IT₀) (ext₀ : extIT t₀) (t₁ : IT₀) (ext₁ : extIT t₁)
        (t₀₁ : IT~ t₀ t₁)
    → P ∋ (γ₀₁ , t₀₁) ⊢ ElimAt.rec-isElim γ₀ t₀ ext₀ .₁ ~ ElimAt.rec-isElim γ₁ t₁ ext₁ .₁
  elim₀₁ γ₀₁ (leaf b) _        (leaf c) _        t₀₁ = pleaf .resp (γ₀₁ , t₀₁)
  elim₀₁ γ₀₁ (node t) (e₀ , _) (node u) (e₁ , _) t₀₁ =
    pnode .resp ((γ₀₁ , (λ _ _ a₀₁ → t₀₁ a₀₁))
                , (λ x y a₀₁ → elim₀₁ γ₀₁ (t x) (e₀ x) (u y) (e₁ y) (t₀₁ a₀₁)) )

  elim : Tm (Γ ▷ IT) P
  ∣ elim ∣ (γ , (t , ext)) = ElimAt.rec-isElim γ t ext .₁
  elim .resp {_ , t₀ , ext₀} {_ , t₁ , ext₁} (γ₀₁ , t₀₁) =
    elim₀₁ γ₀₁ t₀ ext₀ t₁ ext₁ t₀₁

IT[] : IT [ σ ] ≡ IT
IT[] = Eq.refl

lea[] : lea [ σ ⁺ ]t ≡ lea
lea[] = Eq.refl

nod[] : nod [ σ ⁺ ]t ≡ nod
nod[] = Eq.refl

module _
    {Γ : Con ℓΓ}
    (P : Ty (Γ ▷ IT) ℓ)
    (pleaf : Tm (Γ ▷ Bool) (P [ wk ▸ lea ]))
    (pnode
      : Tm (Γ ▷ Nat ⇒ IT ▷ Π Nat (P [ wk∘wk ▸ app' Nat IT vz ]))
           (P [ wk ▸ nod ] [ wk ]))
  where

  elim[]
    : {σ : Sub Δ Γ}
    → elim P pleaf pnode [ σ ⁺ ]t
    ≡ elim (P [ σ ⁺ ]) (pleaf [ σ ⁺ ]t) (pnode [ σ ⁺ ⁺ ]t)
  elim[] = Eq.refl

  elim-lea : elim P pleaf pnode [ wk ▸ lea ]t ≡ pleaf
  elim-lea = Eq.refl

  elim-nod
    : {s : Tm Γ (Nat ⇒ IT)}
    → elim P pleaf pnode ⟨ nodₑ s ⟩t
    ≡ pnode [ idS ▸ s
            ▸ (ƛ (elim P pleaf pnode [ wk∘wk ▸ app' Nat IT vz ]t)) [ idS ▸ s ]t ]t
  elim-nod = Eq.refl
