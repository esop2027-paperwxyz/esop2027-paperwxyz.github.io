open import Prelude

module Model.Prop.Sum.Core where

open import Model.Prop.CwF

module SUM
  {A₀ A₁ : Set ℓA}(A~ : A₀ → A₁ → Prop ℓA)
  {B₀ B₁ : Set ℓB}(B~ : B₀ → B₁ → Prop ℓB) where

  ~ : A₀ ⊎ B₀ → A₁ ⊎ B₁ → Prop (ℓA ⊔ ℓB)
  ~ (inl x₀) (inl x₁) = LiftP ℓB (A~ x₀ x₁)
  ~ (inr y₀) (inr y₁) = LiftP ℓA (B~ y₀ y₁)
  ~ _ _ = ⊥ₚ

module _ (A : Ty Γ ℓA)(B : Ty Γ ℓB) where
  Sum : Ty Γ (ℓA ⊔ ℓB)
  ∣ Sum ∣ γ = ∣ A ∣ γ ⊎ ∣ B ∣ γ
  Sum .heq γ₀₁ = SUM.~ (A .heq γ₀₁) (B .heq γ₀₁)
  Sum .refl* (inl a) = ↑ (A .refl* a)
  Sum .refl* (inr b) = ↑ (B .refl* b)
  Sum .sym* {x₀ = inl a₀} {inl a₁} a₀₁ = ↑ (A .sym* (a₀₁ .↓))
  Sum .sym* {x₀ = inr b₀} {inr b₁} b₀₁ = ↑ (B .sym* (b₀₁ .↓))
  Sum .trans* {x₀ = inl a₀} {inl a₁} {inl a₂} a₀₁ a₁₂ =
    ↑ (A .trans* (a₀₁ .↓) (a₁₂ .↓))
  Sum .trans* {x₀ = inr b₀} {inr b₁} {inr b₂} b₀₁ b₁₂ =
    ↑ (B .trans* (b₀₁ .↓) (b₁₂ .↓))
  Sum .coe γ₀₁ = ⊎-map (A .coe γ₀₁) (B .coe γ₀₁)
  Sum .coh γ₀₁ (inl a) = ↑ (A .coh γ₀₁ a)
  Sum .coh γ₀₁ (inr b) = ↑ (B .coh γ₀₁ b)
  {-# INJECTIVE_FOR_INFERENCE Sum #-}

  Inl : Tm Γ A → Tm Γ Sum
  ∣ Inl a ∣   γ        = inl (∣ a ∣ γ)
  Inl a .resp {γ₀} γ₀₁ = ↑ (a .resp γ₀₁)
  {-# INJECTIVE_FOR_INFERENCE Inl #-}

  Inr : Tm Γ B → Tm Γ Sum
  ∣ Inr b ∣   γ        = inr (∣ b ∣ γ)
  Inr b .resp {γ₀} γ₀₁ = ↑ (b .resp γ₀₁)
  {-# INJECTIVE_FOR_INFERENCE Inr #-}
