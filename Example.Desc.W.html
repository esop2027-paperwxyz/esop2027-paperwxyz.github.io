-- Well-founded type as an inductive type generated by Desc.

private variable
  ℓI : Level

module _ (A : Ty ∅ ℓI) (B : Ty (∅ ▷ A) ℓI) where
  WD : ConD ∅ Unit Unit
  WD = non A (arg B ⋆ (end ⋆))

  WDs : ConDs Unit Unit
  WDs = WD ∷ []

  W : Ty Γ ℓI
  W = μ WDs ⟨ ⋆ ⟩
  {-# INJECTIVE_FOR_INFERENCE W #-}

  module _
      (P : Ty (Γ ▷ Unit {ℓ = ℓI} ▷ μ WDs) ℓ)
      (ps : Tm (Γ ▷ Unit ▷ μArg WD [ ε ⁺ ] ▷ μIH P ε WD)
               (P [ wk ▸ conᵢ WDs ] [ wk ▸ inlArgsᵢ WD [] ] [ wk ]))
      where

    WDs-elim :
      Tm (Γ ▷ Unit ▷ μ WDs) P
    WDs-elim = elim' P (ps , _)

module _ {A : Ty ∅ ℓI} (B : Ty (∅ ▷ A) ℓI) where
  sup
    : (a : Tm Γ (A [ ε ]))
    → (f : Tm Γ (B [ ε ▸ a ] ⇒ W A B))
    → Tm Γ (W A B)
  sup a f =
    con# (# 0) ⋆
      (dpair ((B [ ε ⁺ ] ⇒ W A B) ẋ ⇑ Eq ⋆ ⋆)
         a
         (pair (B [ ε ▸ a ] ⇒ W A B) (⇑ Eq ⋆ ⋆)
           f (reflT' ⋆)))

  module _
      (P : Ty (Γ ▷ Unit {ℓ = ℓI} ▷ μ (WDs A B)) ℓ)
      (ps : Tm (Γ ▷ Unit ▷ μArg (WD A B) [ ε ⁺ ] ▷ μIH P ε (WD A B))
               (P [ wk ▸ conᵢ (WDs A B) ] [ wk ▸ inlArgsᵢ (WD A B) [] ] [ wk ]))
      where

    WDs-elim-sup
      : (a : Tm Γ (A [ ε ]))
      → (f : Tm Γ (B [ ε ▸ a ] ⇒ W A B))
      → WDs-elim _ B P ps [ idS ▸ ⋆ ▸ sup a f ]t
      ≡ ps [ σArgᵢ→Arg ε (WD A B) ⁺ ]t
           -- ^~~~~~~~ or, to expand the goal using IHᵢ, [ σIHᵢ→IH (ε {Γ = Γ}) P (WD A B) ]t
           [ idS ▸ ⋆
           ▸ dpair ((B [ ε ⁺ ] ⇒ W A B) ẋ ⇑ Eq ⋆ ⋆)
               a
               (pair (B [ ε ▸ a ] ⇒ W A B) (⇑ Eq ⋆ ⋆)
                 f
                 (reflT' ⋆))
           -- ^~~~~~~~~~~~~~~~~ taken from `sup a f`
           ▸ pair (Π (B [ ε ▸ a ]) (P [ wk ▸ ⋆ ▸ app f ]))
                  Unit
               (ƛ (WDs-elim _ B P ps [ wk ▸ ⋆ ▸ app f ]t))
               ⋆
           ]t
    WDs-elim-sup a f = Eq.refl
