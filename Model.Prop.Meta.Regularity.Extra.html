sum-elim-η
  : ∀ {a b c} {A : Set a} {B : Set b} {C : Set c}
  → (f : A ⊎ B → C) (x : A ⊎ B)
  → f x
    ≡ [ (λ a → f (inl a)) , (λ b → f (inr b)) ] x
sum-elim-η f (inl x) = Eq.refl
sum-elim-η f (inr y) = Eq.refl

sum-η
  : ∀ {a b} {A : Set a} {B : Set b}
  → (x : A ⊎ B)
  → x ≡ [ inl , inr ] x
sum-η x = sum-elim-η (λ y → y) x

Sum-isReg : IsReg Γ A → IsReg Γ B → IsReg Γ (Sum A B)
Sum-isReg {Γ = Γ} {A = A} {B = B} A-isReg B-isReg =
  (λ γ x → ⊎-map (A .coe (Γ .refl γ)) (B .coe (Γ .refl γ)) x)
    ≡⟨
    Jₚ²
     (λ cast p cast' q →
        _≡_
        {A =
         (γ : ∣ Γ ∣) → ∣ Sum A B ∣ γ → ∣ Sum A B ∣ γ}
        (λ γ x → ⊎-map (CoeLHS Γ A γ) (CoeLHS Γ B γ) x)
        (λ γ x → ⊎-map (λ a → cast γ a) (λ b → cast' γ b) x))
     Eq.refl _ A-isReg _ B-isReg
     ⟩
  (λ _ x → [ inl , inr ] x)
    ≡⟨⟩
  (λ _ → [ inl , inr ])
    ≡⟨ funext (λ γ → funext (λ x → Eq.sym (sum-η x))) ⟩
  (λ _ x → x) ∎
  where open Eq.≡-Reasoning
