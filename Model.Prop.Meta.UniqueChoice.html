-- Unique choice with the truncated existential quantifier
uc
  : (A : Ty Γ ℓA)
  → Tm Γ (Exists A (Π (A [ wk ]) (⇑ Eq (vs vz) vz)) ⇒ A)
  -- (∃ (x : A) (∀ (y : A) → x = y)) → A
∣ uc A ∣ γ = (λ (x , p) → x) , λ {(x₀ , f₀ , p₀)} {(x₁ , _)} _ → f₀ x₁ .↓
uc {Γ = Γ} A .resp γ₀₁ (x₀ , f₀ , p₀) (x₁ , f₁ , p₁) _ =
  A .trans* (A .coh γ₀₁ x₀) (A .sym* (f₁ (A .coe γ₀₁ x₀) .↓))

-- Unique choice with the Prop-valued existential quantifier is equivalent to that of metatheory.

data ∃ₚ {a b} (X : Set a) (P : X → Prop b) : Prop (a ⊔ b) where
  _,∃_ : (x : X) → P x → ∃ₚ X P

UC : Setω
UC = ∀ {ℓA ℓ} (A : Set ℓA) (P : A → A → Prop ℓ) → LiftS ℓ (∃ₚ A λ x → ∀ (y : A) → P x y) → A

Existsₚ : (A : Ty Γ ℓA) → Prp ℓP (Γ ▷ A) → Prp (ℓA ⊔ ℓP) Γ
∣ Existsₚ A P ∣ γ = ∃ₚ (∣ A ∣ γ) (λ x → ∣ P ∣ (γ , x))
Existsₚ A P .coe p (x ,∃ r) =
  A .coe p x ,∃ P .coe (p , A .coh p x) r

UCₚ : (A : Ty Γ ℓA) → Ty Γ ℓA
UCₚ A = (⇑ Existsₚ A (πₚ' (A [ wk ]) (Eq (vs vz) vz))) ⇒ A

module _ (f : UC) where

  ucₚ : (A : Ty Γ ℓA) → Tm Γ (UCₚ A)
  ∣ ucₚ A ∣ γ = (λ { (↑ x) → f (∣ A ∣ γ) _ (↑ x)}) , λ { {↑ (x₀ ,∃ p₀) } {↑ (x₁ ,∃ p₁) } _ →
    A .refl* (f (∣ A ∣ γ) (λ z z₁ → A ∋ γ ~ γ ⊢ z ~ z₁) (↑ (x₀ ,∃ p₀)))}
  ucₚ A .resp {γ₀} {γ₁} γ₀₁ (↑ (x₀ ,∃ f₀)) (↑ (x₁ ,∃ f₁)) tt =
    let a₀ = f (∣ A ∣ γ₀) _ (↑ (x₀ ,∃ f₀))
        a₁ = f (∣ A ∣ γ₁) _ (↑ (x₁ ,∃ f₁))
    in A .trans* (A .coh γ₀₁ a₀) (A. trans* (A .sym* (f₁ (A .coe γ₀₁ a₀))) (f₁ a₁))

inhabited
  : ∀ {ℓA ℓ} {A : Set ℓA} (P : A → A → Prop ℓ)
  → ∃ₚ A (λ x → ∀ (y : A) → P x y)
  → ∃ₚ A (λ _ → (y : A) → ⊤ₚ {ℓA})
inhabited {A = A} P (x ,∃ _) = x ,∃ λ _ → tt

ucₚ→UC : (∀ {ℓA} (A : Ty ∅ ℓA) → Tm ∅ (UCₚ A)) → UC
ucₚ→UC f {ℓA} A P (↑ x) = ∣ f (Indiscrete A) ∣ tt .₁ (↑ (inhabited P x))

-- Alternative formulation

Exists! : (A : Ty Γ ℓA) (P : Ty (Γ ▷ A) ℓB) → Ty Γ (ℓA ⊔ ℓB)
Exists! A P = Exists A (P ẋ Π (A [ wk ]) (P [ wk∘wk ▸ vz ]  ⇒ (⇑ Eq vz (vs vz))))

-- Should work with R : hProp instead of R : SProp as well
UniqueChoice' : (A : Ty Γ ℓA)(B : Ty Γ ℓB)(R : Prp ℓ (Γ ▷ A ▷ B [ wk ]))
   → Ty Γ (ℓA ⊔ ℓB ⊔ ℓ)
UniqueChoice' A B R =
 Π A (Exists! (B [ wk ]) (⇑ R)) ⇒ Exists (A ⇒ B) (Π (A [ wk ]) ((⇑ R) [ wk∘wk ▸ vz ▸ app vz ]))

uc' : (A : Ty Γ ℓA)(B : Ty Γ ℓB)(R : Prp ℓ (Γ ▷ A ▷ B [ wk ]))
   → Tm Γ (UniqueChoice' A B R)
∣ uc' {Γ = Γ} A B R ∣ γ = (λ h → ((λ a → h .₁ a .₁) , fact₀ h) , ((λ a → h .₁ a .₂ .₁) , fact₁ h)) , λ _ → tt
 where
  fact₀ : (h : |Π| A (Exists! (B [ wk ]) (⇑ R)) γ)
        → {x₀ x₁ : ∣ A ∣ γ} → A ∋ Γ .refl γ ⊢ x₀ ~ x₁ → B ∋ Γ .refl γ ⊢ ₁ (₁ h x₀) ~ ₁ (₁ h x₁)
  fact₀ (h₀ , _) {x₀} {x₁} x₀₁ =
    ↓ (h₀ x₁ .₂ .₂ .₁ (h₀ x₀ .₁) .₁ (↑ (R .coe ((Γ .refl γ , x₀₁) , B .refl* (h₀ x₀ .₁)) (↓ (h₀ x₀ .₂ .₁) ))))

  fact₁ : (h : |Π| A (Exists! (B [ wk ]) (⇑ R)) γ)
        → {x₀ x₁ : ∣ A ∣ γ} → (q : A ∋ Γ .refl γ ⊢ x₀ ~ x₁)
        → (⇑ R) ∋ (Γ .refl γ , q) , fact₀ h q ⊢ h .₁ x₀ .₂ .₁ ~ h .₁ x₁ .₂ .₁
  fact₁ (h₀ , _) {x₀} {x₁} x₀₁ = tt
uc' {Γ = Γ} A B R .resp γ₀₁ x₀ x₁ x₀₁ = tt
