-- W-type encoding of natural numbers.

Two : Ty Γ lzero
Two = Sum Unit Unit [ ε ]

false' : Tm Γ Two
false' = Inl _ _ ⋆

true' : Tm Γ Two
true' = Inr _ _ ⋆

module _ {Γ : Con ℓΓ} where
  ∣Ar∣ : ∣ Sum {Γ = ∅} {lzero} {lzero} Unit Unit ∣ tt → Set
  ∣Ar∣ = [ (λ _ → ∣ Empty {Γ = ∅} ∣ tt) , (λ _ → ∣ Unit {Γ = ∅} ∣ tt) ]′

  Ar : Ty (Γ ▷ Two) lzero
  ∣ Ar ∣ (γ , b) = ∣Ar∣ b
  Ar .heq {γ₀ = γ₀ , b₀} {γ₁ , b₁} (γ₀₁ , b₀₁) =
    [_,_] {C = λ b₀ → ∀ b₁ → ∣Ar∣ b₀ → ∣Ar∣ b₁ → Prop}
    (λ _ → [ (λ _ _ _ → ⊤ₚ) , (λ _ → λ where (↑ ())) ])
      (λ _ → [ (λ _ → λ where _ (↑ ())) , (λ _ _ _ → ⊤ₚ) ])
      b₀
      b₁
  Ar .refl* {γ = _ , inr _} _ = tt
  Ar .sym* {γ₀ = _ , inr _} {_ , inr _} _ = tt
  Ar .trans* {γ₀ = _ , inr _} {_ , inr _} {_ , inr _} _ _ = tt
  Ar .coe {γ₀ = γ₀ , b₀} {γ₁ , b₁} (γ₀₁ , b₀₁) =
    [_,_] {C = λ b₀ → ∀ b₁
        → (b₀₁ : Sum {Γ = Γ} Unit Unit ∋ γ₀₁ ⊢ b₀ ~ b₁)
        → (x₀ : ∣Ar∣ b₀) → ∣Ar∣ b₁}
      (λ _ → [ (λ _ _ x → x) , (λ _ ()) ])
      (λ _ → [ (λ _ ()) , (λ _ _ x → x) ])
      b₀
      b₁
      b₀₁
  Ar .coh {γ₀ = γ₀ , inr _} {γ₁ , inr _} (γ₀₁ , b₀₁) x = tt

Ar-Inl : Ar ⟨ false' ⟩ ≡ Empty {Γ = ∅}
Ar-Inl = Eq.refl

Ar-Inr : Ar ⟨ true' ⟩ ≡ Unit {Γ = ∅}
Ar-Inr = Eq.refl

WNat : Ty Γ lzero
WNat = W Two Ar

wze : Tm Γ WNat
wze = sup Ar false' (ƛ elim-Empty _)

wsu : Tm Γ WNat → Tm Γ WNat
wsu t = sup Ar true' (ƛ (t [ wk ]t))

wsu' : Tm (Γ ▷ WNat) (WNat [ wk ])
wsu' = wsu vz

module _
  (B : Ty Γ ℓB)
  (pz : Tm Γ B)
  (ps : Tm (Γ ▷ B) (B [ wk ]))
  where

  alg-b
    : Tm (Γ ▷ Two ▷ Ar ⇒ B [ wk ]) (B [ wk∘wk ])
  alg-b = appₑ'
            (case ((Ar ⇒ B [ wk ]) ⇒ B [ wk ])
                (ƛ:(Empty ⇒ B [ wk ]) -- comes from (Ar ⇒ B [ wk ]) [ wk ▸ false' ]
                  (pz [ wk∘wk ]t))
                (ƛ:(Unit ⇒ B [ wk ])  -- comes from (Ar ⇒ B [ wk ]) [ wk ▸ true' ]
                  (ps [ wk∘wk ▸ appₑ' vz ⋆ ]t))
              [ wk ]t)
            vz

  alg : Tm (Γ ▷ Unit {ℓ = lzero}
              ▷ Argᵢ (WD _ Ar) (μ (WDs _ Ar)) [ ε ⁺ ]
              ▷ IHᵢ ε (WD _ Ar) (μ (WDs _ Ar)) (B [ wk∘wk ]))
           (B [ wk∘wk∘wk ])
  alg = alg-expanded-ty
    where mutual
      alg-expanded-ty
        : Tm (Γ ▷ Unit {ℓ = lzero}
                ▷ Sigma Two ((Ar [ wk ⁺ ] ⇒ WNat) ẋ ⇑ Eq ⋆ (vs vz)) -- the substitution, [ ε ⁺ ], is computed away
                ▷ (Ar [ wk∘wk ▸ fst ((Ar [ wk ⁺ ] ⇒ WNat) ẋ ⇑ Eq ⋆ (vs vz)) ] ⇒ B [ wk∘wk ]) ẋ Unit {ℓ = ℓB})
             (B [ wk∘wk∘wk ])
      alg-expanded-ty = alg-expanded-ty' [ wk ⁺ ⁺ ]t

      alg-expanded-ty'
        : Tm (Γ ▷ Sigma Two ((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ (⋆ {ℓ = lzero}))
                ▷ (Ar [ wk ▸ fst ((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ ⋆) ] ⇒ B [ wk ]) ẋ Unit {ℓ = ℓB})
             (B [ wk∘wk ])
      alg-expanded-ty' =
        alg-only-ih
          [ wk ▸ fst' (Ar [ wk ▸ fst ((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ ⋆) ] ⇒ B [ wk ])
                      (Unit {ℓ = ℓB})
          ]t

      alg-only-ih
        : Tm (Γ ▷ Sigma Two ((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ (⋆ {ℓ = lzero}))
                ▷ Ar [ wk ▸ fst ((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ ⋆) ] ⇒ B [ wk ])
             (B [ wk∘wk ])
      alg-only-ih = alg-only-ih' [ wk∘wk ▸ assoc-Sigma _ ]t

      alg-only-ih'
        : Tm (Γ ▷ Sigma Two (((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ (⋆ {ℓ = lzero})) ẋ (Ar ⇒ B [ wk ])))
             (B [ wk ])
      alg-only-ih' =
        elim-Sigma (B [ wk ])
          (alg-b [ wk∘wk
                 ▸ vs vz
                 ▸ snd' ((Ar ⇒ WNat) ẋ ⇑ Eq ⋆ ⋆) (Ar ⇒ B [ wk ])
                 ]t)

  wrec
    : (n : Tm Γ WNat)
    → Tm Γ B
  wrec =
    elimₑ' (B [ wk∘wk ])
           (alg [ σIH→IHᵢ (ε {Γ = Γ}) (B [ wk∘wk ]) (WD Two Ar) ]t
             -- or alg [ σArg→Argᵢ ε (WD _ Ar) ⁺ ]t
           , _)
           ⋆
