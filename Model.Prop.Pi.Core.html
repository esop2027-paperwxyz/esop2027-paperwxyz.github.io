module _ (A : Ty Γ ℓA) (B : Ty (Γ ▷ A) ℓB) where

  |Π| : ∣ Γ ∣ → Set _ -- (ℓA ⊔ ℓB)
  |Π| γ = Σₚ[ f ∈ ((x : ∣ A ∣ γ) → ∣ B ∣ (γ , x)) ]
    (∀ {x₀ x₁} (q : A ∋ Γ .refl γ ⊢ x₀ ~ x₁) → B ∋ (Γ .refl γ) , q ⊢ f x₀ ~ f x₁)

  ∣_∣f : ∀ {γ} → |Π| γ → (x : ∣ A ∣ γ) → ∣ B ∣ (γ , x)
  ∣ f , _ ∣f = f

  respf
    : ∀ {γ} (f : |Π| γ) {x₀ x₁} (q : A ∋ Γ .refl _ ⊢ x₀ ~ x₁)
    → B ∋ Γ .refl _ , q ⊢ f .₁ x₀ ~ f .₁ x₁
  respf f = f .₂

  Π : Ty Γ _
  ∣ Π ∣ γ = |Π| γ 
  Π .heq p f g =
    ∀ x y (q : A ∋ p ⊢ x ~ y) → B ∋ p , q ⊢ ∣ f ∣f x ~ ∣ g ∣f y
  Π .refl* f _ _   = respf f
  Π .sym*  p y x q = B .sym* (p x y (A .sym* q)) 
  Π .trans* {γ₀} {γ₁} {γ₂} {γ₀₁} {_} {f} {g} {h} f~g g~h x y r =
  -- Check this proof -- I don't understand.
    B .trans*
      (B .trans* (f~g x (coe A γ₀₁ x) (coh A γ₀₁ x)) (respf g (refl* A (coe A _ x))))
      (g~h (coe A γ₀₁ x) y (trans* A (sym* A (coh A γ₀₁ x)) r))
  Π .coe {γ} {γ′} r f =
  -- WTF ...
    let r~ = Γ .sym r in 
    (λ x →  B .coe (r , A .sym* (A .coh r~ x)) (∣ f ∣f (A .coe r~ x)))
    , λ {x₀} {x₁} q → let p = respf f (A .trans* (A .trans* (A .sym* (A .coh r~ x₀)) q) (A .coh r~ x₁)) in
      B .trans*
        (B. trans* (B .sym* (B .coh (r , sym* A (coh A r~ x₀)) (∣ f ∣f (coe A _ x₀))))
          p)
        (B .coh (r , A .sym* (coh A r~ x₁)) (∣ f ∣f (coe A r~ x₁)))
  Π .coh γ₀₁ f x y x~y = let γ₁₀ = Γ .sym γ₀₁ in
    B .trans* (respf f (trans* A x~y (A .coh γ₁₀ y)))
      (B .coh (γ₀₁ , sym* A (coh A γ₁₀ y)) (∣ f ∣f (A .coe _ y)))
{-# INJECTIVE_FOR_INFERENCE Π #-}

ƛ_ : Tm (Γ ▷ A) B → Tm Γ (Π A B)
∣ ƛ_ {Γ = Γ} t ∣ γ = (λ x → ∣ t ∣ (γ , x)) , λ p → t .resp (Γ .refl γ , p)
ƛ_ {Γ} t .resp p = λ x y q → t .resp (p , q)
