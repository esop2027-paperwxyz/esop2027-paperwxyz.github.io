module Prelude.Fin where

-- Standalone replacement for the standard library's `Data.Fin` / `Data.Fin.Base`.
-- Only `Fin` (with constructors `zero`/`suc`) and `#_` are used by the project;
-- the ordering machinery backing `#_` is kept private.

open import Prelude.Base
open import Prelude.Decidable

data Fin : ℕ → Set where
  zero : ∀ {n} → Fin (suc n)
  suc  : ∀ {n} → Fin n → Fin (suc n)

private
  data _≤_ : ℕ → ℕ → Set where
    z≤n : ∀ {n}            → zero  ≤ n
    s≤s : ∀ {m n} → m ≤ n → suc m ≤ suc n

  _<_ : ℕ → ℕ → Set
  m < n = suc m ≤ n

  fromℕ< : ∀ {m n} → m < n → Fin n
  fromℕ< {zero}  (s≤s z≤n) = zero
  fromℕ< {suc m} (s≤s m<n) = suc (fromℕ< m<n)

  _≤?_ : ∀ m n → Dec (m ≤ n)
  zero  ≤? n     = yes z≤n
  suc m ≤? zero  = no λ ()
  suc m ≤? suc n with m ≤? n
  ... | yes p = yes (s≤s p)
  ... | no ¬p = no λ { (s≤s p) → ¬p p }

  _<?_ : ∀ m n → Dec (m < n)
  m <? n = suc m ≤? n

infix 10 #_
#_ : ∀ m {n} {m<n : True (m <? n)} → Fin n
#_ m {m<n = m<n} = fromℕ< (toWitness m<n)
