-- Natural numbers as an inductive type generated by IR Desc.

-- Closed unit type used as the index of Nat, together with parity function
One : Ty Γ lzero
One = Unit {ℓ = lzero}

data ∣Bool∣ : Set where ff tt : ∣Bool∣

Bool : Ty Γ lzero
∣ Bool ∣ γ = ∣Bool∣
Bool .heq _ = _≡_
Bool .refl* a = Eq.refl
Bool .sym* = Eq.sym
Bool .trans* = Eq.trans
Bool .coe p x = x
Bool .coh p x = Eq.refl

False : Tm Γ Bool
∣ False ∣ γ = ff
False .resp = λ γ₀₁ → Eq.refl

True : Tm Γ Bool
∣ True ∣ γ = tt
True .resp = λ γ₀₁ → Eq.refl

if_then_else_ : {A : Set ℓ} → ∣Bool∣ → A → A → A
if tt then x else y = x
if ff then x else y = y

Not : Tm Γ Bool → Tm Γ Bool
∣ Not b ∣ γ = if ∣ b ∣ γ then ff else tt
Not b .resp γ₀₁ = Eq.cong (λ z → if z then ff else tt) (b .resp γ₀₁)

-- Zero constructor description.
NatZeroD : ConD ∅ One Bool One Bool
NatZeroD = end ⋆ False

-- Successor constructor description.
NatSucD : ConD ∅ One Bool One Bool
NatSucD = arg1 ⋆ (end ⋆ (Not vz))

-- Description of natural-number constructors.
NatDs : ConDs One Bool One Bool
NatDs = NatZeroD ∷ NatSucD ∷ []

-- Natural numbers in context Γ.
Nat : Ty Γ lzero
Nat =  μ NatDs ⟨ ⋆ ⟩

-- Zero constructor.
ze : Tm Γ Nat
ze = con# (# 0) ⋆ (reflT' ⋆)

-- Successor constructor.
su : Tm Γ Nat → Tm Γ Nat
su n = con# (# 1) ⋆ (pair Nat (⇑ Eq ⋆ ⋆) n (reflT' ⋆))

parityₑ : Tm Γ Nat → Tm Γ Bool
parityₑ n = T NatDs [ idS ▸ ⋆ ▸ n ]t

private
  _ : parityₑ {Γ = ∅} ze ≡ False
  _ = Eq.refl

  _ : parityₑ {Γ = ∅} (su ze) ≡ True
  _ = Eq.refl

  _ : parityₑ {Γ = ∅} (su (su ze)) ≡ False
  _ = Eq.refl

  _ : (n : Tm Γ Nat) → parityₑ (su n) ≡ Not (parityₑ n)
  _ = λ n → Eq.refl
