open import Prelude

module Model.Prop.Pi.Rules where

open import Model.Prop.CwF
open import Model.Prop.Pi.Core
open import Model.Prop.Pi.Elim

Π[] : ((Π A B) [ σ ]) ≡ (Π (A [ σ ]) (B [ σ ⁺ ]))
Π[] = Eq.refl

ƛ[]
  : {Γ : Con ℓΓ}{A : Ty Γ ℓA}{B : Ty (Γ ▷ A) ℓB} {σ : Sub Δ Γ}
  → (t : Tm (Γ ▷ A) B)
  → (ƛ t) [ σ ]t ≡ ƛ (t [ σ ⁺ ]t)
ƛ[] t = Eq.refl

app[]
  : {Γ : Con ℓΓ}{A : Ty Γ ℓA}{B : Ty (Γ ▷ A) ℓB} {σ : Sub Δ Γ}
  → (t : Tm Γ (Π A B))
  → (app t) [ σ ⁺ ]t ≡ app (t [ σ ]t)
app[] t = Eq.refl

Πβ
  : (t : Tm (Γ ▷ A) B)
  → app (ƛ t) ≡ t
Πβ t = Eq.refl

Πη
  : (t : Tm Γ (Π A B))
  → ƛ (app t) ≡ t
Πη t = Eq.refl
