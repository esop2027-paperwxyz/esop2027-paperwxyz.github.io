module Model.Prop.Desc.Arg where

open import Model.Prop.CwF
open import Model.Prop.HEq
open import Model.Prop.Product

open import Model.Prop.Desc.Core
  hiding (Arg₀; Args₀; Arg~; Args~)

private variable
  ℓI : Level

module _ {I : Ty ∅ ℓI} where
  Arg₀
    : {Δ : Con ℓΔ} (D : ConD Δ I I)
    → {X : (i : ∣ I ∣ tt) → Set ℓ}
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X i₀ → X i₁ → Prop ℓ)
    → (δ : ∣ Δ ∣)
    → ∣ I ∣ tt → Set (ℓI ⊔ ℓ)
  Arg₀ {ℓ = ℓ} (end t)     X~ δ j = LiftS ℓ (I ∋ tt ⊢ ∣ t ∣ δ ~ j)
  Arg₀ (non A D)   X~ δ j =
    Σ[ a ∈ ∣ A ∣ δ ] Arg₀ D X~ (δ , a) j
  Arg₀ (arg1 f D)  {X} X~ δ j =
    X (∣ f ∣ δ) × Arg₀ D X~ δ j
  Arg₀ {Δ = Δ} (arg A f D) {X} X~ δ j =
    Σₚ ((a : ∣ A ∣ δ) → X (∣ f ∣ (δ , a))) (λ x →
      {a₀ a₁ : ∣ A ∣ δ} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁)
        → X~ (f .resp (Δ .refl δ , a₀₁)) (x a₀) (x a₁))
    × Arg₀ D X~ δ j
  {-# INJECTIVE_FOR_INFERENCE Arg₀ #-}
  
  Args₀
    : (Ds : ConDs I I)
    → {X : ∣ I ∣ tt → Set ℓ}
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X i₀ → X i₁ → Prop ℓ)
    → ∣ I ∣ tt → Set (ℓI ⊔ ℓ)
  Args₀ []       X~ j = Lift _ ⊥
  Args₀ (D ∷ Ds) X~ j = Arg₀ D X~ tt j ⊎ Args₀ Ds X~ j

  {-# INJECTIVE_FOR_INFERENCE Args₀ #-}

  Arg~ : {Δ : Con ℓΔ}
    → (D : ConD Δ I I)
    → {δ₀ δ₁ : ∣ Δ ∣} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → {X₀ : ∣ I ∣ tt → Set ℓ}
    → (X₀~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₀ i₁ → Prop ℓ)
    → {X₁ : ∣ I ∣ tt → Set ℓ}
    → (X₁~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₁ i₀ → X₁ i₁ → Prop ℓ)
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → X₀ i₀ → X₁ i₁ → Prop ℓ)
    → {j₀ j₁ : ∣ I ∣ tt}
    → (xs₀ : Arg₀ D X₀~ δ₀ j₀) (xs₁ : Arg₀ D X₁~ δ₁ j₁) → Prop (ℓI ⊔ ℓ)
  Arg~ (end t)     δ₀₁ X₀~ X₁~ X~ xs₀ xs₁ = ⊤ₚ
  Arg~ (non A D)   δ₀₁ X₀~ X₁~ X~ (a₀ , xs₀) (a₁ , xs₁) =
    ∃[ a₀₁ ∈ A ∋ δ₀₁ ⊢ a₀ ~ a₁ ] Arg~ D (δ₀₁ , a₀₁) X₀~ X₁~ X~ xs₀ xs₁
  Arg~ (arg1 f D)  δ₀₁ X₀~ X₁~ X~ (x₀ , xs₀) (x₁ , xs₁) =
    X~ (f .resp δ₀₁) x₀ x₁
    ∧ (Arg~ D δ₀₁  X₀~ X₁~ X~ xs₀ xs₁)
  Arg~ (arg A f D) {δ₀} {δ₁} δ₀₁ X₀~ X₁~ X~ ((x₀ , _) , xs₀) ((x₁ , _) , xs₁) =
    ((a₀ : ∣ A ∣ δ₀)(a₁ : ∣ A ∣ δ₁)(a₀₁ : A ∋ δ₀₁ ⊢ a₀ ~ a₁)
      → X~ (f .resp (δ₀₁ , a₀₁)) (x₀ a₀) (x₁ a₁))
    ∧ Arg~ D δ₀₁ X₀~ X₁~ X~ xs₀ xs₁
  {-# INJECTIVE_FOR_INFERENCE Arg~ #-}

  -- Setoid relation on a constructor list's arguments.
  Args~
    : (Es : ConDs I I)
    → {X₀ : ∣ I ∣ tt → Set ℓ}
    → (X₀~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₀ i₁ → Prop ℓ)
    → {X₁ : ∣ I ∣ tt → Set ℓ}
    → (X₁~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₁ i₀ → X₁ i₁ → Prop ℓ)
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₁ i₁ → Prop ℓ)
    → {j₀ j₁ : ∣ I ∣ tt}
    → (xs₀ : Args₀ Es X₀~ j₀) (xs₁ : Args₀ Es X₁~ j₁) → Prop (ℓI ⊔ ℓ)
  Args~ (D ∷ Es) X₀~ X₁~ X~ (inl xs₀) (inl xs₁) = Arg~ D tt X₀~ X₁~ X~ xs₀ xs₁
  Args~ (D ∷ Es) X₀~ X₁~ X~ (inr xs₀) (inr xs₁) = Args~  Es X₀~ X₁~ X~ xs₀ xs₁
  Args~ (D ∷ Es) X₀~ X₁~ X~ _         _         = ⊥ₚ
  
  coeArg
    : {Δ : Con ℓΔ} (D : ConD Δ I I)
    → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → {X₀ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₀~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₀ i₁ → Prop ℓ)
    → {X₁ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₁~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₁ i₀ → X₁ i₁ → Prop ℓ)
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₁ i₁ → Prop ℓ)
    → (Xtrans₀₀₁
        : ∀ {i₀ i₀' i₁}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁}
        → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
        → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁ x₀' x₁
        → X~ (I .trans* i₀₀' i₀'₁) x₀ x₁)
    → (Xtrans₁₀₀₁
        : ∀ {i₀ i₀' i₁ i₁'}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁} {x₁' : X₁ i₁'}
        → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
        → X~ i₀₁ x₀ x₁ → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁' x₀' x₁'
        → X₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁ x₁')
    → (coeX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀) → X₁ i₁)
    → (cohX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀)
        → X~ i₀₁ x₀ (coeX i₀₁ x₀))
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (x₀ : Arg₀ D X₀~ δ₀ j₀)
    → Arg₀ D X₁~ δ₁ j₁
  coeArg (end t)     δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (↑ x₀)  =
    ↑ (I .trans* (I .sym* (t .resp δ₀₁)) (I .trans* x₀ j₀₁))
  coeArg (non A D)   δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (a₀ , xs₀) =
    A .coe δ₀₁ a₀ , coeArg D (δ₀₁ , A .coh δ₀₁ a₀) X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀
  coeArg (arg1 f D)  δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (x₀ , xs₀) =
    coeX (f .resp δ₀₁) x₀ , coeArg D δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀
  coeArg {Δ = Δ} (arg A f D) δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ ((x₀ , x~) , xs₀) =
    let δ₁₀ = Δ .sym δ₀₁ in

    ((λ a₁ → coeX (f .resp (δ₀₁ , coh' A δ₁₀ a₁)) (x₀ (A .coe δ₁₀ a₁)))
    , λ {a₁} {a₁'} a₀₁ → Xtrans₁₀₀₁
      (cohX ((f .resp (δ₀₁ , coh' A δ₁₀ a₁))) (x₀ (A .coe δ₁₀ a₁)))
      (x~ (A .trans* (coh'' A δ₀₁ a₁) (A .trans* a₀₁ (A .coh δ₁₀ a₁'))))
      (cohX ((f .resp (δ₀₁ , coh' A δ₁₀ a₁'))) (x₀ (A .coe δ₁₀ a₁'))))
    , coeArg D δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀

  cohArg
    : {Δ : Con ℓΔ} (D : ConD Δ I I)
    → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → {X₀ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₀~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₀ i₁ → Prop ℓ)
    → {X₁ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₁~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₁ i₀ → X₁ i₁ → Prop ℓ)
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₁ i₁ → Prop ℓ)
    → (Xtrans₀₀₁
        : ∀ {i₀ i₀' i₁}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁}
        → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
        → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁ x₀' x₁
        → X~ (I .trans* i₀₀' i₀'₁) x₀ x₁)
    → (Xtrans₁₀₀₁
        : ∀ {i₀ i₀' i₁ i₁'}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁} {x₁' : X₁ i₁'}
        → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
        → X~ i₀₁ x₀ x₁ → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁' x₀' x₁'
        → X₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁ x₁')
    → (coeX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀) → X₁ i₁)
    → (cohX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀)
        → X~ i₀₁ x₀ (coeX i₀₁ x₀))
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (x₀ : Arg₀ D X₀~ δ₀ j₀)
    → Arg~ D δ₀₁ X₀~ X₁~ X~ x₀ (coeArg D δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ x₀)
  cohArg (end t)     δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ x₀ = tt
  cohArg (non A D)   δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (a₀ , xs₀) =
    A .coh δ₀₁ a₀ , cohArg D (δ₀₁ , A .coh δ₀₁ a₀) X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀
  cohArg (arg1 f D)  δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (x₀ , xs₀) =
    (cohX (f .resp δ₀₁) x₀) , cohArg D δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀
  cohArg {Δ = Δ} (arg A f D) δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ ((x₀ , x₀~) , xs₀) =
    (let δ₁₀ = Δ .sym δ₀₁ in λ a₀ a₁ a₀₁ →
      Xtrans₀₀₁
        (x₀~ (A .trans* a₀₁ (A .coh δ₁₀ a₁)))
        (cohX (f .resp (δ₀₁ , coh' A δ₁₀ a₁)) (x₀ (coe A δ₁₀ a₁))))
    , (cohArg D δ₀₁ X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀)

  coeArgs
    : (Ds : ConDs I I)
    → {X₀ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₀~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₀ i₁ → Prop ℓ)
    → {X₁ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₁~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₁ i₀ → X₁ i₁ → Prop ℓ)
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₁ i₁ → Prop ℓ)
    → (Xtrans₀₀₁
        : ∀ {i₀ i₀' i₁}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁}
        → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
        → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁ x₀' x₁
        → X~ (I .trans* i₀₀' i₀'₁) x₀ x₁)
    → (Xtrans₁₀₀₁
        : ∀ {i₀ i₀' i₁ i₁'}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁} {x₁' : X₁ i₁'}
        → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
        → X~ i₀₁ x₀ x₁ → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁' x₀' x₁'
        → X₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁ x₁')
    → (coeX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀) → X₁ i₁)
    → (cohX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀)
        → X~ i₀₁ x₀ (coeX i₀₁ x₀))
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (xs₀ : Args₀ Ds X₀~ j₀)
    → Args₀ Ds X₁~ j₁
  coeArgs (D ∷ Ds) X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (inl xs₀) =
    inl (coeArg D tt X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀)
  coeArgs (D ∷ Ds) X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (inr xs₀) =
    inr (coeArgs Ds X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀)

  cohArgs
    : (Ds : ConDs I I)
    → {X₀ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₀~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₀ i₁ → Prop ℓ)
    → {X₁ : (i : ∣ I ∣ tt) → Set ℓ}
    → (X₁~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₁ i₀ → X₁ i₁ → Prop ℓ)
    → (X~ : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → X₀ i₀ → X₁ i₁ → Prop ℓ)
    → (Xtrans₀₀₁
        : ∀ {i₀ i₀' i₁}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁}
        → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
        → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁ x₀' x₁
        → X~ (I .trans* i₀₀' i₀'₁) x₀ x₁)
    → (Xtrans₁₀₀₁
        : ∀ {i₀ i₀' i₁ i₁'}
        → {x₀ : X₀ i₀} {x₀' : X₀ i₀'} {x₁ : X₁ i₁} {x₁' : X₁ i₁'}
        → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
        → X~ i₀₁ x₀ x₁ → X₀~ i₀₀' x₀ x₀' → X~ i₀'₁' x₀' x₁'
        → X₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁ x₁')
    → (coeX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀) → X₁ i₁)
    → (cohX
        : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀ : X₀ i₀)
        → X~ i₀₁ x₀ (coeX i₀₁ x₀))
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (x₀ : Args₀ Ds X₀~ j₀)
    → Args~ Ds X₀~ X₁~ X~ x₀
        (coeArgs Ds X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ x₀)
  cohArgs (D ∷ Ds) X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (inl xs₀) =
    cohArg D tt X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀
  cohArgs (D ∷ Ds) X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ (inr xs₀) =
    cohArgs Ds X₀~ X₁~ X~ Xtrans₀₀₁ Xtrans₁₀₀₁ coeX cohX j₀₁ xs₀

module _ {Γ : Con ℓΓ} {I : Ty ∅ ℓI} (X : Ty (Γ ▷ I [ ε ]) ℓ) where

  private
    Xγ
      : (γ : ∣ Γ ∣)
      → (i : ∣ I ∣ tt)
      → Set ℓ
    Xγ γ i = ∣ X ∣ (γ , i)

    Xγ~
      : ∀ {γ₀ γ₁}
      → Γ ∋ γ₀ ~ γ₁
      → ∀ {i₀ i₁}
      → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
      → Xγ γ₀ i₀ → Xγ γ₁ i₁ → Prop ℓ
    Xγ~ γ₀₁ i₀₁ = X .heq (γ₀₁ , i₀₁)

  ∣Arg∣
    : {Δ : Con ℓΔ} (E : ConD Δ I I)
    → ∀ γ δ i 
    → Set (ℓI ⊔ ℓ)
  ∣Arg∣ E γ δ i = Arg₀ E {Xγ γ} (Xγ~ (Γ .refl γ)) δ i

  ∣Arg~∣
    : {Δ : Con ℓΔ} (E : ConD Δ I I)
    → ∀ {γ₀ γ₁ δ₀ δ₁ i₀ i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (x₀ : ∣Arg∣ E γ₀ δ₀ i₀) (x₁ : ∣Arg∣ E γ₁ δ₁ i₁)
    → Prop (ℓI ⊔ ℓ)
  ∣Arg~∣ E {γ₀} {γ₁} {δ₀} {δ₁} {i₀} {i₁} γ₀₁ δ₀₁ x₀ x₁ =
    Arg~ E δ₀₁ (Xγ~ (Γ .refl γ₀)) (Xγ~ (Γ .refl γ₁)) (Xγ~ γ₀₁) x₀ x₁

  ∣Args∣
    : (Es : ConDs I I)
    → ∀ γ j
    → Set (ℓI ⊔ ℓ)
  ∣Args∣ Es γ j = Args₀ Es (Xγ~ (Γ .refl γ)) j

  ∣Args~∣
    : (Es : ConDs I I)
    → ∀ {γ₀ γ₁ i₀ i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (xs₀ : ∣Args∣ Es γ₀ i₀) (xs₁ : ∣Args∣ Es γ₁ i₁)
    → Prop (ℓI ⊔ ℓ)
  ∣Args~∣ Es {γ₀} {γ₁} γ₀₁ xs₀ xs₁ =
    Args~ Es (Xγ~ (Γ .refl γ₀)) (Xγ~ (Γ .refl γ₁)) (Xγ~ γ₀₁) xs₀ xs₁

  Arg₀-refl*
    : {Δ : Con ℓΔ} (E : ConD Δ I I)
    → ∀ {γ δ i} 
    → (x₀ : ∣Arg∣ E γ δ i)
    → ∣Arg~∣ E (Γ .refl γ) (Δ .refl δ) x₀ x₀
  Arg₀-refl* (end t)     xs₀        = tt
  Arg₀-refl* (non A E)   (a₀ , xs₀) = A .refl* a₀   , Arg₀-refl* E xs₀
  Arg₀-refl* (arg1 f E)  (x₀ , xs₀) = (X .refl* x₀) , Arg₀-refl* E xs₀
  Arg₀-refl* (arg A f E) ((x₀ , x₀~) , xs₀) =
    (λ _ _ a₀₁ → x₀~ a₀₁) , Arg₀-refl* E xs₀

  Arg₀-sym*
    : {Δ : Con ℓΔ} (E : ConD Δ I I)
    → ∀ {γ₀ γ₁ δ₀ δ₁ i₀ i₁}
    → {xs₀ : ∣Arg∣ E γ₀ δ₀ i₀} {xs₁ : ∣Arg∣ E γ₁ δ₁ i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : ∣Arg~∣ E γ₀₁ δ₀₁ xs₀ xs₁)
    → ∣Arg~∣ E (Γ .sym γ₀₁) (Δ .sym δ₀₁)
        xs₁ xs₀
  Arg₀-sym* (end t)     γ₀₁ δ₀₁ _       = tt
  Arg₀-sym* (non A E)   γ₀₁ δ₀₁ (a₀₁ , xs₀₁) = A .sym* a₀₁ , Arg₀-sym* E γ₀₁ (δ₀₁ , a₀₁) xs₀₁
  Arg₀-sym* (arg1 f E)  γ₀₁ δ₀₁ (x₀₁ , xs₀₁) = X .sym* x₀₁ , (Arg₀-sym* E γ₀₁ δ₀₁ xs₀₁)
  Arg₀-sym* (arg A f E) γ₀₁ δ₀₁ (x₀₁ , xs₀₁) =
    (λ a₀ a₁ a₀₁ → X .sym* (x₀₁ a₁ a₀ (A .sym* a₀₁))) , Arg₀-sym* E γ₀₁ δ₀₁ xs₀₁

  Arg₀-trans*
    : {Δ : Con ℓΔ} (E : ConD Δ I I)
    → ∀ {γ₀ γ₁ γ₂ δ₀ δ₁ δ₂ i₀ i₁ i₂}
    → {xs₀ : ∣Arg∣ E γ₀ δ₀ i₀} {xs₁ : ∣Arg∣ E γ₁ δ₁ i₁} {xs₂ : ∣Arg∣ E γ₂ δ₂ i₂}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂) (δ₀₁ : Δ ∋ δ₀ ~ δ₁) (δ₁₂ : Δ ∋ δ₁ ~ δ₂)
    → (xs₀₁ : ∣Arg~∣ E γ₀₁ δ₀₁ xs₀ xs₁) (xs₁₂ : ∣Arg~∣ E γ₁₂ δ₁₂ xs₁ xs₂)
    → ∣Arg~∣ E (Γ .trans γ₀₁ γ₁₂) (Δ .trans δ₀₁ δ₁₂) xs₀ xs₂
  Arg₀-trans* (end t) γ₀₁ γ₁₂ δ₀₁ δ₁₂ _ _ = tt
  Arg₀-trans* (non A E) γ₀₁ γ₁₂ δ₀₁ δ₁₂ (a₀₁ , xs₀₁) (a₁₂ , xs₁₂) =
    A .trans* a₀₁ a₁₂ , Arg₀-trans* E γ₀₁ γ₁₂ (δ₀₁ , a₀₁) (δ₁₂ , a₁₂) xs₀₁ xs₁₂
  Arg₀-trans* (arg1 f E) γ₀₁ γ₁₂ δ₀₁ δ₁₂ (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) =
    X .trans* x₀₁ x₁₂ , Arg₀-trans* E γ₀₁ γ₁₂ δ₀₁ δ₁₂ xs₀₁ xs₁₂
  Arg₀-trans* (arg A f E) γ₀₁ γ₁₂ δ₀₁ δ₁₂ (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) =
    (λ a₀ a₁ a₀₁ → X .trans*
       (x₀₁ a₀ (A .coe δ₀₁ a₀) (A .coh δ₀₁ a₀))
       (x₁₂ (A .coe δ₀₁ a₀) a₁ (A .trans* (A .sym* (A .coh δ₀₁ a₀)) a₀₁)))
    , Arg₀-trans* E γ₀₁ γ₁₂ δ₀₁ δ₁₂ xs₀₁ xs₁₂

  Args₀-refl*
    : (Es : ConDs I I)
    → ∀ {γ i}
    → (x₀ : ∣Args∣ Es γ i)
    → ∣Args~∣ Es (Γ .refl γ) x₀ x₀
  Args₀-refl* (E ∷ Es) (inl xs₀) = Arg₀-refl* E xs₀
  Args₀-refl* (E ∷ Es) (inr xs₁) = Args₀-refl* Es xs₁

  Args₀-sym*
    : (Es : ConDs I I)
    → ∀ {γ₀ γ₁ i₀ i₁}
    → {x₀ : ∣Args∣ Es γ₀ i₀} {x₁ : ∣Args∣ Es γ₁ i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (x₀₁ : ∣Args~∣ Es γ₀₁ x₀ x₁)
    → ∣Args~∣ Es (Γ .sym γ₀₁) x₁ x₀
  Args₀-sym* (D ∷ Es) {x₀ = inl _} {inl _} γ₀₁ x₀₁ = Arg₀-sym* D γ₀₁ tt x₀₁
  Args₀-sym* (D ∷ Es) {x₀ = inr _} {inr _} γ₀₁ x₀₁ = Args₀-sym* Es γ₀₁ x₀₁
   
  Args₀-trans*
    : (Es : ConDs I I)
    → ∀ {γ₀ γ₁ γ₂ i₀ i₁ i₂}
    → {xs₀ : ∣Args∣ Es γ₀ i₀} {xs₁ : ∣Args∣ Es γ₁ i₁} {xs₂ : ∣Args∣ Es γ₂ i₂}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂)
    → (xs₀₁ : ∣Args~∣ Es γ₀₁ xs₀ xs₁) (xs₁₂ : ∣Args~∣ Es γ₁₂ xs₁ xs₂)
    → ∣Args~∣ Es (Γ .trans γ₀₁ γ₁₂) xs₀ xs₂
  Args₀-trans* (E ∷ Es) {xs₀ = inl _} {xs₁ = inl _} {xs₂ = inl _} γ₀₁ γ₁₂ xs₀₁ xs₁₂ =
    Arg₀-trans* E γ₀₁ γ₁₂ tt tt xs₀₁ xs₁₂
  Args₀-trans* (E ∷ Es) {xs₀ = inr _} {xs₁ = inr _} {xs₂ = inr _} γ₀₁ γ₁₂ xs₀₁ xs₁₂ =
    Args₀-trans* Es γ₀₁ γ₁₂ xs₀₁ xs₁₂

  Arg
    : {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (E : ConD Δ I I)
    → Ty (Γ ▷ I [ ε ]) (ℓI ⊔ ℓ)
  ∣ Arg σ E ∣ (γ , i) = ∣Arg∣ E γ (∣ σ ∣ γ) i
  Arg σ E .heq (γ₀₁ , i₀₁) x₀ x₁  = ∣Arg~∣ E γ₀₁ (σ .resp γ₀₁) x₀ x₁
  Arg σ E .refl* {γ , j}          = Arg₀-refl* E
  Arg σ E .sym* {γ₀₁ = γ₀₁ , j₀₁} = Arg₀-sym* E γ₀₁ (σ .resp γ₀₁)
  Arg σ E .trans* {γ₀₁ = γ₀₁ , j₀₁} {γ₁₂ , j₁₂} =
    λ x₀₁ x₁₂ → Arg₀-trans* E γ₀₁ γ₁₂ (σ .resp γ₀₁) (σ .resp γ₁₂) x₀₁ x₁₂
  Arg σ E .coe {γ₀ = γ₀ , _} {γ₁ , _} (γ₀₁ , i₀₁)  =
    coeArg E (σ .resp γ₀₁)
      (Xγ~ (Γ .refl γ₀)) (Xγ~ (Γ .refl γ₁)) (Xγ~ γ₀₁)
      (X .trans*) (λ x₀₁ x₀₀' x₀'₁' → X .trans* (X .sym* x₀₁) (X .trans* x₀₀' x₀'₁'))
      (λ i₀₁ x₀ → X .coe (γ₀₁ , i₀₁) x₀) (λ i₀₁ x₀ → X .coh (γ₀₁ , i₀₁) x₀) i₀₁
  Arg σ E .coh {γ₀ = γ₀ , _} {γ₁ , _} (γ₀₁ , i₀₁) =
    λ x₀ → cohArg E (σ .resp γ₀₁) (Xγ~ (Γ .refl γ₀)) (Xγ~ (Γ .refl γ₁)) (Xγ~ γ₀₁) (X .trans*)
      (λ x₀₁ x₀₀'  x₀'₁' → X .trans* (X .sym* x₀₁) (X .trans* x₀₀' x₀'₁') )
      (λ i₀₁ x₀ → X .coe (γ₀₁ , i₀₁) x₀)
      (λ i₀₁ x₀ → X .coh (γ₀₁ , i₀₁) x₀)
      i₀₁ x₀
  {-# INJECTIVE_FOR_INFERENCE Arg #-}

  Args
    : (Es : ConDs I I)
    → Ty (Γ ▷ I [ ε ]) (ℓI ⊔ ℓ)
  ∣ Args Es ∣ (γ , i) = ∣Args∣ Es γ i
  Args Es .heq (γ₀₁ , i₀₁)        = ∣Args~∣ Es γ₀₁
  Args Es .refl*                  = Args₀-refl* Es
  Args Es .sym* {γ₀₁ = γ₀₁ , i₀₁} = Args₀-sym* Es γ₀₁
  Args Es .trans* {γ₀₁ = γ₀₁ , i₀₁} {γ₁₂ , i₁₂} =
    Args₀-trans* Es γ₀₁ γ₁₂
  Args Es .coe {γ₀ = γ₀ , _} {γ₁ , _} (γ₀₁ , i₀₁) =
    coeArgs Es (Xγ~ (Γ .refl γ₀)) (Xγ~ (Γ .refl γ₁)) (Xγ~ γ₀₁)
      (λ x₀₀' x₀'₁ → X .trans* x₀₀' x₀'₁)
      (λ x₀₁ x₀₀' x₀'₁' →
        X .trans* (X .sym* x₀₁) (X .trans* x₀₀' x₀'₁'))
      (λ i₀₁ → X .coe (γ₀₁ , i₀₁))
      (λ i₀₁ → X .coh (γ₀₁ , i₀₁))
      i₀₁
  Args Es .coh {γ₀ = γ₀ , _} {γ₁ , _} (γ₀₁ , i₀₁) =
     cohArgs Es (Xγ~ (Γ .refl γ₀)) (Xγ~ (Γ .refl γ₁)) (Xγ~ γ₀₁)
      (λ i₀₀' i₀'₁ → X .trans* i₀₀' i₀'₁)
      (λ i₀₁ i₀₀' i₀'₁' → X .trans* (X .sym* i₀₁) (X .trans* i₀₀' i₀'₁'))
      (λ i₀₁ x₀ → X .coe (γ₀₁ , i₀₁) x₀)
      (λ i₀₁ x₀ → X .coh (γ₀₁ , i₀₁) x₀)
      i₀₁ 
  {-# INJECTIVE_FOR_INFERENCE Args #-}
