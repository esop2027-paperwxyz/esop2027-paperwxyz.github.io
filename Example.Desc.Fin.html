-- Finite sets as an inductive type generated by Desc.

-- Description of the two Fin constructors.
FZeroD FSucD : ConD ∅ Nat Nat
FinDs : ConDs Nat Nat

FZeroD = non Nat (end (su vz))
FSucD  = non Nat (arg1 vz (end (su vz)))
FinDs  = FZeroD ∷ FSucD ∷ []

-- Fin family in context Γ.
Fin : Tm Γ Nat → Ty Γ lzero
Fin n = μ FinDs ⟨ n ⟩
{-# INJECTIVE_FOR_INFERENCE Fin #-}

Fin' : Ty (Γ ▷ Nat) lzero
Fin' = μ FinDs

-- Zero constructor for Fin.
fzero : {Γ : Con lzero}
  → (j i : Tm Γ Nat)
  → (eq : Tm Γ (⇑ Eq (su i) j))
  → Tm Γ (Fin j)
fzero {Γ} j i eq =
  con# (# 0) j
    (dpair (⇑ Eq (su vz) (j [ wk ]t))
      i
      eq)
{-# INJECTIVE_FOR_INFERENCE fzero #-}

fzero' : {Γ : Con lzero}
  → Tm (Γ ▷ Nat ▷ Nat ▷ ⇑ Eq (su vz) (vs vz)) (Fin' [ wk∘wk ])
fzero' = fzero (vz [ wk∘wk ]t) (vz [ wk ]t) vz

-- Successor constructor for Fin.
fsuc : {Γ : Con lzero}
  → (j i : Tm Γ Nat)
  → (n : Tm Γ (Fin i))
  → (eq : Tm Γ (⇑ Eq (su i) j))
  → Tm Γ (Fin j)
fsuc {Γ} j i n eq =
  con# (# 1) j
    (dpair (Fin vz ẋ ⇑ Eq (su vz) (j [ wk ]t))
      i
      (pair (Fin' ⟨ i ⟩) (⇑ Eq (su vz) (j [ wk ]t) ⟨ i ⟩)
        n
        eq))
{-# INJECTIVE_FOR_INFERENCE fsuc #-}

fsuc' : {Γ : Con lzero}
  → Tm (Γ ▷ Nat ▷ Nat ▷ Fin' ▷ ⇑ Eq (su (vs vz)) (vz [ wk∘wk ]t))
       (Fin' [ wk∘wk∘wk ])
fsuc' = fsuc (vz [ wk∘wk∘wk ]t) (vz [ wk∘wk ]t) (vz [ wk ]t) vz

elimFin : {Γ : Con lzero} (P : Ty (Γ ▷ Nat ▷ Fin') lzero)
  → (pz : Tm (Γ ▷ Nat ▷ Nat ▷ ⇑ Eq (su vz) (vs vz))
             (P [ wk∘wk ▸ fzero' ]))
  → (ps : Tm (Γ ▷ Nat ▷ Nat ▷ Fin' ▷ ⇑ Eq (su (vs vz)) (vz [ wk∘wk ]t) ▷ P [ wk ⁺ ⁺ ] [ wk ])
             (P [ wk∘wk∘wk ▸ fsuc' ] [ wk ]))
  → Tm (Γ ▷ Nat ▷ Fin') P
elimFin P pz ps =
  elim' P
       ( pz [ wk∘wk ▸ vs vz ▸ fst (⇑ Eq (su vz) (vs vz)) ▸ snd (⇑ Eq (su vz) (vs vz)) ]t
            [ σArg→Argᵢ ε FZeroD ∘ wk ]t
       , ps [
            (let i = fst {A = Nat [ ε ]} (Fin vz ẋ (⇑ Eq (su vz) (vs vz)))
                 n = fstₑ' (Fin i) (⇑ Eq (su i) (vs vz))
                       (snd {A = Nat [ ε ]} (Fin vz ẋ (⇑ Eq (su vz) (vs vz)))) in
               wk∘wk∘wk
             ▸ vz [ wk∘wk ]t
             ▸ i [ wk ]t
             ▸ n [ wk ]t
             ▸ sndₑ' (Fin i [ wk ]) (⇑ Eq (su i) (vs vz) [ wk ])
                 (snd {A = Nat [ ε ]} (Fin vz ẋ (⇑ Eq (su vz) (vs vz))) [ wk ]t)
             ▸ fst' (P [ wk∘wk
                       ▸ i
                       ▸ n
                       ])
                    Unit
             )]t
             [ σIH→IHᵢ ε P FSucD ]t
       , _)

module _ {Γ : Con lzero} where
  -- Motive for adding a natural number to a Fin element.
  add-FinP : Ty (Γ ▷ Nat ▷ Nat ▷ μ FinDs) lzero
  add-FinP = Nat [ wk∘wk ]

  -- Branches for add-Fin.
  add-FinPs : ArgIHs add-FinP
  add-FinPs =
      vs (vs (vs vz))
    , su (fstₑ' _ _ vz)
    , _

  -- Addition defined by the generic eliminator.
  add-Fin : Tm (Γ ▷ Nat ▷ Nat ▷ μ FinDs) add-FinP
  add-Fin = elim' add-FinP add-FinPs

  -- Pointwise form of add-Fin.
  add-Finₑ
    : (m : Tm Γ Nat)
    → {i : Tm Γ Nat}
    → Tm Γ (Fin i)
    → Tm Γ Nat
  add-Finₑ m {i} x = add-Fin [ idS ▸ m ▸ i ▸ x ]t

module _ {Γ : Con lzero} (P : Ty (Γ ▷ Nat ▷ Fin') lzero)
  (pz : Tm (Γ ▷ Nat ▷ μArg FZeroD [ ε ⁺ ] ▷ μIH P ε FZeroD)
           (P [ wk ▸ conᵢ FinDs ] [ wk ▸ inlArgsᵢ FZeroD (FSucD ∷ []) ] [ wk ]))
  (ps : Tm (Γ ▷ Nat ▷ μArg FSucD [ ε ⁺ ] ▷ μIH P ε FSucD)
           (P [ wk ▸ conᵢ FinDs ]
              [ wk ▸ inrArgsᵢ FZeroD (FSucD ∷ []) ]
              [ wk ▸ inlArgsᵢ FSucD [] ]
              [ wk ]))
  where

  FinDs-elim-fzero
    : {i j : Tm Γ Nat} {eq : Tm Γ (⇑ Eq (su j) i)}
    → elimₑ' P (pz , ps , _) i (fzero i j eq)
    ≡ pz [ σArgᵢ→Arg ε FZeroD ⁺ ]t
         [ idS ▸ i
         ▸ dpair (⇑ Eq (su vz) (i [ wk ]t))
             j eq
         ▸ ⋆ ]t
  FinDs-elim-fzero = Eq.refl

  FinDs-elim-fsuc
    : {i j : Tm Γ Nat} {n : Tm Γ (Fin j)} {eq : Tm Γ (⇑ Eq (su j) i)}
    → elimₑ' P (pz , ps , _) i (fsuc i j n eq)
    ≡ ps [ σArgᵢ→Arg ε FSucD ⁺ ]t
         [ idS ▸ i
         ▸ dpair (Fin vz ẋ (⇑ Eq (su vz) (i [ wk ]t)))
             j
             (pair (Fin j) (⇑ Eq (su j) i)
               n eq)
         ▸ pair (P [ idS ▸ j ▸ n ]) Unit
             (elimₑ' P (pz , ps , _) j n) ⋆ ]t
  FinDs-elim-fsuc = Eq.refl
