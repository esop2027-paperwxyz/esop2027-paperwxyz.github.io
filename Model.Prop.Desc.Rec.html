open import Prelude

module Model.Prop.Desc.Rec where

open import Model.Prop.CwF

open import Model.Prop.Desc.Core
import Model.Prop.Desc.Arg as A

private variable
  ℓI : Level

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  module Rec₀
      (C  : ∣ I ∣ tt → Set ℓ)
      (C~ : {i₀ i₁ : ∣ I ∣ tt} → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → C i₀ → C i₁ → Prop ℓ)
      (~refl  : ∀ {i} (v : C i) → C~ (I .refl* i) v v)
      (~sym   : ∀ {i₀ i₁} {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {v₀ : C i₀} {v₁ : C i₁}
              → C~ i₀₁ v₀ v₁ → C~ (I .sym* i₀₁) v₁ v₀)
      (~trans : ∀ {i₀ i₁ i₂} {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₁₂ : I ∋ tt ⊢ i₁ ~ i₂}
                  {v₀ : C i₀} {v₁ : C i₁} {v₂ : C i₂}
              → C~ i₀₁ v₀ v₁ → C~ i₁₂ v₁ v₂ → C~ (I .trans* i₀₁ i₁₂) v₀ v₂)
      (alg  : {i : ∣ I ∣ tt} (xs : A.Args₀ Ds C~ i) → C i)
      (alg~ : {i₀ i₁ : ∣ I ∣ tt} {xs₀ : A.Args₀ Ds C~ i₀} {xs₁ : A.Args₀ Ds C~ i₁}
            → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
            → (xs₀₁ : A.Args~ Ds C~ C~ C~ xs₀ xs₁)
            → C~ i₀₁ (alg xs₀) (alg xs₁))
    where

    -- Candidate outputs together with the recursion witness that reconstructs
    -- the generic arguments over `C`.
    mutual
      isRec
        : {i : ∣ I ∣ tt} (x : μ₀ Ds i) (extx : extμ Ds x) (v : C i)
        → Set (ℓI ⊔ ℓ)
      isRec {i} (con xs) extxs v =
        Σₚ[ wʳ ∈ Argsʳ Ds xs extxs .₁ ]
          C~ (I .refl* i) (alg (Argsʳ Ds xs extxs .₂ wʳ)) v

      Argsʳ
        : (Es : ConDs I I) {j : ∣ I ∣ tt}
        → (xs : Args₀ Es (μ₀ Ds) j) (extxs : extArgs Ds Es xs)
        → Σ[ W ∈ Set (ℓI ⊔ ℓ) ] (W → A.Args₀ Es C~ j)
      Argsʳ (E ∷ Es) (inl xs) extxs =
        Argʳ E xs extxs .₁  , λ w → inl (Argʳ E xs extxs .₂ w)
      Argsʳ (E ∷ Es) (inr xs) extxs =
        Argsʳ Es xs extxs .₁ , λ w → inr (Argsʳ Es xs extxs .₂ w)

      Argʳ
        : {Δ : Con ℓΔ} (E : ConD Δ I I) {δ : ∣ Δ ∣} {j : ∣ I ∣ tt}
        → (xs : Arg₀ E (μ₀ Ds) δ j) (extxs : extArg Ds E xs)
        → Σ[ W ∈ Set (ℓI ⊔ ℓ) ] (W → A.Arg₀ E C~ δ j)
      Argʳ (end t)    xs extxs = Lift _ ⊤ , λ _ → ↑ (xs .↓)
      Argʳ (non A E)  (a , xs) extxs =
        Argʳ E xs extxs .₁ , λ w → a , Argʳ E xs extxs .₂ w
      Argʳ (arg1 f E) {δ} (x , xs) (extx , extxs) =
        (Σ (C (∣ f ∣ δ)) (isRec x extx) × Argʳ E xs extxs .₁)
        , λ (pr , w) → pr .₁ , Argʳ E xs extxs .₂ w
      Argʳ {Δ = Δ} (arg A f E) {δ} (x , xs) (extx , x~ , extxs) =
        (Σₚ ((a : ∣ A ∣ δ) → Σ (C (∣ f ∣ (δ , a))) (isRec (x a) (extx a)))
           (λ pr → {a₀ a₁ : ∣ A ∣ δ} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁)
                     → C~ (f .resp (Δ .refl δ , a₀₁)) (pr a₀ .₁) (pr a₁ .₁))
         × Argʳ E xs extxs .₁)
        , λ ((pr , pr~) , w) → ((λ a → pr a .₁) , pr~) , Argʳ E xs extxs .₂ w

    -- Functionality: any two candidates over related raw elements agree in `C`.
    mutual
      uniq
        : {i₀ i₁ : ∣ I ∣ tt}
          (x₀ : μ₀ Ds i₀) (extx₀ : extμ Ds x₀) (v₀ : C i₀)
          (x₁ : μ₀ Ds i₁) (extx₁ : extμ Ds x₁) (v₁ : C i₁)
          (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁)
        → isRec x₀ extx₀ v₀ → isRec x₁ extx₁ v₁ → C~ i₀₁ v₀ v₁
      uniq (con xs₀) extxs₀ v₀ (con xs₁) extxs₁ v₁ i₀₁ xs₀₁ (wʳ₀ , p₀) (wʳ₁ , p₁) =
        ~trans (~sym p₀)
          (~trans (alg~ i₀₁ (uniqArgs Ds extxs₀ extxs₁ xs₀₁ wʳ₀ wʳ₁)) p₁)

      uniqArg
        : {Δ : Con ℓΔ} (E : ConD Δ I I)
        → {δ₀ δ₁ : ∣ Δ ∣} {j₀ j₁ : ∣ I ∣ tt}
        → {xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀} {xs₁ : Arg₀ E (μ₀ Ds) δ₁ j₁}
        → (extxs₀ : extArg Ds E xs₀) (extxs₁ : extArg Ds E xs₁)
        → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
        → (xs₀₁ : Arg~ Ds E δ₀₁ xs₀ xs₁)
        → (wʳ₀ : Argʳ E xs₀ extxs₀ .₁) (wʳ₁ : Argʳ E xs₁ extxs₁ .₁)
        → A.Arg~ E δ₀₁ C~ C~ C~ (Argʳ E xs₀ extxs₀ .₂ wʳ₀) (Argʳ E xs₁ extxs₁ .₂ wʳ₁)
      uniqArg (end t) extxs₀ extxs₁ δ₀₁ xs₀₁ wʳ₀ wʳ₁ = tt
      uniqArg (non A E) extxs₀ extxs₁ δ₀₁ (a₀₁ , xs₀₁) wʳ₀ wʳ₁ =
        a₀₁ , uniqArg E extxs₀ extxs₁ (δ₀₁ , a₀₁) xs₀₁ wʳ₀ wʳ₁
      uniqArg (arg1 f E) {xs₀ = x₀ , _} {x₁ , _} (extx₀ , extxs₀) (extx₁ , extxs₁) δ₀₁ (x₀₁ , xs₀₁) (pr₀ , wʳ₀) (pr₁ , wʳ₁) =
        uniq _ extx₀ _ _ extx₁ _ (f .resp δ₀₁) x₀₁ (pr₀ .₂) (pr₁ .₂)
        , uniqArg E extxs₀ extxs₁ δ₀₁ xs₀₁ wʳ₀ wʳ₁
      uniqArg (arg A f E) {xs₀ = x₀ , _} {x₁ , _} (extx₀ , x₀~ , extxs₀) (extx₁ , x₁~ , extxs₁) δ₀₁ (x₀₁ , xs₀₁) ((pr₀ , _) , wʳ₀) ((pr₁ , _) , wʳ₁) =
        (λ a₀ a₁ a₀₁ → uniq _ (extx₀ a₀) _ _ (extx₁ a₁) _ (f .resp (δ₀₁ , a₀₁)) (x₀₁ a₀ a₁ a₀₁)
                            (pr₀ a₀ .₂) (pr₁ a₁ .₂))
        , uniqArg E extxs₀ extxs₁ δ₀₁ xs₀₁ wʳ₀ wʳ₁

      uniqArgs
        : (Es : ConDs I I) {i₀ i₁ : ∣ I ∣ tt}
        → {xs₀ : Args₀ Es (μ₀ Ds) i₀} {xs₁ : Args₀ Es (μ₀ Ds) i₁}
        → (extxs₀ : extArgs Ds Es xs₀) (extxs₁ : extArgs Ds Es xs₁)
        → (xs₀₁ : Args~ Ds Es xs₀ xs₁)
        → (wʳ₀ : Argsʳ Es xs₀ extxs₀ .₁) (wʳ₁ : Argsʳ Es xs₁ extxs₁ .₁)
        → A.Args~ Es C~ C~ C~ (Argsʳ Es xs₀ extxs₀ .₂ wʳ₀) (Argsʳ Es xs₁ extxs₁ .₂ wʳ₁)
      uniqArgs (E ∷ Es) {xs₀ = inl _} {inl _} extxs₀ extxs₁ xs₀₁ wʳ₀ wʳ₁ =
        uniqArg E extxs₀ extxs₁ tt xs₀₁ wʳ₀ wʳ₁
      uniqArgs (E ∷ Es) {xs₀ = inr _} {inr _} extxs₀ extxs₁ xs₀₁ wʳ₀ wʳ₁ =
        uniqArgs Es extxs₀ extxs₁ xs₀₁ wʳ₀ wʳ₁

    -- Totality: fold every extensional raw element to an output with a witness.
    mutual
      rec-isRec
        : {i : ∣ I ∣ tt} (x : μ₀ Ds i) (extx : extμ Ds x)
        → Σ (C i) (isRec x extx)
      rec-isRec (con xs) extxs =
        alg (Argsʳ Ds xs extxs .₂ (mapArgsʳ Ds xs extxs))
        , (mapArgsʳ Ds xs extxs , ~refl _)

      mapArgsʳ
        : (Es : ConDs I I) {i : ∣ I ∣ tt}
        → (xs : Args₀ Es (μ₀ Ds) i) (extxs : extArgs Ds Es xs)
        → Argsʳ Es xs extxs .₁
      mapArgsʳ (E ∷ Es) (inl xs) extxs = mapArgʳ E xs extxs
      mapArgsʳ (E ∷ Es) (inr xs) extxs = mapArgsʳ Es xs extxs

      mapArgʳ
        : {Δ : Con ℓΔ} (E : ConD Δ I I) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
        → (xs : Arg₀ E (μ₀ Ds) δ i) (extxs : extArg Ds E xs)
        → Argʳ E xs extxs .₁
      mapArgʳ (end t)    xs extxs = ↑ tt
      mapArgʳ (non A E)  (a , xs) extxs = mapArgʳ E xs extxs
      mapArgʳ (arg1 f E) (x , xs) (extx , extxs) =
        rec-isRec x extx , mapArgʳ E xs extxs
      mapArgʳ (arg A f E) (x , xs) (extx , x~ , extxs) =
        ( (λ a → rec-isRec (x a) (extx a))
        , (λ {a₀} {a₁} a₀₁ →
            uniq _ (extx a₀) _ _ (extx a₁) _ (f .resp (_ , a₀₁)) (x~ a₀₁)
                 (rec-isRec (x a₀) (extx a₀) .₂) (rec-isRec (x a₁) (extx a₁) .₂)) )
        , mapArgʳ E xs extxs

    rec₀
      : {i : ∣ I ∣ tt} (x : μ₀ Ds i) (extx : extμ Ds x) → C i
    rec₀ x extx = rec-isRec x extx .₁

    mapArg
      : {Δ : Con ℓΔ} (E : ConD Δ I I) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
      → (xs : Arg₀ E (μ₀ Ds) δ i) (extxs : extArg Ds E xs)
      → A.Arg₀ E C~ δ i
    mapArg E xs extxs = Argʳ E xs extxs .₂ (mapArgʳ E xs extxs)

    mapArgs
      : (Es : ConDs I I) {i : ∣ I ∣ tt}
      → (xs : Args₀ Es (μ₀ Ds) i) (extxs : extArgs Ds Es xs)
      → A.Args₀ Es C~ i
    mapArgs Es xs extxs = Argsʳ Es xs extxs .₂ (mapArgsʳ Es xs extxs)

  -- The internal recursor over a motive `C : Ty (Γ ▷ I)` and an algebra term.
  module _
      {Γ : Con ℓΓ}
      (C : Ty (Γ ▷ I [ ε ]) ℓ)
      (alg : Tm (Γ ▷ I [ ε ] ▷ A.Args C Ds) (C [ wk ]))
      where

      module RecAt (γ : ∣ Γ ∣) = Rec₀
        (λ i → ∣ C ∣ (γ , i))
        (λ i₀₁ → C .heq (Γ .refl γ , i₀₁))
        (λ v → C .refl* v)
        (λ p → C .sym* p)
        (λ p q → C .trans* p q)
        (λ xs → ∣ alg ∣ ((γ , _) , xs))
        (λ i₀₁ xs₀₁ → alg .resp ((Γ .refl γ , i₀₁) , xs₀₁))

      mutual
        recArg₀₁
          : {Δ : Con ℓΔ} (E : ConD Δ I I)
          → {δ₀ δ₁ : ∣ Δ ∣} {j₀ j₁ : ∣ I ∣ tt}
          → {xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀} {xs₁ : Arg₀ E (μ₀ Ds) δ₁ j₁}
          → (extxs₀ : extArg Ds E xs₀) (extxs₁ : extArg Ds E xs₁)
          → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
          → {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (xs₀₁ : Arg~ Ds E δ₀₁ xs₀ xs₁)
          → A.Arg~ E δ₀₁
              (λ i₀₁ → C .heq (Γ .refl γ₀ , i₀₁))
              (λ i₀₁ → C .heq (Γ .refl γ₁ , i₀₁))
              (λ i₀₁ → C .heq (γ₀₁ , i₀₁))
              (RecAt.mapArg γ₀ E xs₀ extxs₀)
              (RecAt.mapArg γ₁ E xs₁ extxs₁)
        recArg₀₁ (end t) extxs₀ extxs₁ δ₀₁ γ₀₁ xs₀₁ = tt
        recArg₀₁ (non A E) extxs₀ extxs₁ δ₀₁ γ₀₁ (a₀₁ , xs₀₁) =
          a₀₁ , recArg₀₁ E extxs₀ extxs₁ (δ₀₁ , a₀₁) γ₀₁ xs₀₁
        recArg₀₁ (arg1 f E) {xs₀ = x₀ , _} {x₁ , _} (_ , extxs₀) (_ , extxs₁) δ₀₁ γ₀₁ (x₀₁ , xs₀₁) =
          rec₀₁ γ₀₁ (f .resp δ₀₁) x₀₁
          , recArg₀₁ E extxs₀ extxs₁ δ₀₁ γ₀₁ xs₀₁
        recArg₀₁ (arg A f E) {xs₀ = x₀ , _} {x₁ , _} (_ , _ , extxs₀) (_ , _ , extxs₁) δ₀₁ γ₀₁ (x₀₁ , xs₀₁) =
          (λ a₀ a₁ a₀₁ → rec₀₁ {x₀ = x₀ a₀} {x₁ a₁} γ₀₁ (f .resp (_ , a₀₁)) (x₀₁ _ _ a₀₁))
          , recArg₀₁ E extxs₀ extxs₁ δ₀₁ γ₀₁ xs₀₁

        recArgs₀₁
          : (Es : ConDs I I) {i₀ i₁ : ∣ I ∣ tt} {γ₀ γ₁ : ∣ Γ ∣}
          → {xs₀ : Args₀ Es (μ₀ Ds) i₀} {xs₁ : Args₀ Es (μ₀ Ds) i₁}
          → (extxs₀ : extArgs Ds Es xs₀) (extxs₁ : extArgs Ds Es xs₁)
          → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (xs₀₁ : Args~ Ds Es xs₀ xs₁)
          → A.Args~ Es
              (λ i₀₁ → C .heq (Γ .refl γ₀ , i₀₁))
              (λ i₀₁ → C .heq (Γ .refl γ₁ , i₀₁))
              (λ i₀₁ → C .heq (γ₀₁ , i₀₁))
              (RecAt.mapArgs γ₀ Es xs₀ extxs₀)
              (RecAt.mapArgs γ₁ Es xs₁ extxs₁)
        recArgs₀₁ (E ∷ Es) {xs₀ = inl _} {inl _} extxs₀ extxs₁ γ₀₁ xs₀₁ =
          recArg₀₁ E extxs₀ extxs₁ tt γ₀₁ xs₀₁
        recArgs₀₁ (E ∷ Es) {xs₀ = inr _} {inr _} extxs₀ extxs₁ γ₀₁ xs₀₁ =
          recArgs₀₁ Es extxs₀ extxs₁ γ₀₁ xs₀₁

        rec₀₁
          : ∀ {γ₀ γ₁ i₀ i₁ x₀ x₁ extx₀ extx₁}
          → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
          → (x₀₁ : μ~ Ds x₀ x₁)
          → C .heq (γ₀₁ , i₀₁) (RecAt.rec₀ γ₀ x₀ extx₀) (RecAt.rec₀ γ₁ x₁ extx₁)
        rec₀₁ {x₀ = con _} {x₁ = con _} {extx₀ = extxs₀} {extx₁ = extxs₁} γ₀₁ i₀₁ xs₀₁ =
          alg .resp ((γ₀₁ , i₀₁) , recArgs₀₁ Ds extxs₀ extxs₁ γ₀₁ xs₀₁)

      -- Generic recursor for the inductive type μ Ds.
      rec
        : Tm (Γ ▷ I [ ε ] ▷ μ Ds) (C [ wk ])
      ∣ rec ∣ ((γ , i) , (x , extx)) = RecAt.rec₀ γ x extx
      rec .resp ((γ₀₁ , i₀₁) , x₀₁) = rec₀₁ γ₀₁ i₀₁ x₀₁

      -- Recursor specialised to an explicit index and term.
      recₑ
        : (i : Tm Γ (I [ ε ]))
        → (x : Tm Γ (μ Ds ⟨ i ⟩))
        → Tm Γ (C [ idS ▸ i ])
      recₑ i x = rec [ idS ▸ i ▸ x ]t

      -- Recursive constructor arguments as an internal argument term.
      mapArgᵢ
        : (E : ConD ∅ I I)
        → Tm (Γ ▷ I [ ε ] ▷ μArg E [ ε ⁺ ]) (A.Arg C ε E [ wk ])
      ∣ mapArgᵢ E ∣ ((γ , i) , (xs , extxs)) = RecAt.mapArg γ E xs extxs
      mapArgᵢ E .resp ((γ₀₁ , i₀₁) , xs₀₁) = recArg₀₁ E _ _ tt γ₀₁ xs₀₁

      mapArgsᵢ
        : (Es : ConDs I I)
        → Tm (Γ ▷ I [ ε ] ▷ μArgs Es) (A.Args C Es [ wk ])
      ∣ mapArgsᵢ Es ∣ ((γ , i) , (xs , extxs)) = RecAt.mapArgs γ Es xs extxs
      mapArgsᵢ Es .resp ((γ₀₁ , i₀₁) , xs₀₁) = recArgs₀₁ Es _ _ γ₀₁ xs₀₁
