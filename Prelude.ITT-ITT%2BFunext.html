{-
  Sets up function extensionality.
-}

open import Prelude as Prelude
  using (a; b; module Eq; _⊔_; Σₚ; _,_)


module Prelude.ITT-ITT+Funext where

open Eq

postulate
  funext : {A : Set a} {B : A → Set b}
    → {f g : (x : A) → B x}
    → (∀ (x : A) → f x ≡ g x)
    → f ≡ g

funext²
  : ∀ {a} {A : Set a} {b} {B : A → Set b} {c} {C : (a : A) → B a → Set c}
  → (f g : (x : A) → (y : B x) → C x y)
  → ((x : A) (y : B x) → f x y ≡ g x y)
  → f ≡ g
funext² {A = A} {B = B} {C = C} f g p =
  funext λ x → funext (λ y → p x y)
