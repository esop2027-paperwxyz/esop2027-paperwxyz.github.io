-- Infinitary A-branching trees as an inductive type generated by Desc.

-- Closed unit type used as the index of Nat.
One : Ty Γ lzero
One = Unit {ℓ = lzero}

module _ (A : Ty ∅ lzero) where
  -- Leaf constructor description.
  TrLeafD : ConD ∅ One One
  TrLeafD = end ⋆

  -- Node constructor description.
  TrNodeD : ConD ∅ One One
  TrNodeD = arg A ⋆ (end ⋆)

  -- Description of natural-number constructors.
  TrDs : ConDs One One
  TrDs = TrLeafD ∷ TrNodeD ∷ []

  Tr : Ty Γ lzero
  Tr =  μ TrDs ⟨ ⋆ ⟩

  -- Constructors
  lea : Tm Γ Tr
  lea = con# (# 0) ⋆ (reflT' ⋆)

  nod : Tm Γ (A [ ε ] ⇒ Tr)  → Tm Γ Tr
  nod t = con# (# 1) ⋆ (pair (A [ ε ] ⇒ Tr) (⇑ Eq ⋆ ⋆) t (reflT' ⋆))

  -- Check the substitution rule of `elim`.
  module _ {Γ : Con lzero}
    (P : Ty (Γ ▷ One ▷ μ TrDs) lzero)
    (plea : Tm (Γ ▷ One ▷ μArg TrLeafD [ ε ⁺ ] ▷ μIH P ε TrLeafD)
               (P [ wk ▸ conᵢ TrDs ]
                  [ wk ▸ inlArgsᵢ TrLeafD (TrNodeD ∷ []) ]
                  [ wk ]))
    (pnod : Tm (Γ ▷ One ▷ μArg TrNodeD [ ε ⁺ ] ▷ μIH P ε TrNodeD)
               (P [ wk ▸ conᵢ TrDs ]
                  [ wk ▸ inrArgsᵢ TrLeafD (TrNodeD ∷ []) ] [ wk ▸ inlArgsᵢ TrNodeD [] ]
                  [ wk ]))
    where

    TrDs-elim[]
      : {Δ : Con lzero} {σ : Sub Δ Γ}
      → elim' P (plea , pnod , _) [ σ ⁺ ⁺ ]t
      ≡ elim' (P [ σ ⁺ ⁺ ]) (plea [ σ ⁺ ⁺ ⁺ ]t , pnod [ σ ⁺ ⁺ ⁺ ]t , _)
    TrDs-elim[] = Eq.refl

    TrDs-elim-leaf
      : elimₑ' P (plea , pnod , _) ⋆ lea
      ≡ plea [ σArgᵢ→Arg ε TrLeafD ⁺ ]t
             [ idS ▸ ⋆ ▸ reflT' ⋆ ▸ ⋆ ]t
    TrDs-elim-leaf = Eq.refl

    TrDs-elim-nod
      : {t : Tm Γ (A [ ε ] ⇒ Tr)}
      → elimₑ' P (plea , pnod , _) ⋆ (nod t)
      ≡ pnod [ σArgᵢ→Arg ε TrNodeD ⁺ ]t
             [ idS ▸ ⋆
             ▸ pair (A [ ε ] ⇒ μ TrDs ⟨ ⋆ ⟩) (⇑ Eq ⋆ ⋆)
                  t (reflT' ⋆)
             ▸ pair (Π (A [ ε ]) (P [ wk ▸ ⋆ ▸ app t ])) Unit
                 (ƛ elimₑ' (P [ wk ⁺ ⁺ ]) (plea [ wk ⁺ ⁺ ⁺ ]t , pnod [ wk ⁺ ⁺ ⁺ ]t , _) ⋆ (app t)) ⋆ ]t
    TrDs-elim-nod = Eq.refl
