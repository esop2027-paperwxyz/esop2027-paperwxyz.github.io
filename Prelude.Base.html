open import Agda.Primitive    public

variable
  ℓ ℓ' ℓ'' : Level
  a b c d : Level

_∋ₚ_ : ∀ {a} (A : Prop a) → A → A
A ∋ₚ a = a
{-# NOINLINE _∋ₚ_ #-}

infix  0 case_return_of_

case_return_of_ : ∀ {a b} {A : Set a} (x : A) (B : A → Set b) →
                  ((x : A) → B x) → B x
case x return B of f = f x

case_of_ : ∀ {a b} {A : Set a} {B : Set b} → A → (A → B) → B
case x of f = case x return _ of f

infixl 6 _∘f_
infixl 6 _∘p_

_∘f_ : ∀ {a b c} {A : Set a} {B : A → Set b} {C : {x : A} → B x → Set c}
  → (g : {x : A} → (y : B x) → C y)
  → (f : (x : A) → B x)
  → (x : A) → C (f x)
(g ∘f f) x = g (f x)

_∘p_ : ∀ {a b c} {A : Set a} {B : A → Prop b} {C : {x : A} → B x → Prop c}
  → (g : {x : A} → (y : B x) → C y)
  → (f : (x : A) → B x)
  → (x : A) → C (f x)
(g ∘p f) x = g (f x)

₂' : ∀ {a b} {A : Set a} {B : Set b}
  → A × B → B
₂' = ₂

record ∃ {a b} (A : Prop a) (B : A → Prop b) : Prop (a ⊔ b) where
  constructor _,_
  field
    x : A
    y : B x

open ∃ public
  renaming (x to ₁; y to ₂)

∃-syntax : ∀ {a b} (A : Prop a) → (A → Prop b) → Prop (a ⊔ b)
∃-syntax = ∃

syntax ∃-syntax A (λ x → B) = ∃[ x ∈ A ] B

infixr 2 _∧_
_∧_ : ∀ {a b} (A : Prop a) (B : Prop b) → Prop (a ⊔ b)
A ∧ B = ∃ A λ _ → B

infix 1 _↔_
_↔_ : ∀ {a b} (A : Prop a) (B : Prop b) → Prop (a ⊔ b)
A ↔ B = (A → B) ∧ (B → A)

record Σₚ {a b} (A : Set a) (B : A → Prop b) : Set (a ⊔ b) where
  constructor _,_
  field
    x : A
    y : B x
open Σₚ public
  renaming (x to ₁; y to ₂)

Σₚ-syntax : ∀ {a b} (A : Set a) → (A → Prop b) → Set (a ⊔ b)
Σₚ-syntax = Σₚ

syntax Σₚ-syntax A (λ x → B) = Σₚ[ x ∈ A ] B
{-# DISPLAY Σₚ-syntax A B = Σₚ A B #-}


infixr 1 _⊎_ _∨_

data _∨_ (A : Prop a) (B : Prop b) : Prop (a ⊔ b) where
  inl : (x : A) → A ∨ B
  inr : (y : B) → A ∨ B

data _⊎_ (A : Set a) (B : Set b) : Set (a ⊔ b) where
  inl : (x : A) → A ⊎ B
  inr : (y : B) → A ⊎ B

[_,_]ₚ : {A : Set a}{B : Set b}{C : A ⊎ B → Prop c} →
        ((x : A) → C (inl x)) → ((x : B) → C (inr x)) →
        ((x : A ⊎ B) → C x)
[ f , g ]ₚ (inl x) = f x
[ f , g ]ₚ (inr y) = g y

[_,_] : {A : Set a} {B : Set b} {C : A ⊎ B → Set c} →
  ((x : A) → C (inl x)) → ((x : B) → C (inr x)) →
  ((x : A ⊎ B) → C x)
[ f , g ] (inl x) = f x
[ f , g ] (inr y) = g y

[_,_]′ : {A : Set a} {B : Set b} {C : Set c}
  → (A → C) → (B → C) → (A ⊎ B → C)
[_,_]′ = [_,_]

⊎-map : {A : Set a} {B : Set b} {C : Set c} {D : Set d}
  → (f : A → C) (g : B → D)
  → A ⊎ B → C ⊎ D
⊎-map f g = [ inl ∘f f , inr ∘f g ]′
