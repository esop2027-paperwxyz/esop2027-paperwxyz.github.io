ty : (ℓ : Level) → Ty Γ (lsuc ℓ)
∣ ty ℓ ∣ γ = Ty ∅ ℓ
ty ℓ .heq _ A B = A ≡ B
ty ℓ .refl* A   = Eq.refl
ty ℓ .sym*      = Eq.sym
ty ℓ .trans*    = Eq.trans
ty ℓ .coe _ A   = A
ty ℓ .coh _ A   = Eq.refl

-- Seems to be the best we can do with this version
⌊_⌋ : {ℓ : Level} → Ty ∅ ℓ → Tm Γ (ty ℓ)
∣ ⌊_⌋ {Γ = Γ} A ∣ γ = A
⌊ A ⌋ .resp _ = Eq.refl

⌈_⌉ꟳ : {ℓ : Level} → Tm Γ (ty ℓ) → Fam ℓ Γ
⌈ t ⌉ꟳ .EL γ = Ty∅→Con (∣ t ∣ γ)
⌈ t ⌉ꟳ .subst {γ₀} {γ₁} γ₀₁ =
 Tm∅→Sub {B = ∣ t ∣ γ₁} (transp' (λ z → Tm (∅ ▷ ∣ t ∣ γ₀) (z [ wk ])) (t .resp γ₀₁) vz)
⌈ t ⌉ꟳ .refl* x = ∣ t ∣ _ .refl* x
⌈ t ⌉ꟳ .trans* {γ₀} {γ₁} {γ₂} γ₀₁ γ₁₂ x =
 Eq.Jₚ
   (λ tγ₁ tγ₀₁ → (tγ₁₂ : tγ₁ ≡ ∣ t ∣ γ₂) →
      (∣ t ∣ γ₂) ∋ tt ⊢ ∣ transp' (λ z → Tm (∅ ▷ ∣ t ∣ γ₀) (z [ wk ]))
                                (Eq.trans tγ₀₁ tγ₁₂) vz ∣
                                (tt , x)
                    ~ ∣ transp' (λ z → Tm (∅ ▷ tγ₁) (z [ wk ]))
                                tγ₁₂ vz ∣
                                (tt , ∣ transp' (λ z → Tm (∅ ▷ ∣ t ∣ γ₀) (z [ wk ]))
                                                (tγ₀₁) vz ∣
                                                (tt , x)))
   (λ tγ₁₂ → Eq.Jₚ (λ tγ₂ tγ₁₂ →
      tγ₂ ∋ tt ⊢ ∣ transp' (λ z → Tm (∅ ▷ ∣ t ∣ γ₀) (z [ wk ]))
                           (Eq.trans Eq.refl tγ₁₂) vz ∣
                           (tt , x)
               ~ ∣ transp' (λ z → Tm (∅ ▷ ∣ t ∣ γ₀) (z [ wk ]))
                           tγ₁₂ vz ∣
                           (tt , x))
                          (∣ t ∣ γ₀ .refl* x)
                          (∣ t ∣ γ₂)
                          tγ₁₂)
   (∣ t ∣ γ₁)
   (t .resp γ₀₁)
   (t .resp γ₁₂)

⌈_⌉ : {ℓ : Level} → Tm Γ (ty ℓ) → Ty Γ ℓ
⌈ t ⌉ = Fam→Ty ⌈ t ⌉ꟳ

ty[] : ty ℓ [ σ ] ≡ ty ℓ
ty[] = Eq.refl

⌊⌋[]t : ⌊ A ⌋ [ σ ]t ≡ ⌊ A ⌋
⌊⌋[]t = Eq.refl


⌈⌉[] : {t : Tm Γ (ty ℓ)}
  → ⌈ t ⌉ [ σ ] ≡ ⌈ t [ σ ]t ⌉
⌈⌉[] = Eq.refl

∣⌈⌊⌋⌉∣ : {A : Ty ∅ ℓ} → ∣ ⌈_⌉ {Γ = ∅} ⌊ A ⌋ ∣ ≡ ∣ A ∣
∣⌈⌊⌋⌉∣ = Eq.refl

⌈⌊⌋⌉heq : {A : Ty ∅ ℓ} → ⌈_⌉ {Γ = ∅} ⌊ A ⌋ .heq ≡ A .heq
⌈⌊⌋⌉heq = Eq.refl
