open import Prelude

module Model.Prop.IF.Bool where

open Eq.≡-Reasoning

-- Booleans
module B where

  data Bool : Set where
    true false : Bool

  private variable
    b₀ b₁ b₂ : Bool

  _~_ : Bool → Bool → Prop
  true  ~ true  = ⊤ₚ
  false ~ false = ⊤ₚ
  true  ~ false = ⊥ₚ
  false ~ true  = ⊥ₚ

  -- _~_ is an equivalence (the off-diagonal cases are absurd as ⊥ₚ is empty)
  ref : (b : Bool) → b ~ b
  ref true  = tt
  ref false = tt

  sym : b₀ ~ b₁ → b₁ ~ b₀
  sym {true}  {true}  _ = tt
  sym {false} {false} _ = tt

  trans : b₀ ~ b₁ → b₁ ~ b₂ → b₀ ~ b₂
  trans {true}  {true}  {true}  _ _ = tt
  trans {false} {false} {false} _ _ = tt

  elim : (P : Bool → Set ℓ) → P true → P false → (b : Bool) → P b
  elim P Pt Pf true  = Pt
  elim P Pt Pf false = Pf

  elimp : (P : Bool → Prop ℓ) → P true → P false → (b : Bool) → P b
  elimp P Pt Pf true  = Pt
  elimp P Pt Pf false = Pf

open import Model.Prop.CwF

-- the type of Booleans
Bool : Ty Γ lzero
∣ Bool ∣ γ    = B.Bool
Bool .heq _      = B._~_
Bool .refl* b    = B.ref b
Bool .sym* p     = B.sym p
Bool .trans* p q = B.trans p q
Bool .coe _ b    = b
Bool .coh _ b    = B.ref b

-- constructors
true : Tm Γ Bool
∣ true ∣ _   = B.true
true .resp _ = tt

false : Tm Γ Bool
∣ false ∣ _   = B.false
false .resp _ = tt

-- dependent eliminator
module _ (P : Ty (Γ ▷ Bool) ℓ) (pt : Tm Γ (P ⟨ true ⟩)) (pf : Tm Γ (P ⟨ false ⟩)) where
  BoolΓ = Bool {Γ = Γ}

  elim-B-at : (γ : ∣ Γ ∣) (b : ∣ BoolΓ ∣ γ) → ∣ P ∣ (γ , b)
  elim-B-at γ = B.elim (λ b → ∣ P ∣ (γ , b)) (∣ pt ∣ γ) (∣ pf ∣ γ)

  elim-Bool : Tm (Γ ▷ Bool) P
  ∣ elim-Bool ∣ (γ , b)                           = elim-B-at γ b
  elim-Bool .resp {γ , B.true}  {γ′ , B.true}  (γ~γ′ , b~b′) = pt .resp γ~γ′
  elim-Bool .resp {γ , B.false} {γ′ , B.false} (γ~γ′ , b~b′) = pf .resp γ~γ′

Bool[] : Bool [ σ ] ≡ Bool
Bool[] = Eq.refl

true[] : true [ σ ]t ≡ true
true[] = Eq.refl

false[] : false [ σ ]t ≡ false
false[] = Eq.refl

module _ (P : Ty (Γ ▷ Bool) ℓ) (pt : Tm Γ (P ⟨ true ⟩)) (pf : Tm Γ (P ⟨ false ⟩)) where
  elim-true : elim-Bool P pt pf ⟨ true ⟩t ≡ pt
  elim-true = Eq.refl

  elim-false : elim-Bool P pt pf ⟨ false ⟩t ≡ pf
  elim-false = Eq.refl

  elim-Bool[]
    : {σ : Sub Δ Γ}
    → elim-Bool P pt pf [ σ ⁺ ]t ≡ elim-Bool (P [ σ ⁺ ]) (pt [ σ ]t) (pf [ σ ]t)
  elim-Bool[] = Eq.refl
