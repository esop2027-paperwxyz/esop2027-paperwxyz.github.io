FunExt
  : (B : Ty (Γ ▷ A) ℓB) (f g : Tm Γ (Π A B))
  → Tm Γ (Π A (⇑ Eq (app f) (app g)))
  → Tm Γ (⇑ Eq f g)
∣ FunExt B f g f~g ∣ γ = ↑ λ x y x~y →
  B .trans* (∣ f~g ∣ γ .₁ x .↓) (∣ g ∣ γ .₂ x~y)
FunExt B f g f~g .resp _ = tt

PropExt
  : (P Q : Tm Γ (Props ℓ))
  → Tm Γ (⇑ impₚ' (decodeP P) (decodeP Q))
  → Tm Γ (⇑ impₚ' (decodeP Q) (decodeP P))
  → Tm Γ (⇑ Eq P Q)
∣ PropExt P Q P→Q Q→P ∣ γ     = ↑ (↑ (∣ P→Q ∣ γ .↓ , ∣ Q→P ∣ γ .↓))
PropExt P Q P→Q Q→P .resp γ₀₁ = tt
