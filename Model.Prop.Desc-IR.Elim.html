-- Semantic induction hypotheses for one constructor.
IH₀
  : {Δ : Con ℓΔ}
  → {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
  → {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
  → (d : ConD Δ I D J E)
  → (X : ∣ I ∣ tt → Set ℓI)
  → (X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI)
  → (T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i))
  → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
           → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
  → (δ : ∣ Δ ∣)
  → {P : (i : ∣ I ∣ tt) (x : X i) → Set ℓ}
  → (P~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)(x₀₁ : X~ x₀ x₁) → P i₀ x₀ → P i₁ x₁ → Prop ℓ)
  → {j : ∣ J ∣ tt}
  → (xs : Arg₀ d X X~ T Tresp δ j)
  → Set (ℓI ⊔ ℓ)
IH₀ (end t e) X X~ T Tresp δ P~ xs = Lift _ ⊤
IH₀ (non A d) X X~ T Tresp δ P~ (a , xs) = IH₀ d X X~ T Tresp _ P~ xs
IH₀ (arg1 f d) X X~ T Tresp δ {P} P~ (x , xs) =
 P (∣ f ∣ δ) x × IH₀ d X X~ T Tresp _ P~ xs
IH₀ {Δ = Δ} (arg A f d) X X~ T Tresp δ {P} P~ ((g , gresp) , xs) =
 Σₚ[ pg ∈ ((a : ∣ A ∣ δ) → P (∣ f ∣ (δ , a)) (g a)) ] ({a₀ a₁ : ∣ A ∣ δ}(a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁) → P~ (f .resp ((Δ .refl δ) , a₀₁)) (gresp a₀₁) (pg a₀) (pg a₁))
 × IH₀ d X X~ T Tresp _ P~ xs
{-# INJECTIVE_FOR_INFERENCE IH₀ #-}

IHs₀
  : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
  → {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
  → (ds : ConDs I D J E)
  → {X : ∣ I ∣ tt → Set ℓI}
  → (X~ : ∀ {i₀ i₁} → X i₀ → X i₁ → Prop ℓI)
  → (T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i))
  → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
           → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
  → {P : (i : ∣ I ∣ tt) (x : X i) → Set ℓ}
  → (P~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : X~ x₀ x₁) → P i₀ x₀ → P i₁ x₁ → Prop ℓ)
  → {j : ∣ J ∣ tt}
  → (xs : Args₀ ds X X~ T Tresp j )
  → Set (ℓI ⊔ ℓ)
IHs₀ (d ∷ s) {X} X~ T Tresp P~ (inl xs) = IH₀ d X X~ T Tresp tt P~ xs
IHs₀ (d ∷ ds) X~ T Tresp P~ (inr xs) = IHs₀ ds X~ T Tresp P~ xs
{-# INJECTIVE_FOR_INFERENCE IHs₀ #-}

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D} where

 -- Setoid relation on semantic induction hypotheses.
 IH~
   : {Δ : Con ℓΔ}
   → (d : ConD Δ I D I D)
   → {δ₀ δ₁ : ∣ Δ ∣} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
   → {P₀ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₀~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)(x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₀ i₁ x₁ → Prop ℓ)
   → {P₁ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₁~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₁ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (P~ : ∀ {i₀ i₁ x₀ x₁}  (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → {j₀ j₁ : ∣ I ∣ tt}
   → (xs₀  : Arg₀ d _ _ _ _ δ₀ j₀) (xs₁ : Arg₀ d _ _ _ _ δ₁ j₁)
   → (xs₀₁ : Arg~ₛ ds d  δ₀₁ xs₀ xs₁)
   → (ih₀ : IH₀ d _ _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ (λ {x₁ = x₁} → P₀~ {x₁ = x₁}) xs₀) (ih₁ : IH₀ d _ _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ (λ {x₁ = x₁} → P₁~ {x₁ = x₁}) xs₁)
   → Prop (ℓI ⊔ ℓ)
 IH~ (end t e) δ₀₁ P₀~ P₁~ P~ xs₀ xs₁ xs₀₁ ih₀ ih₁ = ⊤ₚ
 IH~ (non A d) δ₀₁ P₀~ P₁~  P~ (a₀ , xs₀) (a₁ , xs₁) (a₀₁ , xs₀₁) ih₀ ih₁
  = IH~ d (δ₀₁ , a₀₁) P₀~ P₁~ P~ xs₀ xs₁ xs₀₁ ih₀ ih₁
 IH~ (arg1 f d) δ₀₁ P₀~ P₁~  P~ (x₀ , xs₀) (x₁ , xs₁) (x₀₁ , xs₀₁) (px₀ , ih₀) (px₁ , ih₁)
  = P~ (f . resp δ₀₁) x₀₁ px₀ px₁
    ∧ IH~ d _ P₀~ P₁~ P~ xs₀ xs₁ xs₀₁ ih₀ ih₁
 IH~ (arg A f d) δ₀₁ P₀~ P₁~ P~ ((g₀ , gresp₀) , xs₀) ((g₁ , gresp₁) , xs₁) (g₀₁ , xs₀₁) ((pg₀ , pgresp₀), ih₀) ((pg₁ , pgresp₁), ih₁)
  = (∀ a₀ a₁ (a₀₁ : A ∋ δ₀₁ ⊢ a₀ ~ a₁) → P~ (f .resp (_ , a₀₁)) (g₀₁ a₀₁) (pg₀ a₀) (pg₁ a₁))
    ∧ IH~ d _ P₀~ P₁~ P~ xs₀ xs₁ xs₀₁ ih₀ ih₁
 {-# INJECTIVE_FOR_INFERENCE IH~ #-}


 IHs~
   : (es : ConDs I D I D)
   → {P₀ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₀~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₀ i₁ x₁ → Prop ℓ)
   → {P₁ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₁~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₁ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (P~ : ∀ {i₀ i₁ x₀ x₁}  (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → {j₀ j₁ : ∣ I ∣ tt}
   → (xs₀ : Args₀ es _ _ _ _ j₀)
   → (xs₁ : Args₀ es _ _ _ _ j₁)
   → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
   → (ih₀ : IHs₀ es _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) (λ {x₁ = x₁} → P₀~ {x₁ = x₁}) xs₀) (ih₁ : IHs₀ es _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) (λ {x₁ = x₁} → P₁~ {x₁ = x₁}) xs₁)
   → Prop (ℓI ⊔ ℓ)
 IHs~ (d ∷ ds) P₀~ P₁~ P~ (inl _) (inl _) xs₀₁ ih₀ ih₁ =
   IH~ d tt P₀~ P₁~ P~ _ _ xs₀₁ ih₀ ih₁
 IHs~ (d ∷ ds) P₀~ P₁~ P~ (inr _) (inr _) xs₀₁ ih₀ ih₁ =
   IHs~ ds P₀~ P₁~ P~ _ _ xs₀₁ ih₀ ih₁
 {-# INJECTIVE_FOR_INFERENCE IHs~ #-}

 -- Transport and coherence for semantic induction hypotheses.
 coeIH
   : {Δ : Con ℓΔ}
   → (d : ConD Δ I D I D)
   → {δ₀ δ₁ : ∣ Δ ∣} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
   → {P₀ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₀~ : ∀ {i₀ i₁ x₀ x₁} i₀₁ (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₀ i₁ x₁ → Prop ℓ)
   → {P₁ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₁~ : ∀ {i₀ i₁ x₀ x₁} i₀₁ (x₀₁ : μ~ ds x₀ x₁) → P₁ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (P~ : ∀ {i₀ i₁ x₀ x₁}  (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (Ptrans₁₀₀₁
       : ∀ {i₀ i₀' i₁ i₁' x₀ x₀' x₁ x₁'}
       → {px₀ : P₀ i₀ x₀} {px₀' : P₀ i₀' x₀'} {px₁ : P₁ i₁ x₁} {px₁' : P₁ i₁' x₁'}
       → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
       → {x₀₁ : μ~ ds x₀ x₁} (x₀₀' : μ~ ds x₀ x₀') (x₀'₁' : μ~ ds x₀' x₁')
       → {x₁₁' : μ~ ds x₁ x₁'}
       → P~ i₀₁ x₀₁ px₀ px₁ → P₀~ i₀₀' x₀₀' px₀ px₀' → P~ i₀'₁' x₀'₁' px₀' px₁'
       → P₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁₁' px₁ px₁')
   → (coeP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀) → P₁ i₁ x₁)
   → (cohP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀)
       → P~ i₀₁ x₀₁ px₀ (coeP i₀₁ x₀₁ px₀))
   → ∀ {j₀ j₁} {xs₀ : Arg₀ d _ _ _ _ δ₀ j₀} {xs₁ : Arg₀ d _ _ _ _ δ₁ j₁}
   → (xs₀₁ : Arg~ₛ ds d δ₀₁ xs₀ xs₁)
   → (ih₀ : IH₀ d _ _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ (λ {x₁ = x₁} → P₀~ {x₁ = x₁}) xs₀)
   → IH₀ d _ _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ (λ {x₁ = x₁} → P₁~ {x₁ = x₁}) xs₁
 coeIH (end t e) δ₀₁ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀ = ih₀
 coeIH (non A d) δ₀₁ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP (a₀₁ , xs₀₁) ih₀ = coeIH d _ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀
 coeIH (arg1 f d) δ₀₁ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP (x₀₁ , xs₀₁) (px₀ , ih₀)
  = coeP (f .resp δ₀₁) x₀₁ px₀ , coeIH d _ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀
 coeIH {Δ = Δ} (arg A f d) δ₀₁ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP {xs₁ = (g₁ , _)} (g₀₁ , xs₀₁) ((pg₀ , pgresp₀) , ih₀)
  = let δ₁₀ = Δ .sym δ₀₁ in
     ((λ a₁ → coeP (f .resp (δ₀₁ , coh' A δ₁₀ a₁)) (g₀₁ (coh' A δ₁₀ a₁)) (pg₀ (A .coe δ₁₀ a₁)))
     , λ {a₁} {a₁'} a₁₁' → Ptrans₁₀₀₁ _ _ ((cohP (f .resp (δ₀₁ , coh' A δ₁₀ a₁)) (g₀₁ (coh' A δ₁₀ a₁)) (pg₀ (A .coe δ₁₀ a₁))))
                                          (pgresp₀ (A .trans* (coh' A δ₁₀ a₁) (A .trans* a₁₁' (A .coh δ₁₀ a₁'))))
                                          (cohP (f .resp (δ₀₁ , coh' A δ₁₀ a₁')) (g₀₁ (coh' A δ₁₀ a₁')) (pg₀ (A .coe δ₁₀ a₁'))))
    , coeIH d _ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀

 cohIH
   : {Δ : Con ℓΔ}
   → (d : ConD Δ I D I D)
   → {δ₀ δ₁ : ∣ Δ ∣} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
   → {P₀ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₀~ : ∀ {i₀ i₁ x₀ x₁} i₀₁ (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₀ i₁ x₁ → Prop ℓ)
   → {P₁ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₁~ : ∀ {i₀ i₁ x₀ x₁} i₀₁ (x₀₁ : μ~ ds x₀ x₁) → P₁ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (P~ : ∀ {i₀ i₁ x₀ x₁}  (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (Ptrans₀₀₁
       : ∀ {i₀ i₀' i₁ x₀ x₀' x₁}
       → {px₀ : P₀ i₀ x₀} {px₀' : P₀ i₀' x₀'} {px₁ : P₁ i₁ x₁}
       → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
       → {x₀₀' : μ~ ds x₀ x₀'} {x₀'₁ : μ~ ds x₀' x₁}
       → {x₀₁ : μ~ ds x₀ x₁}
       → P₀~ i₀₀' x₀₀' px₀ px₀' → P~ i₀'₁ x₀'₁ px₀' px₁
       → P~ (I .trans* i₀₀' i₀'₁) x₀₁ px₀ px₁)
   → (Ptrans₁₀₀₁
       : ∀ {i₀ i₀' i₁ i₁' x₀ x₀' x₁ x₁'}
       → {px₀ : P₀ i₀ x₀} {px₀' : P₀ i₀' x₀'} {px₁ : P₁ i₁ x₁} {px₁' : P₁ i₁' x₁'}
       → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
       → {x₀₁ : μ~ ds x₀ x₁} (x₀₀' : μ~ ds x₀ x₀') (x₀'₁' : μ~ ds x₀' x₁')
       → {x₁₁' : μ~ ds x₁ x₁'}
       → P~ i₀₁ x₀₁ px₀ px₁ → P₀~ i₀₀' x₀₀' px₀ px₀' → P~ i₀'₁' x₀'₁' px₀' px₁'
       → P₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁₁' px₁ px₁')
   → (coeP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀) → P₁ i₁ x₁)
   → (cohP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀)
       → P~ i₀₁ x₀₁ px₀ (coeP i₀₁ x₀₁ px₀))
   → ∀ {j₀ j₁} {xs₀ : Arg₀ d _ _ _ _ δ₀ j₀} {xs₁ : Arg₀ d _ _ _ _ δ₁ j₁}
   → (xs₀₁ : Arg~ₛ ds d δ₀₁ xs₀ xs₁)
   → (ih₀ : IH₀ d _ _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ (λ {x₁ = x₁} → P₀~ {x₁ = x₁}) xs₀)
  → IH~ d δ₀₁ P₀~ P₁~ P~ xs₀ xs₁ xs₀₁ ih₀ (coeIH d δ₀₁ P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀)
 cohIH (end t e) δ₀₁ P₀~ P₁~ P~ Ptrans₀₀₁ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀ = tt
 cohIH (non A d) δ₀₁ P₀~ P₁~ P~ Ptrans₀₀₁ Ptrans₁₀₀₁ coeP cohP (a₀₁ , xs₀₁) ih₀ = cohIH d (δ₀₁ , a₀₁) P₀~ P₁~ P~ Ptrans₀₀₁ (λ {i₀'₁' = i₀'₁'} → Ptrans₁₀₀₁ {i₀'₁' = i₀'₁'}) coeP cohP xs₀₁ ih₀
 cohIH (arg1 f d) δ₀₁ P₀~ P₁~ P~ Ptrans₀₀₁ Ptrans₁₀₀₁ coeP cohP {xs₀ = x₀ , _} (x₀₁ , xs₀₁) (px₀ , ih₀)
  = cohP (f .resp δ₀₁) x₀₁ px₀ , cohIH d (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) x₀₁) P₀~ P₁~ P~ Ptrans₀₀₁ (λ {i₀'₁' = i₀'₁'} → Ptrans₁₀₀₁ {i₀'₁' = i₀'₁'}) coeP cohP xs₀₁ ih₀
 cohIH {Δ = Δ} (arg A f d) δ₀₁ P₀~ P₁~ P~ Ptrans₀₀₁ Ptrans₁₀₀₁ coeP cohP {xs₀ = (g₀ , _) , _} (g₀₁ , xs₀₁) ((pg₀ , pgresp₀) , ih₀)
  = let δ₁₀ = Δ .sym δ₀₁ in
    (λ a₀ a₁ a₀₁ → Ptrans₀₀₁ (pgresp₀ (A .trans* a₀₁ (A .coh δ₁₀ a₁)))
                             (cohP (f .resp (δ₀₁ , coh' A δ₁₀ a₁)) (g₀₁ (coh' A δ₁₀ a₁)) (pg₀ (A .coe δ₁₀ a₁))))
    , cohIH d (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) P₀~ P₁~ P~ Ptrans₀₀₁ (λ {i₀'₁' = i₀'₁'} → Ptrans₁₀₀₁ {i₀'₁' = i₀'₁'}) coeP cohP xs₀₁ ih₀

 coeIHs
   : (es : ConDs I D I D)
   → {P₀ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₀~ : ∀ {i₀ i₁ x₀ x₁} i₀₁ (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₀ i₁ x₁ → Prop ℓ)
   → {P₁ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₁~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₁ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (P~ : ∀ {i₀ i₁ x₀ x₁}  (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (Ptrans₁₀₀₁
       : ∀ {i₀ i₀' i₁ i₁' x₀ x₀' x₁ x₁'}
       → {px₀ : P₀ i₀ x₀} {px₀' : P₀ i₀' x₀'} {px₁ : P₁ i₁ x₁} {px₁' : P₁ i₁' x₁'}
       → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
       → {x₀₁ : μ~ ds x₀ x₁} (x₀₀' : μ~ ds x₀ x₀') (x₀'₁' : μ~ ds x₀' x₁')
       → {x₁₁' : μ~ ds x₁ x₁'}
       → P~ i₀₁ x₀₁ px₀ px₁ → P₀~ i₀₀' x₀₀' px₀ px₀' → P~ i₀'₁' x₀'₁' px₀' px₁'
       → P₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁₁' px₁ px₁')
   → (coeP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀) → P₁ i₁ x₁)
   → (cohP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀)
       → P~ i₀₁ x₀₁ px₀ (coeP i₀₁ x₀₁ px₀))
   → ∀ {j₀ j₁} {xs₀ : Args₀ es _ _ _ _ j₀} {xs₁ : Args₀ es _ _ _ _ j₁}
   → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
   → (ih₀ : IHs₀ es _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) (λ {x₁ = x₁} → P₀~ {x₁ = x₁}) xs₀)
   → IHs₀ es _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) (λ {x₁ = x₁} → P₁~ {x₁ = x₁}) xs₁
 coeIHs (d ∷ ds) P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP {xs₀ = inl _} {inl _} xs₀₁ ih₀
  = coeIH d tt P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀
 coeIHs (d ∷ ds) P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP {xs₀ = inr _} {inr _} xs₀₁ ih₀
  = coeIHs ds P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀

 cohIHs
   : (es : ConDs I D I D)
   → {P₀ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₀~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₀ i₁ x₁ → Prop ℓ)
   → {P₁ : (i : ∣ I ∣ tt) (x : μ₀ ds i) → Set ℓ}
   → (P₁~ : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₁ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (P~ : ∀ {i₀ i₁ x₀ x₁}  (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁) → P₀ i₀ x₀ → P₁ i₁ x₁ → Prop ℓ)
   → (Ptrans₀₀₁
       : ∀ {i₀ i₀' i₁ x₀ x₀' x₁}
       → {px₀ : P₀ i₀ x₀} {px₀' : P₀ i₀' x₀'} {px₁ : P₁ i₁ x₁}
       → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
       → {x₀₀' : μ~ ds x₀ x₀'} {x₀'₁ : μ~ ds x₀' x₁}
       → {x₀₁ : μ~ ds x₀ x₁}
       → P₀~ i₀₀' x₀₀' px₀ px₀' → P~ i₀'₁ x₀'₁ px₀' px₁
       → P~ (I .trans* i₀₀' i₀'₁) x₀₁ px₀ px₁)
   → (Ptrans₁₀₀₁
       : ∀ {i₀ i₀' i₁ i₁' x₀ x₀' x₁ x₁'}
       → {px₀ : P₀ i₀ x₀} {px₀' : P₀ i₀' x₀'} {px₁ : P₁ i₁ x₁} {px₁' : P₁ i₁' x₁'}
       → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
       → {x₀₁ : μ~ ds x₀ x₁} (x₀₀' : μ~ ds x₀ x₀') (x₀'₁' : μ~ ds x₀' x₁')
       → {x₁₁' : μ~ ds x₁ x₁'}
       → P~ i₀₁ x₀₁ px₀ px₁ → P₀~ i₀₀' x₀₀' px₀ px₀' → P~ i₀'₁' x₀'₁' px₀' px₁'
       → P₁~ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁₁' px₁ px₁')
   → (coeP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀) → P₁ i₁ x₁)
   → (cohP
       : ∀ {i₀ i₁ x₀ x₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
       → (px₀ : P₀ i₀ x₀)
       → P~ i₀₁ x₀₁ px₀ (coeP i₀₁ x₀₁ px₀))
   → ∀ {j₀ j₁} {xs₀ : Args₀ es _ _ _ _ j₀} {xs₁ : Args₀ es _ _ _ _ j₁}
   → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
   → (ih₀ : IHs₀ es _ _ (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) (λ {x₁ = x₁} → P₀~ {x₁ = x₁}) xs₀)
   → IHs~ es P₀~ P₁~ P~ xs₀ xs₁ xs₀₁ ih₀ (coeIHs es P₀~ P₁~ P~ Ptrans₁₀₀₁ coeP cohP xs₀₁ ih₀)
 cohIHs (d ∷ ds) P₀~ P₁~ P~ Ptrans₀₀₁ Ptrans₁₀₀₁ coeP cohP {xs₀ = inl _} {inl _} xs₀₁ ih₀
  = cohIH d tt P₀~ P₁~ P~ Ptrans₀₀₁ (λ {i₀'₁' = i₀'₁'} → Ptrans₁₀₀₁ {i₀'₁' = i₀'₁'}) coeP cohP xs₀₁ ih₀
 cohIHs (d ∷ ds) P₀~ P₁~ P~ Ptrans₀₀₁ Ptrans₁₀₀₁ coeP cohP {xs₀ = inr _} {inr _} xs₀₁ ih₀
  = cohIHs ds P₀~ P₁~ P~ Ptrans₀₀₁ (λ {i₀'₁' = i₀'₁'} → Ptrans₁₀₀₁ {i₀'₁' = i₀'₁'}) coeP cohP xs₀₁ ih₀

-- Induction-hypothesis families

module _ {Γ : Con ℓΓ} {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} {ds : ConDs I D I D} (P : Ty (Γ ▷ I [ ε ] ▷ μ ds) ℓ) where
  -- Shorthand carrier for semantic induction hypotheses.
  ∣IH∣
    : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
    → ∀ γ δ {i} (xs : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i)
    → Set (ℓI ⊔ ℓ)
  ∣IH∣ e γ δ {i} xs = IH₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ
                          {λ i x → ∣ P ∣ ((γ , i) , x)}
                          (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
                          xs

  ∣IHs∣
    : (es : ConDs I D I D)
    → ∀ γ {i} (xs : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
    → Set (ℓI ⊔ ℓ)
  ∣IHs∣ es γ {i} xs = IHs₀ es (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀})
                           {P = λ i x → ∣ P ∣ ((γ , i) , x)}
                           (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
                           xs

  -- Shorthand relation for semantic induction hypotheses.
  ∣IH~∣
    : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
    → ∀ {γ₀ γ₁ δ₀ δ₁ i₀ i₁}
    → {xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀}
    → {xs₁ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : Arg~ₛ ds e δ₀₁ xs₀ xs₁)
    → (ih₀ : ∣IH∣ e γ₀ δ₀ xs₀)
    → (ih₁ : ∣IH∣ e γ₁ δ₁ xs₁)
    → Prop (ℓI ⊔ ℓ)
  ∣IH~∣ e {γ₀} {γ₁} γ₀₁ δ₀₁ = IH~ e δ₀₁
                                  (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
                                  (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
                                  ((λ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁)))
                                  _ _

  ∣IHs~∣
    : (es : ConDs I D I D)
    → ∀ {γ₀ γ₁ i₀ i₁}
    → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
    → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
    → (ih₀ : ∣IHs∣ es γ₀ xs₀)
    → (ih₁ : ∣IHs∣ es γ₁ xs₁)
    → Prop (ℓI ⊔ ℓ)
  ∣IHs~∣ es {γ₀} {γ₁} γ₀₁ xs₀₁ =
    IHs~ es
         (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
         (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
         (λ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
         _ _ xs₀₁


  -- Setoid structure on semantic induction hypotheses.
  IH₀-refl*
    : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
    → ∀ {γ δ i} {xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i}
    → (ih₀ : ∣IH∣ e γ δ xs₀)
    → ∣IH~∣ e (Γ .refl γ) (Δ .refl δ) (Arg₀-refl* e xs₀) ih₀ ih₀
  IH₀-refl* (end t e)   ih₀                = tt
  IH₀-refl* (non A E)   ih₀                = IH₀-refl* E ih₀
  IH₀-refl* (arg1 f E)  (px , ih₀)         = P .refl* px , IH₀-refl* E ih₀
  IH₀-refl* (arg A f E) ((px , px~) , ih₀) = (λ a₀ a₁ a₀₁ → px~ a₀₁) , IH₀-refl* E ih₀


  IH₀-sym*
    : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
    → ∀ {γ₀ γ₁ δ₀ δ₁ i₀ i₁}
    → {xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀}
    → {xs₁ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ i₁}
    → {ih₀ : ∣IH∣ e γ₀ δ₀ xs₀} {ih₁ : ∣IH∣ e γ₁ δ₁ xs₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : Arg~ₛ ds e δ₀₁ xs₀ xs₁)
    → (ih₀₁ : ∣IH~∣ e γ₀₁ δ₀₁ xs₀₁ ih₀ ih₁)
    → ∣IH~∣ e (Γ .sym γ₀₁) (Δ .sym δ₀₁) (Arg₀-sym* e xs₀₁) ih₁ ih₀
  IH₀-sym* (end t e) γ₀₁ δ₀₁ xs₀₁ ih₀₁ = tt
  IH₀-sym* (non A E) γ₀₁ δ₀₁ (a₀₁ , xs₀₁) ih₀₁ =
    IH₀-sym* E γ₀₁ (δ₀₁ , a₀₁) xs₀₁ ih₀₁
  IH₀-sym* (arg1 f E) {xs₀ = x₀ , _} γ₀₁ δ₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    P .sym* px₀₁ , IH₀-sym* E γ₀₁ (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) x₀₁) xs₀₁ ih₀₁
  IH₀-sym* (arg A f E) {xs₀ = (g₀ , _) , _} γ₀₁ δ₀₁ (g₀₁ , xs₀₁) (pg₀₁ , ih₀₁) =
    (λ a₀ a₁ a₀₁ → P .sym* (pg₀₁ _ _ (A .sym* a₀₁)))
    , IH₀-sym* E γ₀₁ (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) xs₀₁ ih₀₁

  IH₀-trans*
    : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
    → ∀ {γ₀ γ₁ γ₂ δ₀ δ₁ δ₂ i₀ i₁ i₂}
    → {xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀}
    → {xs₁ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ i₁}
    → {xs₂ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₂ i₂}
    → {ih₀ : ∣IH∣ e γ₀ δ₀ xs₀} {ih₁ : ∣IH∣ e γ₁ δ₁ xs₁} {ih₂ : ∣IH∣ e γ₂ δ₂ xs₂}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂) (δ₀₁ : Δ ∋ δ₀ ~ δ₁) (δ₁₂ : Δ ∋ δ₁ ~ δ₂)
    → (xs₀₁ : Arg~ₛ ds e δ₀₁ xs₀ xs₁)
    → (xs₁₂ : Arg~ₛ ds e δ₁₂ xs₁ xs₂)
    → (ih₀₁ : ∣IH~∣ e γ₀₁ δ₀₁ xs₀₁ ih₀ ih₁)
    → (ih₁₂ : ∣IH~∣ e γ₁₂ δ₁₂ xs₁₂ ih₁ ih₂)
    → ∣IH~∣ e (Γ .trans γ₀₁ γ₁₂) (Δ .trans δ₀₁ δ₁₂) (Arg₀-trans* e xs₀₁ xs₁₂) ih₀ ih₂
  IH₀-trans* (end t d) γ₀₁ γ₁₂ δ₀₁ δ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂ = tt
  IH₀-trans* (non A e) γ₀₁ γ₁₂ δ₀₁ δ₁₂ (a₀₁ , xs₀₁) (a₁₂ , xs₁₂) ih₀₁ ih₁₂ =
    IH₀-trans* e γ₀₁ γ₁₂ (δ₀₁ , a₀₁) (δ₁₂ , a₁₂) xs₀₁ xs₁₂ ih₀₁ ih₁₂
  IH₀-trans* (arg1 f e) {xs₀ = x₀ , _} {xs₁ = x₁ , _} γ₀₁ γ₁₂ δ₀₁ δ₁₂ (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) (px₀₁ , ih₀₁) (px₁₂ , ih₁₂) =
    P .trans* px₀₁ px₁₂
    , IH₀-trans* e γ₀₁ γ₁₂ (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) x₀₁) (δ₁₂ , Tresp₀ ds {x₀ = x₁} (f .resp δ₁₂) x₁₂) xs₀₁ xs₁₂ ih₀₁ ih₁₂
  IH₀-trans* (arg A f e) {xs₀ = (g₀ , _) , _} {xs₁ = (g₁ , _) , _} γ₀₁ γ₁₂ δ₀₁ δ₁₂ (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) (px₀₁ , ih₀₁) (px₁₂ , ih₁₂) =
    (λ a₀ a₂ a₀₂ →
      P .trans* (px₀₁ _ _ (A .coh δ₀₁ a₀))
                (px₁₂ _ _ (A .trans* (coh' A δ₀₁ a₀) a₀₂)))
    , IH₀-trans* e γ₀₁ γ₁₂ (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (x₀₁ a₀₁)) (δ₁₂ , λ a₁ a₂ a₁₂ → Tresp₀ ds {x₀ = g₁ a₁} (f .resp (δ₁₂ , a₁₂)) (x₁₂ a₁₂)) xs₀₁ xs₁₂ ih₀₁ ih₁₂


  IHs₀-refl*
    : (es : ConDs I D I D)
    → ∀ {γ i} {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i}
    → (ih₀ : ∣IHs∣ es γ xs₀)
    → ∣IHs~∣ es (Γ .refl γ) (Args₀-refl* es xs₀) ih₀ ih₀
  IHs₀-refl* (E ∷ Es) {xs₀ = inl _} ih₀ = IH₀-refl* E ih₀
  IHs₀-refl* (E ∷ Es) {xs₀ = inr _} ih₀ = IHs₀-refl* Es ih₀

  IHs₀-sym*
    : (es : ConDs I D I D)
    → ∀ {γ₀ γ₁ i₀ i₁}
    → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
    → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
    → {ih₀ : ∣IHs∣ es γ₀ xs₀} {ih₁ : ∣IHs∣ es γ₁ xs₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
    → (ih₀₁ : ∣IHs~∣ es γ₀₁ xs₀₁ ih₀ ih₁)
    → ∣IHs~∣ es (Γ .sym γ₀₁) (Args₀-sym* es xs₀₁) ih₁ ih₀
  IHs₀-sym* (E ∷ Es) {xs₀ = inl _} {inl _} γ₀₁ xs₀₁ ih₀₁ = IH₀-sym* E γ₀₁ tt xs₀₁ ih₀₁
  IHs₀-sym* (E ∷ Es) {xs₀ = inr _} {inr _} γ₀₁ xs₀₁ ih₀₁ = IHs₀-sym* Es γ₀₁ xs₀₁ ih₀₁

  IHs₀-trans*
    : (es : ConDs I D I D)
    → ∀ {γ₀ γ₁ γ₂ i₀ i₁ i₂}
    → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
    → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
    → {xs₂ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₂}
    → {ih₀ : ∣IHs∣ es γ₀ xs₀} {ih₁ : ∣IHs∣ es γ₁ xs₁} {ih₂ : ∣IHs∣ es γ₂ xs₂}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂)
    → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
    → (xs₁₂ : Args~ₛ ds es xs₁ xs₂)
    → (ih₀₁ : ∣IHs~∣ es γ₀₁ xs₀₁ ih₀ ih₁)
    → (ih₁₂ : ∣IHs~∣ es γ₁₂ xs₁₂ ih₁ ih₂)
    → ∣IHs~∣ es (Γ .trans γ₀₁ γ₁₂) (Args₀-trans* es xs₀₁ xs₁₂) ih₀ ih₂
  IHs₀-trans* (e ∷ es) {xs₀ = inl _} {inl _} {inl _} γ₀₁ γ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂ =
    IH₀-trans* e γ₀₁ γ₁₂ tt tt xs₀₁ xs₁₂ ih₀₁ ih₁₂
  IHs₀-trans* (e ∷ es) {xs₀ = inr _} {inr _} {inr _} γ₀₁ γ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂ =
    IHs₀-trans* es γ₀₁ γ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂

  -- The IH type is the displayed induction hypothesis specialized to μ Ds.
  IH
    : {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (E : ConD Δ I D I D)
    → Ty (Γ ▷ I [ ε ] ▷ Arg E [ σ ↑t I [ ε ] ]) (ℓI ⊔ ℓ)
  ∣ IH σ E ∣ ((γ , i) , xs) = ∣IH∣ E γ (∣ σ ∣ γ) xs
  IH σ E .heq ((γ₀₁ , i₀₁) , xs₀₁) ih₀ ih₁ = ∣IH~∣ E γ₀₁ (σ .resp γ₀₁) xs₀₁ ih₀ ih₁
  IH σ E .refl* {(γ , j) , xs} = IH₀-refl* E
  IH σ E .sym* {γ₀₁ = (γ₀₁ , j₀₁) , xs₀₁} = IH₀-sym* E γ₀₁ (σ .resp γ₀₁) xs₀₁
  IH σ E .trans* {γ₀₁ = (γ₀₁ , j₀₁) , xs₀₁} {(γ₁₂ , j₁₂) , xs₁₂} =
    IH₀-trans* E γ₀₁ γ₁₂ (σ .resp γ₀₁) (σ .resp γ₁₂) xs₀₁ xs₁₂
  IH σ E .coe {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    coeIH E (σ .resp γ₀₁)
          (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
          (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
          (λ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
          (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
          ((λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁)))
          ((λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁)))
          xs₀₁
  IH σ E .coh {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    cohIH E (σ .resp γ₀₁)
          (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
          (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
          (λ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
          (P .trans*)
          (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
          ((λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁)))
          ((λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁)))
          xs₀₁
  {-# INJECTIVE_FOR_INFERENCE IH #-}

  IHs
    : (Es : ConDs I D I D)
    → Ty (Γ ▷ I [ ε ] ▷ Args Es) (ℓI ⊔ ℓ)
  ∣ IHs Es ∣ ((γ , i) , xs) = ∣IHs∣ Es γ xs
  IHs es .heq ((γ₀₁ , i₀₁) , xs₀₁) ih₀ ih₁ = ∣IHs~∣ es γ₀₁ xs₀₁ ih₀ ih₁
  IHs es .refl* = IHs₀-refl* es
  IHs es .sym* {γ₀₁ = (γ₀₁ , i₀₁) , xs₀₁} = IHs₀-sym* es γ₀₁ xs₀₁
  IHs es .trans* {γ₀₁ = (γ₀₁ , i₀₁) , xs₀₁} {(γ₁₂ , i₁₂) , xs₁₂} = IHs₀-trans* es γ₀₁ γ₁₂ xs₀₁ xs₁₂
  IHs es .coe {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    coeIHs es
           (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
           (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
           (λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁))
           xs₀₁
  IHs es .coh {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    cohIHs es
           (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
           (P .trans*)
           (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
           (λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁))
           xs₀₁
  {-# INJECTIVE_FOR_INFERENCE IHs #-}

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} {ds : ConDs I D I D} where
  module Elim₀
      (P  : (i : ∣ I ∣ tt)(x₀ : μ₀ ds i) → Set ℓ)
      (P~ : {i₀ i₁ : ∣ I ∣ tt}
        {x₀ : μ₀ ds i₀} {x₁ : μ₀ ds i₁}
        → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀₁ : μ~ ds x₀ x₁)
        → P _ x₀ → P _ x₁ → Prop ℓ)
      (~refl : ∀ {i x} (v : P i x)
             → P~ (I .refl* i) (μ~-refl* x) v v)
      (~sym : ∀ {i₀ i₁ x₀ x₁}
                {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {x₀₁ : μ~ ds x₀ x₁}
                {v₀ : P i₀ x₀} {v₁ : P i₁ x₁}
            → P~ i₀₁ x₀₁ v₀ v₁
            → P~ (I .sym* i₀₁) (μ~-sym* {ds = ds} {x₀ = x₀} x₀₁) v₁ v₀)
      (~trans : ∀ {i₀ i₁ i₂ x₀ x₁ x₂}
                  {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₁₂ : I ∋ tt ⊢ i₁ ~ i₂}
                  {x₀₁ : μ~ ds x₀ x₁} {x₁₂ : μ~ ds x₁ x₂}
                  {v₀ : P i₀ x₀} {v₁ : P i₁ x₁} {v₂ : P i₂ x₂}
              → P~ i₀₁ x₀₁ v₀ v₁ → P~ i₁₂ x₁₂ v₁ v₂
              → P~ (I .trans* i₀₁ i₁₂) (μ~-trans* {ds = ds} {x₀ = x₀} x₀₁ x₁₂) v₀ v₂)
      (pxs
        : {i : ∣ I ∣ tt} (xs₀ : Args₀ ds (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
        → (ih₀ : IHs₀ ds (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) (λ {x₁ = x₁} → P~ {x₁ = x₁}) xs₀)
        → P i (con xs₀))
      (pxs~
        : {i₀ i₁ : ∣ I ∣ tt}
          {xs₀ : Args₀ ds (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀} {xs₁ : Args₀ ds (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
          {ih₀ : IHs₀ ds (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) P~ xs₀}
          {ih₁ : IHs₀ ds (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) P~ xs₁}
          → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
          → (xs₀₁ : Args~ₛ ds ds xs₀ xs₁)
          → (ih₀₁ : IHs~ ds P~ P~ P~ xs₀ xs₁ xs₀₁ ih₀ ih₁)
          → P~
              i₀₁
              xs₀₁
              (pxs xs₀ ih₀)
              (pxs xs₁ ih₁))
    where

    mutual
      isRec
        : {i : ∣ I ∣ tt} (x : μ₀ ds i) (v : P i x)
        → Set (ℓI ⊔ ℓ)
      isRec {i} (con xs) v =
        Σₚ[ ihʳ ∈ IHsʳ₀ ds xs .₁ ]
          P~ (I .refl* i) (μ~-refl* (con xs))
             (pxs xs (IHsʳ₀ ds xs .₂ ihʳ)) v

      IHsʳ₀
        : (es : ConDs I D I D) {i : ∣ I ∣ tt}
        → (xs : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
        → Σ[ X ∈ Set (ℓI ⊔ ℓ) ] (X → IHs₀ es (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) P~ xs)
      IHsʳ₀ (E ∷ Es) (inl xs) = IHʳ₀ E xs
      IHsʳ₀ (E ∷ Es) (inr xs) = IHsʳ₀ Es xs

      IHʳ₀
        : {Δ : Con ℓΔ} (e : ConD Δ I D I D) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
        → (xs : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i)
        → Σ[ X ∈ Set (ℓI ⊔ ℓ) ] (X → IH₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ (λ {x₁ = x₁} → P~ {x₁ = x₁}) xs)
      IHʳ₀ (end t e)   xs = Lift _ ⊤ , λ _ → ↑ tt
      IHʳ₀ (non A E) {δ} (a , xs) = IHʳ₀ E {δ , a} xs
      IHʳ₀ (arg1 f E) {δ} (x , xs) =
        (Σ (P (∣ f ∣ δ) x) (isRec x) × IHʳ₀ E xs .₁)
        , λ (pr , ihʳ) → pr .₁ , IHʳ₀ E xs .₂ ihʳ
      IHʳ₀ {Δ = Δ} (arg A f E) {δ} ((g , gresp) , xs) =
        (Σₚ ((a : ∣ A ∣ δ) → Σ (P (∣ f ∣ (δ , a)) (g a)) (isRec (g a)))
           (λ pr → {a₀ a₁ : ∣ A ∣ δ} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁)
                     → P~ (f .resp (Δ .refl δ , a₀₁)) (gresp a₀₁) (pr a₀ .₁) (pr a₁ .₁))
        × IHʳ₀ E xs .₁)
        , λ ((pr , pr~) , ihʳ) → ((λ a → pr a .₁) , pr~) , IHʳ₀ E xs .₂ ihʳ

    mutual
      uniq
        : {i₀ i₁ : ∣ I ∣ tt}
          (x₀ : μ₀ ds i₀) (v₀ : P i₀ x₀)
          (x₁ : μ₀ ds i₁) (v₁ : P i₁ x₁)
          (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ ds x₀ x₁)
        → isRec x₀ v₀ → isRec x₁ v₁ → P~ i₀₁ x₀₁ v₀ v₁
      uniq (con xs₀) v₀ (con xs₁) v₁ i₀₁ xs₀₁ (ihʳ₀ , p₀) (ihʳ₁ , p₁) =
        ~trans (~sym p₀)
          (~trans (pxs~ i₀₁ xs₀₁ (uniqIHs ds xs₀₁ ihʳ₀ ihʳ₁)) p₁)

      uniqIH
        : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
        → {δ₀ δ₁ : ∣ Δ ∣} {j₀ j₁ : ∣ I ∣ tt}
        → {xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ j₀}
        → {xs₁ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ j₁}
        → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
        → (xs₀₁ : Arg~ₛ ds e δ₀₁ xs₀ xs₁)
        → (ihʳ₀ : IHʳ₀ e xs₀ .₁) (ihʳ₁ : IHʳ₀ e xs₁ .₁)
        → IH~ e δ₀₁ P~ P~ P~ xs₀ xs₁ xs₀₁
            (IHʳ₀ e xs₀ .₂ ihʳ₀) (IHʳ₀ e xs₁ .₂ ihʳ₁)
      uniqIH (end t e) δ₀₁ xs₀₁ ihʳ₀ ihʳ₁ = tt
      uniqIH (non A E) δ₀₁ (a₀₁ , xs₀₁) ihʳ₀ ihʳ₁ =
        uniqIH E (δ₀₁ , a₀₁) xs₀₁ ihʳ₀ ihʳ₁
      uniqIH (arg1 f E) {xs₀ = x₀ , _} δ₀₁ (x₀₁ , xs₀₁) (pr₀ , ihʳ₀) (pr₁ , ihʳ₁) =
        uniq _ _ _ _ (f .resp δ₀₁) x₀₁ (pr₀ .₂) (pr₁ .₂)
        , uniqIH E (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) x₀₁) xs₀₁ ihʳ₀ ihʳ₁
      uniqIH (arg A f E) {xs₀ = (g₀ , _) , _} δ₀₁ (g₀₁ , xs₀₁) ((pr₀ , _) , ihʳ₀) ((pr₁ , _) , ihʳ₁) =
        (λ a₀ a₁ a₀₁ → uniq _ _ _ _ (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁) (pr₀ a₀ .₂) (pr₁ a₁ .₂))
        , uniqIH E (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) xs₀₁ ihʳ₀ ihʳ₁

      uniqIHs
        : (es : ConDs I D I D) {i₀ i₁ : ∣ I ∣ tt}
        → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
        → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
        → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
        → (ihʳ₀ : IHsʳ₀ es xs₀ .₁) (ihʳ₁ : IHsʳ₀ es xs₁ .₁)
        → IHs~ es P~ P~ P~ xs₀ xs₁ xs₀₁
            (IHsʳ₀ es xs₀ .₂ ihʳ₀) (IHsʳ₀ es xs₁ .₂ ihʳ₁)
      uniqIHs (E ∷ Es) {xs₀ = inl _} {inl _} xs₀₁ ihʳ₀ ihʳ₁ =
        uniqIH E tt xs₀₁ ihʳ₀ ihʳ₁
      uniqIHs (E ∷ Es) {xs₀ = inr _} {inr _} xs₀₁ ihʳ₀ ihʳ₁ =
        uniqIHs Es xs₀₁ ihʳ₀ ihʳ₁

    mutual
      rec-isRec
        : {i : ∣ I ∣ tt} (x : μ₀ ds i)
        → Σ (P i x) (isRec x)
      rec-isRec (con xs) =
        pxs xs (IHsʳ₀ ds xs .₂ (mapIHsʳ ds xs))
        , (mapIHsʳ ds xs , ~refl _)

      mapIHsʳ
        : (es : ConDs I D I D) {i : ∣ I ∣ tt}
        → (xs : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
        → IHsʳ₀ es xs .₁
      mapIHsʳ (E ∷ Es) (inl xs) = mapIHʳ E xs
      mapIHsʳ (E ∷ Es) (inr xs) = mapIHsʳ Es xs

      mapIHʳ
        : {Δ : Con ℓΔ} (e : ConD Δ I D I D) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
        → (xs : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i)
        → IHʳ₀ e xs .₁
      mapIHʳ (end t e)   xs = ↑ tt
      mapIHʳ (non A E) {δ} (a , xs) = mapIHʳ E {δ , a} xs
      mapIHʳ (arg1 f E) {δ} (x , xs) = rec-isRec x , mapIHʳ E xs
      mapIHʳ {Δ = Δ} (arg A f E) {δ} ((g , gresp) , xs) =
        ( (λ a → rec-isRec (g a))
        , (λ {a₀} {a₁} a₀₁ →
            uniq _ _ _ _ (f .resp (Δ .refl δ , a₀₁)) (gresp a₀₁)
                 (rec-isRec (g a₀) .₂) (rec-isRec (g a₁) .₂)) )
        , mapIHʳ E xs

    elim₀
      : {i : ∣ I ∣ tt}
      → (x₀ : μ₀ ds i)
      → P i x₀
    elim₀ x₀ = rec-isRec x₀ .₁

    mapIH₀
      : {Δ : Con ℓΔ} (e : ConD Δ I D I D) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
      → (xs : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i)
      → IH₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ P~ xs
    mapIH₀ e xs = IHʳ₀ e xs .₂ (mapIHʳ e xs)

    mapIHs₀
      : {i : ∣ I ∣ tt} (es : ConDs I D I D)
      → (xs : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
      → IHs₀ es (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) P~ xs
    mapIHs₀ es xs = IHsʳ₀ es xs .₂ (mapIHsʳ es xs)

    elim₀~
      : {i₀ i₁ : ∣ I ∣ tt}
      → (x₀ : μ₀ ds i₀)
      → (x₁ : μ₀ ds i₁)
      → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
      → (x₀₁ : μ~ ds x₀ x₁)
      → P~ i₀₁ x₀₁ (elim₀ x₀) (elim₀ x₁)
    elim₀~ x₀ x₁ i₀₁ x₀₁ =
      uniq x₀ (elim₀ x₀) x₁ (elim₀ x₁) i₀₁ x₀₁ (rec-isRec x₀ .₂) (rec-isRec x₁ .₂)

  module _
      {Γ : Con ℓΓ}
      (P : Ty (Γ ▷ I [ ε ] ▷ μ ds) ℓ)
      (pxs : Tm (Γ ▷ I [ ε ] ▷ Args ds ▷ IHs P ds)
                (P [ wk ▸ conᵢ ds ] [ wk ]))
      where

      module ElimAt (γ : ∣ Γ ∣) = Elim₀
        (λ i x₀ → ∣ P ∣ ((γ , i) , x₀))
        (λ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
        (λ v → P .refl* v)
        (λ p → P .sym* p)
        (λ p q → P .trans* p q)
        (λ xs₀ ih₀ → ∣ pxs ∣ (((γ , _) , xs₀) , ih₀))
        (λ i₀₁ xs₀₁ ih₀₁ → pxs .resp (((Γ .refl γ , i₀₁) , xs₀₁) , ih₀₁))

      mutual
        mapIH₀₁~
          : {Δ : Con ℓΔ}
          → (E : ConD Δ I D I D)
          → {δ₀ δ₁ : ∣ Δ ∣} {i₀ i₁ : ∣ I ∣ tt}
          → {xs₀ : Arg₀ E (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀} {xs₁ : Arg₀ E (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ i₁}
          → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
          → {γ₀ γ₁ : _} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (xs₀₁ : Arg~ₛ ds E δ₀₁ xs₀ xs₁)
          → ∣IH~∣ P E γ₀₁ δ₀₁ xs₀₁
              (ElimAt.mapIH₀ γ₀ E xs₀)
              (ElimAt.mapIH₀ γ₁ E xs₁)
        mapIH₀₁~ (end t d) δ₀₁ γ₀₁ xs₀₁ = tt
        mapIH₀₁~ (non A E) δ₀₁ γ₀₁ (a₀₁ , xs₀₁) =
          mapIH₀₁~ E (δ₀₁ , a₀₁) γ₀₁ xs₀₁
        mapIH₀₁~ (arg1 f E) {xs₀ = x₀ , _} δ₀₁ γ₀₁ (x₀₁ , xs₀₁) =
          elim₀₁~ γ₀₁ (f .resp δ₀₁) x₀₁
          , mapIH₀₁~ E (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) x₀₁) γ₀₁ xs₀₁
        mapIH₀₁~ (arg A f E) {xs₀ = (g₀ , _) , _} {(g₁ , _) , _} δ₀₁ γ₀₁ (x₀₁ , xs₀₁) =
          (λ a₀ a₁ a₀₁ → elim₀₁~ γ₀₁ (f .resp (_ , a₀₁)) (x₀₁ a₀₁))
          , mapIH₀₁~ E (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (x₀₁ a₀₁)) γ₀₁ xs₀₁

        mapIHs₀₁~
          : (Es : ConDs I D I D) {i₀ i₁ : _} {γ₀ γ₁ : _}
          → {xs₀ : Args₀ Es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀} {xs₁ : Args₀ Es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
          → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (xs₀₁ : Args~ₛ ds Es xs₀ xs₁)
          → ∣IHs~∣ P Es γ₀₁ xs₀₁
              (ElimAt.mapIHs₀ γ₀ Es xs₀)
              (ElimAt.mapIHs₀ γ₁ Es xs₁)
        mapIHs₀₁~ (E ∷ Es) {xs₀ = inl _} {inl _} γ₀₁ xs₀₁ =
          mapIH₀₁~ E tt γ₀₁ xs₀₁
        mapIHs₀₁~ (E ∷ Es) {xs₀ = inr _} {inr _} γ₀₁ xs₀₁ =
          mapIHs₀₁~ Es γ₀₁ xs₀₁

        elim-Γ
          : ∀ {γ₀ γ₁}(γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → {i : ∣ I ∣ tt}
          → (x    : μ₀ ds i)
          → P ∋ (γ₀₁ , I .refl* i) , μ~-refl* x
          ⊢ ElimAt.elim₀ γ₀ x
          ~ ElimAt.elim₀ γ₁ x
        elim-Γ {γ₀} {γ₁} γ₀₁ (con xs) =
          pxs .resp
            (((γ₀₁ , I .refl* _) , Args₀-refl* ds xs) , mapIHs₀₁~ ds γ₀₁ (Args₀-refl* ds xs))

        elim₀₁~
          : ∀ {γ₀ γ₁ i₀ i₁ x₀ x₁}
          → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
          → (x₀₁ : μ~ ds x₀ x₁)
          → P ∋ (γ₀₁ , i₀₁) , x₀₁
              ⊢ ElimAt.elim₀ γ₀ x₀
              ~ ElimAt.elim₀ γ₁ x₁
        elim₀₁~ {γ₁ = γ₁} γ₀₁ i₀₁ x₀₁ =
          P .trans*
            (elim-Γ γ₀₁ _)
            (ElimAt.elim₀~ γ₁ _ _ i₀₁ x₀₁)

      -- Induction hypotheses as a term in the displayed IH type.
      mapIHᵢ
        : (E : ConD ∅ I D I D)
        → Tm (Γ ▷ I [ ε ] ▷ Arg E [ ε ⁺ ]) (IH P ε E)
      ∣ mapIHᵢ E ∣ ((γ , i) , xs) = ElimAt.mapIH₀ γ E xs
      mapIHᵢ E .resp ((γ₀₁ , i₀₁) , xs₀₁)   = mapIH₀₁~ E tt γ₀₁ xs₀₁

      mapIHsᵢ
        : (Es : ConDs I D I D)
        → Tm (Γ ▷ I [ ε ] ▷ Args Es) (IHs P Es)
      ∣ mapIHsᵢ Es ∣ ((γ , i) , xs) = ElimAt.mapIHs₀ γ Es xs
      mapIHsᵢ Es .resp ((γ₀₁ , i₀₁) , xs₀₁)    = mapIHs₀₁~ Es γ₀₁ xs₀₁

      -- Generic eliminator for the inductive type μ ds.
      elim
        : Tm (Γ ▷ I [ ε ] ▷ μ ds) P
      ∣ elim ∣ ((γ , i) , x) = ElimAt.elim₀ γ x
      elim .resp ((γ₀₁ , i₀₁) , x₀₁) = elim₀₁~ γ₀₁ i₀₁ x₀₁

      -- Eliminator specialized to an explicit index and term.
      elimₑ
        : (i : Tm Γ (I [ ε ]))
        → (x : Tm Γ (μ ds ⟨ i ⟩))
        → Tm Γ (P [ idS ▸ i ▸ x ])
      elimₑ i x = elim [ idS ▸ i ▸ x ]t
