open import Prelude

module Model.Prop.Desc.Elim where

open import Model.Prop.CwF

open import Model.Prop.Desc.Core
open import Model.Prop.Desc.IH

private variable
  ℓI : Level

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  module Elim₀
      (P  : (i : ∣ I ∣ tt)(x₀ : μ₀ Ds i)(extx : extμ Ds x₀) → Set ℓ)
      (P~ : {i₀ i₁ : ∣ I ∣ tt}
        {x₀ : μ₀ Ds i₀} {x₁ : μ₀ Ds i₁}
        → (extx₀ : extμ Ds x₀) (extx₁ : extμ Ds x₁)
        → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
        → (x₀₁ : μ~ Ds x₀ x₁)
        → P _ x₀ extx₀ → P _ x₁ extx₁ → Prop ℓ)
      -- `P~` is an equivalence (instantiated by `refl*`/`sym*`/`trans*`); needed
      -- by the standalone uniqueness `uniq`, which composes rather than recurses
      -- on the equality.
      (~refl : ∀ {i x extx} (v : P i x extx)
             → P~ extx extx (I .refl* i) (μ~-refl* x extx) v v)
      (~sym : ∀ {i₀ i₁ x₀ x₁ extx₀ extx₁}
                {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {x₀₁ : μ~ Ds x₀ x₁}
                {v₀ : P i₀ x₀ extx₀} {v₁ : P i₁ x₁ extx₁}
            → P~ extx₀ extx₁ i₀₁ x₀₁ v₀ v₁
            → P~ extx₁ extx₀ (I .sym* i₀₁) (μ~-sym* x₀₁) v₁ v₀)
      (~trans : ∀ {i₀ i₁ i₂ x₀ x₁ x₂ extx₀ extx₁ extx₂}
                  {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₁₂ : I ∋ tt ⊢ i₁ ~ i₂}
                  {x₀₁ : μ~ Ds x₀ x₁} {x₁₂ : μ~ Ds x₁ x₂}
                  {v₀ : P i₀ x₀ extx₀} {v₁ : P i₁ x₁ extx₁} {v₂ : P i₂ x₂ extx₂}
              → P~ extx₀ extx₁ i₀₁ x₀₁ v₀ v₁ → P~ extx₁ extx₂ i₁₂ x₁₂ v₁ v₂
              → P~ extx₀ extx₂ (I .trans* i₀₁ i₁₂) (μ~-trans* x₀₁ x₁₂) v₀ v₂)
      (pxs
        : {i : ∣ I ∣ tt} (xs₀ : Args₀ Ds (μ₀ Ds) i)
        → (extxs : extArgs Ds Ds xs₀)
        → (ih₀ : IHs₀ Ds  P~ Ds (xs₀ , extxs))
        → P i (con xs₀) extxs)
      (pxs~
        : {i₀ i₁ : ∣ I ∣ tt}
          {xs₀ : Args₀ Ds (μ₀ Ds) i₀} {xs₁ : Args₀ Ds (μ₀ Ds) i₁}
          (extxs₀ : extArgs Ds Ds xs₀)
          (extxs₁ : extArgs Ds Ds xs₁)
          {ih₀ : IHs₀ Ds P~ Ds (xs₀ , extxs₀)}
          {ih₁ : IHs₀ Ds P~ Ds (xs₁ , extxs₁)}
          → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
          → (xs₀₁ : ∣μArgs~∣ Ds Ds (xs₀ , extxs₀) (xs₁ , extxs₁))
          → (ih₀₁ : IHs~ Ds P~ P~ P~ Ds (xs₀ , extxs₀) (xs₁ , extxs₁) xs₀₁ ih₀ ih₁)
          → P~ extxs₀ extxs₁
              i₀₁
              xs₀₁
              (pxs xs₀ extxs₀ ih₀)
              (pxs xs₁ extxs₁ ih₁))
    where
    mutual
      isElim
        : {i : ∣ I ∣ tt} (x : μ₀ Ds i) (extx : extμ Ds x) (v : P i x extx)
        → Set (ℓI ⊔ ℓ)
      isElim {i} (con xs) extxs v =
        Σₚ[ ihʳ ∈ IHsʳ₀ Ds xs extxs .₁ ]
          P~ extxs extxs (I .refl* i) (μ~-refl* (con xs) extxs)
             (pxs xs extxs (IHsʳ₀ Ds xs extxs .₂ ihʳ)) v

      IHsʳ₀
        : (Es : ConDs I I) {j : ∣ I ∣ tt}
        → (xs : Args₀ Es (μ₀ Ds) j) (extxs : extArgs Ds Es xs) → Σ[ X ∈ Set (ℓI ⊔ ℓ) ] (X → IHs₀ Ds P~ Es (xs , extxs))
      IHsʳ₀ (E ∷ Es) (inl xs) extxs = IHʳ₀ E xs extxs
      IHsʳ₀ (E ∷ Es) (inr xs) extxs = IHsʳ₀ Es xs extxs

      IHʳ₀
        : {Δ : Con ℓΔ} (E : ConD Δ I I) {δ : ∣ Δ ∣} {j : ∣ I ∣ tt}
        → (xs : Arg₀ E (μ₀ Ds) δ j) (extxs : extArg Ds E xs) → Σ[ X ∈ Set (ℓI ⊔ ℓ) ] (X → IH₀ Ds P~ E δ (xs , extxs))
      IHʳ₀ (end t)    xs extxs = Lift _ ⊤ , λ _ → ↑ tt
      IHʳ₀ (non A E)  (a , xs) extxs = IHʳ₀ E xs extxs
      IHʳ₀ (arg1 f E) {δ} (x , xs) (extx , extxs) =
        (Σ (P (∣ f ∣ δ) x extx) (isElim x extx) × IHʳ₀ E xs extxs .₁) , λ (pr , ihʳ) → pr .₁ , IHʳ₀ E xs extxs .₂ ihʳ
      IHʳ₀ {Δ = Δ} (arg A f E) {δ} (x , xs) (extx , x~ , extxs) =
        (Σₚ ((a : ∣ A ∣ δ) → Σ (P (∣ f ∣ (δ , a)) (x a) (extx a)) (isElim (x a) (extx a)))
           (λ pr → {a₀ a₁ : ∣ A ∣ δ} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁)
                     → P~ _ _ (f .resp (_ , a₀₁)) (x~ a₀₁) (pr a₀ .₁) (pr a₁ .₁))
        × IHʳ₀ E xs extxs .₁) , λ ((pr , pr~) , ihʳ) → ((λ a → pr a .₁) , pr~) , IHʳ₀ E xs extxs .₂ ihʳ

    mutual
      uniq
        : {i₀ i₁ : ∣ I ∣ tt}
          (x₀ : μ₀ Ds i₀) (extx₀ : extμ Ds x₀) (v₀ : P i₀ x₀ extx₀)
          (x₁ : μ₀ Ds i₁) (extx₁ : extμ Ds x₁) (v₁ : P i₁ x₁ extx₁)
          (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁)
        → isElim x₀ extx₀ v₀ → isElim x₁ extx₁ v₁ → P~ extx₀ extx₁ i₀₁ x₀₁ v₀ v₁
      uniq (con xs₀) extxs₀ v₀ (con xs₁) extxs₁ v₁ i₀₁ xs₀₁ (ihʳ₀ , p₀) (ihʳ₁ , p₁) =
        ~trans (~sym p₀)
          (~trans (pxs~ extxs₀ extxs₁ i₀₁ xs₀₁ (uniqIHs Ds extxs₀ extxs₁ xs₀₁ ihʳ₀ ihʳ₁)) p₁)

      uniqIH
        : {Δ : Con ℓΔ} (E : ConD Δ I I)
        → {δ₀ δ₁ : ∣ Δ ∣} {j₀ j₁ : ∣ I ∣ tt}
        → {xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀} {xs₁ : Arg₀ E (μ₀ Ds) δ₁ j₁}
        → (extxs₀ : extArg Ds E xs₀) (extxs₁ : extArg Ds E xs₁)
        → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
        → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ (xs₀ , extxs₀) (xs₁ , extxs₁))
        → (ihʳ₀ : IHʳ₀ E xs₀ extxs₀ .₁) (ihʳ₁ : IHʳ₀ E xs₁ extxs₁ .₁)
        → IH~ Ds P~ P~ P~ E δ₀₁ (xs₀ , extxs₀) (xs₁ , extxs₁) xs₀₁
            (IHʳ₀ E xs₀ extxs₀ .₂ ihʳ₀) (IHʳ₀ E xs₁ extxs₁ .₂ ihʳ₁)
      uniqIH (end t) extxs₀ extxs₁ δ₀₁ xs₀₁ ihʳ₀ ihʳ₁ = tt
      uniqIH (non A E) extxs₀ extxs₁ δ₀₁ (a₀₁ , xs₀₁) ihʳ₀ ihʳ₁ =
        uniqIH E extxs₀ extxs₁ (δ₀₁ , a₀₁) xs₀₁ ihʳ₀ ihʳ₁
      uniqIH (arg1 f E) {xs₀ = x₀ , _} {x₁ , _} (extx₀ , extxs₀) (extx₁ , extxs₁) δ₀₁ (x₀₁ , xs₀₁) (pr₀ , ihʳ₀) (pr₁ , ihʳ₁) =
        uniq _ _ _ _ _ _ (f .resp δ₀₁) x₀₁ (pr₀ .₂) (pr₁ .₂)
        , uniqIH E extxs₀ extxs₁ δ₀₁ xs₀₁ ihʳ₀ ihʳ₁
      uniqIH (arg A f E) {xs₀ = x₀ , _} {x₁ , _} (extx₀ , x₀~ , extxs₀) (extx₁ , x₁~ , extxs₁) δ₀₁ (x₀₁ , xs₀₁) ((pr₀ , _) , ihʳ₀) ((pr₁ , _) , ihʳ₁) =
        (λ a₀ a₁ a₀₁ → uniq _ _ _ _ _ _ (f .resp (δ₀₁ , a₀₁)) (x₀₁ a₀ a₁ a₀₁)
                            (pr₀ a₀ .₂) (pr₁ a₁ .₂))
        , uniqIH E extxs₀ extxs₁ δ₀₁ xs₀₁ ihʳ₀ ihʳ₁

      uniqIHs
        : (Es : ConDs I I) {i₀ i₁ : ∣ I ∣ tt}
        → {xs₀ : Args₀ Es (μ₀ Ds) i₀} {xs₁ : Args₀ Es (μ₀ Ds) i₁}
        → (extxs₀ : extArgs Ds Es xs₀) (extxs₁ : extArgs Ds Es xs₁)
        → (xs₀₁ : ∣μArgs~∣ Ds Es (xs₀ , extxs₀) (xs₁ , extxs₁))

        → (ihʳ₀ : IHsʳ₀ Es xs₀ extxs₀ .₁) (ihʳ₁ : IHsʳ₀ Es xs₁ extxs₁ .₁)
        → IHs~ Ds P~ P~ P~ Es (xs₀ , extxs₀) (xs₁ , extxs₁) xs₀₁
            (IHsʳ₀ Es xs₀ extxs₀ .₂ ihʳ₀) (IHsʳ₀ Es xs₁ extxs₁ .₂ ihʳ₁)
      uniqIHs (E ∷ Es) {xs₀ = inl _} {inl _} extxs₀ extxs₁ xs₀₁ ihʳ₀ ihʳ₁ =
        uniqIH E extxs₀ extxs₁ tt xs₀₁ ihʳ₀ ihʳ₁
      uniqIHs (E ∷ Es) {xs₀ = inr _} {inr _} extxs₀ extxs₁ xs₀₁ ihʳ₀ ihʳ₁ =
        uniqIHs Es extxs₀ extxs₁ xs₀₁ ihʳ₀ ihʳ₁

    mutual
      rec-isRec
        : {i : ∣ I ∣ tt} (x : μ₀ Ds i) (extx : extμ Ds x)
        → Σ (P i x extx) (isElim x extx)
      rec-isRec (con xs) extxs =
        pxs xs extxs (IHsʳ₀ Ds xs extxs .₂ (mapIHsʳ Ds xs extxs))
        , (mapIHsʳ Ds xs extxs , ~refl _)

      mapIHsʳ
        : (Es : ConDs I I) {i : ∣ I ∣ tt}
        → (xs : Args₀ Es (μ₀ Ds) i) (extxs : extArgs Ds Es xs)
        → IHsʳ₀ Es xs extxs .₁
      mapIHsʳ (E ∷ Es) (inl xs) extxs = mapIHʳ E xs extxs
      mapIHsʳ (E ∷ Es) (inr xs) extxs = mapIHsʳ Es xs extxs

      mapIHʳ
        : {Δ : Con ℓΔ} (E : ConD Δ I I) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
        → (xs : Arg₀ E (μ₀ Ds) δ i) (extxs : extArg Ds E xs)
        → IHʳ₀ E xs extxs .₁
      mapIHʳ (end t)    xs extxs = ↑ tt
      mapIHʳ (non A E)  {δ} (a , xs) extxs = mapIHʳ E {δ , a} xs extxs
      mapIHʳ (arg1 f E) {δ} (x , xs) (extx , extxs) =
        rec-isRec x extx , mapIHʳ E xs extxs
      mapIHʳ (arg A f E) {δ} (x , xs) (extx , x~ , extxs) =
        ( (λ a → rec-isRec (x a) (extx a))
        , (λ {a₀} {a₁} a₀₁ →
            uniq _ _ _ _ _ _ (f .resp (_ , a₀₁)) (x~ a₀₁)
                 (rec-isRec (x a₀) (extx a₀) .₂) (rec-isRec (x a₁) (extx a₁) .₂)) )
        , mapIHʳ E xs extxs

    elim₀
      : {i : ∣ I ∣ tt} (x₀ : μ₀ Ds i) (extx : extμ Ds x₀) → P i x₀ extx
    elim₀ x₀ extx = rec-isRec x₀ extx .₁

    mapIH₀
      : {Δ : Con ℓΔ} (E : ConD Δ I I) {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
      → (xs : Arg₀ E (μ₀ Ds) δ i) (extxs : extArg Ds E xs)
      → IH₀ Ds P~ E δ (xs , extxs)
    mapIH₀ E xs extxs = IHʳ₀ E xs extxs .₂ (mapIHʳ E xs extxs)

    mapIHs₀
      : {i : ∣ I ∣ tt} (Es : ConDs I I)
      → (xs : Args₀ Es (μ₀ Ds) i) (extxs : extArgs Ds Es xs)
      → IHs₀ Ds P~ Es (xs , extxs)
    mapIHs₀ Es xs extxs = IHsʳ₀ Es xs extxs .₂ (mapIHsʳ Es xs extxs)

  module _
      {Γ : Con ℓΓ}
      (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
      (pxs : Tm (Γ ▷ I [ ε ] ▷ μArgs Ds ▷ μIHs P Ds)
                (P [ wk ▸ conᵢ Ds ] [ wk ]))
      where

      module ElimAt (γ : ∣ Γ ∣) = Elim₀
        (λ i x₀ extx → ∣ P ∣ ((γ , i) , (x₀ , extx)))
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
        (λ v → P .refl* v)
        (λ p → P .sym* p)
        (λ p q → P .trans* p q)
        (λ xs₀ extxs ih₀ → ∣ pxs ∣ (((γ , _) , (xs₀ , extxs)) , ih₀))
        (λ extxs₀ extxs₁ i₀₁ xs₀₁ ih₀₁ → pxs .resp (((Γ .refl γ , i₀₁) , xs₀₁) , ih₀₁))

      mutual
        mapIH₀₁
          : {Δ : Con ℓΔ}
          → (E : ConD Δ I I)
          → {δ₀ δ₁ : ∣ Δ ∣} {j₀ : ∣ I ∣ tt} {j₁ : ∣ I ∣ tt}
          → {xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀} {xs₁ : Arg₀ E (μ₀ Ds) δ₁ j₁}
          → (extxs₀ : extArg Ds E xs₀)
          → (extxs₁ : extArg Ds E xs₁)
          → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
          → {γ₀ γ₁ : _} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ (xs₀ , extxs₀) (xs₁ , extxs₁))
          → ∣μIH~∣ P E γ₀₁ δ₀₁ xs₀₁
              (ElimAt.mapIH₀ γ₀ E xs₀ extxs₀)
              (ElimAt.mapIH₀ γ₁ E xs₁ extxs₁)
        mapIH₀₁ (end t) extxs₀ extxs₁ δ₀₁ γ₀₁ xs₀₁ = tt
        mapIH₀₁ (non A E) extxs₀ extxs₁ δ₀₁ γ₀₁ (a₀₁ , xs₀₁) =
          mapIH₀₁ E extxs₀ extxs₁ (δ₀₁ , a₀₁) γ₀₁ xs₀₁
        mapIH₀₁ (arg1 f E) {xs₀ = x₀ , _} {x₁ , _} (_ , extxs₀) (_ , extxs₁) δ₀₁ γ₀₁ (x₀₁ , xs₀₁) =
          elim₀₁ γ₀₁ (f .resp δ₀₁) x₀₁
          , mapIH₀₁ E extxs₀ extxs₁ δ₀₁ γ₀₁ xs₀₁
        mapIH₀₁ (arg A f E) {xs₀ = x₀ , _} {x₁ , _} (_ , _ , extxs₀) (_ , _ , extxs₁) δ₀₁ γ₀₁ (x₀₁ , xs₀₁) =
          (λ a₀ a₁ a₀₁ → elim₀₁ {x₀ = x₀ a₀} {x₁ a₁} γ₀₁ (f .resp (_ , a₀₁)) (x₀₁ _ _ a₀₁))
          , mapIH₀₁ E extxs₀ extxs₁ δ₀₁ γ₀₁ xs₀₁

        mapIHs₀₁
          : (Es : ConDs I I) {i₀ i₁ : _} {γ₀ γ₁ : _}
          → {xs₀ : Args₀ Es (μ₀ Ds) i₀} {xs₁ : Args₀ Es (μ₀ Ds) i₁}
          → (extxs₀ : extArgs Ds Es xs₀)
          → (extxs₁ : extArgs Ds Es xs₁)
          → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (xs₀₁ : ∣μArgs~∣ Ds Es (xs₀ , extxs₀) (xs₁ , extxs₁))
          → ∣μIHs~∣ P Es γ₀₁ xs₀₁
              (ElimAt.mapIHs₀ γ₀ Es xs₀ extxs₀)
              (ElimAt.mapIHs₀ γ₁ Es xs₁ extxs₁)
        mapIHs₀₁ (E ∷ Es) {xs₀ = inl _} {inl _} extxs₀ extxs₁ γ₀₁ xs₀₁ =
          mapIH₀₁ E extxs₀ extxs₁ tt γ₀₁ xs₀₁
        mapIHs₀₁ (E ∷ Es) {xs₀ = inr _} {inr _} extxs₀ extxs₁ γ₀₁ xs₀₁ =
          mapIHs₀₁ Es extxs₀ extxs₁ γ₀₁ xs₀₁

        elim₀₁
          : ∀ {γ₀ γ₁ i₀ i₁ x₀ x₁ extx₀ extx₁}
          → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
          → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
          → (x₀₁ : μ~ Ds x₀ x₁)
          → P ∋ (γ₀₁ , i₀₁) , x₀₁
              ⊢ ElimAt.elim₀ γ₀ x₀ extx₀
              ~ ElimAt.elim₀ γ₁ x₁ extx₁
        elim₀₁ {γ₁ = γ₁} {x₀ = con _} {x₁ = con _} {extx₀ = extxs₀} {extx₁ = extxs₁} γ₀₁ i₀₁ xs₀₁ =
          pxs .resp (((γ₀₁ , i₀₁) , xs₀₁) , mapIHs₀₁ Ds extxs₀ extxs₁ γ₀₁ xs₀₁)

      -- Generic eliminator for the inductive type μ Ds.
      elim
        : Tm (Γ ▷ I [ ε ] ▷ μ Ds) P
      ∣ elim ∣ ((γ , i) , (x , extx)) = ElimAt.elim₀ γ x extx
      elim .resp ((γ₀₁ , i₀₁) , x₀₁) = elim₀₁ γ₀₁ i₀₁ x₀₁

      -- Eliminator specialized to an explicit index and term.
      elimₑ
        : (i : Tm Γ (I [ ε ]))
        → (x : Tm Γ (μ Ds ⟨ i ⟩))
        → Tm Γ (P [ idS ▸ i ▸ x ])
      elimₑ i x = elim [ idS ▸ i ▸ x ]t

      -- Induction hypotheses as a term in the displayed IH type.
      mapIHᵢ
        : (E : ConD ∅ I I)
        → Tm (Γ ▷ I [ ε ] ▷ μArg E [ ε ⁺ ]) (μIH P ε E)
      ∣ mapIHᵢ E ∣ ((γ , i) , (xs , extxs)) = ElimAt.mapIH₀ γ E xs extxs
      mapIHᵢ E .resp ((γ₀₁ , i₀₁) , xs₀₁)   = mapIH₀₁ E _ _ tt γ₀₁ xs₀₁

      mapIHsᵢ
        : (Es : ConDs I I)
        → Tm (Γ ▷ I [ ε ] ▷ μArgs Es) (μIHs P Es)
      ∣ mapIHsᵢ Es ∣ ((γ , i) , (xs₀ , extxs)) = ElimAt.mapIHs₀ γ Es xs₀ extxs
      mapIHsᵢ Es .resp ((γ₀₁ , i₀₁) , xs₀₁)    = mapIHs₀₁ Es _ _ γ₀₁ xs₀₁
