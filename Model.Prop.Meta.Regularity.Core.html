open Eq hiding (refl)
private variable ℓE : Level

CoeTy : ∀ {ℓΓ ℓA} (Γ : Con ℓΓ) (A : Ty Γ ℓA) → Set (ℓΓ ⊔ ℓA)
CoeTy Γ A = (γ : ∣ Γ ∣) (a : ∣ A ∣ γ) → ∣ A ∣ γ

CoeLHS : ∀ {ℓΓ ℓA} (Γ : Con ℓΓ) (A : Ty Γ ℓA) → CoeTy Γ A
CoeLHS Γ A γ = A .coe (Γ .refl γ)

CoeRHS : ∀ {ℓΓ ℓA} (Γ : Con ℓΓ) (A : Ty Γ ℓA) → CoeTy Γ A
CoeRHS Γ A γ = λ x → x

IsReg : ∀ {ℓΓ ℓA} (Γ : Con ℓΓ) (A : Ty Γ ℓA) → Prop (ℓA ⊔ ℓΓ)
IsReg Γ A = _≡_ {A = CoeTy Γ A} (CoeLHS Γ A) (CoeRHS Γ A)

-- The propositional computation rule for internal coercion along reflexivity,
-- for every substitution and term.
IsRegInternal : ∀ {ℓΔ ℓA} (Δ : Con ℓΔ) (A : Ty Δ ℓA) → Propω
IsRegInternal Δ A =
  ∀ {ℓΓ} {Γ : Con ℓΓ} (σ : Sub Γ Δ) (t : Tm Γ (A [ σ ]))
  → coeT A σ σ (reflS σ) t ≡ t

isReg↑
  : ∀ {ℓΓ ℓA} {Γ : Con ℓΓ} {A : Ty Γ ℓA}
  → IsReg Γ A
  → CoeLHS Γ A ≡ CoeRHS Γ A
isReg↑ {Γ = Γ} {A = A} A-isReg =
  Jₚ
    (λ cast _ →
      CoeLHS Γ A ≡
      λ γ x → cast γ x)
    Eq.refl _ A-isReg

-- Regularity gives the internal computation rule after any substitution.
coeT-isReg
  : ∀ {ℓΓ ℓΔ ℓA} {Γ : Con ℓΓ} {Δ : Con ℓΔ} {A : Ty Δ ℓA}
  → IsReg Δ A
  → (σ : Sub Γ Δ) (t : Tm Γ (A [ σ ]))
  → coeT A σ σ (reflS σ) t ≡ t
coeT-isReg {Δ = Δ} {A = A} A-isReg σ t =
  tm-ext {A = A [ σ ]} (Eq.cong
    (λ cast γ → cast (∣ σ ∣ γ) (∣ t ∣ γ))
    (isReg↑ {Γ = Δ} {A = A} A-isReg))
  where
  -- The respectfulness field is proof irrelevant.
  tm-ext : ∀ {ℓΓ ℓA} {Γ : Con ℓΓ} {A : Ty Γ ℓA} {t u : Tm Γ A}
    → ∣ t ∣ ≡ ∣ u ∣ → t ≡ u
  tm-ext Eq.refl = Eq.refl

isReg→internal : IsReg Γ A → IsRegInternal Γ A
isReg→internal {A = A} A-isReg = coeT-isReg {A = A} A-isReg

-- Specialise to weakening and the generic term in Γ ▷ A, then evaluate
-- at (γ , a) to recover the original law.
internal→isReg : IsRegInternal Γ A → IsReg Γ A
internal→isReg {A = A} A-isReg =
  Eq.cong
    (λ t γ a → ∣ t ∣ (γ , a))
    (A-isReg (wk {A = A}) vz)

Sub-isReg : IsReg Δ A → IsReg Γ (A [ σ ])
Sub-isReg {Δ = Δ} {A = A} {Γ = Γ} {σ = σ} A-isReg =
  Jₚ
    (λ cast _ →
      _≡_ {A = CoeTy Γ (A [ σ ])}
        (CoeLHS Γ (A [ σ ]))
        (λ γ a →
          cast (∣ σ ∣ γ) a))
    Eq.refl _ A-isReg

LiftTy-isReg : IsReg Γ A → IsReg Γ (LiftTy ℓB A)
LiftTy-isReg {Γ = Γ} {A = A} A-isReg =
  Jₚ
    (λ cast _ →
      _≡_ {A = CoeTy Γ (LiftTy _ A)}
        (CoeLHS Γ (LiftTy _ A))
        (λ γ a →
          (↑ (cast γ (a .↓)))))
    Eq.refl _ A-isReg

Props-isReg : IsReg Γ (Props {Γ = Γ} ℓ)
Props-isReg = Eq.refl

ElP-isReg : ∀ {p} → IsReg Γ (ElP {ℓ = ℓ} p)
ElP-isReg = Eq.refl

⇑-isReg : {P : Prp ℓP Γ} → IsReg Γ (⇑ P)
⇑-isReg = Eq.refl

Empty-isReg : IsReg Γ (Empty {ℓE = ℓE})
Empty-isReg = Eq.refl

Unit-isReg : IsReg Γ (Unit {ℓ = ℓ})
Unit-isReg = Eq.refl

Π-isReg : IsReg Γ A → IsReg (Γ ▷ A) B → IsReg Γ (Π A B)
Π-isReg {Γ = Γ} {A = A} {B = B} A-isReg B-isReg =
  Jₚ
    (λ cast _ →
      (cohr : ∀ δ b →
        B ∋ (Γ ▷ A) .refl δ ⊢
        b ~ cast δ b) →
      _≡_ {A = CoeTy Γ (Π A B)}
        (CoeLHS Γ (Π A B))
        (λ γ (f , extf) →
            ((λ x → cast (γ , x) (f x))
            , λ {x₀} {x₁} q →
                B .trans*
                  (B .sym* (cohr (γ , x₀) (f x₀)))
                  (B .trans* (extf q)
                    (cohr (γ , x₁) (f x₁))))))
    (λ _ →
      Jₚ
        (λ cast _ →
          (cohr : ∀ γ a →
            A ∋ Γ .refl γ ⊢
            a ~ cast γ a) →
          _≡_ {A = CoeTy Γ (Π A B)}
            (CoeLHS Γ (Π A B))
            (λ γ (f , extf) →
                ((λ x →
                    B .coe
                      (Γ .refl γ , A .sym* (cohr γ x))
                      (f (cast γ x)))
                , λ {x₀} {x₁} q →
                    B .trans*
                      (coh' B
                        (Γ .refl γ , A .sym* (cohr γ x₀))
                        (f (cast γ x₀)))
                      (B .trans*
                        (extf
                          (A .trans*
                            (A .trans*
                              (A .sym* (cohr γ x₀)) q)
                            (cohr γ x₁)))
                        (B .coh
                          (Γ .refl γ , A .sym* (cohr γ x₁))
                          (f (cast γ x₁)))))))
        (λ _ → Eq.refl)
        _
        A-isReg
        (λ γ a → A .refl* a))
    _
    B-isReg
    (λ δ b → B .refl* b)

Sigma-isReg : IsReg Γ A → IsReg (Γ ▷ A) B → IsReg Γ (Sigma A B)
Sigma-isReg {Γ = Γ} {A = A} {B = B} A-isReg B-isReg =
  Jₚ
    (λ cast _ →
      _≡_ {A = CoeTy Γ (Sigma A B)}
        (CoeLHS Γ (Sigma A B))
        (λ γ (a , b) →
          a , cast (γ , a) b))
    (Jₚ
      (λ cast _ →
        (cohr : ∀ γ a →
          A ∋ Γ .refl γ ⊢
          a ~ cast γ a) →
        _≡_ {A = CoeTy Γ (Sigma A B)}
          (CoeLHS Γ (Sigma A B))
          (λ γ (a , b) →
              (cast γ a
              , B .coe (_ , cohr γ a) b)))
      (λ _ → Eq.refl)
      _
      A-isReg
      (λ γ a → A .refl* a))
    _
    B-isReg

ẋ-isReg : IsReg Γ A → IsReg Γ B → IsReg Γ (A ẋ B)
ẋ-isReg {Γ = Γ} {A = A} {B = B} A-isReg B-isReg =
  Jₚ²
    (λ cast _ cast' _ →
      _≡_ {A = CoeTy Γ (A ẋ B)}
        (CoeLHS Γ (A ẋ B))
        (λ γ (a , b) →
            (cast γ a
            , cast' γ b)))
    Eq.refl _ A-isReg _ B-isReg
