-- A Tarski universe generated by IR Desc.

natD : ConD {ℓI = lzero} ∅ Unit (U lzero [ wk ]) Unit (U lzero [ wk ])
natD = end ⋆ (opaqueᵁ Nat)

piD : ConD ∅ (Unit {lzero}) (U lzero [ wk ]) (Unit {lzero}) (U lzero [ wk ])
piD = arg1 ⋆ (arg (decode vz) ⋆ (end ⋆ (pi (vs vz) vz)))

TarskiD : ConDs Unit (U lzero [ wk ]) Unit (U lzero [ wk ])
TarskiD = natD ∷ piD ∷ []

UU : Ty ∅ lzero
UU = μ TarskiD ⟨ ⋆ ⟩

TT : Tm Γ (UU [ ε ]) → Ty Γ lzero
TT u = decode (T TarskiD [ idS ▸ ⋆ ▸ u ]t)

`nat : Tm Γ (UU [ ε ])
`nat = con# (# 0) ⋆ (reflT' ⋆)

`pi : (a : Tm Γ (UU [ ε ])) → Tm Γ (Π (TT a) (UU [ ε ])) → Tm Γ (UU [ ε ])
`pi a b = con# (# 1) ⋆ (dpair {A = UU [ ε ]} (Sigma (Π (TT vz) (UU [ ε ])) (⇑ Eq ⋆ ⋆)) a (dpair (⇑ Eq ⋆ ⋆) b (reflT' ⋆) ))
