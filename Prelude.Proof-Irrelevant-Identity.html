module Prelude.Proof-Irrelevant-Identity where
  
open import Prelude.Base

private variable
  A B C : Set a

infix 4 _≡_
data _≡_ {A : Set a} (x : A) : A → Prop a where
  refl : x ≡ x

{-# BUILTIN EQUALITY _≡_  #-}

Jₚ
  : {x : A} (P : (y : A) → x ≡ y → Prop ℓ)
  → P x refl → (y : A) (p : x ≡ y) → P y p
Jₚ P px y refl = px

Jₚ²
  : {x : A} {x' : B} (P : (y : A) → x ≡ y → (y' : B) → x' ≡ y' → Prop ℓ)
  → P x refl x' refl → (y : A) (p : x ≡ y) (y' : B) (p' : x' ≡ y')
  → P y p y' p'
Jₚ² {A = A} {B = B} {x = x} {x'} P Prefl = Jₚ
  (λ y eq → ∀ y' (p' : x' ≡ y') → P y eq y' p')
  (Jₚ (λ y'' q → P x refl y'' q) Prefl)

Jₚ³
  : {x : A} {x' : B} {x'' : C}
  → (P : (y : A) → x ≡ y → (y' : B) → x' ≡ y' → (y'' : C) → x'' ≡ y'' → Prop ℓ)
  → P x refl x' refl x'' refl → (y : A) (p : x ≡ y) (y' : B) (q : x' ≡ y') (y'' : C) (r : x'' ≡ y'')
  → P y p y' q y'' r
Jₚ³ {x = x} {x'} {x''} P Prefl = Jₚ
  (λ y eq → ∀ y' (q : x' ≡ y') y'' (r : x'' ≡ y'') → P y eq y' q y'' r)
  (Jₚ² (λ x' q x'' r → P x refl x' q x'' r) Prefl)

sym : {x y : A} → x ≡ y → y ≡ x
sym refl = refl

trans : {x y z : A}
  → x ≡ y → y ≡ z
  → x ≡ z
trans refl p = p

cong : (f : A → B) {x y : A} → x ≡ y
  → f x ≡ f y
cong f refl = refl

cong₂ : (f : A → B → C) {x z : A} {y w : B}
  → x ≡ z → y ≡ w
  → f x y ≡ f z w
cong₂ f refl refl = refl

-- TODO: FIXME: cleanup the types 
congₛₚ
  : ∀ {a} {A : Set a} {b} {B : A → Prop b} {c} {C : Set c}
  → (f : (x : A) → B x → C)
  → {x y : A} → (p : x ≡ y)
  → (z : B x) (w : B y)
  → f x z ≡ f y w
congₛₚ f refl _ _ = refl

transp : {x y : A} (P : A → Prop b) → x ≡ y → P x → P y
transp P refl p = p

module ≡-Reasoning where
  -- Syntax for chains of equational reasoning (from Cubical Agda)
  infix  1 begin_
  infix  3 _∎
  infixr 2 step-≡ _≡⟨⟩_

  begin_ : {x y : A} → x ≡ y → x ≡ y
  begin q = q

  step-≡ : {y z : A} (x : A) → y ≡ z → x ≡ y → x ≡ z
  step-≡ _ p q = trans q p

  syntax step-≡ x y p = x ≡⟨ p ⟩ y

  _≡⟨⟩_ : {y : A} (x : A) → x ≡ y → x ≡ y
  _ ≡⟨⟩ x≡y = x≡y

  _∎ : (x : A) → x ≡ x
  _ ∎ = refl

Σₚ-≡ : {A : Set a} {B : A → Prop b} → ∀ {a b p q} → a ≡ b → _≡_ {_} {Σₚ A B} (a , p) (b , q)
Σₚ-≡ refl = refl
