open import Prelude

module Model.Prop.IF.Nat.Elim where

open import Model.Prop.CwF
open import Model.Prop.IF.Nat.Core
open import Model.Prop.IF.Nat.Derived

module _ (P : Ty (Γ ▷ Nat) ℓ) (pz : Tm Γ (P ⟨ ze ⟩)) (ps : Tm (Γ ▷ Nat ▷ P) (P [ wk ▸ su' ] [ wk ])) where
  elim-N-at : (γ : ∣ Γ ∣) → (m : ℕ) → ∣ P ∣ (γ , m)
  elim-N-at γ = Na.elim (λ n → ∣ P ∣ (γ , n)) (∣ pz ∣ γ) (λ n P[n] → ∣ ps ∣ ((γ , n) , P[n]))

  elim-Nat~
    : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → ∀ {m₀ m₁} (m₀₁ : m₀ Na.≈ m₁)
    → P ∋ γ₀₁ , m₀₁ ⊢ elim-N-at γ₀ m₀ ~ elim-N-at γ₁ m₁
  elim-Nat~ γ₀₁ Na.zero      = pz .resp γ₀₁
  elim-Nat~ γ₀₁ (Na.suc m₀₁) = ps .resp ((γ₀₁ , m₀₁) , elim-Nat~ γ₀₁ m₀₁)

  elim-Nat : Tm (Γ ▷ Nat) P
  ∣ elim-Nat ∣   (γ , m)     = elim-N-at γ m
  elim-Nat .resp (γ₀₁ , m₀₁) = elim-Nat~ γ₀₁ m₀₁
