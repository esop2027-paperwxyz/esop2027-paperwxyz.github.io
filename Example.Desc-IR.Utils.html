private variable
  ℓI : Level

-- Utilities for positions, internal forms, and term construction.

length : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
       → ConDs I D J E → ℕ
length []       = zero
length (_ ∷ Ds) = suc (length Ds)

data _∈_ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ} (d : ConD ∅ I D J E) : ConDs I D J E → Set (lsuc (ℓI ⊔ ℓ)) where
  here  : ∀ {Ds} → d ∈ (d ∷ Ds)
  there : ∀ {e Ds} → d ∈ Ds → d ∈ (e ∷ Ds)

lookupConD : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ} (Ds : ConDs I D J E) → Fin (length Ds) → ConD ∅ I D J E
lookupConD []       ()
lookupConD (D ∷ Ds) fzero    = D
lookupConD (D ∷ Ds) (fsuc i) = lookupConD Ds i

lookupConD∈ : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ} (Ds : ConDs I D J E) → (i : Fin (length Ds)) → lookupConD Ds i ∈ Ds
lookupConD∈ []       ()
lookupConD∈ (D ∷ Ds) fzero    = here
lookupConD∈ (D ∷ Ds) (fsuc i) = there (lookupConD∈ Ds i)

lookupConD∈?
  : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ} (Ds : ConDs I D J E)
  → (i : Fin (length Ds))
  → Dec (lookupConD Ds i ∈ Ds)
lookupConD∈? Ds i = yes (lookupConD∈ Ds i)

toWitness∈
  : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ} (Ds : ConDs I D J E)
  → (i : Fin (length Ds))
  → lookupConD Ds i ∈ Ds
toWitness∈ Ds i = toWitness {a? = lookupConD∈? Ds i} tt

Argᵢ : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}(d : ConD Δ I D J E)(X : Ty (Δ ▷ I [ ε ]) ℓI)(T : Tm (Δ ▷ I [ ε ] ▷ X) (D [ ε ▸ vs vz ]))
  → Ty (Δ ▷ J [ ε ]) ℓI
Argᵢ (end t e)   X T = ⇑ Eq (t [ wk ]t) vz
Argᵢ (non A E)   X T =
 Sigma (A [ wk ]) (Argᵢ E (X [ wk ⁺ ]) (T [ wk ⁺ ⁺ ]t) [ wk∘wk ▸ vz ▸ vs vz ])
Argᵢ (arg1 f E)  X T =
 Sigma (X [ wk ▸ f [ wk ]t ])
       (Argᵢ E (X [ wk ⁺ ]) (T [ wk ⁺ ⁺ ]t)
         [ wk∘wk ▸ T [ wk∘wk ▸ f [ wk∘wk ]t ▸ vz ]t ▸ vs vz ])
Argᵢ {Δ = Δ} {I = I} {D} {J} {E} (arg A f e) X T =
 let f' : Tm (Δ ▷ J [ ε ] ▷ (Π A (X [ wk ▸ f ]) [ wk ]) ▷ A [ wk∘wk ]) (I [ ε ])
     f' = f [ wk∘wk∘wk ▸ vz ]t
     f'' : Tm (Δ ▷ J [ ε ] ▷ A [ wk ]) (I [ ε ])
     f'' = f [ wk∘wk ▸ vz ]t
 in
 Sigma (Π A (X [ wk ▸ f ]) [ wk ])
       (Argᵢ e (X [ wk ⁺ ]) (T [ wk ⁺ ⁺ ]t)
         [ wk∘wk ▸ ƛ: (A [ wk∘wk ]) (T [ wk∘wk∘wk ▸ f' ▸ eval (X [ wk∘wk ▸ f'' ]) ]t) ▸ vs vz ])
{-# INJECTIVE_FOR_INFERENCE Argᵢ #-}

Argsᵢ
  : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ} (Es : ConDs I D J E)
  → (X : Ty (∅ ▷ I) ℓI)(T : Tm (∅ ▷ I ▷ X) (D [ ε ▸ vs vz ]))
  → Ty (∅ ▷ J) ℓI
Argsᵢ []       X T = Empty
Argsᵢ (E ∷ Es) X T = Sum (Argᵢ E X T) (Argsᵢ Es X T)
{-# INJECTIVE_FOR_INFERENCE Argsᵢ #-}


module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} {Ds : ConDs I D I D} where
  ∣Argᵢ∣→Arg₀
    : {Δ : Con ℓΔ}
    → (e : ConD Δ I D I D)
    → (δ : ∣ Δ ∣) {j : ∣ I ∣ tt}
    → ∣ Argᵢ e (μ Ds) (T Ds) ∣ (δ , j)
    → Arg₀ e (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) δ j
  ∣Argᵢ∣→Arg₀ (end t e)     δ q = q
  ∣Argᵢ∣→Arg₀ (non A E)   δ (a , xs) =
    a , ∣Argᵢ∣→Arg₀ E (δ , a) xs
  ∣Argᵢ∣→Arg₀ (arg1 f E)  δ (x , xs) =
    x , ∣Argᵢ∣→Arg₀ E _ xs
  ∣Argᵢ∣→Arg₀ (arg A f E) δ ((g , gresp), xs) =
    (g , gresp) , ∣Argᵢ∣→Arg₀ E _ xs

  Argᵢ~→Arg~
    : {Δ : Con ℓΔ}
    → (e : ConD Δ I D I D)
    → {δ₀ δ₁ : ∣ Δ ∣}{i₀ i₁ : ∣ I ∣ tt}
    → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)(i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → {xs₀ : ∣ Argᵢ e (μ Ds) (T Ds) ∣ (δ₀ , i₀)}
    → {xs₁ : ∣ Argᵢ e (μ Ds) (T Ds) ∣ (δ₁ , i₁)}
    → (xs₀₁ : Argᵢ e (μ Ds) (T Ds) ∋ δ₀₁ , i₀₁ ⊢ xs₀ ~ xs₁)
    → Arg~ₛ Ds e δ₀₁ (∣Argᵢ∣→Arg₀ e δ₀ xs₀) (∣Argᵢ∣→Arg₀ e δ₁ xs₁)
  Argᵢ~→Arg~ (end t e) δ₀₁ j₀₁ xs₀₁ = tt
  Argᵢ~→Arg~ (non A E) δ₀₁ j₀₁ {a₀ , xs₀} {a₁ , xs₁} (a₀₁ , xs₀₁) =
    a₀₁ , (Argᵢ~→Arg~ E (δ₀₁ , a₀₁) j₀₁ xs₀₁)
  Argᵢ~→Arg~ (arg1 f E)  δ₀₁ j₀₁ (x₀₁ , xs₀₁) =
    x₀₁ , (Argᵢ~→Arg~ E _ j₀₁ xs₀₁)
  Argᵢ~→Arg~ (arg A f E) δ₀₁ j₀₁ (g₀₁ , xs₀₁) =
    (g₀₁ _ _) , (Argᵢ~→Arg~ E _ j₀₁ xs₀₁)

  ∣Argsᵢ∣→Args₀
    : {i : ∣ I ∣ tt} (es : ConDs I D I D)
    → (xs : ∣ Argsᵢ es (μ Ds) (T Ds) ∣ (tt , i))
    → Args₀ es (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) i
  ∣Argsᵢ∣→Args₀ (e ∷ es) =
    [ (λ xs → inl (∣Argᵢ∣→Arg₀ e tt xs)) , (λ xs → inr (∣Argsᵢ∣→Args₀ es xs)) ]′

  Argsᵢ~→Args~
    : {i₀ i₁ : ∣ I ∣ tt} (es : ConDs I D I D)
    → {xs₀ : ∣ Argsᵢ es (μ Ds) (T Ds) ∣ (tt , i₀)}
    → {xs₁ : ∣ Argsᵢ es (μ Ds) (T Ds) ∣ (tt , i₁)}
    → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (xs₀₁ : Argsᵢ es (μ Ds) (T Ds) ∋ tt , i₀₁ ⊢ xs₀ ~ xs₁)
    → Args~ₛ Ds es (∣Argsᵢ∣→Args₀ es xs₀) (∣Argsᵢ∣→Args₀ es xs₁)
  Argsᵢ~→Args~ (e ∷ es) {xs₀ = inl _} {inl _} i₀₁ xs₀₁ =
    Argᵢ~→Arg~ e tt i₀₁ (xs₀₁ .↓)
  Argsᵢ~→Args~ (e ∷ es) {xs₀ = inr _} {inr _} i₀₁ xs₀₁ =
    Argsᵢ~→Args~ es i₀₁ (xs₀₁ .↓)

  Arg₀→∣Argᵢ∣
    : {Δ : Con ℓΔ}
    → (e : ConD Δ I D I D)
    → (δ : ∣ Δ ∣) {j : ∣ I ∣ tt}
    → Arg₀ e (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) δ j
    → ∣ Argᵢ e (μ Ds) (T Ds) ∣ (δ , j)
  Arg₀→∣Argᵢ∣ (end t e)     δ q = q
  Arg₀→∣Argᵢ∣ (non A E)   δ (a , xs) =
    a , Arg₀→∣Argᵢ∣ E (δ , a) xs
  Arg₀→∣Argᵢ∣ (arg1 f E)  δ (x , xs) =
    x , Arg₀→∣Argᵢ∣ E _ xs
  Arg₀→∣Argᵢ∣ (arg A f E) δ ((g , gresp), xs) =
    (g , gresp) , Arg₀→∣Argᵢ∣ E _ xs

  Args₀→∣Argsᵢ∣
    : {i : ∣ I ∣ tt} (es : ConDs I D I D)
    → (xs : Args₀ es (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) i)
    → ∣ Argsᵢ es (μ Ds) (T Ds) ∣ (tt , i)
  Args₀→∣Argsᵢ∣ (e ∷ es) =
    [ (λ xs → inl (Arg₀→∣Argᵢ∣ e tt xs)) , (λ xs → inr (Args₀→∣Argsᵢ∣ es xs)) ]′

  Arg~→Argᵢ~
    : {Δ : Con ℓΔ}
    → (e : ConD Δ I D I D)
    → {δ₀ δ₁ : ∣ Δ ∣}{i₀ i₁ : ∣ I ∣ tt}
    → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)(i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → {xs₀ : Arg₀ e (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) δ₀ i₀}
    → {xs₁ : Arg₀ e (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) δ₁ i₁}
    → (xs₀₁ : Arg~ₛ Ds e δ₀₁ xs₀ xs₁)
    → Argᵢ e (μ Ds) (T Ds) ∋ δ₀₁ , i₀₁ ⊢ Arg₀→∣Argᵢ∣ e δ₀ xs₀ ~ Arg₀→∣Argᵢ∣ e δ₁ xs₁
  Arg~→Argᵢ~ (end t e) δ₀₁ j₀₁ xs₀₁ = tt
  Arg~→Argᵢ~ (non A E) δ₀₁ j₀₁ {a₀ , xs₀} {a₁ , xs₁} (a₀₁ , xs₀₁) =
    a₀₁ , (Arg~→Argᵢ~ E (δ₀₁ , a₀₁) j₀₁ xs₀₁)
  Arg~→Argᵢ~ (arg1 f E)  δ₀₁ j₀₁ (x₀₁ , xs₀₁) =
    x₀₁ , (Arg~→Argᵢ~ E _ j₀₁ xs₀₁)
  Arg~→Argᵢ~ (arg A f E) δ₀₁ j₀₁ (g₀₁ , xs₀₁) =
    (λ a₀ a₁ → g₀₁ {a₀} {a₁}) , (Arg~→Argᵢ~ E _ j₀₁ xs₀₁)

  Args~→Argsᵢ~
    : (es : ConDs I D I D)
    → {i₀ i₁ : ∣ I ∣ tt}
    → {xs₀ : Args₀ es (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) i₀}
    → {xs₁ : Args₀ es (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) i₁}
    → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (xs₀₁ : Args~ₛ Ds es xs₀ xs₁)
    → Argsᵢ es (μ Ds) (T Ds) ∋ tt , i₀₁ ⊢ Args₀→∣Argsᵢ∣ es xs₀ ~ Args₀→∣Argsᵢ∣ es xs₁
  Args~→Argsᵢ~ (e ∷ es) {xs₀ = inl _} {inl _} i₀₁ xs₀₁ =
    ↑ (Arg~→Argᵢ~ e tt i₀₁ xs₀₁)
  Args~→Argsᵢ~ (e ∷ es) {xs₀ = inr _} {inr _} i₀₁ xs₀₁ =
    ↑ (Args~→Argsᵢ~ es i₀₁ xs₀₁)

  Argᵢ→Arg
    : {Δ : Con ℓΔ} (E : ConD Δ I D I D)
    → Tm (Δ ▷ I [ ε ] ▷ Argᵢ E (μ Ds) (T Ds)) (Arg {ds = Ds} E [ wk ])
  ∣ Argᵢ→Arg E ∣ ((δ , i) , xs)         = ∣Argᵢ∣→Arg₀ E δ xs
  Argᵢ→Arg E .resp ((δ₀₁ , i₀₁) , xs₀₁) = Argᵢ~→Arg~ E δ₀₁ i₀₁ xs₀₁

  Arg→Argᵢ
    : {Δ : Con ℓΔ} (E : ConD Δ I D I D)
    → Tm (Δ ▷ I [ ε ] ▷ Arg {ds = Ds} E) (Argᵢ E (μ Ds) (T Ds) [ wk ])
  ∣ Arg→Argᵢ E ∣ ((δ , i) , xs)         = Arg₀→∣Argᵢ∣ E δ xs
  Arg→Argᵢ E .resp ((δ₀₁ , i₀₁) , xs₀₁) = Arg~→Argᵢ~ E δ₀₁ i₀₁ xs₀₁


  Argsᵢ→Args
    : (Es : ConDs I D I D)
    → Tm (Γ ▷ I [ ε ] ▷ Argsᵢ Es (μ Ds) (T Ds) [ ε ⁺ ]) (Args {ds = Ds} Es [ wk ])
  ∣ Argsᵢ→Args Es ∣ ((γ , i) , xs₀)        = ∣Argsᵢ∣→Args₀ Es xs₀
  Argsᵢ→Args Es .resp ((γ₀₁ , i₀₁) , xs₀₁) = Argsᵢ~→Args~ Es i₀₁ xs₀₁

  Args→Argsᵢ
    : (Es : ConDs I D I D)
    → Tm (Γ ▷ I [ ε ] ▷ Args {ds = Ds} Es) (Argsᵢ Es (μ Ds) (T Ds) [ (ε ⁺) ∘ wk ])
  ∣ Args→Argsᵢ Es ∣ ((γ , i) , xs₀) = Args₀→∣Argsᵢ∣ Es xs₀
  Args→Argsᵢ Es .resp {γ₀ = _ , xs₀} {_ , xs₁} ((γ₀₁ , i₀₁) , xs₀₁) =
    Args~→Argsᵢ~ Es i₀₁ xs₀₁

  σArgᵢ→Arg
    : {Δ : Con ℓΔ}
    → (τ : Sub Θ Δ) (E : ConD Δ I D I D)
    → Sub (Θ ▷ I [ ε ] ▷ Argᵢ E (μ Ds) (T Ds) [ τ ↑t I [ ε ] ])
          (Θ ▷ I [ ε ] ▷ Arg {ds = Ds} E [ τ ↑t I [ ε ] ])
  σArgᵢ→Arg τ E = wk ▸ Argᵢ→Arg E [ τ ⁺ ⁺ ]t

  σArg→Argᵢ
    : {Δ : Con ℓΔ}
    → (τ : Sub Θ Δ) (E : ConD Δ I D I D)
    → Sub (Θ ▷ I [ ε ] ▷ Arg {ds = Ds} E [ τ ↑t I [ ε ] ])
          (Θ ▷ I [ ε ] ▷ Argᵢ E (μ Ds) (T Ds) [ τ ↑t I [ ε ] ])
  σArg→Argᵢ τ E = wk ▸ Arg→Argᵢ E [ τ ⁺ ⁺ ]t

  σArgsᵢ→Args
    : (Es : ConDs I D I D)
    → Sub (Θ ▷ I [ ε ] ▷ Argsᵢ Es (μ Ds) (T Ds) [ ε ⁺ ]) (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Es)
  σArgsᵢ→Args Es = wk ▸ Argsᵢ→Args Es [ ε ⁺ ⁺ ]t

  σArgs→Argsᵢ
    : (Es : ConDs I D I D)
    → Sub (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Es) (Θ ▷ I [ ε ] ▷ Argsᵢ Es (μ Ds) (T Ds) [ ε ⁺ ])
  σArgs→Argsᵢ Es = wk ▸ Args→Argsᵢ Es [ ε ⁺ ⁺ ]t

  private
    inArgsᵢ'
      : (d : ConD ∅ I D I D)
      → (Es : ConDs I D I D)
      → d ∈ Es
      → (i : Tm Γ (I [ ε ]))
      → Tm Γ (Argᵢ d (μ Ds) (T Ds) [ ε ▸ i ])
      → Tm Γ (Argsᵢ Es (μ Ds) (T Ds) [ ε ▸ i ])
    inArgsᵢ' d (d ∷ Es) here i xs =
      Inl (Argᵢ d (μ Ds) (T Ds) [ ε ▸ i ]) (Argsᵢ Es (μ Ds) (T Ds) [ ε ▸ i ]) xs
    inArgsᵢ' d (e ∷ Es) (there p) i xs =
      Inr (Argᵢ e (μ Ds) (T Ds) [ ε ▸ i ]) (Argsᵢ Es (μ Ds) (T Ds) [ ε ▸ i ]) (inArgsᵢ' d Es p i xs)

  inArgsᵢ
    : {Γ : Con ℓΓ} {d : ConD ∅ I D I D}
    → d ∈ Ds
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ d (μ Ds) (T Ds) [ ε ▸ i ])
    → Tm Γ (Argsᵢ Ds (μ Ds) (T Ds) [ ε ▸ i ])
  inArgsᵢ {d = d} = inArgsᵢ' d Ds

  injSum
    : {d : ConD ∅ I D I D} (es : ConDs I D I D) → d ∈ es
    → ∀ {j} → ∣ Argᵢ d (μ Ds) (T Ds) ∣ (tt , j) → ∣ Argsᵢ es (μ Ds) (T Ds) ∣ (tt , j)
  injSum (d ∷ es) here      v = inl v
  injSum (e ∷ es) (there p) v = inr (injSum es p v)

  ∣inArgsᵢ∣
    : {d : ConD ∅ I D I D} (es : ConDs I D I D) (mem : d ∈ es)
    → (i : Tm Γ (I [ ε ])) (xs : Tm Γ (Argᵢ d (μ Ds) (T Ds) [ ε ▸ i ])) (γ : ∣ Γ ∣)
    → ∣ inArgsᵢ' d es mem i xs ∣ γ ≡ injSum es mem (∣ xs ∣ γ)
  ∣inArgsᵢ∣ (d ∷ es) here      i xs γ = Eq.refl
  ∣inArgsᵢ∣ (e ∷ es) (there p) i xs γ = Eq.cong inr (∣inArgsᵢ∣ es p i xs γ)

  inArgs#
    : {Γ : Con ℓΓ}
    → (k : Fin (length Ds))
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ (lookupConD Ds k) (μ Ds) (T Ds) [ ε ▸ i ])
    → Tm Γ (Argsᵢ Ds (μ Ds) (T Ds) [ ε ▸ i ])
  inArgs# k i xs = inArgsᵢ (toWitness∈ Ds k) i xs

  con∈
    : {Γ : Con ℓΓ} {d : ConD ∅ I D I D}
    → d ∈ Ds
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ d (μ Ds) (T Ds) [ ε ▸ i ])
    → Tm Γ (μ Ds [ ε ▸ i ])
  con∈ {Γ = Γ} p i xs =
    conᵢ {Γ = Γ} Ds
      ⟨ i
      && (Argsᵢ→Args Ds [ ε ⁺ ⁺ ]t ⟨ i && inArgsᵢ p i xs ⟩t)
      ⟩t

  con#
    : {Γ : Con ℓΓ}
    → (k : Fin (length Ds))
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ (lookupConD Ds k) (μ Ds) (T Ds) [ ε ▸ i ])
    → Tm Γ (μ Ds [ ε ▸ i ])
  con# {Γ = Γ} k i xs =
    conᵢ {Γ = Γ} Ds
      ⟨ i
      && (Argsᵢ→Args Ds [ ε ⁺ ⁺ ]t ⟨ i && inArgs# k i xs ⟩t)
      ⟩t

  module _
    {Γ : Con ℓΓ}
    (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
    where

    Pcon : Ty (Γ ▷ I [ ε ] ▷ Args Ds) ℓ
    Pcon = P [ wk ▸ conᵢ Ds ]

    data ArgIHsʳ
      : (Es : ConDs I D I D)
      → (σinj
          : {Θ : Con ℓΓ}
          → Sub (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Es) (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Ds))
      → Set (lsuc (ℓI ⊔ ℓΓ ⊔ ℓ)) where
      -- There are no constructors left to handle.
      []
        : {σcon : {Θ : Con ℓΓ}
            → Sub (Θ ▷ I [ ε ] ▷ Args {ds = Ds} []) (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Ds)}
        → ArgIHsʳ [] σcon
      _∷_
        : {E : ConD ∅ I D I D} {Es : ConDs I D I D}
        → {σcon : {Θ : Con ℓΓ}
            → Sub (Θ ▷ I [ ε ] ▷ Args (E ∷ Es)) (Θ ▷ I [ ε ] ▷ Args Ds)}
        → Tm (Γ ▷ I [ ε ] ▷ Arg E [ ε ⁺ ] ▷ IH P ε E)
             (Pcon [ σcon ∘ (wk ▸ inlArgsᵢ E Es) ∘ wk ])
        → ArgIHsʳ Es (σcon ∘ (wk ▸ inrArgsᵢ E Es))
        → ArgIHsʳ (E ∷ Es) σcon

    liftIHs₀
      : {Es : ConDs I D I D} {η : {Θ : Con ℓΓ} → Sub (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Es) (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Ds)}
      → ArgIHsʳ Es η
      → ∀ γ {i}
      → (xs : Args₀ Es (μ₀ Ds) (μ~ Ds) (T₀ Ds) (λ {x₀ = x₀} → Tresp₀ Ds {x₀ = x₀}) i)
      → (ihs : ∣IHs∣ P Es γ xs)
      → ∣ Pcon [ η ] ∣ ((γ , i) , xs)
    liftIHs₀ [] γ xs = ⊥-elim (xs .↓)
    liftIHs₀ (p ∷ ps) γ {i} =
      [ (λ xs ihs → ∣ p ∣ (((γ , i) , xs) , ihs))
      , liftIHs₀ ps γ ]

    liftIHsʳ
      : {Es : ConDs I D I D}
      → (η : {Θ : Con ℓΓ} → Sub (Θ ▷ I [ ε ] ▷ Args {ds = Ds} Es) (Θ ▷ I [ ε ] ▷ Args Ds))
      → (ps : ArgIHsʳ Es η)
      → Tm (Γ ▷ I [ ε ] ▷ Args Es ▷ IHs P Es)
           (P [ wk ▸ conᵢ Ds ] [ η ] [ wk ])
    ∣ liftIHsʳ _ ps ∣ (((γ , i) , xs) , ihs) =
      liftIHs₀ ps γ xs ihs
    liftIHsʳ _ (p ∷ ps) .resp {γ₀ = (_ , (inl _)) , _} {γ₁ = (_ , (inl _)) , _} (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁) =
      p .resp (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁)
    liftIHsʳ _ (p ∷ ps) .resp {γ₀ = (_ , (inr _)) , _} {γ₁ = (_ , (inr _)) , _} (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁) =
      liftIHsʳ _ ps .resp (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁)

    ArgIHs : Set (lsuc (ℓI ⊔ ℓΓ ⊔ ℓ))
    ArgIHs = ArgIHsʳ Ds idS

    module _ (ps : ArgIHs) where
      liftIHsᵢ
        : Tm (Γ ▷ I [ ε ] ▷ Args Ds ▷ IHs P Ds) (P [ wk ▸ conᵢ Ds ] [ wk ])
      liftIHsᵢ = liftIHsʳ idS ps

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} {Ds : ConDs I D I D} (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ) (ps : ArgIHs P) where
  elim'
    : Tm (Γ ▷ I [ ε ] ▷ μ Ds) P
  elim' = elim P (liftIHsᵢ P ps)

  elimₑ'
    : (i : Tm Γ (I [ ε ])) (xs : Tm Γ (μ Ds ⟨ i ⟩))
    → Tm Γ (P [ idS ▸ i ▸ xs ])
  elimₑ' i xs = elimₑ P (liftIHsᵢ P ps) i xs
