module _ {Γ : Con ℓΓ} (A : Ty Γ ℓA) (B : Ty (Γ ▷ A) ℓB) where
  fstₑ=₁
    : (xs : Tm Γ (Sigma A B))
    → (γ : ∣ Γ ∣)
    → ∣ fstₑ A B xs ∣ γ ≡ ∣ xs ∣ γ .₁
  fstₑ=₁ xs γ = Eq.refl

  sndₑ=₂
    : (xs : Tm Γ (Sigma A B))
    → (γ : ∣ Γ ∣)
    → ∣ sndₑ A B xs ∣ γ ≡ ∣ xs ∣ γ .₂
  sndₑ=₂ xs γ = Eq.refl

Sigma[] : {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} {σ : Sub Δ Γ}
    → Sigma A B [ σ ] ≡ Sigma (A [ σ ]) (B [ σ ⁺ ])
Sigma[] = Eq.refl

dpair[] : {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB} {σ : Sub Δ Γ}
       → (a : Tm Γ A) (b : Tm Γ (B ⟨ a ⟩))
       → dpair B a b [ σ ]t ≡ dpair (B [ σ ⁺ ]) (a [ σ ]t) (b [ σ ]t)
dpair[] a b = Eq.refl

module _ {Γ : Con ℓΓ} {A : Ty Γ ℓA} {B : Ty (Γ ▷ A) ℓB}
         {P : Ty (Γ ▷ Sigma A B) ℓC}
         {t : Tm (Γ ▷ A ▷ B) (P [ wk∘wk ▸ dpair (B [ wk∘wk∘wk ▸ vz ]) (vs vz) vz ])}
         where

  elim-dpair
    : (a : Tm Γ A) (b : Tm Γ (B ⟨ a ⟩))
    → elim-Sigma P t ⟨ dpair B a b ⟩t ≡ t ⟨ a && b ⟩t
  elim-dpair a b = Eq.refl

  elim-Sigma[]
    : {σ : Sub Δ Γ} {x : Tm Γ (Sigma A B)}
    → elim-Sigma P t ⟨ x ⟩t [ σ ]t
    ≡ elim-Sigma (P [ σ ⁺ ]) (t [ σ ⁺ ⁺ ]t) ⟨ x [ σ ]t ⟩t
  elim-Sigma[] = Eq.refl
