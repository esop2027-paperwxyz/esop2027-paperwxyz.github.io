private variable
  ℓI : Level

-- Arg and Args live in the empty context, so only μ needs an explicit rule.

-- Substitution rule for μ.
μ[] : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ) {I : Ty ∅ ℓI} (Ds : ConDs I I)
    → (μ Ds {Γ = Γ}) [ σ ⁺ ] ≡ μ Ds {Γ = Δ}
μ[] σ Ds = Eq.refl

-- Instantiated substitution rule for μ.
μ[]' : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ) {I : Ty ∅ ℓI} (Ds : ConDs I I)
    → (i : Tm Γ (I [ ε ]))
    → μ Ds ⟨ i ⟩ [ σ ] ≡ μ Ds ⟨ i [ σ ]t ⟩
μ[]' σ Ds i = Eq.refl

-- Substitution rule for the generic constructor.
conᵢ[]
  : {Δ : Con ℓΔ} {Γ : Con ℓΓ} {I : Ty ∅ ℓI} (Ds : ConDs I I)
  → (σ : Sub Δ Γ)
  → (conᵢ {Γ = Γ} Ds) [ σ ⁺ ⁺ ]t ≡ conᵢ {Γ = Δ} Ds
conᵢ[] Ds σ = Eq.refl

-- Substitution rule for the pointwise constructor.
conₑ[]
  : {Δ : Con ℓΔ} {Γ : Con ℓΓ} {σ : Sub Δ Γ}
  → {I : Ty ∅ ℓI} (Ds : ConDs I I)
  → {i : Tm Γ (I [ ε ])}
  → {t : Tm Γ (μArgs Ds ⟨ i ⟩)}
  → conₑ Ds i t [ σ ]t ≡ conₑ Ds (i [ σ ]t) (t [ σ ]t)
conₑ[] Ds = Eq.refl

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I}
  {Γ : Con ℓΓ}
  (C : Ty (Γ ▷ I [ ε ]) ℓ)
  (alg : Tm (Γ ▷ I [ ε ] ▷ A.Args C Ds)
            (C [ wk ]))
  where

  -- Computation rule for recursion on a constructor.
  rec-con
    : {i : Tm Γ (I [ ε ])}
    → {xs : Tm Γ (μArgs Ds ⟨ i ⟩)}
    → recₑ C alg i (conₑ Ds i xs)
    ≡ alg [ idS ▸ i ▸ mapArgsᵢ C alg Ds [ idS ▸ i ▸ xs ]t ]t
  rec-con = Eq.refl

  rec[]
    : {σ : Sub Δ Γ}
    → rec C alg [ σ ⁺ ⁺ ]t ≡ rec (C [ σ ⁺ ]) (alg [ σ ⁺ ⁺ ]t)
  rec[] = Eq.refl

-- Substitution rule for IH in Elim.
IH[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    {I : Ty ∅ ℓI} {Ds : ConDs I I} (E : ConD Δ (I [ ε ]) (I [ ε ]))
  → (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
  → {Θ : Con ℓ} (τ : Sub Θ Γ)
  → μIH P             σ       E [ τ ⁺ ⁺ ]
  ≡ μIH (P [ τ ⁺ ⁺ ]) (σ ∘ τ) E
IH[] σ E P τ = Eq.refl

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I}
  {Γ : Con ℓΓ}
  (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
  (pxs : Tm (Γ ▷ I [ ε ] ▷ μArgs Ds ▷ μIHs P Ds)
            (P [ wk ▸ conᵢ Ds ] [ wk ]))
  where

  -- Computation rule for elimination on a constructor.
  elim-con
    : {i : Tm Γ (I [ ε ])}
    → {xs : Tm Γ (μArgs Ds ⟨ i ⟩)}
    → elimₑ P pxs i (conₑ Ds i xs)
    ≡ pxs [ idS ▸ i ▸ xs ▸ mapIHsᵢ P pxs Ds [ idS ▸ i ▸ xs ]t ]t
  elim-con = Eq.refl

  elim-conᵢ
    : elim P pxs [ wk ▸ conᵢ Ds ]t
    ≡ pxs [ idS ▸ mapIHsᵢ P pxs Ds ]t
  elim-conᵢ = Eq.refl

  elim[]
    : {Δ : Con ℓΔ} {σ : Sub Δ Γ}
    → elim P pxs [ σ ⁺ ⁺ ]t
    ≡ elim (P [ σ ⁺ ⁺ ]) (pxs [ σ ⁺ ⁺ ⁺ ]t)
  elim[] = Eq.refl

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I}
  {Γ : Con ℓΓ}
  (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
  (E : ConD ∅ I I) (Es : ConDs I I)
  where

  IHs-inlArgs
    : μIHs {Ds = Ds} P (E ∷ Es) [ wk ▸ inlArgsᵢ E Es ]
    ≡ μIH {Ds = Ds} P ε E
  IHs-inlArgs = Eq.refl

  IHs-inrArgs
    : μIHs {Ds = Ds} P (E ∷ Es) [ wk ▸ inrArgsᵢ E Es ]
    ≡ μIHs {Ds = Ds} P Es
  IHs-inrArgs = Eq.refl
