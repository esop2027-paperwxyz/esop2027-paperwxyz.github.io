{-# OPTIONS --rewriting #-}
module Prelude.StrongJ where
  
open import Prelude.Base
open import Prelude.Proof-Irrelevant-Identity

private variable
  A B C : Set a

{-# BUILTIN REWRITE _≡_ #-}

postulate
  J : {x : A} (P : (y : A) → x ≡ y → Set ℓ)
    → P x refl → (y : A) (p : x ≡ y) → P y p

  Jrefl
    : {x : A} (P : (y : A) → x ≡ y → Set ℓ)
    → (Prefl : P x refl)
    → J P Prefl x refl ≡ Prefl

{-# REWRITE Jrefl #-}

transp' : {x y : A} (P : A → Set b) → x ≡ y → P x → P y
transp' {x = x} {y} P eq =
  J (λ y eq → P x → P y) (λ x → x) y eq

J²
  : {x : A} {x' : B} (P : (y : A) → x ≡ y → (y' : B) → x' ≡ y' → Set ℓ)
  → P x refl x' refl → (y : A) (p : x ≡ y) (y' : B) (p' : x' ≡ y')
  → P y p y' p'
J² {A = A} {B = B} {x = x} {x'} P Prefl = J
  (λ y eq → ∀ y' (p' : x' ≡ y') → P y eq y' p')
  (J (λ y'' q → P x refl y'' q) Prefl)

J³
  : {x : A} {x' : B} {x'' : C}
  → (P : (y : A) → x ≡ y → (y' : B) → x' ≡ y' → (y'' : C) → x'' ≡ y'' → Set ℓ)
  → P x refl x' refl x'' refl → (y : A) (p : x ≡ y) (y' : B) (q : x' ≡ y') (y'' : C) (r : x'' ≡ y'')
  → P y p y' q y'' r
J³ {x = x} {x'} {x''} P Prefl = J
  (λ y eq → ∀ y' (q : x' ≡ y') y'' (r : x'' ≡ y'') → P y eq y' q y'' r)
  (J² (λ x' q x'' r → P x refl x' q x'' r) Prefl)

cong₂d : {B : A → Set b}(f : (x : A) → B x → C) {x z : A} {y : B x}{w : B z}
  → (p : x ≡ z) → transp' B p y ≡ w
  → f x y ≡ f z w
cong₂d f refl refl = refl
