open import Prelude

module Model.Prop.IF.Nat.Core where

-- natural number
module Na where
  open Prelude public
    using (ℕ; zero; suc)

  private variable
    m n o k : ℕ

  -- large elimination into Prop
  data _≈_ : ℕ → ℕ → Prop where
    zero : zero ≈ zero
    suc  : m ≈ n → suc m ≈ suc n

  elim : (P : ℕ → Set ℓ) → P zero → (∀ n → P n → P (suc n)) → (m : ℕ) → P m
  elim P P[z] P∘s zero    = P[z]
  elim P P[z] P∘s (suc m) = P∘s m (elim P P[z] P∘s m)

  elimp : (P : ℕ → Prop ℓ) → P zero → (∀ n → P n → P (suc n)) → (m : ℕ) → P m
  elimp P P[z] P∘s zero    = P[z]
  elimp P P[z] P∘s (suc m) = P∘s m (elimp P P[z] P∘s m)

  refl : ∀ {n} → n ≈ n
  refl {zero}  = zero
  refl {suc n} = suc refl

  sym : ∀ {m n} → m ≈ n → n ≈ m
  sym zero      = zero
  sym (suc m≈n) = suc (sym m≈n)

  trans : ∀ {m n o} → m ≈ n → n ≈ o → m ≈ o
  trans zero      zero      = zero
  trans (suc m≈n) (suc n≈o) = suc (trans m≈n n≈o)

open import Model.Prop.CwF
open Na using (zero; suc)

Nat : Ty Γ lzero
∣ Nat ∣ γ = Na.ℕ
Nat .heq γ₀₁ = Na._≈_
Nat .refl* _ = Na.refl
Nat .sym* = Na.sym
Nat .trans* = Na.trans
Nat .coe γ₀₁ m = m
Nat .coh γ₀₁ m = Na.refl

ze : Tm Γ Nat
∣ ze ∣ _   = zero
ze .resp _ = zero

su : Tm Γ Nat → Tm Γ Nat
∣ su n ∣ γ   = suc (∣ n ∣ γ) 
su n .resp p = suc (n .resp p)
