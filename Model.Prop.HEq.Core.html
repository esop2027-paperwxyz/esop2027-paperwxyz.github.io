open import Prelude

module Model.Prop.HEq.Core where

open import Model.Prop.CwF

EqS : (Δ : Con ℓ) (σ₀ σ₁ : Sub Γ Δ)
  → Prp ℓ Γ 
∣ EqS Δ σ₀ σ₁ ∣ γ =
  Δ .eq (∣ σ₀ ∣ γ) (∣ σ₁ ∣ γ)
EqS Δ σ₀ σ₁ .coe γ₀₁ σ₀₁ =
  Δ .trans (Δ .trans
    (Δ .sym (σ₀ .resp γ₀₁)) σ₀₁)
    (σ₁ .resp γ₀₁)
-- {-# INJECTIVE_FOR_INFERENCE EqS #-}

reflS
  : ∀ σ
  → Tm Γ (⇑ EqS Δ σ σ)
∣ reflS {Δ = Δ} σ ∣ γ =
  ↑ (Δ .refl (∣ σ ∣ γ))
reflS _ .resp _ = tt
{-# INJECTIVE_FOR_INFERENCE reflS #-}

symS
  : ∀ {Δ : Con ℓ} σ₀ σ₁ 
  → Tm Γ (⇑ EqS Δ σ₀ σ₁)
  → Tm Γ (⇑ EqS Δ σ₁ σ₀)
∣ symS {Δ = Δ} σ₀ σ₁ t ∣ γ =
  ↑ (Δ .sym (∣ t ∣ γ .↓))
symS t _ _ .resp _ = tt

transS
  : ∀ (σ₀ σ₁ σ₂ : Sub Γ Δ)
  → Tm Γ (⇑ EqS Δ σ₀ σ₁)
  → Tm Γ (⇑ EqS Δ σ₁ σ₂)
  → Tm Γ (⇑ EqS Δ σ₀ σ₂)
∣ transS {Δ = Δ} _ _ _ σ₀₁ σ₁₂ ∣ γ =
  ↑ (Δ .trans (σ₀₁ .∣_∣ γ .↓) (∣ σ₁₂ ∣ γ .↓))
transS _ _ _ σ₀σ₁ σ₁₂ .resp _ = tt

EqT
  : ∀ (A : Ty Δ ℓ) σ₀ σ₁ (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
  → (t₀ : Tm Γ (A [ σ₀ ])) (t₁ : Tm Γ (A [ σ₁ ]))
  → Prp ℓ Γ
∣ EqT A _ _ σ₀₁ t₀ t₁ ∣ γ = A .heq (∣ σ₀₁ ∣ γ .↓) (∣ t₀ ∣ γ) (∣ t₁ ∣ γ)
EqT A _ _ σ₀₁ t₀ t₁ .coe γ₀₁ x₁ =
  A .trans* (A .trans* (A .sym* (t₀ .resp γ₀₁)) x₁) (t₁ .resp γ₀₁)
{-# INJECTIVE_FOR_INFERENCE EqT #-}

reflT
  : (A : Ty Γ ℓ) (t : Tm Γ (A [ σ ]))
  → Tm Γ (⇑ EqT A σ σ (reflS σ) t t)
∣ reflT A t ∣ γ = ↑
  (A .refl* (∣ t ∣ γ))
reflT _ _ .resp _ = tt
{-# INJECTIVE_FOR_INFERENCE reflT #-}

symT
  : ∀ {Γ : Con ℓ}{Δ : Con ℓ'}(A : Ty Δ ℓA)
    {σ₀ t₀ σ₁ t₁} {σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁)}
  → Tm Γ (⇑ EqT A σ₀ σ₁ σ₀₁        t₀ t₁)
  → Tm Γ (⇑ EqT A σ₁ σ₀ (symS σ₀ σ₁ σ₀₁) t₁ t₀)
∣ symT {Γ = Γ} {Δ} A {σ₀} {t₀} {σ₁} {t₁} t₀₁ ∣ γ = 
  ↑ (A .sym* (∣ t₀₁ ∣ γ .↓))
symT A t₀ .resp _ = tt

transT
  : ∀ {Γ : Con ℓ}{Δ : Con ℓ'}(A : Ty Δ ℓA)
    {σ₀ t₀ σ₁ t₁ σ₂ t₂}
    {σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁)}
    {σ₁₂ : Tm Γ (⇑ EqS Δ σ₁ σ₂)} 
  → Tm Γ (⇑ EqT A σ₀ σ₁ σ₀₁ t₀ t₁)
  → Tm Γ (⇑ EqT A σ₁ σ₂ σ₁₂ t₁ t₂)
  → Tm Γ (⇑ EqT A σ₀ σ₂ (transS σ₀ σ₁ σ₂ σ₀₁ σ₁₂) t₀ t₂)
∣ transT {Γ = Γ} {Δ} A {σ₀} {t₀} {σ₁} {t₁} {σ₂} {t₂} {σ₀₁} {σ₁₂} t₀₁ t₁₂ ∣ γ =
  ↑ (A .trans* (∣ t₀₁ ∣ γ .↓) (∣ t₁₂ ∣ γ .↓))
transT A t₀₁ t₁₂ .resp _ = tt

-- Congruence
congS
  : ∀ τ σ₀ σ₁ (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
  → Tm Γ (⇑ EqS Θ (τ ∘ σ₀) (τ ∘ σ₁))
∣ congS τ _ _ σ₀₁ ∣ γ .↓ = τ .resp (∣ σ₀₁ ∣ γ .↓)
congS τ _ _ σ₀₁ .resp _ = tt

congT
  : ∀ (t : Tm Θ A){σ₀ σ₁}(σ₀₁ : Tm Γ (⇑ EqS Θ σ₀ σ₁))
  → Tm Γ (⇑ EqT A σ₀ σ₁ σ₀₁ (t [ σ₀ ]t) (t [ σ₁ ]t))
∣ congT t σ₀₁ ∣ γ .↓ = t .resp (∣ σ₀₁ ∣ γ .↓)
congT t σ₀₁ .resp _ = tt
