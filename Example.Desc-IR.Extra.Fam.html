coe-coe
  : (A : Ty Γ ℓA)
   → ∀ {γ₀ γ₁ γ₂} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂)
   → (x : ∣ A ∣ γ₀)
   → A ∋ Γ .refl γ₂ ⊢ A .coe (Γ .trans γ₀₁ γ₁₂) x ~ A .coe γ₁₂ (A .coe γ₀₁ x)
coe-coe {Γ = Γ} A γ₀₁ γ₁₂ x =
  A .trans* (A .sym* (A .coh (Γ .trans γ₀₁ γ₁₂) x))
 (A .trans* (A .coh γ₀₁ x)
            (A .coh γ₁₂ (A .coe γ₀₁ x)))

record Fam (ℓA : Level) (Γ : Con ℓΓ) : Set (lsuc (ℓA ⊔ ℓΓ)) where
  field
    A : (γ : ∣ Γ ∣) → Con ℓA
    subst
    -- fmap
      : ∀ {γ₀ γ₁} (γ₀₁ : Γ ∋ γ₀ ~ γ₁) → Sub (A γ₀) (A γ₁)
    refl*
    -- identity preservation
      : ∀ {γ} x
      → A γ ∋ ∣ subst (Γ .refl γ) ∣ x ~ x
    trans*
    -- composition preservation 
      : ∀ {γ₀ γ₁ γ₂}(γ₀₁ : Γ ∋ γ₀ ~ γ₁)(γ₁₂ : Γ ∋ γ₁ ~ γ₂) x
      → A γ₂ ∋
          ∣ subst (Γ .trans γ₀₁ γ₁₂) ∣ x
        ~ ∣ subst γ₁₂ ∣ (∣ subst γ₀₁ ∣ x)

  feq
    : {γ γ′ : ∣ Γ ∣} → Γ ∋ γ ~ γ′ → ∣ A γ ∣ → ∣ A γ′ ∣ → Prop ℓA
  feq {γ} {γ′} γ~γ' x y = A γ′ ∋ ∣ subst γ~γ' ∣ x ~ y

open Fam public
  renaming (A to EL)

_∋f_⊢_~_ = feq

_∋f_~_⊢_~_ : (A : Fam ℓA Γ) (γ γ′ : ∣ Γ ∣) → {Γ ∋ γ ~ γ′} → ∣ A .EL γ ∣ → ∣ A .EL γ′ ∣ → Prop ℓA
_∋f_~_⊢_~_ A _ _ {p} x y = feq A p x y

{-# DISPLAY feq      A {γ} {γ′} _ x y = A ∋f γ ~ γ′ ⊢ x ~ y #-}
{-# DISPLAY _∋f_⊢_~_ A {γ} {γ'} _ x y = A ∋f γ ~ γ' ⊢ x ~ y #-}

record TmF (Γ : Con ℓΓ) (A : Fam ℓA Γ) : Set (ℓA ⊔ ℓΓ) where
  field
    t : (γ : ∣ Γ ∣) → ∣ EL A γ ∣
    resp
      : {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → A ∋f γ₀₁ ⊢ t γ₀ ~ t γ₁
    -- A ∋ γ' ~ γ' ⊢ coe A _ (∣ x ∣ γ) ~ ∣ x ∣ γ'

open TmF public
  renaming (t to ∣_∣)

infixl 10 _[_]' _[_]t'
_[_]' : (A : Fam ℓA Δ) (σ : Sub Γ Δ) → Fam ℓA Γ
(A [ σ ]') .EL δ        = A .EL (∣ σ ∣ δ)
(A [ σ ]') .subst γ₀₁ = A .subst (σ .resp γ₀₁)
(A [ σ ]') .refl* = A .refl*
(A [ σ ]') .trans* γ₀₁ γ₁₂ = A .trans* (σ .resp γ₀₁) (σ .resp γ₁₂)

_[_]t' : {A : Fam ℓA Δ}
  → (t : TmF Δ A) (σ : Sub Γ Δ)
  → TmF Γ (A [ σ ]')
∣ t [ σ ]t' ∣ γ = ∣ t ∣ (∣ σ ∣ γ)
(t [ σ ]t') .resp γ₀₁ = t .resp (σ .resp γ₀₁)

infixl 5 _▷'_ 
_▷'_ : (Γ : Con ℓΓ) (A : Fam ℓA Γ) → Con (ℓΓ ⊔ ℓA)
∣ Γ ▷' A ∣ = Σ ∣ Γ ∣ λ γ → ∣ EL A γ ∣
(Γ ▷' A) .eq (γ₀ , x₀) (γ₁ , x₁) =
  ∃[ γ₀₁ ∈ (Γ ∋ γ₀ ~ γ₁) ] (EL A γ₁ ∋ ∣ A .subst γ₀₁ ∣ x₀ ~ x₁)
  --                        ^~~~~~~~ Abbrev. using `feq`, this is (A ∋f γ~γ' ⊢ x ~ x')
(Γ ▷' A) .refl (γ , x) = (Γ .refl γ) , A .refl* x
(Γ ▷' A) .sym {γ₀ , x₀} {γ₁ , x₁} (γ₀₁ , x₀₁) = let γ₁₀ = Γ .sym γ₀₁ in
  γ₁₀
  , A .EL γ₀ .trans
    (A .EL γ₀ .sym
      (A .EL γ₀ .trans (A .trans* γ₀₁ γ₁₀ x₀) (A .subst _ .resp x₀₁)))
    (A .refl* x₀)
   
(Γ ▷' A) .trans {γ₀ , x₀} {γ₁ , x₁} {γ₂ , x₂} (γ₀₁ , x₀₁) (γ₁₂ , x₁₂) =
  (Γ .trans γ₀₁ γ₁₂) , A .EL γ₂ .trans (A .trans* γ₀₁ γ₁₂ x₀)
                            (A .EL γ₂ .trans (A .subst _ .resp x₀₁) x₁₂)
{-# INJECTIVE_FOR_INFERENCE _▷'_ #-}

infixl 6 _▸'_
_▸'_ : (σ : Sub Γ Δ) {A : Fam ℓΔ Δ} (t : TmF Γ (A [ σ ]')) → Sub Γ (Δ ▷' A)
∣ σ ▸' t ∣ γ     = ∣ σ ∣ γ , ∣ t ∣ γ
(σ ▸' t) .resp p = σ .resp p , t .resp p
{-# INJECTIVE_FOR_INFERENCE _▸'_ #-}

infixl 10 _⟨_⟩' _⟨_⟩t'
_⟨_⟩' : {A : Fam ℓA Γ}(B : Fam ℓB (Γ ▷' A)) (t : TmF Γ A) → Fam ℓB Γ
A ⟨ t ⟩' = A [ idS ▸' t ]' 

_⟨_⟩t' : {A : Fam ℓA Γ}{B : Fam ℓB (Γ ▷' A)}
  → (u : TmF (Γ ▷' A) B) (t : TmF Γ A) → TmF Γ (B ⟨ t ⟩')
u ⟨ t ⟩t' = u [ idS ▸' t ]t'

π₁' : (A : Fam ℓΔ Δ) → Sub Γ (Δ ▷' A)
  → Sub Γ Δ
∣ π₁' A σ ∣ γ   = ∣ σ ∣ γ .₁
π₁' A σ .resp p = σ .resp p .₁
{-# INJECTIVE_FOR_INFERENCE π₁' #-}

π₂' : (A : Fam ℓΔ Δ) (σ : Sub Γ (Δ ▷' A))
  → TmF Γ (A [ π₁' _ σ ]')
∣ π₂' A σ ∣ γ = ∣ σ ∣ γ .₂
π₂' A σ .resp γ~γ' = σ .resp γ~γ' .₂
{-# INJECTIVE_FOR_INFERENCE π₂' #-}

infixl 6 _↑'_
_↑'_ : (σ : Sub Γ Δ) (A : Fam ℓA Δ) → Sub (Γ ▷' (A [ σ ]')) (Δ ▷' A)
σ ↑' A = (σ ∘ π₁' _ idS) ▸' π₂' _ idS
{-# INJECTIVE_FOR_INFERENCE _↑'_ #-}

infixl 40 _⁺'
_⁺' : {A : Fam ℓΔ Δ} (σ : Sub Γ Δ) → Sub (Γ ▷' A [ σ ]') (Δ ▷' A)
σ ⁺' = σ ↑' _
{-# INJECTIVE_FOR_INFERENCE _⁺ #-}

Ty→Fam : Ty Γ ℓA → Fam ℓA Γ
∣ EL (Ty→Fam A) γ ∣ = ∣ A ∣ γ
EL (Ty→Fam {_} {Γ = Γ} A) γ .eq = A ∋ Γ .refl γ ⊢_~_
EL (Ty→Fam A) γ .refl  = A .refl*
EL (Ty→Fam A) γ .sym   = A .sym*
EL (Ty→Fam A) γ .trans = A .trans*
∣ Ty→Fam A .subst γ~γ' ∣  = A .coe γ~γ'
Ty→Fam A .subst γ~γ' .resp {x} {y} = λ x~y →
  A .trans* (A .sym* (A .coh γ~γ' x)) (A .trans* x~y (A .coh γ~γ' _))
Ty→Fam {_} {Γ = Γ} A .refl* {γ} x = A .sym* (A .coh (Γ .refl γ) x)
Ty→Fam {_} {Γ = Γ} A .trans* γ₀₁ γ₁₂ a = coe-coe A γ₀₁ γ₁₂ a

Fam→Ty : Fam ℓA Γ → Ty Γ ℓA
∣ Fam→Ty A ∣ γ = ∣ EL A γ ∣
Fam→Ty A .heq {γ} {γ'} γ~γ' x = EL A γ' ∋ ∣ A .subst γ~γ' ∣ x ~_
Fam→Ty A .refl*  = A .refl*
Fam→Ty {Γ = Γ} A .sym*  {γ} {γ'} {x} {y} {γ~γ'} x~y =
  EL A γ .trans
   (EL A γ .sym
    (EL A γ .trans (A .trans* γ~γ' (Γ .sym γ~γ') x)
     (A .subst (Γ .sym γ~γ') .resp x~y)))
   (A .refl* x)
Fam→Ty A .trans* {γ₀} {γ₁} {γ₂} {γ₀₁} {γ₁₂} {x₀} {x₁} {x₂} x₀₁ x₁₂ =
  EL A γ₂ .trans (A .trans* γ₀₁ γ₁₂ x₀)
   (EL A γ₂ .trans (A .subst γ₁₂ .resp x₀₁) x₁₂)
Fam→Ty A .coe {γ} {γ'} γ~γ' x₀ = ∣ A .subst γ~γ' ∣ x₀
Fam→Ty A .coh {γ} {γ'} γ~γ' x₀ = EL A γ' .refl (∣ A .subst γ~γ'  ∣ x₀)

TmF→Tm : {A : Fam ℓA Γ} → TmF Γ A → Tm Γ (Fam→Ty A)
∣ TmF→Tm t' ∣ γ = ∣ t' ∣ γ
TmF→Tm t' .resp {γ₀} {γ₁} γ₀₁ = t' .resp γ₀₁

Tm→TmF : ∀ {ℓΓ ℓA} {Γ : Con ℓΓ} {A : Ty Γ ℓA} → Tm Γ A → TmF Γ (Ty→Fam A)
∣ Tm→TmF t ∣ γ = ∣ t ∣ γ
Tm→TmF {Γ = Γ} {A} t .resp {γ₀} {γ₁} γ₀₁ =
  A .trans* (A .sym* (A .coh γ₀₁ (∣ t ∣ γ₀)))
            (t .resp γ₀₁)

⇑'_ : Prp ℓP Γ → Fam ℓP Γ
⇑' P = Ty→Fam (CwFPrp.⇑ P)
{-# INJECTIVE_FOR_INFERENCE ⇑'_ #-}

⇓'_ : Fam ℓA Γ → Prp ℓA Γ
⇓' A = CwFPrp.⇓ (Fam→Ty A)
