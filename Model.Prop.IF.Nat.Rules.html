open import Prelude

module Model.Prop.IF.Nat.Rules where

open import Model.Prop.CwF
open import Model.Prop.IF.Nat.Core
open import Model.Prop.IF.Nat.Elim
open import Model.Prop.IF.Nat.Derived

Nat[] : Nat [ σ ] ≡ Nat
Nat[] = Eq.refl

ze[] : ze [ σ ]t ≡ ze
ze[] = Eq.refl

su[]
  : {t : Tm Γ Nat}
  → (su t) [ σ ]t ≡ su (t [ σ ]t)
su[] = Eq.refl

module _ (P : Ty (Γ ▷ Nat) ℓ) (pz : Tm Γ (P ⟨ ze ⟩)) (ps : Tm (Γ ▷ Nat ▷ P) (P [ wk∘wk ▸ su (vs vz) ])) where
  elim-ze : elim-Nat P pz ps ⟨ ze ⟩t ≡ pz
  elim-ze = Eq.refl

  elim-su : (n : Tm Γ Nat) → elim-Nat P pz ps ⟨ su n ⟩t ≡ ps [ idS ▸ n ▸ elim-Nat P pz ps ⟨ n ⟩t ]t
  elim-su n = Eq.refl

  elim-su' : elim-Nat P pz ps [ wk ▸ su' ]t ≡ ps [ idS ▸ elim-Nat P pz ps ]t
  elim-su' = Eq.refl

  elim-Nat[]
    : {σ : Sub Δ Γ}
    → elim-Nat P pz ps [ σ ⁺ ]t ≡ elim-Nat (P [ σ ⁺ ]) (pz [ σ ]t) (ps [ σ ⁺ ⁺ ]t)
  elim-Nat[] = Eq.refl
