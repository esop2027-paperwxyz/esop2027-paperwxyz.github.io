-- Utilities for positions, internal forms, and term construction.

-- Length of a constructor list.
length : {I J : Ty ∅ ℓI} → ConDs I J → ℕ
length []       = zero
length (_ ∷ Ds) = suc (length Ds)

-- Membership witness for a constructor inside a constructor list.
data _∈_ {I J : Ty ∅ ℓI} (D : ConD ∅ I J) : ConDs I J → Set (lsuc ℓI) where
  here  : ∀ {Ds} → D ∈ (D ∷ Ds)
  there : ∀ {E Ds} → D ∈ Ds → D ∈ (E ∷ Ds)

-- Lookup a constructor by position.
lookupConD : {I J : Ty ∅ ℓI} (Ds : ConDs I J) → Fin (length Ds) → ConD ∅ I J
lookupConD []       ()
lookupConD (D ∷ Ds) fzero    = D
lookupConD (D ∷ Ds) (fsuc i) = lookupConD Ds i

-- Canonical membership witness for a positional lookup.
lookupConD∈ : {I J : Ty ∅ ℓI} (Ds : ConDs I J) → (i : Fin (length Ds)) → lookupConD Ds i ∈ Ds
lookupConD∈ []       ()
lookupConD∈ (D ∷ Ds) fzero    = here
lookupConD∈ (D ∷ Ds) (fsuc i) = there (lookupConD∈ Ds i)

-- Decidable membership for positional lookup.
lookupConD∈?
  : {I J : Ty ∅ ℓI} (Ds : ConDs I J)
  → (i : Fin (length Ds))
  → Dec (lookupConD Ds i ∈ Ds)
lookupConD∈? Ds i = yes (lookupConD∈ Ds i)

-- Extract the canonical membership witness from a position.
toWitness∈
  : {I J : Ty ∅ ℓI} (Ds : ConDs I J)
  → (i : Fin (length Ds))
  → lookupConD Ds i ∈ Ds
toWitness∈ Ds i = toWitness {a? = lookupConD∈? Ds i} tt

-- Internal argument type for one constructor.
Argᵢ : {I J : Ty ∅ ℓI}(D : ConD Δ I J)(X : Ty (∅ ▷ I) ℓI)
  → Ty (Δ ▷ J [ ε ]) ℓI
Argᵢ (end t)     X = ⇑ Eq (t [ wk ]t) vz
Argᵢ (non A E)   X = Sigma (A [ wk ]) (Argᵢ E X [ wk∘wk ▸ vz ▸ vs vz ])
Argᵢ (arg1 f E)  X = X [ ε ▸ f [ wk ]t ]      ẋ Argᵢ E X
Argᵢ (arg A f E) X = Π A (X [ ε ▸ f ]) [ wk ] ẋ Argᵢ E X
{-# INJECTIVE_FOR_INFERENCE Argᵢ #-}

-- Internal argument type for a constructor list.
Argsᵢ
  : {I J : Ty ∅ ℓI} (Es : ConDs I J)
  → (X : Ty (∅ ▷ I) ℓI)
  → Ty (∅ ▷ J) ℓI
Argsᵢ []       X = Empty
Argsᵢ (E ∷ Es) X = Sum (Argᵢ E X) (Argsᵢ Es X)
{-# INJECTIVE_FOR_INFERENCE Argsᵢ #-}

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  private
    inArgsᵢ'
      : (D : ConD ∅ I I)
      → (Es : ConDs I I)
      → D ∈ Es
      → (i : Tm Γ (I [ ε ]))
      → Tm Γ (Argᵢ D (μ Ds) [ ε ▸ i ])
      → Tm Γ (Argsᵢ Es (μ Ds) [ ε ▸ i ])
    inArgsᵢ' D (D ∷ Es) here i xs =
      Inl (Argᵢ D (μ Ds) [ ε ▸ i ]) (Argsᵢ Es (μ Ds) [ ε ▸ i ]) xs
    inArgsᵢ' D (E ∷ Es) (there p) i xs =
      Inr (Argᵢ E (μ Ds) [ ε ▸ i ]) (Argsᵢ Es (μ Ds) [ ε ▸ i ]) (inArgsᵢ' D Es p i xs)

  -- Inject one constructor's internal arguments into a full argument list.
  inArgsᵢ
    : {Γ : Con ℓΓ} {D : ConD ∅ I I}
    → D ∈ Ds
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ D (μ Ds) [ ε ▸ i ])
    → Tm Γ (Argsᵢ Ds (μ Ds) [ ε ▸ i ])
  inArgsᵢ = inArgsᵢ' _ Ds

  -- Inject arguments by constructor position.
  inArgs#
    : {Γ : Con ℓΓ}
    → (k : Fin (length Ds))
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ (lookupConD Ds k) (μ Ds) [ ε ▸ i ])
    → Tm Γ (Argsᵢ Ds (μ Ds) [ ε ▸ i ])
  inArgs# k i xs = inArgsᵢ (toWitness∈ Ds k) i xs

  -- Value-level injection into the argument coproduct, indexed by a membership
  -- witness; this is `inArgsᵢ` on the underlying values.
  injSum
    : {D : ConD ∅ I I} (Es : ConDs I I) → D ∈ Es
    → ∀ {j} → ∣ Argᵢ D (μ Ds) ∣ (tt , j) → ∣ Argsᵢ Es (μ Ds) ∣ (tt , j)
  injSum (D ∷ Es) here      v = inl v
  injSum (E ∷ Es) (there p) v = inr (injSum Es p v)

  -- The underlying value of `inArgsᵢ` is the corresponding witness injection.
  -- (Needed to reason about `con#` from the outside, since `inArgsᵢ'` is local.)
  ∣inArgsᵢ∣
    : {D : ConD ∅ I I} (Es : ConDs I I) (mem : D ∈ Es)
    → (i : Tm Γ (I [ ε ])) (xs : Tm Γ (Argᵢ D (μ Ds) [ ε ▸ i ])) (γ : ∣ Γ ∣)
    → ∣ inArgsᵢ' D Es mem i xs ∣ γ ≡ injSum Es mem (∣ xs ∣ γ)
  ∣inArgsᵢ∣ (D ∷ Es) here      i xs γ = Eq.refl
  ∣inArgsᵢ∣ (E ∷ Es) (there p) i xs γ = Eq.cong inr (∣inArgsᵢ∣ Es p i xs γ)

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  -- Convert internal constructor arguments to semantic arguments.
  ∣Argᵢ∣→Arg₀
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → (δ : ∣ Δ ∣) {j : ∣ I ∣ tt}
    → ∣ Argᵢ E (μ Ds) ∣ (δ , j)
    → Arg₀ E (μ₀ Ds) δ j
  ∣Argᵢ∣→Arg₀ (end t)     δ q .↓ = q .↓
  ∣Argᵢ∣→Arg₀ (non A E)   δ (a , xs) =
    a , ∣Argᵢ∣→Arg₀ E (δ , a) xs
  ∣Argᵢ∣→Arg₀ (arg1 f E)  δ (x , xs) =
    x .₁ , ∣Argᵢ∣→Arg₀ E δ xs
  ∣Argᵢ∣→Arg₀ (arg A f E) δ (x , xs) =
    (λ a → x .₁ a .₁) , ∣Argᵢ∣→Arg₀ E δ xs

  -- Recover extensionality from internal constructor arguments.
  ∣Argᵢ∣→extArg
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → (δ : ∣ Δ ∣) {j : ∣ I ∣ tt}
    → (xs : ∣ Argᵢ E (μ Ds) ∣ (δ , j))
    → extArg Ds E (∣Argᵢ∣→Arg₀ E δ xs)
  ∣Argᵢ∣→extArg (end t) δ xs = tt
  ∣Argᵢ∣→extArg (non A E) δ (a , xs) =
    ∣Argᵢ∣→extArg E (δ , a) xs
  ∣Argᵢ∣→extArg (arg1 f E) δ ((∣x∣ , x~) , xs) =
    x~ , (∣Argᵢ∣→extArg E δ xs)
  ∣Argᵢ∣→extArg (arg A f E) δ ((∣x∣ , x~) , xs) =
    (λ a → ∣x∣ a .₂) , x~ , (∣Argᵢ∣→extArg E δ xs)

  -- Convert internal argument equality to semantic argument equality.
  Argᵢ~→Arg~
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → {δ₀ δ₁ : ∣ Δ ∣}    {j₀ j₁ : ∣ I ∣ tt}
    → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)(j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → {xs₀ : ∣ Argᵢ E (μ Ds) ∣ (δ₀ , j₀)} {xs₁ : ∣ Argᵢ E (μ Ds) ∣ (δ₁ , j₁)}
    → (xs₀₁ : Argᵢ E (μ Ds) ∋ δ₀₁ , j₀₁ ⊢ xs₀ ~ xs₁)
    → Arg~ Ds E δ₀₁ (∣Argᵢ∣→Arg₀ E δ₀ xs₀) (∣Argᵢ∣→Arg₀ E δ₁ xs₁)
  Argᵢ~→Arg~ (end t) δ₀₁ j₀₁ xs₀₁ = tt
  Argᵢ~→Arg~ (non A E) δ₀₁ j₀₁ {a₀ , xs₀} {a₁ , xs₁} (a₀₁ , xs₀₁) =
    a₀₁ , (Argᵢ~→Arg~ E (δ₀₁ , a₀₁) j₀₁ xs₀₁)
  Argᵢ~→Arg~ (arg1 f E)  δ₀₁ j₀₁ (x₀₁ , xs₀₁) =
    x₀₁ , (Argᵢ~→Arg~ E δ₀₁ j₀₁ xs₀₁)
  Argᵢ~→Arg~ (arg A f E) δ₀₁ j₀₁ (x₀₁ , xs₀₁) =
    x₀₁ , (Argᵢ~→Arg~ E δ₀₁ j₀₁ xs₀₁)

  -- Convert internal constructor-list arguments to semantic arguments.
  ∣Argsᵢ∣→Args₀
    : {i : ∣ I ∣ tt} (Es : ConDs I I)
    → (xs : ∣ Argsᵢ Es (μ Ds) ∣ (tt , i))
    → Args₀ Es (μ₀ Ds) i
  ∣Argsᵢ∣→Args₀ (E ∷ Es) =
    [ (λ xs → inl (∣Argᵢ∣→Arg₀ E tt xs)) , (λ xs → inr (∣Argsᵢ∣→Args₀ Es xs)) ]′

  -- Recover extensionality from internal constructor-list arguments.
  ∣Argsᵢ∣→extArgs
    : {i : ∣ I ∣ tt} (Es : ConDs I I)
    → (xs : ∣ Argsᵢ Es (μ Ds) ∣ (tt , i))
    → extArgs Ds Es (∣Argsᵢ∣→Args₀ Es xs)
  ∣Argsᵢ∣→extArgs (E ∷ Es) (inl xs) = ∣Argᵢ∣→extArg E tt xs
  ∣Argsᵢ∣→extArgs (E ∷ Es) (inr xs) = ∣Argsᵢ∣→extArgs Es xs

  -- Convert internal argument-list equality to semantic argument equality.
  Argsᵢ~→Args~
    : {i₀ i₁ : ∣ I ∣ tt} (Es : ConDs I I)
    → {xs₀ : ∣ Argsᵢ Es (μ Ds) ∣ (tt , i₀)}
    → {xs₁ : ∣ Argsᵢ Es (μ Ds) ∣ (tt , i₁)}
    → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (xs₀₁ : Argsᵢ Es (μ Ds) ∋ tt , i₀₁ ⊢ xs₀ ~ xs₁)
    → Args~ Ds Es (∣Argsᵢ∣→Args₀ Es xs₀) (∣Argsᵢ∣→Args₀ Es xs₁)
  Argsᵢ~→Args~ (E ∷ Es) {xs₀ = inl _} {inl _} i₀₁ xs₀₁ =
    Argᵢ~→Arg~ E tt i₀₁ (xs₀₁ .↓)
  Argsᵢ~→Args~ (E ∷ Es) {xs₀ = inr _} {inr _} i₀₁ xs₀₁ =
    Argsᵢ~→Args~ Es i₀₁ (xs₀₁ .↓)

  -- Convert semantic constructor arguments back to internal form.
  Arg₀,extArg→∣Argᵢ∣
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → (δ : ∣ Δ ∣) {j : ∣ I ∣ tt}
    → (xs : Arg₀ E (μ₀ Ds) δ j)
    → (extxs : extArg Ds E xs)
    → ∣ Argᵢ E (μ Ds) ∣ (δ , j)
  Arg₀,extArg→∣Argᵢ∣ (end t)     δ q  extx = ↑ (q .↓)
  Arg₀,extArg→∣Argᵢ∣ (non A E)   δ (a , xs) extxs =
    a , (Arg₀,extArg→∣Argᵢ∣ E (δ , a) xs extxs)
  Arg₀,extArg→∣Argᵢ∣ (arg1 f E)  δ (x , xs) (x~ , extxs) =
    (x , x~)
    , Arg₀,extArg→∣Argᵢ∣ E δ xs extxs
  Arg₀,extArg→∣Argᵢ∣ (arg A f E) δ (x , xs) (extx , x~ , extxs) =
    ((λ a → (x a) , extx a) , x~)
    , Arg₀,extArg→∣Argᵢ∣ E δ xs extxs

  -- Convert semantic constructor-list arguments back to internal form.
  Args₀,extArgs→∣Argsᵢ∣
    : {j : ∣ I ∣ tt} (Es : ConDs I I)
    → (xs : Args₀ Es (μ₀ Ds) j)
    → (extxs : extArgs Ds Es xs)
    → ∣ Argsᵢ Es (μ Ds) ∣ (tt , j)
  Args₀,extArgs→∣Argsᵢ∣ (E ∷ Es) (inl xs) extxs =
    inl (Arg₀,extArg→∣Argᵢ∣ E tt xs extxs)
  Args₀,extArgs→∣Argsᵢ∣ (E ∷ Es) (inr xs) extxs =
    inr (Args₀,extArgs→∣Argsᵢ∣ Es xs extxs)

  -- Convert semantic argument equality back to internal equality.
  Arg~→Argᵢ~
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → {δ₀ δ₁ : ∣ Δ ∣}    {j₀ j₁ : ∣ I ∣ tt}
    → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)(j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → {xs₀ : ∣μArg∣ Ds E δ₀ j₀}
    → {xs₁ : ∣μArg∣ Ds E δ₁ j₁}
    → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ xs₀ xs₁)
    → Argᵢ E (μ Ds) ∋ δ₀₁ , j₀₁
        ⊢ Arg₀,extArg→∣Argᵢ∣ E δ₀ (xs₀ .₁) (xs₀ .₂)
        ~ Arg₀,extArg→∣Argᵢ∣ E δ₁ (xs₁ .₁) (xs₁ .₂)
  Arg~→Argᵢ~ (end t) δ₀₁ j₀₁ xs₀₁ = tt
  Arg~→Argᵢ~ (non A E) δ₀₁ j₀₁ {a₀ , xs₀} {a₁ , xs₁} (a₀₁ , xs₀₁) =
    a₀₁ , Arg~→Argᵢ~ E (δ₀₁ , a₀₁) j₀₁ xs₀₁
  Arg~→Argᵢ~ (arg1 f E)  δ₀₁ j₀₁ (x₀₁ , xs₀₁) =
    x₀₁ , Arg~→Argᵢ~ E δ₀₁ j₀₁ xs₀₁
  Arg~→Argᵢ~ (arg A f E) δ₀₁ j₀₁ (x₀₁ , xs₀₁) =
    x₀₁ , Arg~→Argᵢ~ E δ₀₁ j₀₁ xs₀₁

  -- Convert semantic constructor-list equality back to internal equality.
  Args~→Argsᵢ~
    : {i₀ i₁ : ∣ I ∣ tt} (Es : ConDs I I)
    → {xs₀ : ∣μArgs∣ Ds Es i₀}
    → {xs₁ : ∣μArgs∣ Ds Es i₁}
    → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
    → Argsᵢ Es (μ Ds) ∋ tt , i₀₁
        ⊢ Args₀,extArgs→∣Argsᵢ∣ Es (xs₀ .₁) (xs₀ .₂)
        ~ Args₀,extArgs→∣Argsᵢ∣ Es (xs₁ .₁) (xs₁ .₂)
  Args~→Argsᵢ~ (E ∷ Es) {xs₀ = inl _ , _} {inl _ , _} i₀₁ xs₀₁ =
    ↑ (Arg~→Argᵢ~ E tt i₀₁ xs₀₁)
  Args~→Argsᵢ~ (E ∷ Es) {xs₀ = inr _ , _} {inr _ , _} i₀₁ xs₀₁ =
    ↑ (Args~→Argsᵢ~ Es i₀₁ xs₀₁)

  -- Convert internal constructor arguments to the semantic Arg type.
  Argᵢ→Arg
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → Tm (Δ ▷ I [ ε ] ▷ Argᵢ E (μ Ds)) (μArg {Ds = Ds} E [ wk ])
  ∣ Argᵢ→Arg E ∣ ((δ , i) , xs)         = ∣Argᵢ∣→Arg₀ E δ xs , ∣Argᵢ∣→extArg E δ xs
  Argᵢ→Arg E .resp ((δ₀₁ , i₀₁) , xs₀₁) = Argᵢ~→Arg~ E δ₀₁ i₀₁ xs₀₁

  -- Convert semantic Arg terms to internal constructor arguments.
  Arg→Argᵢ
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → Tm (Δ ▷ I [ ε ] ▷ μArg {Ds = Ds} E) (Argᵢ E (μ Ds) [ wk ])
  ∣ Arg→Argᵢ E ∣ ((δ , i) , (xs , extxs)) = Arg₀,extArg→∣Argᵢ∣ E δ xs extxs
  Arg→Argᵢ E .resp ((δ₀₁ , i₀₁) , xs₀₁)   = Arg~→Argᵢ~ E δ₀₁ i₀₁ xs₀₁

  -- Convert internal argument lists to the semantic Args type.
  Argsᵢ→Args
    : (Es : ConDs I I)
    → Tm (Γ ▷ I [ ε ] ▷ Argsᵢ Es (μ Ds) [ ε ⁺ ]) (μArgs {Ds = Ds} Es [ wk ])
  ∣ Argsᵢ→Args Es ∣ ((γ , i) , xs₀)        = ∣Argsᵢ∣→Args₀ Es xs₀ , ∣Argsᵢ∣→extArgs Es xs₀
  Argsᵢ→Args Es .resp ((γ₀₁ , i₀₁) , xs₀₁) = Argsᵢ~→Args~ Es i₀₁ xs₀₁

  -- Convert semantic Args terms to internal argument lists.
  Args→Argsᵢ
    : (Es : ConDs I I)
    → Tm (Γ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es) (Argsᵢ Es (μ Ds) [ (ε ⁺) ∘ wk ])
  ∣ Args→Argsᵢ Es ∣ ((γ , i) , (xs₀ , extxs)) =
    Args₀,extArgs→∣Argsᵢ∣ Es xs₀ extxs
  Args→Argsᵢ Es .resp {γ₀ = _ , (xs₀ , extxs₀)} {_ , (xs₁ , extxs₁)} ((γ₀₁ , i₀₁) , xs₀₁) =
    Args~→Argsᵢ~ Es i₀₁ xs₀₁

  -- Subsitution from internal constructor arguments to semantic Arg terms.
  σArgᵢ→Arg
    : {Δ : Con ℓΔ} → let I[ε] = I [ ε ]
   in (τ : Sub Θ Δ) (E : ConD Δ I I)
    → Sub (Θ ▷ I [ ε ] ▷ Argᵢ E (μ Ds [ ε ⁺ ]) [ τ ↑t I[ε] ])
          (Θ ▷ I [ ε ] ▷ μArg {Ds = Ds} E [ τ ↑t I[ε] ])
  σArgᵢ→Arg τ E = wk ▸ Argᵢ→Arg E [ τ ⁺ ⁺ ]t

  -- Subsitution from semantic Arg terms to internal constructor arguments.
  σArg→Argᵢ
    : {Δ : Con ℓΔ} → let I[ε] = I [ ε ]
   in (τ : Sub Θ Δ) (E : ConD Δ I I)
    → Sub (Θ ▷ I [ ε ] ▷ μArg {Ds = Ds} E [ τ ↑t I[ε] ])
          (Θ ▷ I [ ε ] ▷ Argᵢ E (μ Ds [ ε ⁺ ]) [ τ ↑t I[ε] ])
  σArg→Argᵢ τ E = wk ▸ Arg→Argᵢ E [ τ ⁺ ⁺ ]t

  -- Substitution from internal argument lists to semantic Args type.
  σArgsᵢ→Args
    : (Es : ConDs I I)
    → Sub (Θ ▷ I [ ε ] ▷ Argsᵢ Es (μ Ds) [ ε ⁺ ]) (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es)
  σArgsᵢ→Args Es = wk ▸ Argsᵢ→Args Es [ ε ⁺ ⁺ ]t

  -- Substitution from semantic Args type to internal argument lists.
  σArgs→Argsᵢ
    : (Es : ConDs I I)
    → Sub (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es) (Θ ▷ I [ ε ] ▷ Argsᵢ Es (μ Ds) [ ε ⁺ ])
  σArgs→Argsᵢ Es = wk ▸ Args→Argsᵢ Es [ ε ⁺ ⁺ ]t

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  -- Inject one constructor's arguments into a larger argument list.
  inlArgsᵢ
    : {Γ : Con ℓΓ} (E : ConD ∅ I I) (Es : ConDs I I)
    → Tm (Γ ▷ I [ ε ] ▷ μArg {Ds = Ds} E [ ε ⁺ ])
         (μArgs {Ds = Ds} (E ∷ Es) [ wk ])
  ∣ inlArgsᵢ E Es ∣ ((γ , i) , (xs , extxs)) = inl xs , extxs
  inlArgsᵢ E Es .resp ((γ₀₁ , i₀₁) , xs₀₁) = xs₀₁

  -- Inject tail arguments into a larger argument list.
  inrArgsᵢ
    : {Γ : Con ℓΓ} (E : ConD ∅ I I) (Es : ConDs I I)
    → Tm (Γ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es)
         (μArgs {Ds = Ds} (E ∷ Es) [ wk ])
  ∣ inrArgsᵢ E Es ∣ ((γ , i) , (xs , extxs)) = inr xs , extxs
  inrArgsᵢ E Es .resp ((γ₀₁ , i₀₁) , xs₀₁) = xs₀₁
  -- Generic constructor using an explicit membership witness.
  con∈
    : {Γ : Con ℓΓ} {D : ConD ∅ I I}
    → D ∈ Ds
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ D (μ Ds) [ ε ▸ i ])
    → Tm Γ (μ Ds [ ε ▸ i ])
  con∈ p i xs =
    conᵢ Ds ⟨ i && (Argsᵢ→Args Ds [ ε ⁺ ⁺ ]t ⟨ i && inArgsᵢ p i xs ⟩t) ⟩t

  -- Generic constructor using a positional witness `# k`.
  con#
    : {Γ : Con ℓΓ}
    → (k : Fin (length Ds))
    → (i : Tm Γ (I [ ε ]))
    → Tm Γ (Argᵢ (lookupConD Ds k) (μ Ds) [ ε ▸ i ])
    → Tm Γ (μ Ds [ ε ▸ i ])
  con# k i xs = conᵢ Ds ⟨ i && (Argsᵢ→Args Ds [ ε ⁺ ⁺ ]t ⟨ i && inArgs# k i xs ⟩t) ⟩t

  module _
    {Γ : Con ℓΓ}
    (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
    where

    Pcon : Ty (Γ ▷ I [ ε ] ▷ μArgs Ds) ℓ
    Pcon = P [ wk ▸ conᵢ Ds ]

    -- `ArgIHsʳ Es σcon` is the eliminator data for the suffix `Es` of the
    -- constructor list `Ds`. A full premise sequence reads as `p₀ , ⋯ , pₙ , _`.
    --
    -- The argument `σcon` records how a value of `μArgs Es` is reinterpreted as
    -- an element of `μArgs Ds`. For the full list this is `idS`; descending into
    -- the tail it is restricted along the right injection `inrArgsᵢ E Es`, and the
    -- head premise sees it restricted along the left injection `inlArgsᵢ E Es`.
    -- This mirrors the coproduct structure of `μArgs (E ∷ Es)`.
    ArgIHsʳ
      : (Es : ConDs I I)
      → (σcon
          : {Θ : Con ℓΓ}
          → Sub (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es) (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Ds))
      → Set (ℓI ⊔ ℓΓ ⊔ ℓ)
    ArgIHsʳ []       σcon = Lift _ ⊤
    ArgIHsʳ (E ∷ Es) σcon =
        (Tm (Γ ▷ I [ ε ] ▷ μArg E [ ε ⁺ ] ▷ μIH P ε E)
            (Pcon [ σcon ∘ (wk ▸ inlArgsᵢ E Es) ∘ wk ]))
         ×
        (ArgIHsʳ Es (σcon ∘ (wk ▸ inrArgsᵢ E Es)))

    -- Carrier action of the premises on semantic constructor arguments.
    liftIHs₀
      : (Es : ConDs I I)
      → {η : {Θ : Con ℓΓ}
          → Sub (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es) (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Ds)}
      → ArgIHsʳ Es η
      → ∀ γ {i}
      → (xs : Args₀ Es (μ₀ Ds) i)
      → (extxs : extArgs Ds Es xs)
      → (ihs : ∣μIHs∣ P Es γ (xs , extxs))
      → ∣ Pcon [ η ] ∣ ((γ , i) , (xs , extxs))
    liftIHs₀ []       _        γ xs  = ⊥-elim (xs .↓)
    liftIHs₀ (E ∷ Es) (p , ps) γ {i} =
      [ (λ xs extxs ihs → ∣ p ∣ (((γ , i) , (xs , extxs)) , ihs))
      , liftIHs₀ Es ps γ ]

    -- The premises packaged as a term producing `P` on `con`-images.
    liftIHsʳ
      : (Es : ConDs I I)
      → (η : {Θ : Con ℓΓ}
          → Sub (Θ ▷ I [ ε ] ▷ μArgs {Ds = Ds} Es) (Θ ▷ I [ ε ] ▷ μArgs Ds))
      → (ps : ArgIHsʳ Es η)
      → Tm (Γ ▷ I [ ε ] ▷ μArgs Es ▷ μIHs P Es)
           (P [ wk ▸ conᵢ Ds ] [ η ] [ wk ])
    ∣ liftIHsʳ Es _ ps ∣ (((γ , i) , (xs , extxs)) , ihs) =
      liftIHs₀ Es ps γ xs extxs ihs
    liftIHsʳ (E ∷ Es) _ (p , ps) .resp {γ₀ = (_ , (inl _ , _)) , _} {γ₁ = (_ , (inl _ , _)) , _} (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁) =
      p .resp (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁)
    liftIHsʳ (E ∷ Es) _ (p , ps) .resp {γ₀ = (_ , (inr _ , _)) , _} {γ₁ = (_ , (inr _ , _)) , _} (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁) =
      liftIHsʳ Es _ ps .resp (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁)

    -- The premise sequence for the full constructor list `Ds`.
    ArgIHs : Set (ℓI ⊔ ℓΓ ⊔ ℓ)
    ArgIHs = ArgIHsʳ Ds idS

    module _ (ps : ArgIHs) where
      -- Build the single eliminator method from a fixed sequence of premises.
      liftIHsᵢ
        : Tm (Γ ▷ I [ ε ] ▷ μArgs Ds ▷ μIHs P Ds) (P [ wk ▸ conᵢ Ds ] [ wk ])
      liftIHsᵢ = liftIHsʳ Ds idS ps

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ) (ps : ArgIHs P) where
  -- Generic eliminator taking a sequence of premises (one per constructor).
  elim'
    : Tm (Γ ▷ I [ ε ] ▷ μ Ds) P
  elim' = elim P (liftIHsᵢ P ps)

  elimₑ'
    : (i : Tm Γ (I [ ε ])) (xs : Tm Γ (μ Ds ⟨ i ⟩))
    → Tm Γ (P [ idS ▸ i ▸ xs ])
  elimₑ' i xs = elimₑ P (liftIHsᵢ P ps) i xs

-- Internal induction-hypothesis type.
IHᵢ
  : {Γ : Con ℓΓ}{Δ : Con ℓΔ}(σ : Sub Γ Δ)
    {I J : Ty ∅ ℓI}(E : ConD Δ I J)
  → (X : Ty (∅ ▷ I) ℓI)
  → (P : Ty (Γ ▷ I [ ε ] ▷ X [ ε ⁺ ]) ℓ)
  → Ty (Γ ▷ J [ ε ] ▷ Argᵢ E X [ σ ↑t J [ ε ] ]) (ℓI ⊔ ℓ)
IHᵢ σ (end t) X P = Unit
IHᵢ {Γ = Γ} σ {J = J} (non A E)   X P =
  IHᵢ {Γ = Γ ▷ A [ σ ]} (σ ↑t A) E X (P [ wk ⁺ ⁺ ])
    [ wk∘wk
    ▸ fst (Argᵢ E X [ wk∘wk ▸ vz ▸ vs vz ] [ σ ↑t J [ ε ] ↑t _ ])
    ▸ vs vz
    ▸ snd (Argᵢ E X [ wk∘wk ▸ vz ▸ vs vz ] [ σ ↑t J [ ε ] ↑t _ ]) ]
IHᵢ σ {J = J} (arg1 f E)  X P =
  P [ wk∘wk
    ▸ f [ σ ∘ wk∘wk ]t
    ▸ fst' (X [ (ε ▸ f [ wk {A = J [ ε ]} ]t) ∘ σ ⁺ ]) (Argᵢ E X [ σ ⁺ ])
    ]
  ẋ IHᵢ σ E X P
    [ wk
    ▸ snd' (X [ (ε ▸ f [ wk {A = J [ ε ]} ]t) ∘ σ ⁺ ]) (Argᵢ E X [ σ ⁺ ])
    ]
IHᵢ σ {J = J} (arg A f E) X P =
  Π (A [ σ ∘ wk∘wk ])
    (P [ wk∘wk∘wk
       ▸ f [ (σ ∘ wk∘wk) ⁺ ]t
       ▸ appₑ
         (X [ ε ⁺ ∘ (wk∘wk∘wk ▸ f [ (σ ∘ wk∘wk) ⁺ ]t) ] [ wk ⁺ ])
         (fst' (Π A (X [ ε ▸ f ]) [ wk {A = J [ ε ]} ] [ σ ⁺ ]) (Argᵢ E X [ σ ⁺ ]) [ wk ]t)
         vz
       ])
  ẋ IHᵢ σ E X P [ wk ▸ snd' (Π A (X [ ε ▸ f ]) [ wk {A = J [ ε ]} ] [ σ ⁺ ]) (Argᵢ E X [ σ ⁺ ]) ]
{-# INJECTIVE_FOR_INFERENCE IHᵢ #-}

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  -- Convert semantic IH type to internal induction hypothesis.
  IH₀→∣IHᵢ∣
    : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds [ ε ⁺ ]) ℓ)
    → (E : ConD Δ I I)
    → (γ : ∣ Γ ∣)
    → let δ = ∣ σ ∣ γ in {j : ∣ I ∣ tt}
    → (xs : Arg₀ E (μ₀ Ds) δ j)
    → (extxs : extArg Ds E xs)
    → (ih : IH₀ Ds {P = λ i x extx → ∣ P ∣ ((γ , i) , (x , extx))}
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
        E (∣ σ ∣ γ)
        (xs , extxs))
    → ∣ IHᵢ σ E (μ Ds [ ε ⁺ ]) P ∣ ((γ , j) , Arg₀,extArg→∣Argᵢ∣ E δ xs extxs)
  IH₀→∣IHᵢ∣ σ P (end t) γ xs extxs ih₀ = ih₀
  IH₀→∣IHᵢ∣ σ P (non A E) γ (a , xs) extxs ih₀ =
    IH₀→∣IHᵢ∣ (σ ↑t A) (P [ wk ⁺ ⁺ ]) E (γ , a) xs extxs ih₀
  IH₀→∣IHᵢ∣ σ P (arg1 f E) γ (x , xs) (extx , extxs) (px , ih₀) =
    px , IH₀→∣IHᵢ∣ σ P E γ xs extxs ih₀
  IH₀→∣IHᵢ∣ σ P (arg A f E) γ (x , xs) (extx , x~ , extxs) (px , ih₀) =
    px , IH₀→∣IHᵢ∣ σ P E γ xs extxs ih₀

  -- Convert semantic IH equality to internal induction-hypothesis equality.
  IH~→IHᵢ~
    : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds [ ε ⁺ ]) ℓ)
    → (E : ConD Δ I I)
    → ∀ {γ₀ γ₁ j₀ j₁ xs₀ xs₁ extxs₀ extxs₁ ih₀ ih₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → let δ₀ = ∣ σ ∣ γ₀; δ₁ = ∣ σ ∣ γ₁; δ₀₁ = σ .resp γ₀₁
   in (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (xs₀₁ : Arg~ Ds E δ₀₁ xs₀ xs₁)
    → (ih₀₁ : IH~ Ds
        {P₀ = λ i x extx → ∣ P ∣ ((γ₀ , i) , (x , extx))}
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl _ , i₀₁) , x₀₁))
        {P₁ = λ i x extx → ∣ P ∣ ((γ₁ , i) , (x , extx))}
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl _ , i₀₁) , x₀₁))
        (λ _ _ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
        E δ₀₁
        (xs₀ , extxs₀) (xs₁ , extxs₁)
        xs₀₁
        ih₀ ih₁)
    → IHᵢ σ E (μ Ds [ ε ⁺ ]) P ∋ (γ₀₁ , j₀₁) , Arg~→Argᵢ~ E δ₀₁ j₀₁ xs₀₁
        ⊢ IH₀→∣IHᵢ∣ σ P E γ₀ xs₀ extxs₀ ih₀
        ~ IH₀→∣IHᵢ∣ σ P E γ₁ xs₁ extxs₁ ih₁
  IH~→IHᵢ~ σ P (end t) γ₀₁ j₀₁ xs₀₁ ih₀₁ = ih₀₁
  IH~→IHᵢ~ σ P (non A E) γ₀₁ j₀₁ (a₀₁ , xs₀₁) ih₀₁ =
    IH~→IHᵢ~ (σ ↑t A) (P [ wk ⁺ ⁺ ]) E (γ₀₁ , a₀₁) j₀₁ xs₀₁ ih₀₁
  IH~→IHᵢ~ σ P (arg1 f E) γ₀₁ j₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    px₀₁ , IH~→IHᵢ~ σ P E γ₀₁ j₀₁ xs₀₁ ih₀₁
  IH~→IHᵢ~ σ P (arg A f E) γ₀₁ j₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    px₀₁ , IH~→IHᵢ~ σ P E γ₀₁ j₀₁ xs₀₁ ih₀₁

  -- Convert internal induction hypothesis to semantic IH type.
  ∣IHᵢ∣→IH₀
    : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds [ ε ⁺ ]) ℓ)
    → (E : ConD Δ I I)
    → (γ : ∣ Γ ∣)
    → let δ = ∣ σ ∣ γ in {j : ∣ I ∣ tt}
    → (xs : ∣ Argᵢ E (μ Ds [ ε ⁺ ]) ∣ (δ , j))
    → (ih : ∣ IHᵢ σ E (μ Ds [ ε ⁺ ]) P ∣ ((γ , j) , xs))
    → IH₀ Ds {P = λ i x extx → ∣ P ∣ ((γ , i) , (x , extx))}
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
        E (∣ σ ∣ γ)
        (∣Argᵢ∣→Arg₀ E δ xs , ∣Argᵢ∣→extArg E δ xs)
  ∣IHᵢ∣→IH₀ σ P (end t) γ xs ih = ih
  ∣IHᵢ∣→IH₀ σ P (non A E) γ (a , xs) ih =
    ∣IHᵢ∣→IH₀ (σ ↑t A) (P [ wk ⁺ ⁺ ]) E (γ , a) xs ih
  ∣IHᵢ∣→IH₀ σ P (arg1 f E) γ (x , xs) (px , ih) =
    px , ∣IHᵢ∣→IH₀ σ P E γ xs ih
  ∣IHᵢ∣→IH₀ σ P (arg A f E) γ (x , xs) (px , ih) =
    px , ∣IHᵢ∣→IH₀ σ P E γ xs ih

  -- Convert internal induction-hypothesis equality to semantic IH equality.
  IHᵢ~→IH~
    : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds [ ε ⁺ ]) ℓ)
    → (E : ConD Δ I I)
    → ∀ {γ₀ γ₁ j₀ j₁ xs₀ xs₁ ih₀ ih₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → let δ₀ = ∣ σ ∣ γ₀; δ₁ = ∣ σ ∣ γ₁; δ₀₁ = σ . resp γ₀₁
   in (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (xs₀₁ : Argᵢ E (μ Ds [ ε ⁺ ]) ∋ δ₀₁ , j₀₁ ⊢ xs₀ ~ xs₁)
    → (ih₀₁ : IHᵢ σ E (μ Ds [ ε ⁺ ]) P ∋ (γ₀₁ , j₀₁) , xs₀₁ ⊢ ih₀ ~ ih₁)
    → IH~ Ds
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
        (λ _ _ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
        (λ _ _ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
        E δ₀₁
        _ _ (Argᵢ~→Arg~ E δ₀₁ j₀₁ xs₀₁)
        (∣IHᵢ∣→IH₀ σ P E γ₀ xs₀ ih₀)
        (∣IHᵢ∣→IH₀ σ P E γ₁ xs₁ ih₁)
  IHᵢ~→IH~ σ P (end t) γ₀₁ j₀₁ xs₀₁ ih₀₁ = tt
  IHᵢ~→IH~ σ P (non A E) γ₀₁ j₀₁ (a₀₁ , xs₀₁) ih₀₁ =
    IHᵢ~→IH~ (σ ↑t A) (P [ wk ⁺ ⁺ ]) E (γ₀₁ , a₀₁) j₀₁ xs₀₁ ih₀₁
  IHᵢ~→IH~ σ P (arg1 f E) γ₀₁ j₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    px₀₁ , IHᵢ~→IH~ σ P E γ₀₁ j₀₁ xs₀₁ ih₀₁
  IHᵢ~→IH~ σ P (arg A f E) γ₀₁ j₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    px₀₁ , IHᵢ~→IH~ σ P E γ₀₁ j₀₁ xs₀₁ ih₀₁

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I}
    {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds [ ε ⁺ ]) ℓ)
    (E : ConD Δ (I [ ε ]) (I [ ε ]))
    where

  private I[ε] = I [ ε {Γ = Δ} ]

  -- Convert the semantic IH terms to internal induction hypothesis.
  IH→IHᵢ
    : Tm (Γ ▷ I [ ε ] ▷ μArg E [ σ ↑t I[ε] ] ▷ μIH P σ E)
         (IHᵢ σ E (μ Ds [ ε ⁺ ]) P [ σArg→Argᵢ σ E ∘ wk ])
  ∣ IH→IHᵢ ∣ (((γ , i) , (xs , extxs)) , ih) = IH₀→∣IHᵢ∣ σ P E γ xs extxs ih
  IH→IHᵢ .resp (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁) = IH~→IHᵢ~ σ P E γ₀₁ i₀₁ xs₀₁ ih₀₁

  -- Convert internal induction hypothesis to the semantic IH terms.
  IHᵢ→IH
    : Tm (Γ ▷ I [ ε ] ▷ Argᵢ E (μ Ds [ ε ⁺ ]) [ σ ↑t I[ε] ] ▷ IHᵢ σ E (μ Ds [ ε ⁺ ]) P)
         (μIH P σ E [ σArgᵢ→Arg σ E ∘ wk ])
  ∣ IHᵢ→IH ∣ (((γ , i) , xs) , ih)           = ∣IHᵢ∣→IH₀ σ P E γ xs ih
  IHᵢ→IH .resp (((γ₀₁ , i₀₁) , xs₀₁) , ih₀₁) = IHᵢ~→IH~ σ P E γ₀₁ i₀₁ xs₀₁ ih₀₁

  -- Substitution from semantic IH terms to internal induction hypothesis.
  σIH→IHᵢ
    : Sub (Γ ▷ I [ ε ] ▷ μArg E [ σ ↑t I[ε] ] ▷ μIH P σ E)
          (Γ ▷ I [ ε ] ▷ Argᵢ E (μ Ds [ ε ⁺ ]) [ σ ↑t I[ε] ] ▷ IHᵢ σ E (μ Ds [ ε ⁺ ]) P)
  σIH→IHᵢ = wk∘wk ▸ Arg→Argᵢ E [ σ ⁺ ⁺ ∘ wk ]t ▸ IH→IHᵢ

  -- Substitution from internal induction hypothesis to semantic IH terms.
  σIHᵢ→IH
    : Sub (Γ ▷ I [ ε ] ▷ Argᵢ E (μ Ds [ ε ⁺ ]) [ σ ↑t I[ε] ] ▷ IHᵢ σ E (μ Ds [ ε ⁺ ]) P)
          (Γ ▷ I [ ε ] ▷ μArg E [ σ ↑t I[ε] ] ▷ μIH P σ E)
  σIHᵢ→IH = wk∘wk ▸ Argᵢ→Arg E [ σ ⁺ ⁺ ∘ wk ]t ▸ IHᵢ→IH
