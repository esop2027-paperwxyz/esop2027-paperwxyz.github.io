open import Prelude

module Model.Prop.Desc.IH where

open import Model.Prop.CwF

open import Model.Prop.Desc.Core

private variable
  ℓX : Level
  ℓI : Level

module _ {I : Ty ∅ ℓI} (Ds : ConDs I I)
  {P : (i : ∣ I ∣ tt) (x : μ₀ Ds i) (extx : extμ Ds x) → Set ℓ}
  (P~ : ∀ {i₀ i₁ x₀ x₁} extx₀ extx₁ (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁) → P i₀ x₀ extx₀ → P i₁ x₁ extx₁ → Prop ℓ)
  where
  -- Semantic induction hypotheses for one constructor.
  IH₀
    : {Δ : Con ℓΔ}
    → (D : ConD Δ I I)
    → (δ : ∣ Δ ∣)
    → {j : ∣ I ∣ tt}
    → (xs : ∣μArg∣ Ds D δ j)
    → Set (ℓI ⊔ ℓ)
  IH₀ (end t)     δ (xs , _) = Lift _ ⊤
  IH₀ (non A D)   δ ((a , xs) , extxs) =
    IH₀ D (δ , a) (xs , extxs)
  IH₀ (arg1 f D)  δ ((x , xs) , (extx , extxs)) =
    P (∣ f ∣ δ) x extx × IH₀ D δ (xs , extxs)
  IH₀ {Δ = Δ} (arg A f D) δ ((x , xs) , (extx , x~ , extxs)) =
    Σₚ ((a : ∣ A ∣ δ) → P (∣ f ∣ (δ , a)) (x a) (extx a)) (λ px →
      {a₀ a₁ : ∣ A ∣ δ}(a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁)
        → P~ _ _ (f .resp (_ , a₀₁)) (x~ a₀₁) (px a₀) (px a₁))
    × IH₀ D δ (xs , extxs)
  {-# INJECTIVE_FOR_INFERENCE IH₀ #-}

  IHs₀
    : (Es : ConDs I I)
    → {j : ∣ I ∣ tt}
    → (xs : ∣μArgs∣ Ds Es j)
    → Set (ℓI ⊔ ℓ)
  IHs₀ (D ∷ Ds) (inl xs , extxs) = IH₀ D tt (xs , extxs)
  IHs₀ (D ∷ Ds) (inr xs , extxs) = IHs₀ Ds (xs , extxs)
  {-# INJECTIVE_FOR_INFERENCE IHs₀ #-}

module _ {I : Ty ∅ ℓI} (Ds : ConDs I I)
  --   Technically, the full generic definition needs two copies of X (i.e. X₀ and X₁)
  --   and three copies of X~ where one has type X₀ → X₀ → Prop, another has type
  --   X₁ → X₁ → Prop and the last one has type X₀ → X₁ → Prop just like Pᵢs and P[ᵢ]~s.
  --
  --   However, since the IH[s] types later are specialised to μ, X is instantiated
  --   with μ and hence all copies of X and X~ are identical.
  --
  --   For the full generic definition of IH[s]~, see Model.Prop.Desc-QIT.IH.
  {P₀ : (i : ∣ I ∣ tt) (x : μ₀ Ds i) (extx : extμ Ds x) → Set ℓ}
  (P₀~ : ∀ {i₀ i₁ x₀ x₁} extx₀ extx₁ i₀₁ (x₀₁ : μ~ Ds x₀ x₁) → P₀ i₀ x₀ extx₀ → P₀ i₁ x₁ extx₁ → Prop ℓ)
  {P₁ : (i : ∣ I ∣ tt) (x : μ₀ Ds i) (extx : extμ Ds x) → Set ℓ}
  (P₁~ : ∀ {i₀ i₁ x₀ x₁} extx₀ extx₁ i₀₁ (x₀₁ : μ~ Ds x₀ x₁) → P₁ i₀ x₀ extx₀ → P₁ i₁ x₁ extx₁ → Prop ℓ)
  (P~ : ∀ {i₀ i₁ x₀ x₁} extx₀ extx₁ (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁) → P₀ i₀ x₀ extx₀ → P₁ i₁ x₁ extx₁ → Prop ℓ)
  where

  -- Setoid relation on semantic induction hypotheses.
  IH~
    : {Δ : Con ℓΔ}
    → (D : ConD Δ I I)
    → {δ₀ δ₁ : ∣ Δ ∣} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → {j₀ : ∣ I ∣ tt}{j₁ : ∣ I ∣ tt}
    → (xs₀  : ∣μArg∣ Ds D δ₀ j₀) (xs₁ : ∣μArg∣ Ds D δ₁ j₁)
    -- The extArg part cannot be inferred, thus xs₀ and xs₁ are not implicit.
    → (xs₀₁ : ∣μArg~∣ Ds D δ₀₁ xs₀ xs₁)
    → (ih₀ : IH₀ Ds P₀~ D δ₀ xs₀) (ih₁ : IH₀ Ds P₁~ D δ₁ xs₁)
    → Prop (ℓI ⊔ ℓ)
  IH~ (end t)     δ₀₁ _ _ xs₀₁         ih₀         ih₁         = ⊤ₚ
  IH~ (non A D)   δ₀₁ _ _ (a₀₁ , xs₀₁) ih₀         ih₁         = IH~ D (δ₀₁ , a₀₁) _ _ xs₀₁ ih₀ ih₁
  IH~ (arg1 f D)  δ₀₁ _ _ (x₀₁ , xs₀₁) (px₀ , ih₀) (px₁ , ih₁) =
    P~ _ _ (f .resp δ₀₁) x₀₁ px₀ px₁
    ∧ IH~ D δ₀₁ _ _ xs₀₁ ih₀ ih₁
  IH~ (arg A f D) δ₀₁ _ _ (x₀₁ , xs₀₁) ((px₀ , px₀~) , ih₀) ((px₁ , px₁~) , ih₁) =
    (∀ a₀ a₁ (a₀₁ : A ∋ δ₀₁ ⊢ a₀ ~ a₁)
      → P~ _ _ (f .resp (_ , a₀₁)) (x₀₁ a₀ a₁ a₀₁) (px₀ a₀) (px₁ a₁))
    ∧ IH~ D δ₀₁ _ _ xs₀₁ ih₀ ih₁
  {-# INJECTIVE_FOR_INFERENCE IH~ #-}

  IHs~
    : (Es : ConDs I I)
    → {j₀ j₁ : ∣ I ∣ tt} (xs₀ : ∣μArgs∣ Ds Es j₀) (xs₁ : ∣μArgs∣ Ds Es j₁)
    -- The extArgs part cannot be inferred, thus xs₀ and xs₁ are not implicit.
    → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
    → (ih₀ : IHs₀ Ds P₀~ Es xs₀) (ih₁ : IHs₀ Ds P₁~ Es xs₁)
    → Prop (ℓI ⊔ ℓ)
  IHs~ (D ∷ Ds) (inl _ , _) (inl _ , _) xs₀₁ ih₀ ih₁ = IH~ D tt _ _ xs₀₁ ih₀ ih₁
  IHs~ (D ∷ Ds) (inr _ , _) (inr _ , _) xs₀₁ ih₀ ih₁ = IHs~ Ds _ _ xs₀₁ ih₀ ih₁
  {-# INJECTIVE_FOR_INFERENCE IHs~ #-}


  -- Transport and coherence for semantic induction hypotheses.
  module _
    (Ptrans₁₀₀₁
      : ∀ {i₀ i₀' i₁ i₁' x₀ x₀' x₁ x₁' extx₀ extx₀' extx₁ extx₁'}
      → {px₀ : P₀ i₀ x₀ extx₀} {px₀' : P₀ i₀' x₀' extx₀'} {px₁ : P₁ i₁ x₁ extx₁} {px₁' : P₁ i₁' x₁' extx₁'}
      → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
      → {x₀₁ : μ~ Ds x₀ x₁} (x₀₀' : μ~ Ds x₀ x₀') (x₀'₁' : μ~ Ds  x₀' x₁')
      → {x₁₁' : μ~ Ds x₁ x₁'}
      → P~ _ _ i₀₁ x₀₁ px₀ px₁ → P₀~ _ _ i₀₀' x₀₀' px₀ px₀' → P~ _ _ i₀'₁' x₀'₁' px₀' px₁'
      → P₁~ _ _ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁₁' px₁ px₁')
    (coeP
      : ∀ {i₀ i₁ x₀ x₁ extx₀ extx₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁)
      → (px₀ : P₀ i₀ x₀ extx₀) → P₁ i₁ x₁ extx₁)
    (cohP
      : ∀ {i₀ i₁ x₀ x₁ extx₀ extx₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁)
      → (px₀ : P₀ i₀ x₀ extx₀)
      → P~ extx₀ extx₁ i₀₁ x₀₁ px₀ (coeP i₀₁ x₀₁ px₀))
    where

    coeIH
      : {Δ : Con ℓΔ} (E : ConD Δ I I)
      → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
      → ∀ {j₀ j₁} {xs₀ : ∣μArg∣ Ds E δ₀ j₀} {xs₁ : ∣μArg∣ Ds E δ₁ j₁}
      → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ xs₀ xs₁)
      → (ih₀ : IH₀ Ds P₀~ E δ₀ xs₀)
      → IH₀ Ds P₁~ E δ₁ xs₁
    coeIH (end t)    δ₀₁ xs₀₁         ih₀        = ih₀
    coeIH (non A D)  δ₀₁ (a₀₁ , xs₀₁) ih₀        = coeIH D (δ₀₁ , a₀₁) xs₀₁ ih₀
    coeIH (arg1 f D) δ₀₁ (x₀₁ , xs₀₁) (px , ih₀) = coeP (f .resp δ₀₁) x₀₁ px , coeIH D δ₀₁ xs₀₁ ih₀
    coeIH {Δ = Δ} (arg A f D) δ₀₁ (x₀₁ , xs₀₁) ((px , px~) , ih₀) =
      let δ₁₀ = Δ .sym δ₀₁ in
      ((λ a₁ →
          coeP (f .resp (δ₀₁ , coh' A δ₁₀ a₁))
              (x₀₁ (A .coe δ₁₀ a₁) a₁ (coh' A δ₁₀ a₁))
              (px (A .coe δ₁₀ a₁)))
      , λ {a₁} {a₁'} a₁₁' →
          Ptrans₁₀₀₁ _ _
            (cohP _ (x₀₁ (A .coe δ₁₀ a₁)  a₁  (coh' A δ₁₀ a₁))  (px (A .coe δ₁₀ a₁)))
            (px~ (A .trans* (coh' A δ₁₀ a₁)
                   (A .trans* a₁₁'
                     (A .coh δ₁₀ a₁'))))
            (cohP _ (x₀₁ (A .coe δ₁₀ a₁') a₁' (coh' A δ₁₀ a₁')) (px (A .coe δ₁₀ a₁'))))
      , coeIH D δ₀₁ xs₀₁ ih₀

    coeIHs
      : (Es : ConDs I I)
      → ∀ {i₀ i₁} {xs₀ : ∣μArgs∣ Ds Es i₀} {xs₁ : ∣μArgs∣ Ds Es i₁}
      → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
      → (ih₀ : IHs₀ Ds P₀~ Es xs₀)
      → IHs₀ Ds P₁~ Es xs₁
    coeIHs (E ∷ Es) {xs₀ = inl _ , _} {inl _ , _} xs₀₁ ih₀ = coeIH E tt xs₀₁ ih₀
    coeIHs (E ∷ Es) {xs₀ = inr _ , _} {inr _ , _} xs₀₁ ih₀ = coeIHs Es xs₀₁ ih₀

  -- The coherence of IH coercion
  module _
    (Ptrans₀₀₁
      : ∀ {i₀ i₀' i₁ x₀ x₀' x₁ extx₀ extx₀' extx₁}
      → {px₀ : P₀ i₀ x₀ extx₀} {px₀' : P₀ i₀' x₀' extx₀'} {px₁ : P₁ i₁ x₁ extx₁}
      → {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁ : I ∋ tt ⊢ i₀' ~ i₁}
      → {x₀₀' : μ~ Ds x₀ x₀'} {x₀'₁ : μ~ Ds x₀' x₁}
      → {x₀₁ : μ~ Ds x₀ x₁}
      → P₀~ _ _ i₀₀' x₀₀' px₀ px₀' → P~ _ _ i₀'₁ x₀'₁ px₀' px₁
      → P~ _ _ (I .trans* i₀₀' i₀'₁) x₀₁ px₀ px₁)
    (Ptrans₁₀₀₁
      : ∀ {i₀ i₀' i₁ i₁' x₀ x₀' x₁ x₁' extx₀ extx₀' extx₁ extx₁'}
      → {px₀ : P₀ i₀ x₀ extx₀} {px₀' : P₀ i₀' x₀' extx₀'} {px₁ : P₁ i₁ x₁ extx₁} {px₁' : P₁ i₁' x₁' extx₁'}
      → {i₀₁ : I ∋ tt ⊢ i₀ ~ i₁} {i₀₀' : I ∋ tt ⊢ i₀ ~ i₀'} {i₀'₁' : I ∋ tt ⊢ i₀' ~ i₁'}
      → {x₀₁ : μ~ Ds x₀ x₁} (x₀₀' : μ~ Ds x₀ x₀') (x₀'₁' : μ~ Ds x₀' x₁')
      → {x₁₁' : μ~ Ds x₁ x₁'}
      → P~ _ _ i₀₁ x₀₁ px₀ px₁ → P₀~ _ _ i₀₀' x₀₀' px₀ px₀' → P~ _ _ i₀'₁' x₀'₁' px₀' px₁'
      → P₁~ _ _ (I .trans* (I .sym* i₀₁) (I .trans* i₀₀' i₀'₁')) x₁₁' px₁ px₁')
    (coeP
      : ∀ {i₀ i₁ x₀ x₁ extx₀ extx₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁)
      → (px₀ : P₀ i₀ x₀ extx₀) → P₁ i₁ x₁ extx₁)
    (cohP
      : ∀ {i₀ i₁ x₀ x₁ extx₀ extx₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) (x₀₁ : μ~ Ds x₀ x₁)
      → (px₀ : P₀ i₀ x₀ extx₀)
      → P~ extx₀ extx₁ i₀₁ x₀₁ px₀ (coeP i₀₁ x₀₁ px₀))
    where

    cohIH
      : {Δ : Con ℓΔ} (D : ConD Δ I I)
      → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
      → ∀ {j₀ j₁} {xs₀ : ∣μArg∣ Ds D δ₀ j₀} {xs₁ : ∣μArg∣ Ds D δ₁ j₁}
      → (xs₀₁ : ∣μArg~∣ Ds D δ₀₁ xs₀ xs₁)
      → (ih₀ : IH₀ Ds P₀~ D δ₀ xs₀)
      → IH~ D δ₀₁ xs₀ xs₁ xs₀₁
            ih₀
            (coeIH Ptrans₁₀₀₁ coeP cohP D δ₀₁ xs₀₁ ih₀)
    cohIH (end t)    δ₀₁ xs₀₁         ih₀        = tt
    cohIH (non A D)  δ₀₁ (a₀₁ , xs₀₁) ih₀        = cohIH D (δ₀₁ , a₀₁) xs₀₁ ih₀
    cohIH (arg1 f D) δ₀₁ (x₀₁ , xs₀₁) (px , ih₀) = cohP (f .resp δ₀₁) x₀₁ px , cohIH D δ₀₁ xs₀₁ ih₀
    cohIH {Δ = Δ} (arg A f D) δ₀₁ (x₀₁ , xs₀₁) ((px , px~) , ih₀) =
      let δ₁₀ = Δ .sym δ₀₁ in
      (λ a₀ a₁ a₀₁ →
        Ptrans₀₀₁
          (px~ (A .trans* a₀₁ (A .coh δ₁₀ a₁)))
          (cohP (f .resp (δ₀₁ , coh' A δ₁₀ a₁)) (x₀₁ (A .coe δ₁₀ a₁) a₁ (coh' A δ₁₀ a₁)) (px (A .coe δ₁₀ a₁))))
      , cohIH D δ₀₁ xs₀₁ ih₀

    cohIHs
      : (Es : ConDs I I)
      → ∀ {i₀ i₁} {xs₀ : ∣μArgs∣ Ds Es i₀} {xs₁ : ∣μArgs∣ Ds Es i₁}
      → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
      → (ih₀ : IHs₀ Ds P₀~ Es xs₀)
      → IHs~ Es xs₀ xs₁ xs₀₁ ih₀ (coeIHs Ptrans₁₀₀₁ coeP cohP Es xs₀₁ ih₀)
    cohIHs (E ∷ Es) {xs₀ = inl _ , _} {inl _ , _} xs₀₁ ih₀ = cohIH E tt xs₀₁ ih₀
    cohIHs (E ∷ Es) {xs₀ = inr _ , _} {inr _ , _} xs₀₁ ih₀ = cohIHs Es xs₀₁ ih₀

-- Induction-hypothesis families

module _ {Γ : Con ℓΓ} {I : Ty ∅ ℓI} {Ds : ConDs I I} (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ) where
  -- Shorthand carrier for semantic induction hypotheses.
  ∣μIH∣
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → ∀ γ δ {i} (xs : ∣μArg∣ Ds E δ i)
    → Set (ℓI ⊔ ℓ)
  ∣μIH∣ E γ δ {i} xs =
    IH₀ Ds {P = λ i x extx → ∣ P ∣ ((γ , i) , (x , extx))}
        (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
        E δ
        xs

  ∣μIHs∣
    : (Es : ConDs I I)
    → ∀ γ {i} (xs : ∣μArgs∣ Ds Es i)
    → Set (ℓI ⊔ ℓ)
  ∣μIHs∣ Es γ {i} xs =
    IHs₀ Ds {P = λ i x extx → ∣ P ∣ ((γ , i) , (x , extx))}
         (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ , i₀₁) , x₀₁))
         Es
         xs

  -- Shorthand relation for semantic induction hypotheses.
  ∣μIH~∣
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → ∀ {γ₀ γ₁ δ₀ δ₁ i₀ i₁}
    → {xs₀ : ∣μArg∣ Ds E δ₀ i₀}
    → {xs₁ : ∣μArg∣ Ds E δ₁  i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ xs₀ xs₁)
    → (ih₀ : ∣μIH∣ E γ₀ δ₀ xs₀)
    → (ih₁ : ∣μIH∣ E γ₁ δ₁ xs₁)
    → Prop (ℓI ⊔ ℓ)
  ∣μIH~∣ E {γ₀} {γ₁} γ₀₁ δ₀₁ xs₀₁ =
    IH~ Ds (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
        (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
        (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
        E δ₀₁
        _ _ xs₀₁

  ∣μIHs~∣
    : (Es : ConDs I I)
    → ∀ {γ₀ γ₁ i₀ i₁}
    → {xs₀ : ∣μArgs∣ Ds Es i₀}
    → {xs₁ : ∣μArgs∣ Ds Es i₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
    → (ih₀ : ∣μIHs∣ Es γ₀ xs₀)
    → (ih₁ : ∣μIHs∣ Es γ₁ xs₁)
    → Prop (ℓI ⊔ ℓ)
  ∣μIHs~∣ Es {γ₀} {γ₁} γ₀₁ xs₀₁ =
    IHs~ Ds (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
         (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
         (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
         Es
         _ _ xs₀₁

  -- Setoid structure on semantic induction hypotheses.
  μIH₀-refl*
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → ∀ {γ δ i} {xs₀ : ∣μArg∣ Ds E δ i}
    → (ih₀ : ∣μIH∣ E γ δ xs₀)
    → ∣μIH~∣ E (Γ .refl γ) (Δ .refl δ) (μArg₀-refl* E (xs₀ .₁) (xs₀ .₂)) ih₀ ih₀
  μIH₀-refl* (end t)     ih₀                = tt
  μIH₀-refl* (non A E)   ih₀                = μIH₀-refl* E ih₀
  μIH₀-refl* (arg1 f E)  (px , ih₀)         = P .refl* px , μIH₀-refl* E ih₀
  μIH₀-refl* (arg A f E) ((px , px~) , ih₀) = (λ a₀ a₁ a₀₁ → px~ a₀₁) , μIH₀-refl* E ih₀

  μIH₀-sym*
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → ∀ {γ₀ γ₁ δ₀ δ₁ i₀ i₁}
    → {xs₀ : ∣μArg∣ Ds E δ₀ i₀}
    → {xs₁ : ∣μArg∣ Ds E δ₁ i₁}
    → {ih₀ : ∣μIH∣ E γ₀ δ₀ xs₀} {ih₁ : ∣μIH∣ E γ₁ δ₁ xs₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ xs₀ xs₁)
    → (ih₀₁ : ∣μIH~∣ E γ₀₁ δ₀₁ xs₀₁ ih₀ ih₁)
    → ∣μIH~∣ E (Γ .sym γ₀₁) (Δ .sym δ₀₁) (μArg₀-sym* E xs₀₁) ih₁ ih₀
  μIH₀-sym* (end t) γ₀₁ δ₀₁ xs₀₁ ih₀₁ = tt
  μIH₀-sym* (non A E) γ₀₁ δ₀₁ (a₀₁ , xs₀₁) ih₀₁ =
    μIH₀-sym* E γ₀₁ (δ₀₁ , a₀₁) xs₀₁ ih₀₁
  μIH₀-sym* (arg1 f E) γ₀₁ δ₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    P .sym* px₀₁ , μIH₀-sym* E γ₀₁ δ₀₁ xs₀₁ ih₀₁
  μIH₀-sym* (arg A f E) γ₀₁ δ₀₁ (x₀₁ , xs₀₁) (px₀₁ , ih₀₁) =
    (λ a₀ a₁ a₀₁ → P .sym* (px₀₁ _ _ (A .sym* a₀₁)))
    , μIH₀-sym* E γ₀₁ δ₀₁ xs₀₁ ih₀₁

  μIH₀-trans*
    : {Δ : Con ℓΔ} (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → ∀ {γ₀ γ₁ γ₂ δ₀ δ₁ δ₂ i₀ i₁ i₂}
    → {xs₀ : ∣μArg∣ Ds E δ₀ i₀}
    → {xs₁ : ∣μArg∣ Ds E δ₁ i₁}
    → {xs₂ : ∣μArg∣ Ds E δ₂ i₂}
    → {ih₀ : ∣μIH∣ E γ₀ δ₀ xs₀} {ih₁ : ∣μIH∣ E γ₁ δ₁ xs₁} {ih₂ : ∣μIH∣ E γ₂ δ₂ xs₂}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂) (δ₀₁ : Δ ∋ δ₀ ~ δ₁) (δ₁₂ : Δ ∋ δ₁ ~ δ₂)
    → (xs₀₁ : ∣μArg~∣ Ds E δ₀₁ xs₀ xs₁)
    → (xs₁₂ : ∣μArg~∣ Ds E δ₁₂ xs₁ xs₂)
    → (ih₀₁ : ∣μIH~∣ E γ₀₁ δ₀₁ xs₀₁ ih₀ ih₁)
    → (ih₁₂ : ∣μIH~∣ E γ₁₂ δ₁₂ xs₁₂ ih₁ ih₂)
    → ∣μIH~∣ E (Γ .trans γ₀₁ γ₁₂) (Δ .trans δ₀₁ δ₁₂) (μArg₀-trans* E xs₀₁ xs₁₂) ih₀ ih₂
  μIH₀-trans* (end t) γ₀₁ γ₁₂ δ₀₁ δ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂ = tt
  μIH₀-trans* (non A E) γ₀₁ γ₁₂ δ₀₁ δ₁₂ (a₀₁ , xs₀₁) (a₁₂ , xs₁₂) ih₀₁ ih₁₂ =
    μIH₀-trans* E γ₀₁ γ₁₂ (δ₀₁ , a₀₁) (δ₁₂ , a₁₂) xs₀₁ xs₁₂ ih₀₁ ih₁₂
  μIH₀-trans* (arg1 f E) {xs₁ = extx₁ , _} γ₀₁ γ₁₂ δ₀₁ δ₁₂ (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) (px₀₁ , ih₀₁) (px₁₂ , ih₁₂) =
    P .trans* px₀₁ px₁₂
    , μIH₀-trans* E γ₀₁ γ₁₂ δ₀₁ δ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂
  μIH₀-trans* (arg A f E) {xs₁ = extx₁ , x₁~ , _} γ₀₁ γ₁₂ δ₀₁ δ₁₂ (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) (px₀₁ , ih₀₁) (px₁₂ , ih₁₂) =
    (λ a₀ a₂ a₀₂ →
      P .trans* (px₀₁ _ _ (A .coh δ₀₁ a₀))
                (px₁₂ _ _ (A .trans* (coh' A δ₀₁ a₀) a₀₂)))
    , μIH₀-trans* E γ₀₁ γ₁₂ δ₀₁ δ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂

  μIHs₀-refl*
    : (Es : ConDs I I)
    → ∀ {γ i} {xs₀ : ∣μArgs∣ Ds Es i}
    → (ih₀ : ∣μIHs∣ Es γ xs₀)
    → ∣μIHs~∣ Es (Γ .refl γ) (μArgs₀-refl* Es (xs₀ .₁) (xs₀ .₂)) ih₀ ih₀
  μIHs₀-refl* (E ∷ Es) {xs₀ = inl _ , _} ih₀ = μIH₀-refl* E ih₀
  μIHs₀-refl* (E ∷ Es) {xs₀ = inr _ , _} ih₀ = μIHs₀-refl* Es ih₀

  μIHs₀-sym*
    : (Es : ConDs I I)
    → ∀ {γ₀ γ₁ i₀ i₁}
    → {xs₀ : ∣μArgs∣ Ds Es i₀}
    → {xs₁ : ∣μArgs∣ Ds Es i₁}
    → {ih₀ : ∣μIHs∣ Es γ₀ xs₀} {ih₁ : ∣μIHs∣ Es γ₁ xs₁}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
    → (ih₀₁ : ∣μIHs~∣ Es γ₀₁ xs₀₁ ih₀ ih₁)
    → ∣μIHs~∣ Es (Γ .sym γ₀₁) (μArgs₀-sym* Es xs₀₁) ih₁ ih₀
  μIHs₀-sym* (E ∷ Es) {xs₀ = inl _ , _} {inl _ , _} γ₀₁ xs₀₁ ih₀₁ = μIH₀-sym* E γ₀₁ tt xs₀₁ ih₀₁
  μIHs₀-sym* (E ∷ Es) {xs₀ = inr _ , _} {inr _ , _} γ₀₁ xs₀₁ ih₀₁ = μIHs₀-sym* Es γ₀₁ xs₀₁ ih₀₁

  μIHs₀-trans*
    : (Es : ConDs I I)
    → ∀ {γ₀ γ₁ γ₂ i₀ i₁ i₂}
    → {xs₀ : ∣μArgs∣ Ds Es i₀}
    → {xs₁ : ∣μArgs∣ Ds Es i₁}
    → {xs₂ : ∣μArgs∣ Ds Es i₂}
    → {ih₀ : ∣μIHs∣ Es γ₀ xs₀} {ih₁ : ∣μIHs∣ Es γ₁ xs₁} {ih₂ : ∣μIHs∣ Es γ₂ xs₂}
    → (γ₀₁ : Γ ∋ γ₀ ~ γ₁) (γ₁₂ : Γ ∋ γ₁ ~ γ₂)
    → (xs₀₁ : ∣μArgs~∣ Ds Es xs₀ xs₁)
    → (xs₁₂ : ∣μArgs~∣ Ds Es xs₁ xs₂)
    → (ih₀₁ : ∣μIHs~∣ Es γ₀₁ xs₀₁ ih₀ ih₁)
    → (ih₁₂ : ∣μIHs~∣ Es γ₁₂ xs₁₂ ih₁ ih₂)
    → ∣μIHs~∣ Es (Γ .trans γ₀₁ γ₁₂) (μArgs₀-trans* Es xs₀₁ xs₁₂) ih₀ ih₂
  μIHs₀-trans* (E ∷ Es) {xs₀ = inl _ , _} {inl _ , _} {inl _ , _} γ₀₁ γ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂ =
    μIH₀-trans* E γ₀₁ γ₁₂ tt tt xs₀₁ xs₁₂ ih₀₁ ih₁₂
  μIHs₀-trans* (E ∷ Es) {xs₀ = inr _ , _} {inr _ , _} {inr _ , _} γ₀₁ γ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂ =
    μIHs₀-trans* Es γ₀₁ γ₁₂ xs₀₁ xs₁₂ ih₀₁ ih₁₂

  -- The IH type is the displayed induction hypothesis specialized to μ Ds.
  μIH
    : {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    → (E : ConD Δ (I [ ε ]) (I [ ε ]))
    → Ty (Γ ▷ I [ ε ] ▷ μArg E [ σ ↑t I [ ε ] ]) (ℓI ⊔ ℓ)
  ∣ μIH σ E ∣ ((γ , i) , xs) = ∣μIH∣ E γ (∣ σ ∣ γ) xs
  μIH σ E .heq ((γ₀₁ , i₀₁) , xs₀₁) ih₀ ih₁ = ∣μIH~∣ E γ₀₁ (σ .resp γ₀₁) xs₀₁ ih₀ ih₁
  μIH σ E .refl* {(γ , j) , xs} = μIH₀-refl* E
  μIH σ E .sym* {γ₀₁ = (γ₀₁ , j₀₁) , xs₀₁} = μIH₀-sym* E γ₀₁ (σ .resp γ₀₁) xs₀₁
  μIH σ E .trans* {γ₀₁ = (γ₀₁ , j₀₁) , xs₀₁} {(γ₁₂ , j₁₂) , xs₁₂} =
    μIH₀-trans* E γ₀₁ γ₁₂ (σ .resp γ₀₁) (σ .resp γ₁₂) xs₀₁ xs₁₂
  μIH σ E .coe {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    coeIH Ds (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
          (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
          (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
          (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
          (λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁))
          (λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁))
          E (σ .resp γ₀₁)
          xs₀₁
  μIH σ E .coh {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    cohIH Ds (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
          (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
          (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
          (P .trans*)
          (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
          (λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁))
          (λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁))
          E (σ .resp γ₀₁)
          xs₀₁
  {-# INJECTIVE_FOR_INFERENCE μIH #-}

  μIHs
    : (Es : ConDs I I)
    → Ty (Γ ▷ I [ ε ] ▷ μArgs Es) (ℓI ⊔ ℓ)
  ∣ μIHs Es ∣ ((γ , i) , xs) = ∣μIHs∣ Es γ xs
  μIHs Es .heq ((γ₀₁ , i₀₁) , xs₀₁) ih₀ ih₁ = ∣μIHs~∣ Es γ₀₁ xs₀₁ ih₀ ih₁
  μIHs Es .refl* = μIHs₀-refl* Es
  μIHs Es .sym* {γ₀₁ = (γ₀₁ , i₀₁) , xs₀₁} = μIHs₀-sym* Es γ₀₁ xs₀₁
  μIHs Es .trans* {γ₀₁ = (γ₀₁ , i₀₁) , xs₀₁} {(γ₁₂ , i₁₂) , xs₁₂} = μIHs₀-trans* Es γ₀₁ γ₁₂ xs₀₁ xs₁₂
  μIHs Es .coe {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    coeIHs Ds (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
           (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
           (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
           (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
           (λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁))
           Es
           xs₀₁
  μIHs Es .coh {γ₀ = (γ₀ , _) , _} {(γ₁ , _) , _} ((γ₀₁ , i₀₁) , xs₀₁) =
    cohIHs Ds (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₀ , i₀₁) , x₀₁))
           (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((Γ .refl γ₁ , i₀₁) , x₀₁))
           (λ extx₀ extx₁ i₀₁ x₀₁ → P .heq ((γ₀₁ , i₀₁) , x₀₁))
           (P .trans*)
           (λ _ _ px₀₁ px₀₀' px₀'₁' → P .trans* (P .sym* px₀₁) (P .trans* px₀₀' px₀'₁'))
           (λ i₀₁ x₀₁ → P .coe ((γ₀₁ , i₀₁) , x₀₁))
           (λ i₀₁ x₀₁ → P .coh ((γ₀₁ , i₀₁) , x₀₁))
           Es
           xs₀₁
  {-# INJECTIVE_FOR_INFERENCE μIHs #-}
