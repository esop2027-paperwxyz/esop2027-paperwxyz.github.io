private variable
  ℓI ℓJ : Level

-- Description code for one constructor.
data ConD (Δ : Con ℓΔ)
          (I : Ty ∅ ℓI)(D : Ty (∅ ▷ I) ℓ)
          (J : Ty ∅ ℓI)(E : Ty (∅ ▷ J) ℓ) : Set (lsuc (ℓ ⊔ ℓI ⊔ ℓΔ)) where
  end : (t : Tm Δ (J [ ε ]))(e : Tm Δ (E [ ε ▸ t ]))
    → ConD Δ I D J E
  non : (A : Ty Δ ℓI)(d : ConD (Δ ▷ A) I D J E)
    → ConD Δ I D J E
  arg1 : (f : Tm Δ (I [ ε ]))(d : ConD (Δ ▷ D [ ε ▸ f ]) I D J E)
    → ConD Δ I D J E
  arg : (A : Ty Δ ℓI)(f : Tm (Δ ▷ A) (I [ ε ])) (d : ConD (Δ ▷ Π A (D [ ε ▸ f ])) I D J E)
    → ConD Δ I D J E

-- Description codes for a list of constructors.
data ConDs (I : Ty ∅ ℓI)(D : Ty (∅ ▷ I) ℓ)(J : Ty ∅ ℓI)(E : Ty (∅ ▷ J) ℓ) : Set (lsuc (ℓ ⊔ ℓI)) where
  [] : ConDs I D J E
  _∷_  : (d : ConD ∅ I D J E) (Ds : ConDs I D J E) → ConDs I D J E

{-
-- Reindex a description along a substitution.
_[_]d : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {I J : Ty Γ ℓI} (d : ConD Γ I J)(σ : Sub Δ Γ)
  → ConD Δ (I [ σ ]) (J [ σ ]) 
end t     [ σ ]d = end (t [ σ ]t)
non A   D [ σ ]d = non (A [ σ ]) (D [ σ ↑ A ]d)
arg A f D [ σ ]d = arg (A [ σ ]) (f [ σ ⁺ ]t) (D [ σ ]d)
arg1 f  D [ σ ]d = arg1 (f [ σ ]t) (D [ σ ]d)

-- Instantiate a description in an extended context.
_⟨_⟩d : {Γ : Con ℓΓ} {A : Ty Γ ℓA} {I J : Ty (Γ ▷ A) ℓI}
  (d : ConD (Γ ▷ A) I J)(t : Tm Γ A)
  → ConD Γ (I ⟨ t ⟩)  (J ⟨ t ⟩)
D ⟨ t ⟩d = D [ idS ▸ t ]d
-}

-- Carrier type of one constructor's arguments.
Arg₀ : {Δ : Con ℓΔ}
       {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
       {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
       (d : ConD Δ I D J E)
  → (X : ∣ I ∣ tt → Set ℓI)
  → (X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI)
  → (T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i))
  → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
           → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
  → (δ : ∣ Δ ∣) → ∣ J ∣ tt → Set ℓI
Arg₀ {ℓI = ℓI} {J = J} (end t e) X X~ T Tresp δ j =
  LiftS lzero (J ∋ tt ⊢ ∣ t ∣ δ ~ j)
Arg₀ (non A d) X X~ T Tresp δ i = Σ[ a ∈ ∣ A ∣ δ ]
  Arg₀ d X X~ T Tresp ((δ , a)) i
Arg₀ (arg1 f d) X X~ T Tresp δ i =
  Σ[ x ∈ X (∣ f ∣ δ) ] Arg₀ d X X~ T Tresp (δ , T (∣ f ∣ δ) x) i
Arg₀ {ℓI = ℓI} {Δ = Δ} (arg A f d) X X~ T Tresp δ i =
  Σ (Σₚ[ g ∈ ((a : ∣ A ∣ δ) → X (∣ f ∣ (δ , a))) ] ({a₀ a₁ : ∣ A ∣ δ} → (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁) → X~ (g a₀) (g a₁))) (λ (g , gresp) → 
    Arg₀ d X X~ T Tresp (δ , ((λ a → T _ ((g a))) , λ {a₀} {a₁} a₀₁ → Tresp (f .resp (Δ .refl δ , a₀₁)) (gresp a₀₁))) i)
{-# INJECTIVE_FOR_INFERENCE Arg₀ #-}

ArgT₀ : {Δ : Con ℓΔ}
       {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
       {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
       (d : ConD Δ I D J E)
     → {X : ∣ I ∣ tt → Set ℓI}
     → {X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI}
     → {T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i)}
     → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
              → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
     → (δ : ∣ Δ ∣) → (j : ∣ J ∣ tt) → Arg₀ d X X~ T Tresp δ j → ∣ E ∣ (tt , j)
ArgT₀ {E = E} (end t e) Tresp δ j (↑ r) = E .coe (tt , r) (∣ e ∣ δ)
ArgT₀ (non A d) Tresp δ j (a , xs) = ArgT₀ d Tresp (δ , a) j xs
ArgT₀ (arg1 f d) Tresp δ j (x , xs) = ArgT₀ d Tresp (δ , _) j xs
ArgT₀ (arg A f d) Tresp δ j ((g , gresp) , xs) = ArgT₀ d Tresp (δ , _) j xs




-- Carrier type of a constructor list's arguments.
Args₀
  : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
    {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
    (ds : ConDs I D J E)
    (X : ∣ I ∣ tt → Set ℓI)
    (X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI)
    (T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i))
    (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
           → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
    → ∣ J ∣ tt → Set ℓI
Args₀ []       X X~ T Tresp j = Lift _ ⊥
Args₀ (d ∷ ds) X X~ T Tresp j = Arg₀ d X X~ T Tresp tt j ⊎ Args₀ ds X X~ T Tresp j
{-# INJECTIVE_FOR_INFERENCE Args₀ #-}

ArgsT₀ : {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
         {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
         (d : ConDs I D J E)
       → {X : ∣ I ∣ tt → Set ℓI}
       → {X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI}
       → {T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i)}
       → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
               → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
       → (j : ∣ J ∣ tt) → Args₀ d X X~ T Tresp j → ∣ E ∣ (tt , j)
ArgsT₀ (d ∷ ds) Tresp j (inl xs) = ArgT₀ d Tresp tt j xs
ArgsT₀ (d ∷ ds) Tresp j (inr y) = ArgsT₀ ds Tresp j y




-- Setoid relation on one constructor's arguments.
Arg~ : {Δ : Con ℓΔ}
       {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
       {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
       (d : ConD Δ I D J E)
     → (X : ∣ I ∣ tt → Set ℓI)
     → (X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI)
     → (T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i))
     → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
              → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
     → {δ₀ δ₁ : ∣ Δ ∣}(δ₀₁ : Δ ∋ δ₀ ~ δ₁)
     → {j₀ j₁ : ∣ J ∣ tt}
     → (xs₀ : Arg₀ d X X~ T Tresp δ₀ j₀) (xs₁ : Arg₀ d X X~ T Tresp δ₁ j₁) → Prop ℓI
Arg~ (end t e) X X~ T Tresp δ₀₁ xs₀ xs₁ = ⊤ₚ
Arg~ (non A d) X X~ T Tresp δ₀₁ (a₀ , xs₀) (a₁ , xs₁) =
 ∃[ a₀₁ ∈ (A ∋ δ₀₁ ⊢ a₀ ~ a₁) ]
   Arg~ d X X~ T Tresp (δ₀₁ , a₀₁) xs₀ xs₁
Arg~ (arg1 f d) X X~ T Tresp δ₀₁ (x₀ , xs₀) (x₁ , xs₁) =
 ∃[ x₀₁ ∈ X~ x₀ x₁ ]
  Arg~ d X X~ T Tresp ((δ₀₁ , Tresp (f .resp δ₀₁) x₀₁)) xs₀ xs₁
Arg~ (arg A f d) X X~ T Tresp {δ₀} {δ₁} δ₀₁ ((g₀ , gresp₀) , xs₀) ((g₁ , gresp₁) , xs₁) =
 ∃[ g₀₁ ∈ ({a₀ : ∣ A ∣ δ₀}{a₁ : ∣ A ∣ δ₁}(a₀₁ : A ∋ δ₀₁ ⊢ a₀ ~ a₁) → X~ (g₀ a₀) (g₁ a₁)) ]
   Arg~ d X X~ T Tresp (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) xs₀ xs₁
{-# INJECTIVE_FOR_INFERENCE Arg~ #-}

-- Setoid relation on a constructor list's arguments.
Args~ :{I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
       {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
       (ds : ConDs I D J E)
     → (X : ∣ I ∣ tt → Set ℓI)
     → {X~ : {i₀ i₁ : ∣ I ∣ tt} → X i₀ → X i₁ → Prop ℓI}
     → (T : (i : ∣ I ∣ tt) → X i → ∣ D ∣ (tt , i))
     → (Tresp : {i₀ i₁ : ∣ I ∣ tt}{x₀ : X i₀}{x₁ : X i₁}
              → (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁) → (x₀₁ : X~ x₀ x₁) → D ∋ tt , i₀₁ ⊢ T i₀ x₀ ~ T i₁ x₁)
     → {j₀ j₁ : ∣ J ∣ tt}
     → (xs₀ : Args₀ ds X X~ T Tresp j₀) (xs₁ : Args₀ ds X X~ T Tresp j₁) → Prop ℓI
Args~ (d ∷ ds) X {X~} T Tresp (inl xs₀) (inl xs₁) = Arg~ d X X~ T Tresp tt xs₀ xs₁
Args~ (d ∷ ds) X T Tresp (inr xs₀) (inr xs₁) = Args~ ds X T Tresp xs₀ xs₁
Args~ (d ∷ ds) X T Tresp (inl xs₀) (inr xs₁) = ⊥ₚ
Args~ (d ∷ ds) X T Tresp (inr xs₀) (inl xs₁) = ⊥ₚ

--                  Fixed-point: the inductive type                   --
-----------------------------------------------------------------------

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}(ds : ConDs I D I D) where

  -- Underlying inductive-recursive family generated by Ds.
  mutual

    data μ₀ (i : ∣ I ∣ tt) : Set ℓI where
      con : Args₀ ds μ₀ μ~ T₀ Tresp₀ i → μ₀ i

    -- Equivalence relation by recursion
    μ~ : ∀ {i₀ i₁} → μ₀ i₀ → μ₀ i₁ → Prop ℓI
    μ~ (con xs₀) (con xs₁) = Args~ₛ ds xs₀ xs₁

    T₀ : (i : ∣ I ∣ tt) → μ₀ i → ∣ D ∣ (tt , i)
    T₀ i (con xs) = ArgsTₛ ds i xs

    Tresp₀ : {i₀ i₁ : ∣ I ∣ tt} {x₀ : μ₀ i₀} {x₁ : μ₀ i₁}
           → (i₀₁ : I ∋ tt ~ tt ⊢ i₀ ~ i₁) → μ~ x₀ x₁
           → D ∋ tt , i₀₁ ⊢ T₀ i₀ x₀ ~ T₀ i₁ x₁
    Tresp₀ {x₀ = con xs₀} {con xs₁} i₀₁ xs₀₁ = ArgsTrespₛ ds i₀₁ xs₀₁

    -- specialised version of ArgT₀, for the benefit of the termination checker
    ArgTₛ : {Δ : Con ℓΔ}(d : ConD Δ I D I D)
          → (δ : ∣ Δ ∣) → (i : ∣ I ∣ tt)
          → Arg₀ d μ₀ μ~ T₀ Tresp₀ δ i  → ∣ D ∣ (tt , i)
    ArgTₛ (end t e) δ i (↑ r) = D .coe (tt , r) (∣ e ∣ δ)
    ArgTₛ (non A d) δ i (a , xs) = ArgTₛ d (δ , a) i xs
    ArgTₛ (arg1 f d) δ i (x , xs) = ArgTₛ d (δ , _) i xs
    ArgTₛ (arg A f d) δ i ((g , gresp) , xs) = ArgTₛ d (δ , _) i xs

    ArgsTₛ : (ds : ConDs I D I D) →
             (i : ∣ I ∣ tt) → Args₀ ds μ₀ μ~ T₀ Tresp₀ i → ∣ D ∣ (tt , i)
    ArgsTₛ (d ∷ ds) i (inl xs) = ArgTₛ d tt i xs
    ArgsTₛ (d ∷ ds) i (inr xs) = ArgsTₛ ds i xs

    Arg~ₛ : {Δ : Con ℓΔ}
           (d : ConD Δ I D I D)
         → {δ₀ δ₁ : ∣ Δ ∣}(δ₀₁ : Δ ∋ δ₀ ~ δ₁)
         → {j₀ j₁ : ∣ I ∣ tt}
         → (xs₀ : Arg₀ d μ₀ μ~ T₀ Tresp₀ δ₀ j₀) (xs₁ : Arg₀ d μ₀ μ~ T₀ Tresp₀ δ₁ j₁) → Prop ℓI
    Arg~ₛ (end t e) δ₀₁ xs₀ xs₁ = ⊤ₚ
    Arg~ₛ (non A d) δ₀₁ (a₀ , xs₀) (a₁ , xs₁) =
     ∃[ a₀₁ ∈ (A ∋ δ₀₁ ⊢ a₀ ~ a₁) ]
       Arg~ₛ d (δ₀₁ , a₀₁) xs₀ xs₁
    Arg~ₛ (arg1 f d) δ₀₁ (x₀ , xs₀) (x₁ , xs₁) =
     ∃[ x₀₁ ∈ μ~ x₀ x₁ ]
      Arg~ₛ d ((δ₀₁ , Tresp₀ {x₀ = x₀} {x₁} (f .resp δ₀₁) x₀₁)) xs₀ xs₁
    Arg~ₛ (arg A f d) {δ₀} {δ₁} δ₀₁ ((g₀ , gresp₀) , xs₀) ((g₁ , gresp₁) , xs₁) =
     ∃[ g₀₁ ∈ ({a₀ : ∣ A ∣ δ₀}{a₁ : ∣ A ∣ δ₁}(a₀₁ : A ∋ δ₀₁ ⊢ a₀ ~ a₁) → μ~ (g₀ a₀) (g₁ a₁)) ]
       Arg~ₛ d (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ {x₀ = g₀ a₀} {g₁ a₁} (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) xs₀ xs₁
    {-# INJECTIVE_FOR_INFERENCE Arg~ₛ #-}

    -- Setoid relation on a constructor list's arguments.
    Args~ₛ : (ds : ConDs I D I D)
         → {j₀ j₁ : ∣ I ∣ tt}
         → (xs₀ : Args₀ ds μ₀ μ~ T₀ Tresp₀ j₀) (xs₁ : Args₀ ds μ₀ μ~ T₀ Tresp₀ j₁) → Prop ℓI
    Args~ₛ (d ∷ ds) (inl xs₀) (inl xs₁) = Arg~ₛ d tt xs₀ xs₁
    Args~ₛ (d ∷ ds) (inr xs₀) (inr xs₁) = Args~ₛ ds xs₀ xs₁
    Args~ₛ (d ∷ ds) (inl xs₀) (inr xs₁) = ⊥ₚ
    Args~ₛ (d ∷ ds) (inr xs₀) (inl xs₁) = ⊥ₚ
    {-# INJECTIVE_FOR_INFERENCE Args~ₛ #-}

    ArgTrespₛ : {Δ : Con ℓΔ}(d : ConD Δ I D I D)
              → {δ₀ δ₁ : ∣ Δ ∣}(δ₀₁ : Δ ∋ δ₀ ~ δ₁)
              → {i₀ i₁ : ∣ I ∣ tt}(i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
              → {xs₀ : Arg₀ d μ₀ μ~ T₀ Tresp₀ δ₀ i₀}
              → {xs₁ : Arg₀ d μ₀ μ~ T₀ Tresp₀ δ₁ i₁}
              → Arg~ₛ d δ₀₁ xs₀ xs₁
              → D ∋ tt , i₀₁ ⊢ ArgTₛ d δ₀ i₀ xs₀ ~ ArgTₛ d δ₁ i₁ xs₁
    ArgTrespₛ (end t e) {δ₀} {δ₁} δ₀₁ i₀₁ {↑ r₀} {↑ r₁} _ = D .trans* (coh' D (tt , r₀) (∣ e ∣ δ₀)) (D .trans* (e .resp δ₀₁) (coh D (tt , r₁) (∣ e ∣ δ₁)))
    ArgTrespₛ (non A d) δ₀₁ i₀₁ (a₀₁ , xs₀₁) = ArgTrespₛ d (δ₀₁ , a₀₁) i₀₁ xs₀₁
    ArgTrespₛ (arg1 f d) δ₀₁ i₀₁ {x₀ , _} (x₀₁ , xs₀₁) = ArgTrespₛ d (δ₀₁ , Tresp₀ {x₀ = x₀} (f .resp δ₀₁) x₀₁) i₀₁ xs₀₁
    ArgTrespₛ (arg A f d) δ₀₁ i₀₁ {(g₀ , _) , _} (g₀₁ , xs₀₁) = ArgTrespₛ d (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) i₀₁ xs₀₁

    ArgsTrespₛ : (ds : ConDs I D I D)
               → {i₀ i₁ : ∣ I ∣ tt}(i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
               → {xs₀ : Args₀ ds μ₀ μ~ T₀ Tresp₀ i₀}
               → {xs₁ : Args₀ ds μ₀ μ~ T₀ Tresp₀ i₁}
               → Args~ₛ ds xs₀ xs₁
               → D ∋ tt , i₀₁ ⊢ ArgsTₛ ds i₀ xs₀ ~ ArgsTₛ ds i₁ xs₁
    ArgsTrespₛ (d ∷ ds) i₀₁ {inl xs₀} {inl xs₁} xs₀₁ = ArgTrespₛ d tt i₀₁ xs₀₁
    ArgsTrespₛ (d ∷ ds) i₀₁ {inr xs₀} {inr xs₁} xs₀₁ = ArgsTrespₛ ds i₀₁ xs₀₁

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D} where
  mutual
    μ~-refl* : ∀ {i} x → μ~ ds {i} {i} x x
    μ~-refl* (con xs) = Args₀-refl* ds xs

    Args₀-refl* : (es : ConDs I D I D)
      → ∀ {i} (xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
      → Args~ₛ ds es xs₀ xs₀
    Args₀-refl* (e ∷ es) (inl xs) = Arg₀-refl* e xs
    Args₀-refl* (e ∷ es) (inr ys) = Args₀-refl* es ys

    Arg₀-refl*
      : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
      → ∀ {δ i} (xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i)
      → Arg~ₛ ds e (Δ .refl δ) xs₀ xs₀
    Arg₀-refl* (end t d) xs = tt
    Arg₀-refl* (non A e) (a , xs) = A .refl* a , Arg₀-refl* e xs
    Arg₀-refl* (arg1 f e) (x , xs) = (μ~-refl* x) , Arg₀-refl* e xs
    Arg₀-refl* (arg A f e) ((g , gresp) , xs) = gresp , Arg₀-refl* e xs

  mutual
    μ~-sym*
      : ∀ {i₀ x₀}{i₁ x₁}
      → (x₀₁ : μ~ ds {i₀} {i₁} x₀ x₁)
      → μ~ ds x₁ x₀
    μ~-sym* {x₀ = con xs₀} {x₁ = con xs₁} x₀₁ = Args₀-sym* ds x₀₁

    Args₀-sym*
      : (es : ConDs I D I D)
      → ∀ {i₀ i₁}
      → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
      → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
      → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
      → Args~ₛ ds es xs₁ xs₀
    Args₀-sym* (e ∷ es) {xs₀ = inl _} {inl _} xs₀₁ = Arg₀-sym* e xs₀₁
    Args₀-sym* (e ∷ es) {xs₀ = inr _} {inr _} xs₀₁ = Args₀-sym* es xs₀₁

    Arg₀-sym*
      : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
      → ∀ {δ₀ δ₁ i₀ i₁ xs₀ xs₁} {δ₀₁ : Δ ∋ δ₀ ~ δ₁}
      → (xs₀₁ : Arg~ₛ ds e δ₀₁ {i₀} {i₁} xs₀ xs₁)
      → Arg~ₛ ds e (Δ .sym δ₀₁) xs₁ xs₀
    Arg₀-sym* (end t d) xs₀₁ = tt
    Arg₀-sym* (non A e) (a₀₁ , xs₀₁) = A .sym* a₀₁ , Arg₀-sym* e xs₀₁
    Arg₀-sym* (arg1 f e) {xs₀ = x₀ , _} (x₀₁ , xs₀₁) = μ~-sym* {x₀ = x₀} x₀₁ , Arg₀-sym* e xs₀₁
    Arg₀-sym* (arg A f e) {xs₀ = (g₀ , _) , _} (g₀₁ , xs₀₁) = (λ {a₀} {a₁} a₀₁ → μ~-sym* {x₀ = g₀ a₁} (g₀₁ (A .sym* a₀₁))) , Arg₀-sym* e xs₀₁

  mutual
    μ~-trans*
      : ∀ {i₀ i₁ i₂ x₀ x₂}
      → {x₁ : μ₀ ds i₁}
      → (x₀₁ : μ~ ds {i₀} {i₁} x₀ x₁)
      → (x₁₂ : μ~ ds {i₁} {i₂} x₁ x₂)
      → μ~ ds x₀ x₂
    μ~-trans* {x₀ = con xs₀} {x₂ = con xs₂} {x₁ = con xs₁} xs₀₁ xs₁₂ =
      Args₀-trans* ds xs₀₁ xs₁₂

    Args₀-trans*
      : (es : ConDs I D I D)
      → ∀ {i₀ i₁ i₂}
      → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
      → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
      → {xs₂ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₂}
      → (xs₀₁ : Args~ₛ ds es xs₀ xs₁)
      → (xs₁₂ : Args~ₛ ds es xs₁ xs₂)
      → Args~ₛ ds es xs₀ xs₂
    Args₀-trans* (e ∷ es) {xs₀ = inl _} {inl _} {inl _} xs₀₁ xs₁₂ = Arg₀-trans* e xs₀₁ xs₁₂
    Args₀-trans* (e ∷ es) {xs₀ = inr _} {inr _} {inr _} xs₀₁ xs₁₂ = Args₀-trans* es xs₀₁ xs₁₂

    Arg₀-trans*
      : {Δ : Con ℓΔ} (e : ConD Δ I D I D)
      → ∀ {δ₀ δ₁ δ₂ i₀ i₁ i₂ xs₀ xs₁ xs₂} {δ₀₁ : Δ ∋ δ₀ ~ δ₁} {δ₁₂ : Δ ∋ δ₁ ~ δ₂}
      → (xs₀₁ : Arg~ₛ ds e δ₀₁ {i₀} {i₁} xs₀ xs₁)
      → (xs₁₂ : Arg~ₛ ds e δ₁₂ {i₁} {i₂} xs₁ xs₂)
      → Arg~ₛ ds e (Δ .trans δ₀₁ δ₁₂) xs₀ xs₂
    Arg₀-trans* (end t d) xs₀₁ xs₁₂ = tt
    Arg₀-trans* (non A e) (a₀₁ , xs₀₁) (a₁₂ , xs₁₂) = A .trans* a₀₁ a₁₂ , Arg₀-trans* e xs₀₁ xs₁₂
    Arg₀-trans* (arg1 f e) {xs₀ = x₀ , _} (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) = μ~-trans* {x₀ = x₀}  x₀₁ x₁₂ , Arg₀-trans* e xs₀₁ xs₁₂
    Arg₀-trans* (arg A f e) {xs₀ = (g₀ , _) , _} {δ₀₁ = δ₀₁} (g₀₁ , xs₀₁) (g₁₂ , xs₁₂) =
     (λ {a₀} a₀₂ → μ~-trans* {x₀ = g₀ a₀} (g₀₁ (A .coh δ₀₁ _)) (g₁₂ (A .trans* (coh' A δ₀₁ _) a₀₂)))
     , Arg₀-trans* e xs₀₁ xs₁₂

-- The inductive type μ

-- To build μ, we specialize coercion and coherence to μ₀.

  mutual
   -- These definitions are specialized to μ Ds so that recursion terminates.
   coeArg₀
     : {Δ : Con ℓΔ}
     → (e : ConD Δ I D I D)
     → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
     → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
     → (xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀)
     → Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ i₁
   coeArg₀ {Δ = Δ} (end t d) δ₀₁ i₀₁ (↑ q) =
    ↑ (I .trans* (t .resp (Δ .sym δ₀₁)) (I .trans* q i₀₁))
   coeArg₀ (non A e) δ₀₁ i₀₁ (a₀ , xs₀) =
    A .coe δ₀₁ a₀ , coeArg₀ e (δ₀₁ , A .coh δ₀₁ a₀) i₀₁ xs₀
   coeArg₀ (arg1 f e) δ₀₁ i₀₁ (x₀ , xs₀) =
    coeμ₀ (f .resp δ₀₁) x₀ , coeArg₀ e (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) (cohμ₀ _ x₀)) i₀₁ xs₀
   coeArg₀ {Δ = Δ} (arg A f e) δ₀₁ i₀₁ ((g₀ , gresp₀) , xs₀) =
    ((λ a₁ → coeμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁)) (g₀ (A .coe (Δ .sym δ₀₁) a₁)))
    , (λ {a₁} {a₁'} a₁₁ → μ~-trans* {x₀ = coeμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁)) (g₀ (A .coe (Δ .sym δ₀₁) a₁))} {x₁ = g₀ (A .coe (Δ .sym δ₀₁) a₁')}
                                 (μ~-trans* {x₀ = coeμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁)) (g₀ (A .coe (Δ .sym δ₀₁) a₁))} {x₁ = g₀ (A .coe (Δ .sym δ₀₁) a₁)}
                                   (μ~-sym* {x₀ = g₀ (A .coe (Δ .sym δ₀₁) a₁)} (cohμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁)) (g₀ (A .coe (Δ .sym δ₀₁) a₁))))
                                   (gresp₀ (A .trans* (coh' A (Δ .sym δ₀₁) a₁) (A .trans* a₁₁ (coh A (Δ .sym δ₀₁) a₁')))))
                                 (cohμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁')) (g₀ (A .coe (Δ .sym δ₀₁) a₁')))))
    , coeArg₀ e (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (μ~-trans* {x₀ = g₀ a₀} (gresp₀ (A .trans* a₀₁ (coh A (Δ .sym δ₀₁) a₁))) (cohμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁)) (g₀ (A .coe (Δ .sym δ₀₁) a₁))))) i₀₁ xs₀

   coeArgs₀
     : (es : ConDs I D I D)
     → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
     → (xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀)
     → Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁
   coeArgs₀ (e ∷ es) i₀₁ (inl xs₀) = inl (coeArg₀ e tt i₀₁ xs₀)
   coeArgs₀ (e ∷ es) i₀₁ (inr xs₀) = inr (coeArgs₀ es i₀₁ xs₀)

   coeμ₀
     : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
     → (x₀ : μ₀ ds i₀)
     → μ₀ ds i₁
   coeμ₀ i₀₁ (con xs₀) = con (coeArgs₀ ds i₀₁ xs₀)

   cohArg₀
     : {Δ : Con ℓΔ}
     → (e : ConD Δ I D I D)
     → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
     → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
     → (xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀)
     → Arg~ₛ ds e δ₀₁ xs₀ (coeArg₀ e δ₀₁ i₀₁ xs₀)
   cohArg₀ (end t e) δ₀₁ i₀₁ xs₀ = tt
   cohArg₀ (non A e) δ₀₁ i₀₁ (a₀ , xs₀) = coh A δ₀₁ a₀ , cohArg₀ e (δ₀₁ , coh A δ₀₁ a₀) i₀₁ xs₀
   cohArg₀ (arg1 f e) δ₀₁ i₀₁ (x₀ , xs₀) =
    (cohμ₀ (f .resp δ₀₁) x₀) , cohArg₀ e (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) (cohμ₀ (f .resp δ₀₁) x₀)) i₀₁ xs₀
   cohArg₀ {Δ = Δ} (arg A f e) δ₀₁ i₀₁ ((g₀ , gresp₀) , xs₀) =
    (λ {a₀} {a₁} a₀₁ → μ~-trans* {x₀ = g₀ a₀} (gresp₀ (A .trans* a₀₁ (coh A (Δ .sym δ₀₁) a₁))) (cohμ₀ (f .resp (δ₀₁ , coh' A (Δ .sym δ₀₁) a₁)) (g₀ (A .coe (Δ .sym δ₀₁) a₁))))
    , cohArg₀ e _ i₀₁ xs₀

   cohArgs₀
     : (es : ConDs I D I D)
     → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
     → (xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀)
     → Args~ₛ ds es xs₀ (coeArgs₀ es i₀₁ xs₀)
   cohArgs₀ (e ∷ es) i₀₁ (inl xs₀) = cohArg₀ e tt i₀₁ xs₀
   cohArgs₀ (e ∷ es) i₀₁ (inr xs₀) = cohArgs₀ es i₀₁ xs₀

   cohμ₀
     : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
     → (x₀ : μ₀ ds i₀)
     → μ~ ds x₀ (coeμ₀ i₀₁ x₀)
   cohμ₀ i₀₁ (con xs₀) = cohArgs₀ ds i₀₁ xs₀

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}(ds : ConDs I D I D) where

  -- Inductive type generated by ss.
  μ : {Γ : Con ℓΓ} → Ty (Γ ▷ I [ ε ]) ℓI
  ∣ μ ∣ (γ , i)        = μ₀ ds i
  μ .heq _             = μ~ ds
  μ .refl*             = μ~-refl*
  μ .sym* {x₀ = x₀}    = μ~-sym* {x₀ = x₀}
  μ .trans*  {x₀ = x₀} = μ~-trans* {x₀ = x₀}
  μ .coe (_ , i₀₁)     = coeμ₀ i₀₁
  μ .coh (_ , i₀₁)     = cohμ₀ i₀₁
  {-# INJECTIVE_FOR_INFERENCE μ #-}

  T : {Γ : Con ℓΓ} → Tm (Γ ▷ I [ ε ] ▷ μ) (D [ ε ▸ vs vz ])
  T .∣_∣ ((γ , i) , x) = T₀ ds i x
  T .resp {(γ₀ , i₀) , x₀} ((γ₀₁ , i₀₁) , x₀₁) = Tresp₀ ds {x₀ = x₀} i₀₁ x₀₁

-- Constructor arguments specialized to μ.

module _ {Δ : Con ℓΔ} {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D} (e : ConD Δ I D I D) where

  -- Type of arguments for one constructor.
  Arg : Ty (Δ ▷ I [ ε ]) ℓI
  ∣ Arg ∣ (δ , i) = Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i
  Arg .heq (δ₀₁ , i₀₁) = Arg~ₛ ds e δ₀₁
  Arg .refl* = Arg₀-refl* e
  Arg .sym* = Arg₀-sym* e
  Arg .trans* = Arg₀-trans* e
  Arg .coe (δ₀₁ , i₀₁) = coeArg₀ e δ₀₁ i₀₁
  Arg .coh (δ₀₁ , i₀₁) = cohArg₀ e δ₀₁ i₀₁

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D} (es : ConDs I D I D) where

  -- Type of arguments for a constructor list.
  Args : Ty (Γ ▷ I [ ε ]) ℓI
  ∣ Args ∣ (_ , i) = Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i
  Args .heq (_ , i₀₁) = Args~ₛ ds es
  Args .refl* = Args₀-refl* es
  Args .sym* = Args₀-sym* es
  Args .trans* = Args₀-trans* es
  Args .coe (_ , i₀₁) = coeArgs₀ es i₀₁
  Args .coh (_ , i₀₁) = cohArgs₀ es i₀₁

  ArgsT : Tm (Γ ▷ I [ ε ] ▷ Args) (D [ ε ▸ vs vz ])
  ∣ ArgsT ∣ ((γ , i) , xs) = ArgsTₛ ds es i xs
  ArgsT .resp ((γ₀₁ , i₀₁) , xs₀₁) = ArgsTrespₛ ds es i₀₁ xs₀₁ 

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D} where

  -- The empty list-of-constructors is interpreted as Empty.
  emptyArgsᵢ : {Γ : Con ℓΓ} {A : Ty _ (ℓA ⊔ ℓI)} → Tm (Γ ▷ I [ ε ] ▷ Args {ds = ds} []) A
  ∣ emptyArgsᵢ ∣ ((γ , i) , ())
  emptyArgsᵢ .resp {γ₀ = (γ₀ , i₀) , ()}

  -- Inject one constructor's arguments into a larger argument list.
  inlArgsᵢ
    : {Γ : Con ℓΓ} (e : ConD ∅ I D I D) (es : ConDs I D I D)
    → Tm (Γ ▷ I [ ε ] ▷ Arg {ds = ds} e [ ε ⁺ ])
         (Args {ds = ds} (e ∷ es) [ wk ])
  ∣ inlArgsᵢ e es ∣ ((γ , i) , xs) = inl xs
  inlArgsᵢ e es .resp ((γ₀₁ , i₀₁) , xs₀₁) = xs₀₁

  -- Inject tail arguments into a larger argument list.
  inrArgsᵢ
    : {Γ : Con ℓΓ} (e : ConD ∅ I D I D) (es : ConDs I D I D)
    → Tm (Γ ▷ I [ ε ] ▷ Args {ds = ds} es)
         (Args {ds = ds} (e ∷ es) [ wk ])
  ∣ inrArgsᵢ e es ∣ ((γ , i) , xs) = inr xs
  inrArgsᵢ e es .resp ((γ₀₁ , i₀₁) , xs₀₁) = xs₀₁

-- Constructors for the inductive type μ

-- Generic constructor from semantic argument lists.
conᵢ
  : {Γ : Con ℓΓ}{I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}(ds : ConDs I D I D)
  → Tm (Γ ▷ I [ ε ] ▷ Args ds [ ε ⁺ ]) (μ ds [ wk ])
∣ conᵢ ds ∣ ((γ , i) , xs) = con xs
conᵢ ds .resp ((γ₀₁ , i₀₁) , t₀₁) = t₀₁

T-β : {Γ : Con ℓΓ}{I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}(ds : ConDs I D I D)
    → T ds [ wk ▸ conᵢ {Γ = Γ} ds ]t ≡ ArgsT ds
T-β ds = Eq.refl

-- Pointwise form of the generic constructor.
conₑ
  : {Γ : Con ℓΓ}{I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}(ds : ConDs I D I D)
  → (i : Tm Γ (I [ ε ]))
  → (xs : Tm Γ (Args ds ⟨ i ⟩))
  → Tm Γ (μ ds ⟨ i ⟩)
conₑ ds i t = conᵢ ds [ idS ▸ i ▸ t ]t

-- Same as T-β, written in external form
T-βₑ : {Γ : Con ℓΓ}{I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}(ds : ConDs I D I D)
    → (i : Tm Γ (I [ ε ]))
    → (xs : Tm Γ (Args ds ⟨ i ⟩))
    → T ds [ idS ▸ _ ▸ conₑ ds i xs ]t ≡ ArgsT ds [ idS ▸ i ▸ xs ]t
T-βₑ ds i xs = Eq.refl

-- Recover index equality from equal constructor arguments.
module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D} where

  Arg~→I~
    : {Δ : Con ℓΔ}
    → (e : ConD Δ I D I D)
    → ∀ {δ₀ δ₁ i₀ i₁}
    → {xs₀ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀ i₀}
    → {xs₁ : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₁ i₁}
    → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : Arg~ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ₀₁ {i₀} {i₁} xs₀ xs₁)
    → I ∋ tt ⊢ i₀ ~ i₁
  Arg~→I~ (end t e) {xs₀ = ↑ q₀} {↑ q₁} δ₀₁ xs₀₁ =
   I .trans* (I .sym* q₀) (I .trans* (t .resp δ₀₁) q₁)
  Arg~→I~ (non A e)   δ₀₁ (a₀₁ , xs₀₁) = Arg~→I~ e (δ₀₁ , a₀₁) xs₀₁
  Arg~→I~ (arg1 f e)  {xs₀ = x₀ , _} δ₀₁ (x₀₁ , xs₀₁) = Arg~→I~ e (δ₀₁ , Tresp₀ ds {x₀ = x₀} (f .resp δ₀₁) x₀₁) xs₀₁
  Arg~→I~ (arg A f e) {xs₀ = (g₀ , _) , _} δ₀₁ (g₀₁ , xs₀₁) = Arg~→I~ e (δ₀₁ , λ a₀ a₁ a₀₁ → Tresp₀ ds {x₀ = g₀ a₀} (f .resp (δ₀₁ , a₀₁)) (g₀₁ a₀₁)) xs₀₁

  -- Recover index equality from equal constructor-list arguments.
  Args~→I~
    : ∀ {i₀ i₁} (es : ConDs I D I D)
   → {xs₀ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₀}
   → {xs₁ : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i₁}
   → (xs₀₁ : Args~ es (μ₀ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) {i₀} {i₁} xs₀ xs₁)
   → I ∋ tt ⊢ i₀ ~ i₁
  Args~→I~ (e ∷ es) {xs₀ = inl xs₀} {inl xs₁} xs₀₁ = Arg~→I~ e {xs₀ = xs₀} {xs₁} tt xs₀₁
  Args~→I~ (e ∷ es) {xs₀ = inr _} {inr _} xs₀₁ = Args~→I~ es xs₀₁
