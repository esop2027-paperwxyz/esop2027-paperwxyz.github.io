-- used for the Regularity of μ, μArg, IH, etc.
open import Prelude.ITT-ITT+Funext
  using (funext; funext²)

open Eq using (Jₚ; Jₚ²)

private variable
  ℓI ℓE : Level

data ConDIsReg {Δ : Con ℓΔ} {I J : Ty ∅ ℓI}
  : (D : ConD Δ I J) → Set (lsuc (ℓI ⊔ ℓΔ)) where

  end  : ∀ {t} → ConDIsReg (end t)
  non  : ∀ {A D} (A-isReg : IsReg Δ A) → ConDIsReg D → ConDIsReg (non A D)
  arg1 : ∀ {f D} → ConDIsReg D → ConDIsReg (arg1 f D)
  arg : ∀ {A f D} → (A-isReg : IsReg Δ A)
    → ConDIsReg D → ConDIsReg (arg A f D)

data ConDsAreReg {I : Ty ∅ ℓI}
  : (Ds : ConDs I I) → Set (lsuc ℓI)
  where

  [] : ConDsAreReg []
  _∷_ : ∀ {D Ds} → (D-isReg : ConDIsReg D) → ConDsAreReg Ds → ConDsAreReg (D ∷ Ds)

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} (Ds-areReg : ConDsAreReg Ds)
  where
  
  mutual
    coeμ₀-refl
      : {i : ∣ I ∣ tt}
      → (x : μ₀ Ds i)
      → coeμ₀ (I .refl* i) x ≡ x
    coeμ₀-refl {i} (con xs) =
      Jₚ
        (λ ys _ →
          coeμ₀ (I .refl* i) (con xs) ≡
          con ys)
        Eq.refl
        _
        (coeArgs₀-refl Ds-areReg xs)

    -- Pattern match on `_⊎_` to reveal the decreasing argument.
    coeArgs₀-refl
      : {Es : ConDs I I}
      → ConDsAreReg Es
      → {i : ∣ I ∣ tt}
      → (xs : Args₀ Es (μ₀ Ds) i)
      → coeArgs₀ Es (I .refl* i) xs ≡ xs
    coeArgs₀-refl {E ∷ Es} (E-isReg ∷ Es-areReg) {i} (inr xs) =
      Eq.cong inr (coeArgs₀-refl Es-areReg xs)
    coeArgs₀-refl {E ∷ Es} (E-isReg ∷ Es-areReg) {i} (inl xs) =
      Eq.cong inl (coeArg₀-refl E E-isReg xs)

    coeArg₀-refl
      : {Δ : Con ℓΔ}
      → (E : ConD Δ (I [ ε ]) (I [ ε ]))
      → ConDIsReg E
      → {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
      → (xs : Arg₀ E (μ₀ Ds) δ i)
      → coeArg₀ E (Δ .refl δ) (I .refl* i) xs ≡ xs
    coeArg₀-refl (end t) end xs = Eq.refl
    coeArg₀-refl {Δ = Δ} (non A E) (non A-isReg E-isReg) {δ} {i} (a , xs) =
      Jₚ
        (λ ys _ →
            (A .coe (Δ .refl δ) a
            , coeArg₀ E
                (Δ .refl δ , A .coh (Δ .refl δ) a)
                (I .refl* i) xs)
          ≡
          (a , ys))
        (Jₚ
          (λ cast _ →
            (cohr : ∀ δ a →
              A ∋ Δ .refl δ ⊢
              a ~ cast δ a) →
            _≡_
              {A =
                (Σ[ a ∈ ∣ A ∣ δ ]
                  Arg₀ E (μ₀ Ds) (δ , a) i)}
              (A .coe (Δ .refl δ) a      , coeArg₀ E (Δ .refl δ , A .coh (Δ .refl δ) a) (I .refl* i) xs)
              (cast δ a , coeArg₀ E (Δ .refl δ , cohr δ a)             (I .refl* i) xs))
          (λ _ → Eq.refl)
          _
          A-isReg
          (λ δ a → A .refl* a))
        _
        (coeArg₀-refl E E-isReg xs)

    coeArg₀-refl {Δ = Δ} (arg1 f E) (arg1 E-isReg) {δ} {i} (x , xs) =
      Jₚ²
        (λ cast _ ys _ →
          _≡_
            {A =
              (Σ[ _ ∈ μ₀ Ds (∣ f ∣ δ) ]
                Arg₀ E (μ₀ Ds) δ i)}
            (coeμ₀ (f .resp (Δ .refl δ)) x , coeArg₀ E (Δ .refl δ) (I .refl* i) xs)
            (cast , ys))
        Eq.refl
        _
        (coeμ₀-refl x)
        _
        (coeArg₀-refl E E-isReg xs)

    coeArg₀-refl {Δ = Δ} (arg A f E) (arg A-isReg E-isReg) {δ} {i} (x , xs) =
      Jₚ²
        (λ cast _ ys _ →
          _≡_
            {A =
              (Σ[ _ ∈
                    ((a : ∣ A ∣ δ) →
                      μ₀ Ds (∣ f ∣ (δ , a))) ]
                Arg₀ E (μ₀ Ds) δ i)}
            ((λ a → coeμ₀ (f .resp (_ , coh' A (Δ .refl δ) a)) (x (A .coe (Δ .refl δ) a)))
              , coeArg₀ E (Δ .refl δ) (I .refl* i) xs)
            (cast , ys))
        Eq.refl
        _
        (Jₚ
          (λ cast _ →
              (λ a →
                coeμ₀
                  (f .resp
                    (_ , coh' A (Δ .refl δ) a))
                  (x (A .coe (Δ .refl δ) a)))
            ≡ cast)
          (Jₚ
            (λ cast _ →
              (cohr : ∀ δ a →
                A ∋ Δ .refl δ ⊢
                a ~ cast δ a) →
                (λ a →
                  coeμ₀
                    (f .resp
                      (_ , coh' A (Δ .refl δ) a))
                    (x (A .coe (Δ .refl δ) a)))
              ≡
                (λ a →
                  (coeμ₀
                    (f .resp
                      (_ , A .sym* (cohr δ a)))
                    (x
                      (cast δ a)))))
            (λ _ → Eq.refl)
            _
            A-isReg
            (λ δ a → A .refl* a))
          _
          (funext (λ a → coeμ₀-refl (x a))))
        _
        (coeArg₀-refl E E-isReg xs)

μArg-isReg
  : {I : Ty ∅ ℓI} {Ds : ConDs I I}
  → (Ds-areReg : ConDsAreReg Ds)
  → {Δ : Con ℓΔ} {E : ConD Δ (I [ ε ]) (I [ ε ])}
  → (E-isReg : ConDIsReg E)
  → IsReg (Δ ▷ I [ ε ]) (μArg {Ds = Ds} E)
μArg-isReg {I = I} {Ds} Ds-areReg {Δ} {E} E-isReg =
  Jₚ
    {A = Arg₀CoeTyˢ}
    {x = λ (δ , i) xs →
        (coeArg₀ E
          (Δ .refl δ) (I .refl* i) xs)}
    (λ cast _ →
      (extcast :
        ∀ δ i xs →
        extArg Ds E xs →
        extArg Ds E
          (cast (δ , i) xs)) →
      _≡_
        {A = CoeTy
          (Δ ▷ I [ ε ]) (μArg {Ds = Ds} E)}
        (CoeLHS
          (Δ ▷ I [ ε ]) (μArg {Ds = Ds} E))
        (λ (δ , i) (xs , extxs) →
            (cast (δ , i) xs
            , extcast δ i xs extxs)))
    (λ _ → Eq.refl)
    _
    (funext²
      (λ (δ , i) xs →
          (coeArg₀ E
            (Δ .refl δ) (I .refl* i) xs))
      (λ _ xs → xs)
      (λ (δ , i) xs →
        coeArg₀-refl Ds-areReg E E-isReg xs))
    (λ _ _ _ extxs → extxs)
  where
    Arg₀CoeTyˢ : Set _
    Arg₀CoeTyˢ =
      (γ : ∣ Δ ▷ I [ ε ] ∣) →
      (xs :
        (Arg₀ E
          (μ₀ Ds) (γ .₁) (γ .₂))) →
        (Arg₀ E
          (μ₀ Ds) (γ .₁) (γ .₂))

μ-isReg
  : {Γ : Con ℓΓ} {I : Ty ∅ ℓI} {Ds : ConDs I I}
  → (Ds-areReg : ConDsAreReg Ds)
  → IsReg (Γ ▷ I [ ε ]) (μ Ds)
μ-isReg {Γ = Γ} {I = I} {Ds} Ds-areReg =
  Jₚ
    {A = μ₀CoeTyˢ}
    {x = λ (_ , i) x →
      coeμ₀ (I .refl* i) x}
    (λ cast _ →
      (extcast :
        ∀ γ i x →
        extμ Ds x →
        extμ Ds
          (cast (γ , i) x)) →
      _≡_
        {A = CoeTy
          (Γ ▷ I [ ε ]) (μ Ds)}
        (CoeLHS
          (Γ ▷ I [ ε ]) (μ Ds))
        (λ (γ , i) (x , extx) →
            (cast (γ , i) x
            , extcast γ i x extx)))
    (λ _ → Eq.refl)
    _
    (funext²
      (λ (_ , i) x →
        coeμ₀ (I .refl* i) x)
      (λ _ x → x)
      (λ (_ , i) x →
        coeμ₀-refl Ds-areReg x))
    (λ _ _ _ extx → extx)
  where
    μ₀CoeTyˢ : Set _
    μ₀CoeTyˢ =
      (γ : ∣ Γ ▷ I [ ε ] ∣) →
      (x : (μ₀ Ds (γ .₂))) →
      μ₀ Ds (γ .₂)
