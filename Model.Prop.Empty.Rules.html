open import Prelude

module Model.Prop.Empty.Rules where

open import Model.Prop.CwF
open import Model.Prop.Empty.Core
open import Model.Prop.Empty.Elim

Empty[] : ∀ {ℓΓ ℓE} {Γ Δ : Con ℓΓ} {σ : Sub Δ Γ}
  → (Empty {ℓE = ℓE}) [ σ ] ≡ Empty {ℓE = ℓE}
Empty[] = Eq.refl

elim-Empty[]
  : ∀ {ℓΓ ℓE} {Γ Δ : Con ℓΓ} {σ : Sub Δ Γ}
  → {P : Ty (Γ ▷ Empty {ℓE = ℓE}) ℓ}
  → elim-Empty {ℓΓ = ℓΓ} {ℓE = ℓE} {ℓP = ℓ} {Γ = Γ} P
      [ σ ⁺ ]t
    ≡ elim-Empty {ℓΓ = ℓΓ} {ℓE = ℓE} {ℓP = ℓ} {Γ = Δ}
        (P [ σ ⁺ ])
elim-Empty[] = Eq.refl
