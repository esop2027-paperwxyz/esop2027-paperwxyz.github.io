-- Proof-Relevant Identity Type
module Prelude.Proof-Relevant-Identity where
  
open import Prelude.Base

private variable
  A B C : Set a

infix 4 _≡_
data _≡_ {A : Set a} (x : A) : A → Set a where
  refl : x ≡ x

sym : {x y : A} → x ≡ y → y ≡ x
sym refl = refl

trans : {x y z : A}
  → x ≡ y → y ≡ z
  → x ≡ z
trans refl p = p

cong : (f : A → B) {x y : A} → x ≡ y
  → f x ≡ f y
cong f refl = refl

cong₂ : (f : A → B → C) {x z : A} {y w : B}
  → x ≡ z → y ≡ w
  → f x y ≡ f z w
cong₂ f refl refl = refl

transp : ∀ {ℓ} {x y : A} (P : A → Set ℓ) → x ≡ y → P x → P y
transp P refl p = p

-- transp whose motive depends on the equality (and is in Prop)
dtranspₚ
  : ∀ {ℓ} {x : A}
  → (P : (y : A) → (eq : x ≡ y) → Prop ℓ)
  → {y : A} (eq : x ≡ y)
  → P x refl → P y eq
dtranspₚ P refl x = x

dcong
  : {A : Set a} {B : A → Set b}
  → (f : (x : A) → B x)
  → {x y : A} → (p : x ≡ y)
  → transp B p (f x) ≡ f y
dcong f refl = refl

Σₚ-≡ : {A : Set a} {B : A → Prop b} → ∀ {a b p q} → a ≡ b
  → _≡_ {_} {Σₚ A B} (a , p) (b , q)
Σₚ-≡ refl = refl

-- UIP/K used here.
transp-ir : ∀ {a b} {A : Set a} {x y : A} {eq eq' : x ≡ y}
  → (P : A → Set b)
  → (p : P x)
  → transp P eq p ≡ transp P eq' p
transp-ir {eq = refl} {refl} P p = refl

module ≡-Reasoning where
  -- Syntax for chains of equational reasoning (from Cubical Agda)
  infix  1 begin_
  infix  3 _∎
  infixr 2 step-≡ _≡⟨⟩_

  begin_ : {x y : A} → x ≡ y → x ≡ y
  begin q = q

  step-≡ : {y z : A} (x : A) → y ≡ z → x ≡ y → x ≡ z
  step-≡ _ p q = trans q p

  syntax step-≡ x y p = x ≡⟨ p ⟩ y

  _≡⟨⟩_ : {y : A} (x : A) → x ≡ y → x ≡ y
  _ ≡⟨⟩ x≡y = x≡y

  _∎ : (x : A) → x ≡ x
  _ ∎ = refl
