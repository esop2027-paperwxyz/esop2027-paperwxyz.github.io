open import Prelude

module Example.Universe.ClosedPi where

open import Prelude.StrongJ
open import Model.Prop.CwF
open import Model.Prop.Pi

mutual
  data Code (ℓ : Level) : Set (lsuc ℓ) where
    opaque₀ : (A : Ty ∅ ℓ) → Code ℓ
    pi₀
      : (A : Code ℓ) (B : El A → Code ℓ)
      → (Bext : ∀ {a₀ a₁} → HRel (reflU A) a₀ a₁ → EqU (B a₀) (B a₁))
      → Code ℓ

  El : Code ℓ → Set ℓ
  El (opaque₀ A) = ∣ A ∣ tt
  El (pi₀ A B Bext) =
    Σₚ[ f ∈ ((a : El A) → El (B a)) ]
      (∀ {a₀ a₁} (a₀₁ : HRel (reflU A) a₀ a₁) →
        HRel (Bext a₀₁) (f a₀) (f a₁))

  EqU : Code ℓ → Code ℓ → Prop (lsuc ℓ)
  EqU (opaque₀ A) (opaque₀ B) = A ≡ B
  EqU (opaque₀ A) (pi₀ B Q Qext) = ⊥ₚ
  EqU (pi₀ A P Pext) (opaque₀ B) = ⊥ₚ
  EqU (pi₀ A P Pext) (pi₀ B Q Qext) =
    ∃[ eAB ∈ EqU A B ]
      (∀ a b (ab : HRel eAB a b) → EqU (P a) (Q b))

  HRel : {A B : Code ℓ} → EqU A B → El A → El B → Prop ℓ
  HRel {A = opaque₀ A} {opaque₀ B} e x y =
    J (λ B e → ∣ B ∣ tt → Prop _) (A .heq tt x) B e y
  HRel {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ) f g =
    ∀ a b (ab : HRel eAB a b) → HRel (ePQ a b ab) (f .₁ a) (g .₁ b)

  reflU : (A : Code ℓ) → EqU A A
  reflU (opaque₀ A) = Eq.refl
  reflU (pi₀ A B Bext) =
    reflU A , λ a b ab → Bext ab

Rel : (A : Code ℓ) → El A → El A → Prop ℓ
Rel A = HRel (reflU A)

reflEl : (A : Code ℓ) (a : El A) → Rel A a a
reflEl (opaque₀ A) a = A .refl* a
reflEl (pi₀ A B Bext) f a b ab = f .₂ ab

mutual
  symU : {A B : Code ℓ} → EqU A B → EqU B A
  symU {A = opaque₀ A} {opaque₀ .A} Eq.refl = Eq.refl
  symU {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ) =
    symU eAB , λ b a ba → symU (ePQ a b (symEl (symU eAB) ba))

  symEl : ∀ {A B : Code ℓ} (e : EqU A B) {a b} → HRel e a b → HRel (symU e) b a
  symEl {A = opaque₀ A} {opaque₀ .A} Eq.refl ab = A .sym* ab
  symEl {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ) fg b a ba =
    symEl (ePQ a b (symEl (symU eAB) ba))
      (fg a b (symEl (symU eAB) ba))

record Ops {A B : Code ℓ} (eAB : EqU A B) : Set (lsuc ℓ)  where
  field
    comp-f : ∀ {C} → EqU B C → EqU A C
    comp-b : ∀ {C} → EqU A C → EqU B C

    cast-f : El A → El B
    cast-b : El B → El A

    trans-f
      : ∀ {C} (eBC : EqU B C) {a b c}
      → HRel eAB a b → HRel eBC b c → HRel (comp-f eBC) a c
    trans-b
      : ∀ {C} (eAC : EqU A C) {a b c}
      → HRel (symU eAB) b a → HRel eAC a c → HRel (comp-b eAC) b c

    cast-f-coh : (a : El A) → HRel eAB a (cast-f a)
    cast-b-coh : (b : El B) → HRel (symU eAB) b (cast-b b)

open Ops

opaqueOps : (A : Ty ∅ ℓ) → Ops {A = opaque₀ A} {opaque₀ A} Eq.refl
opaqueOps A = record
  { comp-f = ofwd
  ; comp-b = obwd
  ; cast-f = λ x → A .coe tt x
  ; cast-b = λ x → A .coe tt x
  ; trans-f = λ e → tf e
  ; trans-b = λ e → tb e
  ; cast-f-coh = λ x → A .coh tt x
  ; cast-b-coh = λ x → A .coh tt x
  }
  where
  ofwd : ∀ {C} → EqU (opaque₀ A) C → EqU (opaque₀ A) C
  ofwd e = e

  obwd : ∀ {C} → EqU (opaque₀ A) C → EqU (opaque₀ A) C
  obwd e = e

  tf
    : ∀ {C} (eAC : EqU (opaque₀ A) C) {a b c}
    → HRel {A = opaque₀ A} {opaque₀ A} Eq.refl a b
    → HRel eAC b c → HRel (ofwd eAC) a c
  tf {opaque₀ C} eAC {a} {b} {c} ab bc =
    Eq.Jₚ (λ C e → ∀ {c : ∣ C ∣ tt} →
              HRel {A = opaque₀ A} {opaque₀ C} e b c →
              HRel {A = opaque₀ A} {opaque₀ C} e a c)
      (λ bc → A .trans* ab bc) C eAC bc

  tb
    : ∀ {C} (eAC : EqU (opaque₀ A) C) {a b c}
    → HRel {A = opaque₀ A} {opaque₀ A} Eq.refl b a
    → HRel eAC a c → HRel (obwd eAC) b c
  tb {opaque₀ C} eAC {a} {b} {c} ba ac =
    Eq.Jₚ (λ C e → ∀ {c : ∣ C ∣ tt} →
              HRel {A = opaque₀ A} {opaque₀ C} e a c →
              HRel {A = opaque₀ A} {opaque₀ C} e b c)
      (λ ac → A .trans* ba ac) C eAC ac

ops : {A B : Code ℓ} (eAB : EqU A B) → Ops eAB
ops {A = opaque₀ A} {opaque₀ B} eAB =
  J (λ B e → Ops {A = opaque₀ A} {opaque₀ B} e) (opaqueOps A) B eAB
ops {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ) = record
  { comp-f = compF
  ; comp-b = compB
  ; cast-f = cf
  ; cast-b = cb
  ; trans-f = λ e → tf e
  ; trans-b = λ e → tb e
  ; cast-f-coh = λ f → cf-coh f
  ; cast-b-coh = λ f → cb-coh f
  }
  where
  IHA = ops eAB

  -- Model.Prop.Pi transports an argument with A.coe along the reversed
  -- context equality.  Using the forward operation for symU eAB here makes
  -- the decoded Pi coercion judgmentally the model's Pi coercion.
  IHA⁻ = ops (symU eAB)

  IHP : ∀ a b (eab : HRel eAB a b) → Ops (ePQ a b eab)
  IHP a b eab = ops (ePQ a b eab)

  eab : (b : El B) → HRel eAB (cast-f IHA⁻ b) b
  eab b = symEl (symU eAB) (cast-f-coh IHA⁻ b)

  eba : (a : El A) → HRel (symU eAB) (cast-f IHA a) a
  eba a = symEl eAB (cast-f-coh IHA a)

  compF
    : ∀ {C} → EqU (pi₀ B Q Qext) C → EqU (pi₀ A P Pext) C
  compF {opaque₀ C} ()
  compF {pi₀ C R Rext} (eBC , eQR) =
    comp-f IHA eBC , λ a c eac →
      let b = cast-f IHA a
          ab = cast-f-coh IHA a
          bc = trans-b IHA (comp-f IHA eBC) (symEl eAB ab) eac
      in comp-f (IHP a b ab) (eQR b c bc)

  compB
    : ∀ {C} → EqU (pi₀ A P Pext) C → EqU (pi₀ B Q Qext) C
  compB {opaque₀ C} ()
  compB {pi₀ C R Rext} (eAC , ePR) =
    comp-b IHA eAC , λ b c ebc →
      let a = cast-b IHA b
          ba = cast-b-coh IHA b
          ab = symEl (symU eAB) ba
          ac = trans-f IHA (comp-b IHA eAC) ab ebc
      in comp-b (IHP a b ab) (ePR a c ac)

  cf₀ : ((a : El A) → El (P a)) → (b : El B) → El (Q b)
  cf₀ f b = cast-f (IHP (cast-f IHA⁻ b) b (eab b)) (f (cast-f IHA⁻ b))

  cf-resp
    : (f : El (pi₀ A P Pext)) {b₀ b₁ : El B}
    → (bb : Rel B b₀ b₁)
    → HRel (Qext bb) (cf₀ (f .₁) b₀) (cf₀ (f .₁) b₁)
  cf-resp f {b₀} {b₁} bb =
    let e₀ = trans-f IHA (reflU B) (eab b₁) (symEl (reflU B) bb)
        e₁ = trans-f IHA (symU (comp-f IHA (reflU B)))
               (eab b₀) (symEl (comp-f IHA (reflU B)) e₀)
        e₂ = f .₂ e₁
        e₃ = cast-f-coh (IHP (cast-f IHA⁻ b₁) b₁ (eab b₁))
               (f .₁ (cast-f IHA⁻ b₁))
        e₄ = cast-f-coh (IHP (cast-f IHA⁻ b₀) b₀ (eab b₀))
               (f .₁ (cast-f IHA⁻ b₀))
        e₅ = trans-b (IHP (cast-f IHA⁻ b₀) b₀ (eab b₀))
               (Pext e₁)
               (symEl (ePQ (cast-f IHA⁻ b₀) b₀ (eab b₀)) e₄) e₂
        e₆ = trans-b (IHP (cast-f IHA⁻ b₁) b₀ e₀)
               (ePQ (cast-f IHA⁻ b₁) b₁ (eab b₁))
               e₅ e₃
    in e₆

  cf : El (pi₀ A P Pext) → El (pi₀ B Q Qext)
  cf f = cf₀ (f .₁) , cf-resp f

  cb₀ : ((b : El B) → El (Q b)) → (a : El A) → El (P a)
  cb₀ f a = cast-b (IHP a (cast-f IHA a) (cast-f-coh IHA a))
              (f (cast-f IHA a))

  cb-resp
    : (f : El (pi₀ B Q Qext)) {a₀ a₁ : El A}
    → (aa : Rel A a₀ a₁)
    → HRel (Pext aa) (cb₀ (f .₁) a₀) (cb₀ (f .₁) a₁)
  cb-resp f {a₀} {a₁} aa =
    let e₀ = trans-b IHA (reflU A) (eba a₀) aa
        e₁ = trans-b IHA eAB e₀ (cast-f-coh IHA a₁)
        e₂ = f .₂ e₁
        e₃ = cast-b-coh (IHP a₀ (cast-f IHA a₀) (cast-f-coh IHA a₀))
               (f .₁ (cast-f IHA a₀))
        e₄ = cast-b-coh (IHP a₁ (cast-f IHA a₁) (cast-f-coh IHA a₁))
               (f .₁ (cast-f IHA a₁))
        e₅ = trans-f (IHP a₁ (cast-f IHA a₁) (cast-f-coh IHA a₁))
               (symU (Qext e₁))
               (symEl (symU (ePQ a₁ (cast-f IHA a₁) (cast-f-coh IHA a₁))) e₄)
               (symEl (Qext e₁) e₂)
        e₆ = trans-f (IHP a₁ (cast-f IHA a₀) (symEl (comp-b IHA (reflU A)) e₀))
               (symU (ePQ a₀ (cast-f IHA a₀) (cast-f-coh IHA a₀)))
               e₅ e₃
    in symEl (comp-f (IHP a₁ (cast-f IHA a₀)
           (symEl (comp-b IHA (reflU A)) e₀))
           (symU (ePQ a₀ (cast-f IHA a₀) (cast-f-coh IHA a₀)))) e₆

  cb : El (pi₀ B Q Qext) → El (pi₀ A P Pext)
  cb f = cb₀ (f .₁) , cb-resp f

  tf
    : ∀ {C} (eBC : EqU (pi₀ B Q Qext) C)
      {f : El (pi₀ A P Pext)} {g : El (pi₀ B Q Qext)} {h : El C}
    → HRel {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ) f g
    → HRel eBC g h → HRel (compF eBC) f h
  tf {opaque₀ C} () fg gh
  tf {pi₀ C R Rext} (eBC , eQR) fg gh a c eac =
    let b = cast-f IHA a
        ab = cast-f-coh IHA a
        bc = trans-b IHA (comp-f IHA eBC) (symEl eAB ab) eac
    in trans-f (IHP a b ab) (eQR b c bc) (fg a b ab) (gh b c bc)

  tb
    : ∀ {C} (eAC : EqU (pi₀ A P Pext) C)
      {f : El (pi₀ A P Pext)} {g : El (pi₀ B Q Qext)} {h : El C}
    → HRel {A = pi₀ B Q Qext} {pi₀ A P Pext}
        (symU {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ)) g f
    → HRel eAC f h → HRel (compB eAC) g h
  tb {opaque₀ C} () gf fh
  tb {pi₀ C R Rext} (eAC , ePR) gf fh b c ebc =
    let a = cast-b IHA b
        ba = cast-b-coh IHA b
        ab = symEl (symU eAB) ba
        ac = trans-f IHA (comp-b IHA eAC) ab ebc
    in trans-b (IHP a b ab) (ePR a c ac) (gf b a ba) (fh a c ac)

  cf-coh
    : (f : El (pi₀ A P Pext))
    → HRel {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ) f (cf f)
  cf-coh f a b ab =
    let e₀ = trans-f IHA (symU eAB) ab (cast-f-coh IHA⁻ b)
        e₁ = f .₂ e₀
        e₂ = cast-f-coh (IHP (cast-f IHA⁻ b) b (eab b))
               (f .₁ (cast-f IHA⁻ b))
        e₃ = trans-b (IHP (cast-f IHA⁻ b) b (eab b))
               (symU (Pext e₀))
               (symEl (ePQ (cast-f IHA⁻ b) b (eab b)) e₂)
               (symEl (Pext e₀) e₁)
    in symEl (comp-b (IHP (cast-f IHA⁻ b) b (eab b))
         (symU (Pext e₀))) e₃

  cb-coh
    : (f : El (pi₀ B Q Qext))
    → HRel {A = pi₀ B Q Qext} {pi₀ A P Pext}
        (symU {A = pi₀ A P Pext} {pi₀ B Q Qext} (eAB , ePQ)) f (cb f)
  cb-coh f b a ba =
    let e₀ = trans-b IHA eAB ba (cast-f-coh IHA a)
        e₁ = f .₂ e₀
        e₂ = cast-b-coh (IHP a (cast-f IHA a) (cast-f-coh IHA a))
               (f .₁ (cast-f IHA a))
        e₃ = trans-f (IHP a (cast-f IHA a) (cast-f-coh IHA a))
               (symU (Qext e₀))
               (symEl (symU (ePQ a (cast-f IHA a) (cast-f-coh IHA a))) e₂)
               (symEl (Qext e₀) e₁)
    in symEl (comp-f (IHP a (cast-f IHA a) (cast-f-coh IHA a))
         (symU (Qext e₀))) e₃

------------------------------------------------------------------------
-- The operations needed by the setoid model

transU : {A B C : Code ℓ} → EqU A B → EqU B C → EqU A C
transU eAB = comp-f (ops eAB)

transEl
  : ∀ {A B C : Code ℓ} (eAB : EqU A B) (eBC : EqU B C) {a b c}
  → HRel eAB a b → HRel eBC b c → HRel (transU eAB eBC) a c
transEl eAB = trans-f (ops eAB)

cast : {A B : Code ℓ} → EqU A B → El A → El B
cast eAB = cast-f (ops eAB)

cast-coh : {A B : Code ℓ} (eAB : EqU A B) (a : El A) → HRel eAB a (cast eAB a)
cast-coh eAB = cast-f-coh (ops eAB)

------------------------------------------------------------------------
-- The universe and its displayed decoding family

U : (ℓ : Level) → Ty ∅ (lsuc ℓ)
∣ U ℓ ∣ tt = Code ℓ
U ℓ .heq tt A B = EqU A B
U ℓ .refl* A = reflU A
U ℓ .sym* eAB = symU eAB
U ℓ .trans* eAB eBC = transU eAB eBC
U ℓ .coe tt A = A
U ℓ .coh tt A = reflU A

ElU : Ty (∅ ▷ U ℓ) ℓ
∣ ElU ∣ (tt , A) = El A
ElU .heq {γ₀ = tt , A} {γ₁ = tt , B} (tt , eAB) x y = HRel eAB x y
ElU .refl* {γ = tt , A} x = reflEl A x
ElU .sym* {γ₀ = tt , A} {γ₁ = tt , B} {γ₀₁ = tt , eAB} e = symEl eAB e
ElU .trans* {γ₀₁ = tt , eAB} {γ₁₂ = tt , eBC} e₀₁ e₁₂ =
  transEl eAB eBC e₀₁ e₁₂
ElU .coe (tt , eAB) x = cast eAB x
ElU .coh (tt , eAB) x = cast-coh eAB x

decode : ∀ {ℓΓ} {Γ : Con ℓΓ} → Tm Γ (U ℓ [ ε ]) → Ty Γ ℓ
decode a = ElU [ ε ▸ a ]

------------------------------------------------------------------------
-- Public codes

-- Agda 2.9 reserves the bare word `opaque`, so the public spelling carries a
-- universe superscript.
opaqueᵁ : ∀ {ℓΓ} {Γ : Con ℓΓ} → Ty ∅ ℓ → Tm Γ (U ℓ [ ε ])
∣ opaqueᵁ A ∣ γ = opaque₀ A
opaqueᵁ A .resp γ₀₁ = reflU (opaque₀ A)

pi
  : ∀ {ℓΓ} {Γ : Con ℓΓ}
  → (a : Tm Γ (U ℓ [ ε ]))
  → Tm Γ (Π (decode a) (U ℓ [ ε ]))
  → Tm Γ (U ℓ [ ε ])
∣ pi a b ∣ γ = pi₀ (∣ a ∣ γ) (∣ b ∣ γ .₁) (∣ b ∣ γ .₂)
pi a b .resp γ₀₁ =
  a .resp γ₀₁ , λ x y xy → b .resp γ₀₁ x y xy

private
  pi' : Tm (∅ ▷ U ℓ ▷ Π (decode vz) ((U ℓ [ ε ]))) (U ℓ [ ε ])
  pi' = pi (vs vz) vz

  _ : decode (pi' {ℓ}) ≡ Π (decode (vs vz)) (decode (app vz))
  _ = Eq.refl

------------------------------------------------------------------------
-- Judgmental computation and substitution checks

opaque-β
  : ∀ {ℓΓ} {Γ : Con ℓΓ} (A : Ty ∅ lzero)
  → decode (opaqueᵁ {Γ = Γ} A) ≡ A [ ε ]
opaque-β A = Eq.refl

opaque-carrier
  : ∀ {ℓΓ} {Γ : Con ℓΓ} (A : Ty ∅ lzero) (γ : ∣ Γ ∣)
  → ∣ decode (opaqueᵁ {Γ = Γ} A) ∣ γ ≡ ∣ A ∣ tt
opaque-carrier A γ = Eq.refl

opaque-relation
  : ∀ {ℓΓ} {Γ : Con ℓΓ} (A : Ty ∅ lzero)
    {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    (x y : ∣ A ∣ tt)
  → (decode (opaqueᵁ {Γ = Γ} A)) .heq γ₀₁ x y
    ≡ A .heq tt x y
opaque-relation A γ₀₁ x y = Eq.refl

pi-β
  : ∀ {ℓΓ} {Γ : Con ℓΓ}
    (a : Tm Γ (U ℓ [ ε ]))
    (b : Tm Γ (Π (decode a) (U ℓ [ ε ])))
  → decode (pi a b) ≡ Π (decode a) (decode (app b))
pi-β a b = Eq.refl

pi-carrier
  : ∀ {ℓΓ} {Γ : Con ℓΓ}
    (a : Tm Γ (U ℓ [ ε ]))
    (b : Tm Γ (Π (decode a) (U ℓ [ ε ])))
    (γ : ∣ Γ ∣)
  → ∣ decode (pi a b) ∣ γ
    ≡ ∣ Π (decode a) (decode (app b)) ∣ γ
pi-carrier a b γ = Eq.refl

pi-relation
  : ∀ {ℓΓ} {Γ : Con ℓΓ}
    (a : Tm Γ (U ℓ [ ε ]))
    (b : Tm Γ (Π (decode a) (U ℓ [ ε ])))
    {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    {f : ∣ decode (pi a b) ∣ γ₀} {g : ∣ decode (pi a b) ∣ γ₁}
  → (decode (pi a b)) .heq γ₀₁ f g
    ≡ (Π (decode a) (decode (app b))) .heq γ₀₁ f g
pi-relation a b γ₀₁ = Eq.refl

pi-coe
  : ∀ {ℓΓ} {Γ : Con ℓΓ}
    (a : Tm Γ (U ℓ [ ε ]))
    (b : Tm Γ (Π (decode a) (U ℓ [ ε ])))
    {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
    (f : ∣ decode (pi a b) ∣ γ₀)
  → (decode (pi a b)) .coe γ₀₁ f
    ≡ (Π (decode a) (decode (app b))) .coe γ₀₁ f
pi-coe a b γ₀₁ f = Eq.refl

opaque≉pi
  : ∀ (A : Ty ∅ ℓ) (B : Code ℓ) (C : El B → Code ℓ)
    (Cext : ∀ {x y} → HRel (reflU B) x y → EqU (C x) (C y))
  → EqU (opaque₀ A) (pi₀ B C Cext) → ⊥ₚ {lsuc lzero}
opaque≉pi A B C Cext ()

coerce-refl-coherent
  : ∀ (A : Code ℓ) (x : El A)
  → HRel (reflU A) x (cast (reflU A) x)
coerce-refl-coherent A = cast-coh (reflU A)

decode[]
  : ∀ {ℓΓ ℓΔ} {Γ : Con ℓΓ} {Δ : Con ℓΔ}
    {a : Tm Γ (U ℓ [ ε ])} {σ : Sub Δ Γ}
  → decode (a [ σ ]t) ≡ decode a [ σ ]
decode[] = Eq.refl

opaque[]
  : ∀ {ℓΓ ℓΔ} {Γ : Con ℓΓ} {Δ : Con ℓΔ}
    (A : Ty ∅ lzero) (σ : Sub Δ Γ)
  → opaqueᵁ A [ σ ]t ≡ opaqueᵁ A
opaque[] A σ = Eq.refl

pi[]
  : ∀ {ℓΓ ℓΔ} {Γ : Con ℓΓ} {Δ : Con ℓΔ}
    (a : Tm Γ (U ℓ [ ε ]))
    (b : Tm Γ (Π (decode a) (U ℓ [ ε ])))
    (σ : Sub Δ Γ)
  → pi a b [ σ ]t ≡ pi (a [ σ ]t) (b [ σ ]t)
pi[] a b σ = Eq.refl
