-- Natural numbers as an inductive type generated by Desc.

-- Closed unit type used as the index of Nat.
One : Ty Γ lzero
One = Unit {ℓ = lzero}

-- Zero constructor description.
NatZeroD : ConD ∅ One One
NatZeroD = end ⋆

-- Successor constructor description.
NatSucD : ConD ∅ One One
NatSucD = arg1 ⋆ (end ⋆)

-- Description of natural-number constructors.
NatDs : ConDs One One
NatDs = NatZeroD ∷ NatSucD ∷ []

Nat : Ty Γ lzero
Nat =  μ NatDs ⟨ ⋆ ⟩

-- Constructors
ze : Tm Γ Nat
ze = con# (# 0) ⋆ (reflT' ⋆)

su : Tm Γ Nat → Tm Γ Nat
su n = con# (# 1) ⋆ (pair Nat (⇑ Eq ⋆ ⋆) n (reflT' ⋆))

-- Check the substitution rule of `elim`.
module _ {Γ : Con lzero}
  (P : Ty (Γ ▷ One ▷ μ NatDs) lzero)
  (pz : Tm (Γ ▷ One ▷ μArg NatZeroD [ ε ⁺ ] ▷ μIH P ε NatZeroD)
           (P [ wk ▸ conᵢ NatDs ]
              [ wk ▸ inlArgsᵢ NatZeroD (NatSucD ∷ []) ]
              [ wk ]))
  (ps : Tm (Γ ▷ One ▷ μArg NatSucD [ ε ⁺ ] ▷ μIH P ε NatSucD)
           (P [ wk ▸ conᵢ NatDs ]
              [ wk ▸ inrArgsᵢ NatZeroD (NatSucD ∷ []) ] [ wk ▸ inlArgsᵢ NatSucD [] ]
              [ wk ]))
  where

  NatDs-elim[]
    : {Δ : Con lzero} {σ : Sub Δ Γ}
    → elim' P (pz , ps , _) [ σ ⁺ ⁺ ]t
    ≡ elim' (P [ σ ⁺ ⁺ ]) (pz [ σ ⁺ ⁺ ⁺ ]t , ps [ σ ⁺ ⁺ ⁺ ]t , _)
  NatDs-elim[] = Eq.refl

  NatDs-elim-ze
    : elimₑ' P (pz , ps , _) ⋆ ze
    ≡ pz [ σArgᵢ→Arg ε NatZeroD ⁺ ]t
         [ idS ▸ ⋆ ▸ reflT' ⋆ ▸ ⋆ ]t
  NatDs-elim-ze = Eq.refl

  NatDs-elim-su
    : {n : Tm Γ Nat}
    → elimₑ' P (pz , ps , _) ⋆ (su n)
    ≡ ps [ σArgᵢ→Arg ε NatSucD ⁺ ]t
         [ idS ▸ ⋆
         ▸ pair (μ NatDs ⟨ ⋆ ⟩) (⇑ Eq ⋆ ⋆)
              n (reflT' ⋆)
         ▸ pair (P [ idS ▸ ⋆ ▸ n ]) Unit
              (elimₑ' P (pz , ps , _) ⋆ n) ⋆ ]t
  NatDs-elim-su = Eq.refl

module _ {Γ : Con lzero} (P : Ty (Γ ▷ Nat) lzero) where
  -- The semantic induction hypothesis is strictly equivalent to
  -- the internal induction-hypothesis type, modulo the conversion
  -- between the semantic constructor arguments and the internal
  -- constructor type.
  IH≡IHᵢ-NatSucD
    : μIH (P [ wk ⁺ ]) ε NatSucD [ σArgᵢ→Arg (ε {Γ = Γ}) NatSucD ]
    ≡ IHᵢ ε NatSucD (μ NatDs) (P [ wk ⁺ ])
  IH≡IHᵢ-NatSucD = Eq.refl

  IHᵢ≡IH-NatSucD
    : IHᵢ ε NatSucD (μ NatDs) (P [ wk ⁺ ]) [ σArg→Argᵢ (ε {Γ = Γ}) NatSucD ]
    ≡ μIH (P [ wk ⁺ ]) ε NatSucD
  IHᵢ≡IH-NatSucD = Eq.refl

elimNat : {Γ : Con lzero} (P : Ty (Γ ▷ Nat) lzero)
  → (pz : Tm Γ (P ⟨ ze ⟩))
  → (ps : Tm (Γ ▷ Nat ▷ P) (P [ wk ▸ su vz ] [ wk ]))
  → (n : Tm Γ Nat)
  → Tm Γ (P ⟨ n ⟩)
elimNat P pz ps =
  elimₑ' (P [ wk ⁺ ])
       ( pz [ wk∘wk∘wk ]t -- wk∘wk∘wk: from Γ to Γ ▷ I ▷ Arg ▷ IH
       , ps [ ( let n = fst' Nat (Argᵢ (end ⋆) (μ NatDs) [ ε ⁺ ]) [ σArg→Argᵢ ε NatSucD ]t in
              wk∘wk∘wk
              ▸ n [ wk ]t
              -- The substitution σArg→Argᵢ unpacks Arg as Argᵢ. Since Argᵢ NatSucD [ ε ⁺ ]
              -- is a pair, we fill in the types of its two components in fst'.
              ▸ fst' (P [ wk ⁺ ] [ wk∘wk ▸ ⋆ ▸ n ])
                     (IHᵢ ε (end ⋆) (μ NatDs) (P [ wk ⁺ ]) [ wk ▸ reflT' ⋆ ])
               )]t
              -- The types of the two components of IHᵢ ε NatSucD.
              -- Here _ after P is, from the arg1 case of IHᵢ,
              -- fst' (X [ (wk ▸ f [ wk ]t) ∘ σ ⁺ ]) (Argᵢ E X [ σ ⁺ ]).
       , _)
    ⋆

-- `su` is a congruence for the internal equality of `Nat`: from `a ~ b` build
-- `su a ~ su b`. As `Nat`'s heterogeneous equality ignores the context path,
-- `congT` lands directly in the homogeneous `⇑ Eq` (cf. `HEq.Example.fst-Eq`).
cong-su
  : {Γ : Con lzero} {a b : Tm Γ Nat}
  → Tm Γ (⇑ Eq a b) → Tm Γ (⇑ Eq (su a) (su b))
cong-su {a = a} {b} p = congT (su vz) {σ₀ = ⟨ a ⟩s} {σ₁ = ⟨ b ⟩s} (sgC p)

-- Addition `m + n`, defined internally by the eliminator of `Nat`, recursing on
-- the second argument `n`:
--
--   m + ze     = m
--   m + su k   = su (m + k)
--
-- The motive is the constant family `Nat`; the zero branch returns the first
-- argument `m = vs vz`, and the successor branch returns `su` of the induction
-- hypothesis `vz` (the value of `m + k`).
add : {Γ : Con lzero} → Tm (Γ ▷ Nat ▷ Nat) Nat
add {Γ} = elimNat {Γ = Γ ▷ Nat ▷ Nat} Nat (vs vz) (su vz) vz

-- Pointwise addition of two terms.
addₑ : {Γ : Con lzero} (m n : Tm Γ Nat) → Tm Γ Nat
addₑ m n = add [ idS ▸ m ▸ n ]t

-- The recursion equations hold definitionally.
add-ze : {Γ : Con lzero} (m : Tm Γ Nat) → addₑ m ze ≡ m
add-ze m = Eq.refl

add-su : {Γ : Con lzero} (m n : Tm Γ Nat) → addₑ m (su n) ≡ su (addₑ m n)
add-su m n = Eq.refl


-- Associativity of addition.
--
-- Mathematically, the proof is induction on the third argument `k`:
--
--   (m + n) + ze     = m + n              = m + (n + ze)            -- reflexivity
--   (m + n) + su k   = su ((m + n) + k)
--                    = su (m + (n + k))   = m + (n + su k)          -- cong-su (IH)
--
-- i.e. the well-typed term
--
--   add-assoc
--     : {Γ : Con lzero} (m n k : Tm Γ Nat)
--     → Tm Γ (⇑ Eq (addₑ (addₑ m n) k) (addₑ m (addₑ n k)))
--   add-assoc m n k =
--     elimNat
--       (⇑ Eq (addₑ (addₑ (m [ wk ]t) (n [ wk ]t)) vz)
--             (addₑ (m [ wk ]t) (addₑ (n [ wk ]t) vz)))
--       (reflT' (addₑ m n))   -- k = ze   : reflexivity (both sides ≡ m + n)
--       (cong-su vz)          -- k = su _ : su of the induction hypothesis
--       k
--
-- This is type-correct but NOT machine-checkable here: Agda's conversion
-- checker exhausts memory (>12 GB). `add` is built from the generic `Desc`
-- eliminator, whose `liftIHs`/`mapIHs` machinery embeds the recursor's first
-- argument several times in a neutral normal form, so *nested* additions grow
-- the normal form exponentially. Associativity inherently nests additions
-- (`(m + n) + k`), so the successor branch blows up. The blowup is already
-- visible in the bare computation rule with a compound first argument:
--
--   addₑ (addₑ p q) (su n) ≡ su (addₑ (addₑ p q) n)   -- `Eq.refl`, OOMs
--
-- whereas `add-su` (variable first argument) is instant. This is the same wall
-- behind the postulated `add-comm` elsewhere in this repo — note its motive
-- `add m n ~ add n m` uses only *non-nested* `add`s, which is why it survives.
