open import Prelude

module Model.Prop.CwF.Core where

open Eq using (_≡_) public

variable
  ℓΓ ℓΔ ℓA ℓB ℓC ℓP : Level

record Con (ℓΓ : Level) : Set (lsuc ℓΓ) where
  field
    Γ     : Set ℓΓ
    _~_   : Γ → Γ → Prop ℓΓ
    refl  : (γ : Γ) → γ ~ γ
    sym   : {γ γ′ : Γ} → γ ~ γ′ → γ′ ~ γ
    trans : {γ γ′ γ″ : Γ} → γ ~ γ′ → γ′ ~ γ″ → γ ~ γ″

open Con public
  renaming (Γ to ∣_∣; _~_ to eq)

infix 5 _∋_~_
_∋_~_ = eq

{-# DISPLAY eq Γ γ γ′ = Γ ∋ γ ~ γ′ #-}

variable
  Γ Δ Θ : Con ℓΓ

record Sub (Γ : Con ℓΓ) (Δ : Con ℓΔ) : Set (ℓΓ ⊔ ℓΔ) where
  constructor _,_
  field
    σ
      : (γ : ∣ Γ ∣) → ∣ Δ ∣
    resp
      : {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → Δ ∋ σ γ₀ ~ σ γ₁
open Sub public
  renaming (σ to ∣_∣)

variable
  σ τ δ : Sub Γ Δ

record Ty (Γ : Con ℓΓ) (ℓA : Level) : Set (lsuc (ℓA ⊔ ℓΓ)) where
  field
    A : (γ : ∣ Γ ∣) → Set ℓA
    _⊢_~_
      : {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → (A γ₀ → A γ₁ → Prop ℓA)
    refl*
      : {γ : ∣ Γ ∣}(a : A γ) → (refl Γ γ) ⊢ a ~ a
    sym*
      : ∀ {γ₀ γ₁ x₀ x₁} {γ₀₁ : Γ ∋ γ₀ ~ γ₁}
      → (x₀₁ : γ₀₁ ⊢ x₀ ~ x₁)
      → let γ₁₀ = Γ .sym γ₀₁ in γ₁₀ ⊢ x₁ ~ x₀
    trans*
      : ∀ {γ₀ γ₁ γ₂}{γ₀₁ : Γ ∋ γ₀ ~ γ₁}{γ₁₂ : Γ ∋ γ₁ ~ γ₂}
      → {x₀ : A γ₀}{x₁ : A γ₁}{x₂ : A γ₂}
      → (x₀₁ : γ₀₁ ⊢ x₀ ~ x₁) (x₁₂ : γ₁₂ ⊢ x₁ ~ x₂)
      → (Γ .trans γ₀₁ γ₁₂) ⊢ x₀ ~ x₂
    coe
      : ∀ {γ₀ γ₁}(γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → (x₀ : A γ₀) → A γ₁
    coh
      : ∀ {γ₀ γ₁}(γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → (x₀ : A γ₀)
      → γ₀₁ ⊢ x₀ ~ (coe γ₀₁ x₀)
  coh'
   : ∀ {γ₀ γ₁}(γ₀₁ : Γ ∋ γ₀ ~ γ₁)
   → (x₀ : A γ₀)
   → let γ₁₀ = Γ .sym γ₀₁ in γ₁₀ ⊢ (coe γ₀₁ x₀) ~ x₀
  coh' γ₀₁ x₀ = sym* (coh γ₀₁ x₀)

  coh'' -- [TODO] Reverse γ₀₁ in advance 
   : ∀ {γ₀ γ₁}(γ₁₀ : Γ ∋ γ₁ ~ γ₀)
   → (x₀ : A γ₀)
   → γ₁₀ ⊢ (coe (Γ .sym γ₁₀) x₀) ~ x₀
  coh'' γ₀₁ x₀ = sym* (coh (Γ .sym γ₀₁) x₀)

open Ty public
  renaming (A to ∣_∣; _⊢_~_ to heq)

variable
  A B C : Ty Γ ℓA

_∋_⊢_~_ = heq

_∋_~_⊢_~_ : (A : Ty Γ ℓA) (γ γ′ : ∣ Γ ∣) → {Γ ∋ γ ~ γ′} → ∣ A ∣ γ → ∣ A ∣ γ′ → Prop ℓA
_∋_~_⊢_~_ A _ _ {p} x y = A ∋ p ⊢ x ~ y

{-# DISPLAY heq A {γ} {γ′} _ x y     = A ∋ γ ~ γ′ ⊢ x ~ y #-}
{-# DISPLAY _∋_⊢_~_ A {γ} {γ′} _ x y = A ∋ γ ~ γ′ ⊢ x ~ y #-}

record Tm (Γ : Con ℓΓ) (A : Ty Γ ℓA) : Set (ℓA ⊔ ℓΓ) where
  constructor _,_
  field
    t : (γ : ∣ Γ ∣) → ∣ A ∣ γ
    respt
      : {γ₀ γ₁ : ∣ Γ ∣} (γ₀₁ : Γ ∋ γ₀ ~ γ₁)
      → A ∋ γ₀₁ ⊢ t γ₀ ~ t γ₁
open Tm public
  renaming (t to ∣_∣; respt to resp) 

infixl 10 _[_] _[_]t
infixr 8 _∘_
infixl 6 _▸_
infixl 5 _▷_

_[_] : (A : Ty Δ ℓA) (σ : Sub Γ Δ) → Ty Γ ℓA
{- This definition is somewhat faster :thinking:
A [ σ ] = record
           { A = λ γ → ∣ A ∣ (∣ σ ∣ γ)
           ; _⊢_~_ = λ p → A .heq (σ .resp p)
           ; refl* = A .refl*
           ; sym* = A .sym*
           ; trans* = A .trans*
           ; coe = λ p → A .coe (σ .resp p)
           ; coh = λ p → A .coh (σ .resp p)
           }
-}
∣ A [ σ ] ∣ γ = ∣ A ∣ (∣ σ ∣ γ)
(A [ σ ]) .heq p x y = A .heq (σ .resp p) x y
(A [ σ ]) .refl*  = A .refl*
(A [ σ ]) .sym*   = A .sym*
(A [ σ ]) .trans* = A .trans*
(A [ σ ]) .coe    = λ p → A .coe (σ .resp p)
(A [ σ ]) .coh    = λ p → A .coh (σ .resp p)

_[_]t : (t : Tm Δ A)(σ : Sub Γ Δ) → Tm Γ (A [ σ ])
-- This definition is somewhat faster
-- t [ σ ]t = (λ γ → ∣ t ∣ (∣ σ ∣ γ)) , λ p → t .resp (σ .resp p)
∣ t [ σ ]t ∣ γ = ∣ t ∣ (∣ σ ∣ γ)
(t [ σ ]t) .resp p = t .resp (σ .resp p)

∅ : Con lzero
∣ ∅ ∣        = ⊤
∅ .eq _ _    = ⊤ₚ
∅ .refl _    = tt
∅ .sym  _    = tt
∅ .trans _ _ = tt

_▷_ : (Γ : Con ℓΓ) (A : Ty Γ ℓA) → Con (ℓΓ ⊔ ℓA)
∣ Γ ▷ A ∣ = Σ[ γ ∈ ∣ Γ ∣ ] ∣ A ∣ γ
(Γ ▷ A) .eq = λ (γ₀ , x₀) (γ₁ , x₁)
  → ∃[ γ₀₁ ∈ Γ ∋ γ₀ ~ γ₁ ] (A ∋ γ₀₁ ⊢ x₀ ~ x₁)
(Γ ▷ A) .refl (γ , x) = Γ .refl γ , A .refl* x
(Γ ▷ A) .sym  (γ₀₁ , a₀₁) = Γ .sym γ₀₁ , A .sym* a₀₁
(Γ ▷ A) .trans (γ₀₁ , a₀₁) (γ₁₂ , a₁₂) =
  Γ .trans γ₀₁ γ₁₂ , A .trans* a₀₁ a₁₂
{-# INJECTIVE_FOR_INFERENCE _▷_ #-}

ε : Sub Γ ∅
∣ ε ∣ _   = tt
ε .resp _ = tt

_▸_ : (σ : Sub Γ Δ) {A : Ty Δ ℓΔ} (t : Tm Γ (A [ σ ])) → Sub Γ (Δ ▷ A)
∣ σ ▸ t ∣ γ     = ∣ σ ∣ γ , ∣ t ∣ γ
(σ ▸ t) .resp γ₀₁ = σ .resp γ₀₁ , t .resp γ₀₁
{-# INJECTIVE_FOR_INFERENCE _▸_ #-}

idS : Sub Γ Γ
idS = (λ x → x) , λ p → p
{-# INJECTIVE_FOR_INFERENCE idS #-}

_∘_ : Sub Δ Θ → Sub Γ Δ → Sub Γ Θ
∣ τ ∘ σ ∣ γ   = ∣ τ ∣ (∣ σ ∣ γ)
(τ ∘ σ) .resp γ₀₁ = τ .resp (σ .resp γ₀₁)

π₁ : (A : Ty Δ ℓΔ) → Sub Γ (Δ ▷ A)
  → Sub Γ Δ
∣ π₁ A σ ∣ γ   = ∣ σ ∣ γ .₁
π₁ A σ .resp γ₀₁ = σ .resp γ₀₁ .₁

π₂ : (A : Ty Δ ℓΔ) (σ : Sub Γ (Δ ▷ A))
  → Tm Γ (A [ π₁ _ σ ])
∣ π₂ A σ ∣ γ = ∣ σ ∣ γ .₂
π₂ A σ .resp γ₀₁ = σ .resp γ₀₁ .₂

-- Equations

-- (Con, Sub, idS, _∘_) is a category
idl : idS ∘ σ ≡ σ
idl = Eq.refl

idr : σ ∘ idS ≡ σ
idr = Eq.refl

∘-assoc : (δ ∘ τ) ∘ σ ≡ δ ∘ (τ ∘ σ)
∘-assoc = Eq.refl

-- Functoriality

[id] : (A [ idS ]) ≡ A
[id] = Eq.refl

[][] : A [ σ ] [ τ ] ≡ A [ σ ∘ τ ]
[][] = Eq.refl

[id]t
  : {Γ : Con ℓΓ} {A : Ty Γ ℓA} {t : Tm Γ A}
  → t [ idS ]t ≡ t
[id]t = Eq.refl

[][]t
  : {A : Ty Θ ℓA}{x : Tm Θ A}{σ : Sub Δ Θ}{τ : Sub Γ Δ}
-- Lossy unification does not work, doh.
  → x [ σ ]t [ τ ]t ≡ x [ σ ∘ τ ]t
[][]t = Eq.refl

-- Projections
π₁β
  : {t : Tm Γ (A [ σ ])}
  → π₁ A (σ ▸ t) ≡ σ
π₁β = Eq.refl

π₂β
  : {t : Tm Γ (A [ σ ])}
  → π₂ A (σ ▸ t) ≡ t
π₂β = Eq.refl

ϵη : σ ≡ ε
ϵη = Eq.refl

πη : σ ≡ (π₁ _ σ) ▸ π₂ _ σ
πη = Eq.refl

▸∘ : {t : Tm Δ (A [ τ ])}
  → (_▸_ τ {A} t) ∘ σ ≡ (τ ∘ σ) ▸ t [ σ ]t
▸∘ = Eq.refl

-- Closed types are contexts

Ty∅→Con : Ty ∅ ℓ → Con ℓ
Ty∅→Con A .∣_∣ = ∣ A ∣ tt
Ty∅→Con A .eq = A .heq tt
Ty∅→Con A .refl = A .refl*
Ty∅→Con A .sym = A .sym*
Ty∅→Con A .trans = A .trans*

Tm∅→Sub : {A B : Ty ∅ ℓ} → Tm (∅ ▷ A) (B [ π₁ A idS ]) → Sub (Ty∅→Con A) (Ty∅→Con B)
∣ Tm∅→Sub t ∣ a = ∣ t ∣ (tt , a)
Tm∅→Sub t .resp a₀₁ = t .resp (tt , a₀₁)

-- Lifting Ty

LiftTy : ∀ {ℓa} ℓb → (A : Ty Γ ℓa) → Ty Γ (ℓa ⊔ ℓb)
∣ LiftTy ℓB A ∣ γ           = Lift ℓB (∣ A ∣ γ) -- Lift ℓB (∣ A ∣ γ)
LiftTy ℓB A .heq γ₀₁ a₀ a₁  = LiftP ℓB (A .heq γ₀₁ (↓ a₀) (↓ a₁))
LiftTy ℓB A .refl* a        = ↑ (A .refl* (↓ a))
LiftTy ℓB A .sym* a₀₁       = ↑ (A .sym* (↓ a₀₁))
LiftTy ℓB A .trans* a₀₁ a₁₂ = ↑ (A .trans* (↓ a₀₁) (↓ a₁₂))
LiftTy ℓB A .coe γ₀₁ a₀     = ↑ (A .coe γ₀₁ (↓ a₀))
LiftTy ℓB A .coh γ₀₁ a₀     = ↑ (A .coh γ₀₁ (↓ a₀))
