open import Prelude

module Model.Prop.Desc.Core where

open import Model.Prop.CwF

private variable
  ℓI ℓJ : Level

-- Description code for one constructor.
data ConD (Δ : Con ℓΔ) (I : Ty ∅ ℓI) (J : Ty ∅ ℓI)
  : Set (lsuc (ℓI ⊔ ℓΔ)) where
  end : (t : Tm Δ (J [ ε ]))
    → ConD Δ I J
  non : (A : Ty Δ ℓI)(D : ConD (Δ ▷ A) I J)
    → ConD Δ I J
  arg1 : (f : Tm Δ (I [ ε ]))(D : ConD Δ I J)
    → ConD Δ I J
  arg : (A : Ty Δ ℓI)(f : Tm (Δ ▷ A) (I [ ε ])) (D : ConD Δ I J)
    → ConD Δ I J

-- Description codes for a list of constructors.
data ConDs (I J : Ty ∅ ℓI) : Set (lsuc ℓI) where
  [] : ConDs I J
  _∷_  : (D : ConD ∅ I J) (Ds : ConDs I J) → ConDs I J

-- Reindex a description along a substitution.
_[_]d : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {I J : Ty ∅ ℓI} (D : ConD Γ I J)(σ : Sub Δ Γ)
  → ConD Δ I J
end t     [ σ ]d = end (t [ σ ]t)
non A   D [ σ ]d = non (A [ σ ]) (D [ σ ↑t A ]d)
arg A f D [ σ ]d = arg (A [ σ ]) (f [ σ ⁺ ]t) (D [ σ ]d)
arg1 f  D [ σ ]d = arg1 (f [ σ ]t) (D [ σ ]d)

-- Instantiate a description in an extended context.
_⟨_⟩d : {Γ : Con ℓΓ} {A : Ty Γ ℓA} {I J : Ty ∅ ℓI}
  (D : ConD (Γ ▷ A) I J)(t : Tm Γ A)
  → ConD Γ I J
D ⟨ t ⟩d = D [ idS ▸ t ]d




-----------------------------------------------------------------------
--                         Constructor arguments                        --
-----------------------------------------------------------------------


-- Carrier type of one constructor's arguments.
Arg₀ : {Δ : Con ℓΔ}{I J : Ty ∅ ℓI}(D : ConD Δ I J)
  →          (X : ∣ I ∣ tt → Set ℓ)
  → (δ : ∣ Δ ∣) → ∣ J ∣ tt → Set (ℓ ⊔ ℓI)
Arg₀ {ℓ = ℓ} {Δ = Δ} {I} {J} (end t) X δ i =
  LiftS ℓ (J ∋ tt ⊢ ∣ t ∣ δ ~ i)
 -- it forces the level of Arg₀ D X δ i to be at least ℓ
Arg₀ (non A D) X δ i =
  Σ[ a ∈ ∣ A ∣ δ ] Arg₀ D X (δ , a) i
Arg₀ (arg1 f D) X δ i =
  X (∣ f ∣ δ) × Arg₀ D X δ i
Arg₀ {Δ = Δ} (arg A f D) X δ i =
  ((a : ∣ A ∣ δ) → X (∣ f ∣ (δ , a))) × Arg₀ D X δ i
-- An over-approximation, which requires an extensionality predicate to fix it
-- If the extension predicate extX : (x : X) → Prop ℓ were incorporated,
-- the definition of μ, μ~ and extμ would have to be inductive-inductive to instantiate
-- X and extX when using Arg₀.
{-# INJECTIVE_FOR_INFERENCE Arg₀ #-}

-- Carrier type of a constructor list's arguments.
Args₀
  : {I J : Ty ∅ ℓI} (Ds : ConDs I J)
  → (X : ∣ I ∣ tt → Set ℓ)
  → ∣ J ∣ tt → Set (ℓ ⊔ ℓI)
Args₀ []       X j = Lift _ ⊥
Args₀ (D ∷ Ds) X j = Arg₀ D X tt j ⊎ Args₀ Ds X j
{-# INJECTIVE_FOR_INFERENCE Args₀ #-}


-- Note: the setoid relations `Arg~`/`Args~` on constructor arguments are
-- defined further down, recursively and mutually with `μ~`, specialised to
-- the carrier `μ₀` (see the `mutual` block below `μ₀`).



-----------------------------------------------------------------------
--                  Fixed-point: the inductive type                   --
-----------------------------------------------------------------------

module _ {I : Ty ∅ ℓI} (Ds : ConDs I I) where

  -- Underlying inductive family generated by Ds.
  data μ₀ (i : ∣ I ∣ tt) : Set ℓI where
    con : Args₀ Ds μ₀ i → μ₀ i

  -- Setoid relation on μ₀ generated by constructor equality.
  -- This is defined recursively, mutually with `Arg~`/`Args~` specialised
  -- to work with `μ₀`/`μ~` (just like `extμ`/`extArg`/`extArgs` below).
  mutual
    μ~ : ∀ {i₀ i₁} → μ₀ i₀ → μ₀ i₁ → Prop ℓI
    μ~ (con xs₀) (con xs₁) = Args~ Ds xs₀ xs₁

    -- Setoid relation on one constructor's arguments.
    Arg~
      : {Δ : Con ℓΔ}(E : ConD Δ I I)
      → {δ₀ δ₁ : ∣ Δ ∣}(δ₀₁ : Δ ∋ δ₀ ~ δ₁)
      → {j₀ j₁ : ∣ I ∣ tt}
      → (xs₀ : Arg₀ E μ₀ δ₀ j₀) (xs₁ : Arg₀ E μ₀ δ₁ j₁) → Prop ℓI
    Arg~ (end t)     δ₀₁ xs₀ xs₁ = ⊤ₚ
    Arg~ (non A E)   δ₀₁ (a₀ , xs₀) (a₁ , xs₁) =
      ∃[ a₀₁ ∈ (A ∋ δ₀₁ ⊢ a₀ ~ a₁) ]
        Arg~ E (δ₀₁ , a₀₁) xs₀ xs₁
    Arg~ (arg1 f E)  δ₀₁ (x₀ , xs₀) (x₁ , xs₁) =
      μ~ x₀ x₁
      ∧ Arg~ E δ₀₁ xs₀ xs₁
    Arg~ (arg A f E) {δ₀} {δ₁} δ₀₁ (∣x₀∣ , xs₀) (∣x₁∣ , xs₁) =
      ((a₀ : ∣ A ∣ δ₀)(a₁ : ∣ A ∣ δ₁) (a₀₁ : A ∋ δ₀₁ ⊢ a₀ ~ a₁)
         → μ~ (∣x₀∣ a₀) (∣x₁∣ a₁))
      ∧ Arg~ E δ₀₁ xs₀ xs₁

    -- Setoid relation on a constructor list's arguments.
    Args~
      : (Ds : ConDs I I)
      → {j₀ j₁ : ∣ I ∣ tt}
      → (xs₀ : Args₀ Ds μ₀ j₀) (xs₁ : Args₀ Ds μ₀ j₁) → Prop ℓI
    Args~ (D ∷ Ds) (inl xs₀) (inl xs₁) = Arg~ D tt xs₀ xs₁
    Args~ (D ∷ Ds) (inr xs₀) (inr xs₁) = Args~ Ds xs₀ xs₁
    Args~ (D ∷ Ds) (inr xs₀) (inl xs₁) = ⊥ₚ
    Args~ (D ∷ Ds) (inl xs₀) (inr xs₁) = ⊥ₚ
  {-# INJECTIVE_FOR_INFERENCE μ~ #-}
  {-# INJECTIVE_FOR_INFERENCE Arg~ #-}

  -- Extensionality predicate on μ₀.
  mutual
    extμ : {i : ∣ I ∣ tt} → μ₀ i → Prop ℓI
    extμ (con xs) = extArgs Ds xs

    -- Extensionality predicate on one constructor's arguments.
    extArg
      : {Δ : Con ℓΔ}(E : ConD Δ I I)
      → (∀ {δ j} → Arg₀ E μ₀ δ j → Prop ℓI)
    extArg (end t)   xs       = ⊤ₚ
    extArg (non A E)  (_ , xs) = extArg E xs
    extArg (arg1 f E) (x , xs) = extμ x ∧ extArg E xs
    extArg {Δ = Δ} (arg A f E) {δ} (x , xs) =
      (∀ (a : ∣ A ∣ δ) → extμ (x a))
      ∧ (∀ {a₀ a₁} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁)
          → μ~ (x a₀) (x a₁)) -- μ~ (f .resp (_ , a₀₁)) (x a₀) (x a₁))
      ∧ extArg E xs

    -- Extensionality predicate on a constructor list's arguments.
    extArgs
      : (Ds : ConDs I I)
      → (∀ {j} → Args₀ Ds μ₀ j → Prop ℓI)
    extArgs (D ∷ Ds) (inl xs) = extArg D xs
    extArgs (D ∷ Ds) (inr xs) = extArgs Ds xs

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  mutual
    μ~-refl* : ∀ {i} x
      → (extx : extμ Ds x)
      → μ~ Ds {i} {i} x x
    μ~-refl* (con xs) extxs = μArgs₀-refl* Ds xs extxs

    μArgs₀-refl* : (Es : ConDs I I)
      → ∀ {i} (xs₀ : Args₀ Es (μ₀ Ds) i)
      → (extxs : extArgs Ds Es xs₀)
      → Args~ Ds Es xs₀ xs₀
    μArgs₀-refl* (D ∷ Es) (inl xs) extxs = μArg₀-refl* D xs extxs
    μArgs₀-refl* (D ∷ Es) (inr ys) extxs = μArgs₀-refl* Es ys extxs

    μArg₀-refl*
      : {Δ : Con ℓΔ} (E : ConD Δ I I)
      → ∀ {δ i} (xs₀ : Arg₀ E (μ₀ Ds) δ i)
      → (extxs : extArg Ds E xs₀)
      → Arg~ Ds E (Δ .refl δ) xs₀ xs₀
    μArg₀-refl* (end t) xs₀ extxs = tt
    μArg₀-refl* (non A E) {δ} (a , xs₀) extxs =
      A .refl* a , μArg₀-refl* E {δ , a} xs₀ extxs
    μArg₀-refl* (arg1 f E) (x₀ , xs₀) (extx , extxs) =
      μ~-refl* x₀ extx , μArg₀-refl* E xs₀ extxs
    μArg₀-refl* (arg A f E) (x₀ , xs₀) (extx , x₀~ , extxs) =
      (λ a₀ a₁ a₀₁ → x₀~ a₀₁) , μArg₀-refl* E xs₀ extxs

  mutual
    μ~-sym*
      : ∀ {i₀ x₀}{i₁ x₁}
      → (x₀₁ : μ~ Ds {i₀} {i₁} x₀ x₁)
      → μ~ Ds x₁ x₀
    μ~-sym* {x₀ = con _} {x₁ = con _} x₀₁ = μArgs₀-sym* Ds x₀₁

    μArgs₀-sym*
      : (Es : ConDs I I)
      → ∀ {i₀ i₁ xs₀ xs₁}
      → (xs₀₁ : Args~ Ds Es {i₀} {i₁} xs₀ xs₁)
      → Args~ Ds Es xs₁ xs₀
    μArgs₀-sym* (E ∷ Es) {xs₀ = inl _} {inl _} xs₀₁ = μArg₀-sym* E xs₀₁
    μArgs₀-sym* (E ∷ Es) {xs₀ = inr _} {inr _} xs₀₁ = μArgs₀-sym* Es xs₀₁

    μArg₀-sym*
      : {Δ : Con ℓΔ} (E : ConD Δ I I)
      → ∀ {δ₀ δ₁ i₀ i₁ xs₀ xs₁} {δ₀₁ : Δ ∋ δ₀ ~ δ₁}
      → (xs₀₁ : Arg~ Ds E δ₀₁ {i₀} {i₁} xs₀ xs₁)
      → Arg~ Ds E (Δ .sym δ₀₁) xs₁ xs₀
    μArg₀-sym* (end t)     xs₀₁         = tt
    μArg₀-sym* (non A E)   (a₀₁ , xs₀₁) = A .sym* a₀₁ , μArg₀-sym* E xs₀₁
    μArg₀-sym* (arg1 f E)  (x₀₁ , xs₀₁) = μ~-sym* x₀₁ , μArg₀-sym* E xs₀₁
    μArg₀-sym* (arg A f E) (x₀₁ , xs₀₁) = (λ a₀ a₁ a₀₁ → μ~-sym* (x₀₁ _ _ (A .sym* a₀₁)))
      , μArg₀-sym* E xs₀₁

  mutual
    μ~-trans*
      : ∀ {i₀ i₁ i₂ x₀ x₂}
      → {x₁ : μ₀ Ds i₁}
      → (x₀₁ : μ~ Ds {i₀} {i₁} x₀ x₁)
      → (x₁₂ : μ~ Ds {i₁} {i₂} x₁ x₂)
      → μ~ Ds x₀ x₂
    μ~-trans* {x₀ = con _} {x₂ = con _} {x₁ = con _} xs₀₁ xs₁₂ =
      μArgs₀-trans* Ds xs₀₁ xs₁₂

    μArgs₀-trans*
      : (Es : ConDs I I)
      → ∀ {i₀ i₁ i₂ xs₀ xs₁ xs₂}
      → (xs₀₁ : Args~ Ds Es {i₀} {i₁} xs₀ xs₁)
      → (xs₁₂ : Args~ Ds Es {i₁} {i₂} xs₁ xs₂)
      → Args~ Ds Es xs₀ xs₂
    μArgs₀-trans* (E ∷ Es) {xs₀ = inl _} {inl _} {inl _} xs₀₁ xs₁₂ = μArg₀-trans*  E  xs₀₁ xs₁₂
    μArgs₀-trans* (E ∷ Es) {xs₀ = inr _} {inr _} {inr _} xs₀₁ xs₁₂ = μArgs₀-trans* Es xs₀₁ xs₁₂

    μArg₀-trans*
      : {Δ : Con ℓΔ} (E : ConD Δ I I)
      → ∀ {δ₀ δ₁ δ₂ i₀ i₁ i₂ xs₀ xs₁ xs₂} {δ₀₁ : Δ ∋ δ₀ ~ δ₁} {δ₁₂ : Δ ∋ δ₁ ~ δ₂}
      → (xs₀₁ : Arg~ Ds E δ₀₁ {i₀} {i₁} xs₀ xs₁)
      → (xs₁₂ : Arg~ Ds E δ₁₂ {i₁} {i₂} xs₁ xs₂)
      → Arg~ Ds E (Δ .trans δ₀₁ δ₁₂) xs₀ xs₂
    μArg₀-trans* (end t)    xs₀₁         xs₁₂         = tt
    μArg₀-trans* (non A E)  (a₀₁ , xs₀₁) (a₁₂ , xs₁₂) =
      A .trans* a₀₁ a₁₂ , μArg₀-trans* E xs₀₁ xs₁₂
    μArg₀-trans* (arg1 f E) (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) =
      μ~-trans* x₀₁ x₁₂
      , μArg₀-trans* E xs₀₁ xs₁₂
    μArg₀-trans* (arg A f E) {δ₀₁ = δ₀₁} (x₀₁ , xs₀₁) (x₁₂ , xs₁₂) =
      (λ a₀ a₂ a₀₂ → μ~-trans* (x₀₁ _ _ (A .coh δ₀₁ a₀)) (x₁₂ _ _ (A .trans* (coh' A δ₀₁ a₀) a₀₂)))
      , μArg₀-trans* E xs₀₁ xs₁₂

-- The inductive type μ

-- To build μ, we specialize coercion and coherence to μ₀.

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where mutual
  -- These definitions are specialized to μ Ds so that recursion terminates.
  coeArg₀
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀)
    → Arg₀ E (μ₀ Ds) δ₁ j₁
  coeArg₀ {Δ = Δ} (end t) δ₀₁ j₀₁ (↑ q) .↓ =
    I .trans* (t .resp (Δ .sym δ₀₁)) (I .trans* q j₀₁)
  coeArg₀ (non A E) δ₀₁ j₀₁ (a₀ , xs₀) =
    A .coe δ₀₁ a₀ , coeArg₀ E (δ₀₁ , A .coh δ₀₁ a₀) j₀₁ xs₀
  coeArg₀ (arg1 f E) δ₀₁ j₀₁ (x₀ , xs₀) =
    coeμ₀ (f .resp δ₀₁) x₀ , coeArg₀ E δ₀₁ j₀₁ xs₀
  coeArg₀ {Δ = Δ} (arg A f E) δ₀₁ j₀₁ (x₀ , xs₀) =
    let δ₁₀ = Δ .sym δ₀₁ in
    (λ a₁ → coeμ₀ (f .resp (_ , coh' A δ₁₀ a₁)) (x₀ (A .coe δ₁₀ a₁)))
    , coeArg₀ E δ₀₁ j₀₁ xs₀

  coeArgs₀
    : (Es : ConDs I I)
    → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (xs₀ : Args₀ Es (μ₀ Ds) i₀)
    → Args₀ Es (μ₀ Ds) i₁
  coeArgs₀ (E ∷ Es) j₀₁ (inl xs) = inl (coeArg₀ E tt j₀₁ xs)
  coeArgs₀ (E ∷ Es) j₀₁ (inr xs) = inr (coeArgs₀ Es j₀₁ xs)

  coeμ₀
    : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (x₀ : μ₀ Ds i₀)
    → μ₀ Ds i₁
  coeμ₀ i₀₁ (con xs₀) = con (coeArgs₀ Ds i₀₁ xs₀ )

module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where mutual
  cohArg₀
    : {Δ : Con ℓΔ}
    → (E : ConD Δ I I)
    → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → (xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀)
    → (extxs : extArg Ds E  xs₀)
    → Arg~ Ds E δ₀₁ xs₀ (coeArg₀ E δ₀₁ j₀₁ xs₀)
  cohArg₀ (end t) δ₀₁ j₀₁ xs₀ extxs = tt
  cohArg₀ (non A E) δ₀₁ j₀₁ (a₀ , xs₀) extxs =
    A .coh δ₀₁ a₀ , cohArg₀ E (δ₀₁ , A .coh δ₀₁ a₀) j₀₁ xs₀ extxs
  cohArg₀ (arg1 f E) δ₀₁ j₀₁ (x₀ , xs₀) (extx , extxs) =
    cohμ₀ (f .resp δ₀₁) x₀ extx , cohArg₀ E δ₀₁ j₀₁ xs₀ extxs
  cohArg₀ {Δ = Δ} (arg A f E) δ₀₁ j₀₁ (x₀ , xs₀) (extx , x~ , extxs) =
    let δ₁₀ = Δ .sym δ₀₁ in
    (λ a₀ a₁ a₀₁ →
      let a₀' = A .coe δ₁₀ a₁ in
      μ~-trans* --  (extx a₀')
        (x~ (A .trans* a₀₁ (A .coh δ₁₀ a₁)))
        (cohμ₀ (f .resp (_ , coh' A δ₁₀ a₁)) (x₀ a₀') (extx a₀')))
    , cohArg₀ E δ₀₁ j₀₁ xs₀ extxs

  cohArgs₀
    : (Es : ConDs I I)
    → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (xs₀ : Args₀ Es (μ₀ Ds) i₀)
    → (extxs : extArgs Ds Es xs₀)
    → Args~ Ds Es xs₀ (coeArgs₀ Es i₀₁ xs₀)
  cohArgs₀ (E ∷ Es) i₀₁ (inl xs₀) extxs = cohArg₀ E tt i₀₁ xs₀ extxs
  cohArgs₀ (E ∷ Es) i₀₁ (inr xs₀) extxs = cohArgs₀ Es i₀₁ xs₀ extxs

  cohμ₀
    : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (x₀ : μ₀ Ds i₀)
    → (extx : extμ Ds x₀)
    → μ~ Ds x₀ (coeμ₀ i₀₁ x₀)
  cohμ₀ i₀₁ (con xs₀) extxs = cohArgs₀ Ds i₀₁ xs₀ extxs

-- Coercions preserve extensionality.
module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where mutual
  extArg-coeArg₀
    : {Δ : Con ℓΔ} → (E : ConD Δ I I)
    → ∀ {δ₀ δ₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → ∀ {j₀ j₁} (j₀₁ : I ∋ tt ⊢ j₀ ~ j₁)
    → {xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀}
    → (extxs : extArg Ds E xs₀)
    → extArg Ds E (coeArg₀ E δ₀₁ j₀₁ xs₀)
  extArg-coeArg₀ (end t) δ₀₁ j₀₁ extxs = tt
  extArg-coeArg₀ (non A E) δ₀₁ j₀₁ {a , _} extxs =
    extArg-coeArg₀ E (δ₀₁ , A .coh δ₀₁ a) j₀₁ extxs
  extArg-coeArg₀ (arg1 f E) δ₀₁ j₀₁ {x , xs} (extx , extxs) =
    extμ-coeμ₀ (f .resp δ₀₁) x extx
    , extArg-coeArg₀ E δ₀₁ j₀₁ extxs
  extArg-coeArg₀ {Δ = Δ} (arg A f E) δ₀₁ j₀₁ {x₀ , xs₀} (extx , x~ , extxs) =
    let δ₁₀ = Δ .sym δ₀₁ in
    (λ a₁ → extμ-coeμ₀ (f .resp (_ , coh' A δ₁₀ a₁)) (x₀ (A .coe δ₁₀ a₁)) (extx (A .coe δ₁₀ a₁)))
    , (λ {a₀} {a₁} a₀₁ →
        let a₀' = A .coe δ₁₀ a₀
            a₁' = A .coe δ₁₀ a₁ in
        μ~-trans*
          (μ~-sym* (cohμ₀ (f .resp (_ , coh' A δ₁₀ a₀)) _ (extx a₀')))
          (μ~-trans*
            (x~ (A .trans* (coh' A δ₁₀ a₀) (A .trans* a₀₁ (A .coh δ₁₀ a₁))))
            (cohμ₀ (f .resp (_ , coh' A δ₁₀ a₁)) _ (extx a₁'))))
    , extArg-coeArg₀ E δ₀₁ j₀₁ extxs

  extArgs-coeArgs₀
    : (Es : ConDs I I)
    → ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → {xs₀ : Args₀ Es (μ₀ Ds) i₀}
    → (extxs : extArgs Ds Es xs₀)
    → extArgs Ds Es (coeArgs₀ Es i₀₁ xs₀)
  extArgs-coeArgs₀ (E ∷ Es) i₀₁ {inl xs₀} extxs = extArg-coeArg₀ E tt i₀₁ extxs
  extArgs-coeArgs₀ (E ∷ Es) i₀₁ {inr xs₀} extxs = extArgs-coeArgs₀ Es i₀₁ extxs

  extμ-coeμ₀
    : ∀ {i₀ i₁} (i₀₁ : I ∋ tt ⊢ i₀ ~ i₁)
    → (x₀ : μ₀ Ds i₀)
    → (exts : extμ Ds x₀)
    → extμ Ds (coeμ₀ i₀₁ x₀)
  extμ-coeμ₀ i₀₁ (con xs) extxs = extArgs-coeArgs₀ Ds i₀₁ extxs -- (extArgs-coeArgs₀ Ds i₀₁ extxs)

-- Carrier of the inductive type μ.
∣μ∣
  : {I : Ty ∅ ℓI}
  → (Ds : ConDs I I) (i : ∣ I ∣ tt)
  → Set ℓI
∣μ∣ Ds i = Σₚ (μ₀ Ds i) (extμ Ds)

-- Setoid relation on the carrier of μ.
∣μ~∣
  : {I : Ty ∅ ℓI} (Ds : ConDs I I)
  → {i₀ i₁ : ∣ I ∣ tt}
  → ∣μ∣ Ds i₀ → ∣μ∣ Ds i₁ → Prop ℓI
∣μ~∣ Ds (x₀ , _) (x₁ , _) = μ~ Ds x₀ x₁

module _ {I : Ty ∅ ℓI} (Ds : ConDs I I) where
  -- Inductive type generated by Ds.
  μ : {Γ : Con ℓΓ} → Ty (Γ ▷ I [ ε ]) ℓI
  ∣ μ ∣ (γ , i)       = ∣μ∣ Ds i
  μ .heq _            = ∣μ~∣ Ds
  μ .refl* (x , extx) = μ~-refl* x extx
  μ .sym*             = μ~-sym*
  μ .trans*           = μ~-trans*
  μ .coe (_ , i₀₁) (x₀ , extx) = coeμ₀ i₀₁ x₀ , extμ-coeμ₀ i₀₁ x₀ extx
  μ .coh (_ , i₀₁) (x₀ , extx) = cohμ₀ i₀₁ x₀ extx
  {-# INJECTIVE_FOR_INFERENCE μ #-}

-- Constructor arguments specialized to μ.

-- Carrier of constructor arguments specialized to μ.
module _ {I : Ty ∅ ℓI} (Ds : ConDs I I) where
  ∣μArg∣
    : {Δ : Con ℓΔ} → (E : ConD Δ I I)
    → (δ : ∣ Δ ∣) (i : ∣ I ∣ tt)
    → Set ℓI
  ∣μArg∣ E δ i = Σₚ (Arg₀ E (μ₀ Ds) δ i) (extArg Ds E)
  {-# INJECTIVE_FOR_INFERENCE ∣μArg∣ #-}

  -- Carrier of constructor-list arguments specialized to μ.
  ∣μArgs∣
    : (Es : ConDs I I)
    → (i : ∣ I ∣ tt)
    → Set ℓI
  ∣μArgs∣ Es  i = Σₚ (Args₀ Es (μ₀ Ds) i) (extArgs Ds Es)
  {-# INJECTIVE_FOR_INFERENCE ∣μArgs∣ #-}

-- Setoid relation on constructor arguments specialized to μ.
  ∣μArg~∣
    : {Δ : Con ℓΔ} (E : ConD Δ I I)
    → ∀ {δ₀ δ₁ i₀ i₁} (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → ∣μArg∣ E δ₀ i₀ → ∣μArg∣ E δ₁ i₁ → Prop ℓI
  ∣μArg~∣ E δ₀₁ (xs₀ , _) (xs₁ , _) = Arg~ Ds E δ₀₁ xs₀ xs₁

  -- Setoid relation on constructor-list arguments specialized to μ.
  ∣μArgs~∣
    : (Es : ConDs I I)
    → {i₀ i₁ : ∣ I ∣ tt}
    → ∣μArgs∣ Es i₀ → ∣μArgs∣ Es i₁ → Prop ℓI
  ∣μArgs~∣ Es (xs₀ , _) (xs₁ , _) = Args~ Ds Es xs₀ xs₁

module _ {Δ : Con ℓΔ} {I : Ty ∅ ℓI} {Ds : ConDs I I} (E : ConD Δ I I) where
  -- Type of arguments for one constructor.
  μArg : Ty (Δ ▷ I [ ε ]) ℓI
  ∣ μArg ∣ (δ , i) = ∣μArg∣ Ds E δ i
  μArg .heq (δ₀₁ , i₀₁) = ∣μArg~∣ Ds E δ₀₁
  μArg .refl* (xs₀ , extxs) = μArg₀-refl* E xs₀ extxs
  μArg .sym* = μArg₀-sym* E
  μArg .trans* {x₁ = _ , extxs₁} = μArg₀-trans* E
  μArg .coe (δ₀₁ , i₀₁) (xs₀ , extxs) = coeArg₀ E δ₀₁ i₀₁ xs₀ , extArg-coeArg₀ {Ds = Ds} E δ₀₁ i₀₁ extxs
  μArg .coh (δ₀₁ , i₀₁) (xs₀ , extxs) = cohArg₀ E δ₀₁ i₀₁ xs₀ extxs

module _ {Γ : Con ℓΓ} {I : Ty ∅ ℓI} {Ds : ConDs I I} (Es : ConDs I I) where
  -- Type of arguments for a constructor list.
  μArgs : Ty (Γ ▷ I [ ε ]) ℓI
  ∣ μArgs ∣ (_ , i) = ∣μArgs∣ Ds Es  i
  μArgs .heq (_ , i₀₁) = ∣μArgs~∣ Ds Es 
  μArgs .refl* (xs₀ , extxs) = μArgs₀-refl* Es xs₀ extxs
  μArgs .sym* = μArgs₀-sym* Es
  μArgs .trans* {x₁ = _ , extxs₁} = μArgs₀-trans* Es
  μArgs .coe (_ , i₀₁) (xs₀ , extxs) = coeArgs₀ Es i₀₁ xs₀ , extArgs-coeArgs₀ Es i₀₁ extxs
  μArgs .coh (_ , i₀₁) (xs₀ , extxs) = cohArgs₀ Es i₀₁ xs₀ extxs

-- Generic constructor from semantic argument lists.
conᵢ
  : {Γ : Con ℓΓ}{I : Ty ∅ ℓI}(Ds : ConDs I I)
  → Tm (Γ ▷ I [ ε ] ▷ μArgs Ds [ ε ⁺ ]) (μ Ds [ wk ])
∣ conᵢ Ds ∣ ((γ , i) , (xs , extxs)) = con xs , extxs
conᵢ Ds .resp ((γ₀₁ , i₀₁) , t₀₁)    = t₀₁

-- Pointwise form of the generic constructor.
conₑ
  : {Γ : Con ℓΓ}{I : Ty ∅ ℓI}(Ds : ConDs I I)
  → (i : Tm Γ (I [ ε ]))
  → (xs : Tm Γ (μArgs Ds ⟨ i ⟩))
  → Tm Γ (μ Ds ⟨ i ⟩)
conₑ Ds i t = conᵢ Ds [ idS ▸ i ▸ t ]t

-- Recover index equality from equal constructor arguments.
module _ {I : Ty ∅ ℓI} {Ds : ConDs I I} where
  Arg~→I~
    : {Δ : Con ℓΔ}
    → let J = I [ ε ] in (E : ConD Δ J J)
    → ∀ {δ₀ δ₁ j₀ j₁}
    → {xs₀ : Arg₀ E (μ₀ Ds) δ₀ j₀} {xs₁ : Arg₀ E (μ₀ Ds) δ₁ j₁}
    → (δ₀₁ : Δ ∋ δ₀ ~ δ₁)
    → (xs₀₁ : Arg~ Ds E δ₀₁ xs₀ xs₁)
    → J ∋ tt ⊢ j₀ ~ j₁
  Arg~→I~ (end t) {xs₀ = q₀} {q₁} δ₀₁ _ =
    I .trans* (I .sym* (q₀ .↓)) (I .trans* (t .resp δ₀₁) (q₁ .↓))
  Arg~→I~ (non A E)   δ₀₁ (a₀₁ , xs₀₁) = Arg~→I~ E (δ₀₁ , a₀₁) xs₀₁
  Arg~→I~ (arg1 f E)  δ₀₁ (x₀₁ , xs₀₁) = Arg~→I~ E δ₀₁ xs₀₁
  Arg~→I~ (arg A f E) δ₀₁ (x₀₁ , xs₀₁) = Arg~→I~ E δ₀₁ xs₀₁

  -- Recover index equality from equal constructor-list arguments.
  Args~→I~
    : ∀ {i₀ i₁} (Es : ConDs I I)
    → {xs₀ : Args₀ Es (μ₀ Ds) i₀} {xs₁ : Args₀ Es (μ₀ Ds) i₁}
    → (xs₀₁ : Args~ Ds Es xs₀ xs₁)
    → I ∋ tt ⊢ i₀ ~ i₁
  Args~→I~ (E ∷ Es) {xs₀ = inl _} {inl _} xs₀₁ = Arg~→I~ E tt xs₀₁
  Args~→I~ (E ∷ Es) {xs₀ = inr _} {inr _} xs₀₁ = Args~→I~ Es xs₀₁
