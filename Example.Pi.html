idNat : Tm Γ (Nat ⇒ Nat)
idNat = ƛ vz

idNatₑ : Tm Γ Nat → Tm Γ Nat
idNatₑ n = appₑ (Nat [ wk ]) idNat n

succNat : Tm Γ (Nat ⇒ Nat)
succNat = ƛ su'

succNatₑ : Tm Γ Nat → Tm Γ Nat
succNatₑ n = appₑ (Nat [ wk ]) succNat n
