open import Prelude

module Model.Prop.Unit.Rules where

open import Model.Prop.CwF
open import Model.Prop.Unit.Core
open import Model.Prop.Unit.Elim

private variable
  ℓU : Level

Unit[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {σ : Sub Δ Γ}
  → Unit {Γ = Γ} {ℓ = ℓU} [ σ ] ≡ Unit {Γ = Δ} {ℓ = ℓU}
Unit[] = Eq.refl

⋆[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {σ : Sub Δ Γ}
  → ⋆ {Γ = Γ} {ℓ = ℓU} [ σ ]t ≡ ⋆ {Γ = Δ} {ℓ = ℓU}
⋆[] = Eq.refl

Unit-η : {Γ : Con ℓΓ} {t : Tm Γ (Unit {ℓ = ℓU})} → t ≡ ⋆
Unit-η = Eq.refl

elim-Unit[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} {σ : Sub Δ Γ}
  → {P : Ty (Γ ▷ Unit {ℓ = ℓU}) ℓ}
  → {t : Tm Γ (P ⟨ ⋆ {ℓ = ℓU} ⟩)}
  → elim-Unit P t [ σ ⁺ ]t
    ≡ elim-Unit (P [ σ ⁺ ]) (t [ σ ]t)
elim-Unit[] = Eq.refl
