module Prelude.Decidable where

-- Standalone replacement for the standard library's
-- `Relation.Nullary.Decidable.Core`.
--
-- Imports `Prelude.Base` so that `True`/`toWitness` use the *same* `⊤`/`tt`/`⊥`
-- as the rest of the project: defining a local `⊤` would introduce a third `tt`
-- and break type-directed disambiguation of the bare `tt` at use sites.

open import Prelude.Base

private variable
  p : Level
  P : Set p

data Dec (P : Set p) : Set p where
  yes : P → Dec P
  no  : (P → ⊥) → Dec P

True : Dec P → Set
True (yes _) = ⊤
True (no  _) = ⊥

toWitness : {a? : Dec P} → True a? → P
toWitness {a? = yes p} _ = p
