coeT
  : ∀ (A : Ty Δ ℓA) σ₀ σ₁ (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
  → (t₀ : Tm Γ (A [ σ₀ ]))
  → Tm Γ (A [ σ₁ ])
∣ coeT A _ _ σ₀₁ t₀ ∣ γ = A .coe (∣ σ₀₁ ∣ γ .↓) (∣ t₀ ∣ γ)
coeT {Δ = Δ} A _ _ σ₀₁ t₀ .resp {γ₀} {γ₁} γ₀₁ =
  A .trans* (A .sym* (A .coh (∣ σ₀₁ ∣ γ₀ .↓) (∣ t₀ ∣ γ₀)))
    (A .trans* (t₀ .resp γ₀₁) (A .coh (∣ σ₀₁ ∣ γ₁ .↓) (∣ t₀ ∣ γ₁)))

cohT
  : ∀ (A : Ty Δ ℓA) σ₀ σ₁ (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
  → (t₀ : Tm Γ (A [ σ₀ ]))
  → Tm Γ (⇑ EqT A σ₀ σ₁ σ₀₁ t₀ (coeT A σ₀ σ₁ σ₀₁ t₀))
∣ cohT A _ _ σ₀₁ t₀ ∣ γ = ↑
  (A .coh (∣ σ₀₁ ∣ γ .↓) (∣ t₀ ∣ γ))
cohT A _ _ σ₀₁ t₀ .resp _ = tt
