open import Prelude.ITT-ITT+Funext
  using (funext; funext²)

open Eq using (Jₚ)

private variable
  ℓI ℓE : Level

-- Which non-inductive arguments of a constructor are (strictly) regular.
data ConDIsReg {Δ : Con ℓΔ}
               {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
               {J : Ty ∅ ℓI}{E : Ty (∅ ▷ J) ℓ}
  : (d : ConD Δ I D J E) → Set (lsuc (ℓ ⊔ ℓI ⊔ ℓΔ))
  where

  end  : ∀ {t e} → ConDIsReg (end t e)
  non  : ∀ {A d} → (A-isReg : IsReg Δ A) → ConDIsReg d → ConDIsReg (non A d)
  arg1 : ∀ {f d} → ConDIsReg d → ConDIsReg (arg1 f d)
  arg  : ∀ {A f d} → (A-isReg : IsReg Δ A) → ConDIsReg d → ConDIsReg (arg A f d)

data ConDsAreReg {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}
  : (ds : ConDs I D I D) → Set (lsuc (ℓ ⊔ ℓI))
  where

  []  : ConDsAreReg []
  _∷_ : ∀ {d ds} → (d-isReg : ConDIsReg d) → ConDsAreReg ds → ConDsAreReg (d ∷ ds)

module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ}{ds : ConDs I D I D}
         (ds-areReg : ConDsAreReg ds)
  where

  open Eq.≡-Reasoning

  mutual
    coeμ₀-refl
      : {i : ∣ I ∣ tt}
      → (x : μ₀ ds i)
      → coeμ₀ (I .refl* i) x ≡ x
    coeμ₀-refl (con x) = Eq.cong con (coeArgs₀-refl ds-areReg x)

    coeArgs₀-refl
      : {es : ConDs I D I D}
      → ConDsAreReg es
      → {i : ∣ I ∣ tt}
      → (xs : Args₀ es (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) i)
      → coeArgs₀ es (I .refl* i) xs ≡ xs
    coeArgs₀-refl {E ∷ Es} (E-isReg ∷ Es-areReg) {i} (inr xs) =
      Eq.cong inr (coeArgs₀-refl Es-areReg xs)
    coeArgs₀-refl {E ∷ Es} (E-isReg ∷ Es-areReg) {i} (inl xs) =
      Eq.cong inl (coeArg₀-refl E E-isReg xs)

    coeArg₀-refl
      : {Δ : Con ℓΔ}
      → (e : ConD Δ I D I D)
      → ConDIsReg e
      → {δ : ∣ Δ ∣} {i : ∣ I ∣ tt}
      → (xs : Arg₀ e (μ₀ ds) (μ~ ds) (T₀ ds) (λ {x₀ = x₀} → Tresp₀ ds {x₀ = x₀}) δ i)
      → coeArg₀ e (Δ .refl δ) (I .refl* i) xs ≡ xs
    coeArg₀-refl (end t e) end xs = Eq.refl

    coeArg₀-refl {Δ = Δ} (non A e) (non A-isReg e-isReg) {δ} {i} (a , xs) =
      (A .coe (Δ .refl δ) a , coeArg₀ e (Δ .refl δ , A .coh (Δ .refl δ) a) (I .refl* i) xs)
        ≡⟨ Jₚ
             (λ cast _ →
                (cohr : ∀ δ a →
                  A ∋ Δ .refl δ ⊢
                  a ~ cast δ a) →
                _≡_
                  {A =
                    (Σ[ a ∈ ∣ A ∣ δ ]
                      Arg₀ e
                        (μ₀ ds) (μ~ ds)
                        (T₀ ds)
                        (λ {x₀ = x₀} →
                          Tresp₀ ds {x₀ = x₀})
                        (δ , a) i)}
                    (A .coe (Δ .refl δ) a
                    , coeArg₀ e
                        (Δ .refl δ , A .coh (Δ .refl δ) a)
                        (I .refl* i) xs)
                    (cast δ a
                    , coeArg₀ e
                        (Δ .refl δ , cohr δ a)
                        (I .refl* i) xs))
             (λ _ → Eq.refl)
             _
             A-isReg
             (λ δ a → A .refl* a) ⟩
      (a , coeArg₀ e (Δ .refl δ , A .refl* a) (I .refl* i) xs)
        ≡⟨ Eq.cong (λ ys → (a , ys))
             (coeArg₀-refl e e-isReg xs) ⟩
      (a , xs) ∎

    coeArg₀-refl {Δ = Δ} (arg1 f e) (arg1 e-isReg) {δ} {i} (x₀ , xs₀) =
          (coeμ₀ (f .resp (Δ .refl δ)) x₀
          , coeArg₀ e (Δ .refl δ , Tresp₀ ds {x₀ = x₀} (f .resp (Δ .refl δ)) (cohμ₀ (f .resp (Δ .refl δ)) x₀)) (I .refl* i) xs₀)
        ≡⟨ Eq.cong (λ (cast , coh) →
                  (cast
                  , coeArg₀ e (Δ .refl δ , Tresp₀ ds {x₀ = x₀} (f .resp (Δ .refl δ)) coh) (I .refl* i) xs₀))
             (Eq.Σₚ-≡ {B = λ cast → μ~ ds x₀ cast}
               {p = cohμ₀ (I .refl* (∣ f ∣ δ)) x₀}
               {q = μ~-refl* x₀}
               (coeμ₀-refl x₀)) ⟩
      (x₀ , coeArg₀ e (Δ .refl δ , Tresp₀ ds {x₀ = x₀} (f .resp (Δ .refl δ)) (μ~-refl* x₀)) (I .refl* i) xs₀)
        ≡⟨ Eq.cong (λ ys → x₀ , ys)
             (coeArg₀-refl e e-isReg xs₀) ⟩
      (x₀ , xs₀) ∎

    coeArg₀-refl {Δ = Δ} (arg A f e) (arg A-isReg e-isReg) {δ} {i} ((g₀ , gresp₀) , xs₀) =
      (coeArg₀ (arg A f e) (Δ .refl δ) (I .refl* i) ((g₀ , gresp₀) , xs₀))
        ≡⟨ Eq.cong
             (λ (gcast , gcoh) →
                    ( (gcast , λ {a₀} {a₁} a₀₁ →
                          μ~-trans* {x₀ = gcast a₀} {x₁ = g₀ a₀}
                            (μ~-sym* {x₀ = g₀ a₀} (gcoh (A .refl* a₀))) (gcoh a₀₁))
                    , coeArg₀ e
                        (Δ .refl δ , λ a₀ a₁ a₀₁ →
                           Tresp₀ ds {x₀ = g₀ a₀} (f .resp (Δ .refl δ , a₀₁)) (gcoh a₀₁))
                        (I .refl* i) xs₀))
             bundle-refl ⟩
          ((g₀ , gresp₀)
          , coeArg₀ e
              (Δ .refl δ , λ a₀ a₁ a₀₁ →
                 Tresp₀ ds {x₀ = g₀ a₀} (f .resp (Δ .refl δ , a₀₁)) (gresp₀ a₀₁))
              (I .refl* i) xs₀)
        ≡⟨ Eq.cong (λ ys → (g₀ , gresp₀) , ys) (coeArg₀-refl e e-isReg xs₀) ⟩
      ((g₀ , gresp₀) , xs₀) ∎
      where
        g₁ : (a₁ : ∣ A ∣ δ) → μ₀ ds (∣ f ∣ (δ , a₁))
        g₁ a₁ = coeμ₀ (f .resp (Δ .refl δ , coh' A (Δ .sym (Δ .refl δ)) a₁)) (g₀ (A .coe (Δ .sym (Δ .refl δ)) a₁))

        coh-g₁ : ∀ {a₀ a₁} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁) → μ~ ds (g₀ a₀) (g₁ a₁)
        coh-g₁ {a₀} {a₁} a₀₁ =
          μ~-trans* {x₀ = g₀ a₀}
            (gresp₀ (A .trans* a₀₁ (coh A (Δ .sym (Δ .refl δ)) a₁)))
            (cohμ₀ (f .resp (Δ .refl δ , coh' A (Δ .sym (Δ .refl δ)) a₁)) (g₀ (A .coe (Δ .sym (Δ .refl δ)) a₁)))

        g₁≡g₀ : g₁ ≡ g₀
        g₁≡g₀ = funext λ a₁ →
          coeμ₀ (f .resp (Δ .refl δ , coh' A (Δ .sym (Δ .refl δ)) a₁)) (g₀ (A .coe (Δ .sym (Δ .refl δ)) a₁))
            ≡⟨ Jₚ
                 (λ cast _ →
                    (cohr : ∀ δ a →
                      A ∋ Δ .refl δ ⊢
                      a ~ cast δ a) →
                      (coeμ₀
                        (f .resp
                          (Δ .refl δ
                          , coh' A (Δ .sym (Δ .refl δ)) a₁))
                        (g₀
                          (A .coe
                            (Δ .sym (Δ .refl δ)) a₁)))
                    ≡
                      (coeμ₀
                        (f .resp
                          (Δ .refl δ
                          , A .sym* (cohr δ a₁)))
                        (g₀
                          (cast δ a₁))))
                 (λ _ → Eq.refl)
                 _
                 A-isReg
                 (λ δ a → A .refl* a) ⟩
          coeμ₀ (f .resp (Δ .refl δ , A .refl* a₁)) (g₀ a₁)
            ≡⟨ coeμ₀-refl (g₀ a₁) ⟩
          g₀ a₁ ∎

        bundle-refl
          : _≡_ {A = (Σₚ[ gcast ∈ ((a₁ : ∣ A ∣ δ) → μ₀ ds (∣ f ∣ (δ , a₁))) ]
                       (∀ {a₀ a₁} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁) → μ~ ds (g₀ a₀) (gcast a₁)))}
              (g₁ , λ {a₀} {a₁} → coh-g₁ {a₀} {a₁})
              (g₀ , λ {a₀} {a₁} a₀₁ → gresp₀ a₀₁)
        bundle-refl =
          Eq.Σₚ-≡ {B = λ gcast → ∀ {a₀ a₁} (a₀₁ : A ∋ Δ .refl δ ⊢ a₀ ~ a₁) → μ~ ds (g₀ a₀) (gcast a₁)}
            {p = λ {a₀} {a₁} → coh-g₁ {a₀} {a₁}}
            {q = λ {a₀} {a₁} a₀₁ → gresp₀ a₀₁}
            g₁≡g₀

  -- The inductive-recursive type μ₀ is regular.
  μ-isReg
    : {Γ : Con ℓΓ}
    → IsReg (Γ ▷ I [ ε ]) (μ ds)
  μ-isReg =
    funext² _ _ λ (γ , i) x →
      coeμ₀-refl x

  -- The type of one constructor's arguments is regular.
  Args-isReg
    : {Γ : Con ℓΓ}
    → IsReg (Γ ▷ I [ ε ]) (Args {ds = ds} ds)
  Args-isReg =
    funext² _ _ λ (γ , i) xs →
      coeArgs₀-refl ds-areReg xs

  Arg-isReg
    : {Δ : Con ℓΔ} {e : ConD Δ I D I D}
    → (e-isReg : ConDIsReg e)
    → IsReg (Δ ▷ I [ ε ]) (Arg {ds = ds} e)
  Arg-isReg {Δ = Δ} {e = e} e-isReg =
    funext² _ _ λ (δ , i) xs →
      coeArg₀-refl e e-isReg xs
