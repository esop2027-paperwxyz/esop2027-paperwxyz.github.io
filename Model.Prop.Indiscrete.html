open import Prelude

module Model.Prop.Indiscrete where

open import Model.Prop.CwF

Indiscrete : (A : Set ℓA) → Ty ∅ ℓA
∣ Indiscrete A ∣ _ = A
Indiscrete A .heq _ _ _ = ⊤ₚ
Indiscrete A .refl* _ = tt
Indiscrete A .sym* _ = tt
Indiscrete A .trans* _ _ = tt
Indiscrete A .coe _ x = x
Indiscrete A .coh _ _ = tt
