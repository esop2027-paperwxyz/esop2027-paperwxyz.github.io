private variable
  ℓI : Level

-- Arg and Args live in the empty context, so only μ needs an explicit rule.

-- Substitution rule for μ.
μ[] : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ) {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} (Ds : ConDs I D I D)
    → (μ Ds {Γ = Γ}) [ σ ⁺ ] ≡ μ Ds {Γ = Δ}
μ[] σ Ds = Eq.refl

-- Instantiated substitution rule for μ.
μ[]' : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ) {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} (Ds : ConDs I D I D)
    → (i : Tm Γ (I [ ε ]))
    → μ Ds ⟨ i ⟩ [ σ ] ≡ μ Ds ⟨ i [ σ ]t ⟩
μ[]' σ Ds i = Eq.refl

-- Substitution rule for T.
T[] : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ) {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} (Ds : ConDs I D I D)
    → (T Ds {Γ = Γ}) [ σ ⁺ ⁺ ]t ≡ T Ds {Γ = Δ}
T[] σ Ds = Eq.refl

-- Substitution rule for the generic constructor.
conᵢ[]
  : {Δ : Con ℓΔ} {Γ : Con ℓΓ} {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} (Ds : ConDs I D I D)
  → (σ : Sub Δ Γ)
  → (conᵢ {Γ = Γ} Ds) [ σ ⁺ ⁺ ]t ≡ conᵢ {Γ = Δ} Ds
conᵢ[] Ds σ = Eq.refl

-- Substitution rule for the pointwise constructor.
conₑ[]
  : {Δ : Con ℓΔ} {Γ : Con ℓΓ} {σ : Sub Δ Γ}
  → {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} (Ds : ConDs I D I D)
  → {i : Tm Γ (I [ ε ])}
  → {t : Tm Γ (Args Ds ⟨ i ⟩)}
  → conₑ Ds i t [ σ ]t ≡ conₑ Ds (i [ σ ]t) (t [ σ ]t)
conₑ[] Ds = Eq.refl

-- Substitution rule for IH in Elim.
IH[]
  : {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Γ Δ)
    {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} {Ds : ConDs I D I D} (E : ConD Δ I D I D)
  → (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
  → {Θ : Con ℓ} (τ : Sub Θ Γ)
  → IH P             σ       E [ τ ⁺ ⁺ ]
  ≡ IH (P [ τ ⁺ ⁺ ]) (σ ∘ τ) E
IH[] σ E P τ = Eq.refl


module _ {I : Ty ∅ ℓI}{D : Ty (∅ ▷ I) ℓ} {Ds : ConDs I D I D}
  {Γ : Con ℓΓ} {Δ : Con ℓΔ} (σ : Sub Δ Γ)
  (P : Ty (Γ ▷ I [ ε ] ▷ μ Ds) ℓ)
  (pxs : Tm (Γ ▷ I [ ε ] ▷ Args Ds ▷ IHs P Ds)
            (P [ wk ▸ conᵢ Ds ] [ wk ]))
  where

  elim[]
    : elim P pxs [ σ ⁺ ⁺ ]t
    ≡ elim (P [ σ ⁺ ⁺ ]) (pxs [ σ ⁺ ⁺ ⁺ ]t)
  elim[] = Eq.refl

  -- Computation rule for elimination on a constructor.
  elim-con
    : {i : Tm Γ (I [ ε ])}
    → {xs : Tm Γ (Args Ds ⟨ i ⟩)}
    → elimₑ P pxs i (conₑ Ds i xs)
    ≡ pxs [ idS ▸ i ▸ xs ▸ mapIHsᵢ P pxs Ds [ idS ▸ i ▸ xs ]t ]t
  elim-con = Eq.refl
