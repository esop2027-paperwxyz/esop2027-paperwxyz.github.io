open import Prelude

module Model.Prop.Pi.Derived where

infixr 6 _⇒_

ƛ: : ∀ {ℓΓ ℓA ℓB} {Γ : Con ℓΓ} (A : Ty Γ ℓA) {B : Ty (Γ ▷ A) ℓB} → Tm (Γ ▷ A) B → Tm Γ (Π A B)
ƛ: _ = ƛ_
{-# INJECTIVE_FOR_INFERENCE ƛ: #-}

appₑ : (B : Ty (Γ ▷ A) ℓB) (t : Tm Γ (Π A B)) (u : Tm Γ A) → Tm Γ (B ⟨ u ⟩)
appₑ B t u = app t ⟨ u ⟩t

eval : (B : Ty (Γ ▷ A) ℓB) → Tm (Γ ▷ (Π A B) ▷ A [ wk ]) (B [ wk∘wk ▸ vz ])
eval B = appₑ (B [ wk∘wk ⁺ ]) (vs vz) vz

_⇒_ : (A : Ty Γ ℓA) (B : Ty Γ ℓB) → Ty Γ (ℓA ⊔ ℓB)
A ⇒ B = Π A (B [ wk ])
{-# INJECTIVE_FOR_INFERENCE _⇒_ #-}

app' : (A : Ty Γ ℓA) (B : Ty Γ ℓB) → Tm Γ (A ⇒ B) → Tm (Γ ▷ A) (B [ wk ])
app' A B = app
{-# INJECTIVE_FOR_INFERENCE app' #-}

appₑ' : Tm Γ (A ⇒ B) → Tm Γ A → Tm Γ B
appₑ' t u = appₑ _ t u
{-# INJECTIVE_FOR_INFERENCE appₑ' #-}

-- `app` is also expressible using `_∙_`
app-∙ : Tm Γ (Π A B) → Tm (Γ ▷ A) B
app-∙ {B = B} t = appₑ (B [ wk ⁺ ]) (t [ wk ]t) vz
