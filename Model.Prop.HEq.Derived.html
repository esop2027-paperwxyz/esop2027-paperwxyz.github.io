Eq : {A : Ty Γ ℓ} (t u : Tm Γ A)
  → Prp ℓ Γ
Eq {A = A} t u = EqT A idS idS (reflS idS) t u
{-# INJECTIVE_FOR_INFERENCE Eq #-}

reflT'
  : {Γ : Con ℓΓ} {A : Ty Γ ℓA}
  → (t : Tm Γ A)
  → Tm Γ (⇑ EqT A idS idS (reflS idS) t t)
reflT' {A = A} t = reflT A t
{-# INJECTIVE_FOR_INFERENCE reflT' #-}

congSʳ
  : ∀ (τ : Sub Θ Γ) (σ₀ σ₁ : Sub Γ Δ) (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
  → Tm Θ (⇑ EqS Δ (σ₀ ∘ τ) (σ₁ ∘ τ))
congSʳ τ σ₀ σ₁ σ₀₁ = σ₀₁ [ τ ]t

eqS-∅ : (σ₀ σ₁ : Sub Γ ∅) → Tm Γ (⇑ EqS ∅ σ₀ σ₁)
eqS-∅ σ₀ σ₁ = reflS σ₀

eqS-▸ : (A : Ty Δ ℓA)(σ₀ σ₁ : Sub Γ Δ)(t₀ : Tm Γ (A [ σ₀ ]))(t₁ : Tm Γ (A [ σ₁ ]))
      → (σ₀₁ : Tm Γ (⇑ EqS Δ σ₀ σ₁))
      → Tm Γ (⇑ EqT A σ₀ σ₁ σ₀₁ t₀ t₁)
      → Tm Γ (⇑ EqS (Δ ▷ A) (σ₀ ▸ t₀) (σ₁ ▸ t₁))
∣ eqS-▸ A σ₀ σ₁ t₀ t₁ σ₀₁ t₀₁ ∣ γ = ↑ (∣ σ₀₁ ∣ γ .↓ , ∣ t₀₁ ∣ γ .↓)
eqS-▸ A σ₀ σ₁ t₀ t₁ σ₀₁ t₀₁ .resp _ = tt

sgC
  : {A : Ty Γ ℓA} {t u : Tm Γ A}
  → (t~u : Tm Γ (⇑ Eq t u))
  → Tm Γ (⇑ EqS (Γ ▷ A) ⟨ t ⟩s ⟨ u ⟩s)
sgC = eqS-▸ _ idS idS _ _ (reflS idS)
